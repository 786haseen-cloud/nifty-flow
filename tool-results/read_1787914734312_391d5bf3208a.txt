    14→/**
    15→ * 4-Color OI classification for each bar:
    16→ *  CE: Buy  = ΔOI>0 & ΔLTP>0 (bullish)  → green
    17→ *      Write = ΔOI>0 & ΔLTP<0 (bearish)  → red
    18→ *  PE: Buy  = ΔOI>0 & ΔLTP<0 (bearish)  → red
    19→ *      Write = ΔOI>0 & ΔLTP>0 (bullish)  → green
    20→ *  Unwinding (ΔOI<0) → gray
    21→ *  No prev data       → neutral
    22→ */
    23→type OIColor = {
    24→  bg: string;      // tailwind bg class for the bar
    25→  text: string;     // tailwind text class for the OI label
    26→  label?: string;   // short label shown on hover ("Buy", "Write", "Unwind")
    27→};
    28→
    29→function getOIColor(oiChange: number, ltpChange: number, isPE: boolean): OIColor {
    30→  if (oiChange === 0 && ltpChange === 0) {
    31→    // No previous data — neutral
    32→    return { bg: 'bg-zinc-500/25', text: 'text-zinc-400/80' };
    33→  }
    34→  if (oiChange < 0) {
    35→    // Unwinding / short covering
    36→    return { bg: 'bg-zinc-500/30', text: 'text-zinc-500/70', label: 'Unwind' };
    37→  }
    38→  // oiChange > 0 — new positions opened
    39→  if (isPE) {
    40→    // PE: Write = bullish (price up + OI up), Buy = bearish (price down + OI up)
    41→    if (ltpChange > 0) return { bg: 'bg-emerald-500/50', text: 'text-emerald-300', label: 'Write' };
    42→    if (ltpChange < 0) return { bg: 'bg-red-500/50', text: 'text-red-300', label: 'Buy' };
    43→    return { bg: 'bg-zinc-500/30', text: 'text-zinc-400/80', label: 'Write' };
    44→  } else {
    45→    // CE: Buy = bullish (price up + OI up), Write = bearish (price down + OI up)
    46→    if (ltpChange > 0) return { bg: 'bg-emerald-500/50', text: 'text-emerald-300', label: 'Buy' };
    47→    if (ltpChange < 0) return { bg: 'bg-red-500/50', text: 'text-red-300', label: 'Write' };
    48→    return { bg: 'bg-zinc-500/30', text: 'text-zinc-400/80', label: 'Buy' };
    49→  }
    50→}

   109→function formatLakhs(n: number): string {
   110→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
   111→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
   112→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
   113→  return n.toLocaleString('en-IN');
   114→}

   120→// Compute Max Pain: strike where option buyers lose the most
   121→function computeMaxPain(strikes: StrikeFlowData[]): number {
   122→  if (strikes.length === 0) return 0;
   123→  let minLoss = Infinity;
   124→  let maxPainStrike = strikes[0].strike;
   125→
   126→  for (const k of strikes) {
   127→    let totalLoss = 0;
   128→    for (const s of strikes) {
   129→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
   130→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
   131→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
   132→      // But we only have OI, not premium. Standard Max Pain uses:
   133→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
   134→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
   135→      // Simplified: use intrinsic value approach
   136→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
   137→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
   138→    }
   139→    if (totalLoss < minLoss) {
   140→      minLoss = totalLoss;
   141→      maxPainStrike = k.strike;
   142→    }
   143→  }
   144→  return maxPainStrike;
   145→}

   147→// Generate demo data for 11 strikes
   148→function generateDemoData(symbol: string): StrikeFlowSnapshot {
   149→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   150→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   151→  const atmStrike = Math.round(spot / step) * step;
   152→  const strikes: StrikeFlowData[] = [];
   153→
   154→  for (let i = -5; i <= 5; i++) {
   155→    const strike = atmStrike + i * step;
   156→    const dist = Math.abs(i);
   157→    // Heavier OI near ATM, with some randomness
   158→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   159→    const noise = 0.6 + Math.random() * 0.8;
   160→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   161→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   162→    const moneyness = (spot - strike) / spot;
   163→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   164→    const ceITM = spot > strike;
   165→    const peITM = spot < strike;
   166→
   167→    strikes.push({
   168→      strike,
   169→      isATM: i === 0,
   170→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   171→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   172→      ceOI,
   173→      peOI,
   174→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   175→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   176→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   177→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   178→      ceToken: 100 + i,
   179→      peToken: 200 + i,
   180→    });
   181→  }
   182→
   183→  return {
   184→    timestamp: new Date().toISOString(),
   185→    symbol,
   186→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   187→    atmStrike,
   188→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   189→    strikeStep: step,
   190→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   191→    strikes,
   192→  };
   193→}

   195→// ═══════════════════════════════════════════
   196→// PCR HISTORY (localStorage)
   197→// ═══════════════════════════════════════════
   198→
   199→function getPCRHistoryKey(symbol: string): string {
   200→  const today = new Date().toISOString().split('T')[0];
   201→  return `oi-walls-pcr-history-${today}-${symbol}`;
   202→}
   203→
   204→function loadPCRHistory(symbol: string): number[] {
   205→  if (typeof window === 'undefined') return [];
   206→  try {
   207→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   208→    return raw ? JSON.parse(raw) : [];
   209→  } catch {
   210→    return [];
   211→  }
   212→}
   213→
   214→function savePCRHistory(symbol: string, history: number[]) {
   215→  if (typeof window === 'undefined') return;
   216→  try {
   217→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   218→  } catch {
   219→    // localStorage full or unavailable
   220→  }
   221→}

   227→export default function OIWallsTab() {
   228→  const [symbol, setSymbol] = useState('NIFTY');
   229→  const [snapshot, setSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   230→  const [loading, setLoading] = useState(false);
   231→  const [error, setError] = useState<string | null>(null);
   232→  const [isLive, setIsLive] = useState(false);
   233→  const [lastUpdate, setLastUpdate] = useState<string>('');
   234→  const [pcrHistory, setPcrHistory] = useState<number[]>([]);
   235→  const [demoData, setDemoData] = useState<StrikeFlowSnapshot | null>(null);
   236→
   237→  const prevSnapshotRef = useRef<StrikeFlowSnapshot | null>(null);
   238→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   239→
   240→  // Load PCR history on mount / symbol change
   241→  useEffect(() => {
   242→    setPcrHistory(loadPCRHistory(symbol));
   243→  }, [symbol]);
   244→
   245→  const fetchData = useCallback(async () => {
   246→    setLoading(true);
   247→    setError(null);
   248→    try {
   249→      const res = await fetch(withCreds(`/api/kite/strike-flow?symbol=${symbol}`));
   250→      const data = await res.json();
   251→
   252→      if (data.mode !== 'live') {
   253→        // Generate demo data
   254→        setIsLive(false);
   255→        const demo = generateDemoData(symbol);
   256→        setDemoData(demo);
   257→        setLastUpdate(new Date().toLocaleTimeString('en-IN', {
   258→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   259→        }));
   260→        setLoading(false);
   261→        return;
   262→      }
   263→
   264→      setIsLive(true);
   265→      setDemoData(null);
   266→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   267→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   268→      }));
   269→
   270→      // Compute OI change before updating ref
   271→      const prev = prevSnapshotRef.current;
   272→      if (prev && prev.symbol === data.symbol) {
   273→        // OI change is computed in metrics; we just need to store prev for diff
   274→      }
   275→      prevSnapshotRef.current = data;
   276→      setSnapshot(data);
   277→    } catch (err) {
   278→      setError(err instanceof Error ? err.message : 'Fetch failed');
   279→    } finally {
   280→      setLoading(false);
   281→    }
   282→  }, [symbol]);
   283→
   284→  // Initial fetch + polling
   285→  useEffect(() => {
   286→    fetchData();
   287→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   288→    return () => {
   289→      if (intervalRef.current) clearInterval(intervalRef.current);
   290→    };
   291→  }, [fetchData]);
   292→
   293→  // Compute OI change using ref to prev snapshot
   294→  const oiChange = (() => {
   295→    const prev = prevSnapshotRef.current;
   296→    const curr = snapshot;
   297→    if (!prev || !curr || prev.symbol !== curr.symbol) {
   298→      // Check demo data diff
   299→      if (demoData) return { ce: 0, pe: 0 };
   300→      return { ce: 0, pe: 0 };
   301→    }
   302→    let ceChange = 0, peChange = 0;
   303→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   304→    for (const s of curr.strikes) {
   305→      const p = prevMap.get(s.strike);
   306→      if (p) {
   307→        ceChange += s.ceOI - p.ceOI;
   308→        peChange += s.peOI - p.peOI;
   309→      }
   310→    }
   311→    return { ce: ceChange, pe: peChange };
   312→  })();
   313→
   314→  // Use demo or live data
   315→  const data = isLive ? snapshot : demoData;
   316→
   317→  // Compute walls and metrics
   318→  const { walls, metrics } = (() => {
   319→    if (!data || !data.strikes?.length) {
   320→      return { walls: [] as OIWallsStrike[], metrics: null as ComputedMetrics | null };
   321→    }
   322→
   323→    const maxOI = Math.max(
   324→      ...data.strikes.map(s => Math.max(s.ceOI, s.peOI)),
   325→      1
   326→    );
   327→
   328→    // Find ATM strike for ITM determination
   329→    const atmStrike = data.atmStrike || data.strikes.find(s => s.isATM)?.strike || 0;
   330→
   331→    // Build prev strike map for per-strike diff
   332→    const prev = prevSnapshotRef.current;
   333→    const prevMap = new Map<number, StrikeFlowData>();
   334→    if (prev && prev.symbol === data.symbol) {
   335→      for (const s of prev.strikes) prevMap.set(s.strike, s);
   336→    }
   337→
   338→    const walls: OIWallsStrike[] = data.strikes.map(s => {
   339→      const p = prevMap.get(s.strike);
   340→      const ceOiChg = p ? s.ceOI - p.ceOI : 0;
   341→      const peOiChg = p ? s.peOI - p.peOI : 0;
   342→      const ceLtpChg = p ? s.ceLTP - p.ceLTP : 0;
   343→      const peLtpChg = p ? s.peLTP - p.peLTP : 0;
   344→
   345→      return {
   346→        strike: s.strike,
   347→        isATM: s.isATM,
   348→        isCeITM: !s.isATM && s.strike < atmStrike,
   349→        isPeITM: !s.isATM && s.strike > atmStrike,
   350→        ceOI: s.ceOI,
   351→        peOI: s.peOI,
   352→        ceLTP: s.ceLTP,
   353→        peLTP: s.peLTP,
   354→        cePct: (s.ceOI / maxOI) * 100,
   355→        pePct: (s.peOI / maxOI) * 100,
   356→        ceColor: getOIColor(ceOiChg, ceLtpChg, false),
   357→        peColor: getOIColor(peOiChg, peLtpChg, true),
   358→      };
   359→    });
   360→
   361→    const totalCEOI = data.strikes.reduce((sum, s) => sum + s.ceOI, 0);
   362→    const totalPEOI = data.strikes.reduce((sum, s) => sum + s.peOI, 0);
   363→    const pcr = totalCEOI > 0 ? totalPEOI / totalCEOI : 0;
   364→    const maxPain = computeMaxPain(data.strikes);
   365→    const maxPainDist = data.spotPrice - maxPain;
   366→
   367→    const m: ComputedMetrics = {
   368→      pcr,
   369→      maxPain,
   370→      maxPainDist,
   371→      totalCEOI,
   372→      totalPEOI,
   373→      totalOI: totalCEOI + totalPEOI,
   374→      ceOIChange: oiChange.ce,
   375→      peOIChange: oiChange.pe,
   376→    };
   377→
   378→    return { walls, metrics: m };
   379→  })();
   380→
   381→  // Append PCR to history when metrics change
   382→  useEffect(() => {
   383→    if (!metrics) return;
   384→    setPcrHistory(prev => {
   385→      const updated = [...prev, metrics.pcr];
   386→      const trimmed = updated.slice(-60);
   387→      savePCRHistory(symbol, trimmed);
   388→      return trimmed;
   389→    });
   390→  }, [isLive ? snapshot?.timestamp : demoData?.timestamp, metrics?.pcr, symbol]);
   391→
   392→  // PCR label
   393→  function getPCRLabel(pcr: number): { text: string; color: string } {
   394→    if (pcr >= 1.5) return { text: 'Extreme Bearish', color: 'text-emerald-400' };
   395→    if (pcr >= 1.1) return { text: 'Bearish', color: 'text-emerald-300' };
   396→    if (pcr >= 0.9) return { text: 'Neutral', color: 'text-zinc-400' };
   397→    if (pcr >= 0.6) return { text: 'Bullish', color: 'text-red-400' };
   398→    return { text: 'Extreme Bullish', color: 'text-red-500' };
   399→  }

   404→  function PCRSparkline() {
   405→    if (pcrHistory.length < 2) {
   406→      return (
   407→        <div className="flex items-center justify-center h-10 text-zinc-600 text-[10px]">
   408→          Waiting for data points...
   409→        </div>
   410→      );
   411→    }
   412→
   413→    const w = 280, h = 36;
   414→    const vals = pcrHistory;
   415→    const minV = Math.min(...vals) * 0.95;
   416→    const maxV = Math.max(...vals) * 1.05;
   417→    const rangeV = maxV - minV || 1;
   418→    const step = w / (vals.length - 1);
   419→
   420→    const points = vals.map((v, i) => {
   421→      const x = i * step;
   422→      const y = h - 2 - ((v - minV) / rangeV) * (h - 4);
   423→      return `${x},${y}`;
   424→    }).join(' ');
   425→
   426→    // PCR = 1 reference line
   427→    const pcr1Y = h - 2 - ((1 - minV) / rangeV) * (h - 4);
   428→    const lastVal = vals[vals.length - 1];
   429→    const lastX = (vals.length - 1) * step;
   430→    const lastY = h - 2 - ((lastVal - minV) / rangeV) * (h - 4);
   431→    const lineColor = lastVal >= 1 ? '#34d399' : '#f87171';
   432→
   433→    return (
   434→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   435→        {/* PCR = 1 line */}
   436→        {pcr1Y > 0 && pcr1Y < h && (
   437→          <line x1="0" y1={pcr1Y} x2={w} y2={pcr1Y}
   438→            stroke="#525252" strokeWidth="0.5" strokeDasharray="3,3" />
   439→        )}
   440→        {/* Area fill */}
   441→        <polygon
   442→          points={`0,${h} ${points} ${lastX},${h}`}
   443→          fill={lastVal >= 1 ? 'rgba(52,211,153,0.1)' : 'rgba(248,113,113,0.1)'}
   444→        />
   445→        {/* Line */}
   446→        <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" />
   447→        {/* End dot */}
   448→        <circle cx={lastX} cy={lastY} r="2.5" fill={lineColor} />
   449→      </svg>
   450→    );
   451→  }

   469→            }}
   470→            className={
   471→              `text-[10px] px-2 py-1 rounded border transition-all duration-150 cursor-pointer whitespace-nowrap ` +

   478→          </button>

   480→      </div>
   481→
   482→      {/* ─── Status Bar ─── */}
   483→      <div className="flex items-center justify-between">
   484→        <div className="flex items-center gap-2">
   485→          {isLive ? (
   486→            <Badge variant="outline" className="gap-1 text-emerald-400 border-emerald-500/30 bg-emerald-500/5">
   487→              <Wifi className="w-3 h-3" /> LIVE

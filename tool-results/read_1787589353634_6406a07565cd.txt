   102→// ─── 4-Color Flow Engine ───
   103→// Computes net options flow from consecutive snapshots
   104→// Delta-weighted: (deltaOI * delta * lotSize) / 1Cr
   105→// OI decrease: multiply by 0.3 (short covering)
   106→
   107→function computeSymbolFlow(
   108→  prev: StrikeData[],
   109→  curr: StrikeData[],
   110→  lotSize: number
   111→): { bullish: number; bearish: number; net: number } {
   112→  let bullish = 0;
   113→  let bearish = 0;
   114→
   115→  const CR = 10000000; // 1 Crore
   116→
   117→  for (const currStrike of curr) {
   118→    const prevStrike = prev.find((s) => s.strike === currStrike.strike);
   119→    if (!prevStrike) continue;
   120→
   121→    const ceDeltaOI = currStrike.ceOI - prevStrike.ceOI;
   122→    const peDeltaOI = currStrike.peOI - prevStrike.peOI;
   123→    const ceDeltaPrice = currStrike.ceLTP - prevStrike.ceLTP;
   124→    const peDeltaPrice = currStrike.peLTP - prevStrike.peLTP;
   125→
   126→    // CE Flow
   127→    if (ceDeltaOI > 0) {
   128→      const val = (Math.abs(ceDeltaOI) * currStrike.ceDelta * lotSize) / CR;
   129→      if (ceDeltaPrice > 0) {
   130→        bullish += val; // CE Buy (bullish)
   131→      } else {
   132→        bearish += val; // CE Write (bearish)
   133→      }
   134→    } else if (ceDeltaOI < 0) {
   135→      // Short covering: 0.3 factor
   136→      const val = (Math.abs(ceDeltaOI) * 0.3 * currStrike.ceDelta * lotSize) / CR;
   137→      if (ceDeltaPrice > 0) {
   138→        bullish += val; // CE Buy (short cover)
   139→      } else {
   140→        bearish += val; // CE Write (short cover)
   141→      }
   142→    }
   143→
   144→    // PE Flow
   145→    if (peDeltaOI > 0) {
   146→      const val = (Math.abs(peDeltaOI) * currStrike.peDelta * lotSize) / CR;
   147→      if (peDeltaPrice > 0) {
   148→        bullish += val; // PE Write (bullish)
   149→      } else {
   150→        bearish += val; // PE Buy (bearish)
   151→      }
   152→    } else if (peDeltaOI < 0) {
   153→      const val = (Math.abs(peDeltaOI) * 0.3 * currStrike.peDelta * lotSize) / CR;
   154→      if (peDeltaPrice > 0) {
   155→        bullish += val; // PE Write (short cover)
   156→      } else {
   157→        bearish += val; // PE Buy (short cover)
   158→      }
   159→    }
   160→  }
   161→
   162→  return { bullish, bearish, net: bullish - bearish };
   163→}

   233→export default function TrendAnalysisTab() {
   234→  const isLive = hasKiteCreds();
   235→
   236→  // Trends API data (candles + cash flow)
   237→  const [niftyCandles, setNiftyCandles] = useState<NiftyCandle[]>([]);
   238→  const [stockCashFlow, setStockCashFlow] = useState<StockCashFlow[]>([]);
   239→  const [trendMode, setTrendMode] = useState<string>('demo');
   240→
   241→  // Highest-bet snapshots for options flow computation
   242→  const prevSnapshot = useRef<Map<string, StrikeData[]>>(new Map());
   243→  const [flowTrend, setFlowTrend] = useState<FlowTrendPoint[]>([]);
   244→  const cumulativeFlow = useRef<Record<string, number>>({
   245→    NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0, stockAggregate: 0,
   246→  });
   247→
   248→  // Cash flow trend accumulation
   249→  const [cashFlowTrend, setCashFlowTrend] = useState<CashFlowTrendPoint[]>([]);
   250→  const cumulativeCash = useRef({ nse: 0, bse: 0, net: 0, weighted: 0 });
   251→  const prevStockCashFlow = useRef<StockCashFlow[]>([]);
   252→
   253→  // Current flow values (for header cards)
   254→  const [currentIdxFlows, setCurrentIdxFlows] = useState<Record<string, number>>({
   255→    NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0,
   256→  });
   257→  const [currentStockFlow, setCurrentStockFlow] = useState(0);
   258→  const [currentIntervalCashFlow, setCurrentIntervalCashFlow] = useState(0);
   259→
   260→  // ─── Fetch trends data (candles + cash) ───
   261→
   262→  const fetchTrends = useCallback(async () => {
   263→    try {
   264→      let url = '/api/kite/trends';
   265→      url = withCreds(url);
   266→      const res = await fetch(url);
   267→      const data = await res.json();
   268→
   269→      setTrendMode(data.mode || 'demo');
   270→      if (data.niftyCandles) setNiftyCandles(data.niftyCandles);
   271→
   272→      // Build cash flow trend from consecutive snapshots
   273→      if (data.stockCashFlow && data.stockCashFlow.length > 0) {
   274→        setStockCashFlow(data.stockCashFlow);
   275→
   276→        const time = new Date().toLocaleTimeString('en-IN', {
   277→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   278→        });
   279→
   280→        const nseTotal = data.stockCashFlow.reduce((s, v) => s + v.nseCashFlow, 0);
   281→        const bseTotal = data.stockCashFlow.reduce((s, v) => s + v.bseCashFlow, 0);
   282→        const netTotal = nseTotal + bseTotal;
   283→        const weightedTotal = data.stockCashFlow.reduce((s, v) => s + v.weightedFlow, 0);
   284→
   285→        // Compute interval deltas (this poll minus previous poll)
   286→        // cashFlow = (ltp - open) * volume is CUMULATIVE since market open,
   287→        // so we must diff consecutive polls to get the 15s delta.
   288→        let nseDelta = 0;
   289→        let bseDelta = 0;
   290→        let weightedDelta = 0;
   291→        if (prevStockCashFlow.current.length > 0) {
   292→          const prevNse = prevStockCashFlow.current.reduce((s, v) => s + v.nseCashFlow, 0);
   293→          const prevBse = prevStockCashFlow.current.reduce((s, v) => s + v.bseCashFlow, 0);
   294→          const prevWeighted = prevStockCashFlow.current.reduce((s, v) => s + v.weightedFlow, 0);
   295→          nseDelta = nseTotal - prevNse;
   296→          bseDelta = bseTotal - prevBse;
   297→          weightedDelta = weightedTotal - prevWeighted;
   298→        }
   299→        const intervalDelta = nseDelta + bseDelta;
   300→
   301→        prevStockCashFlow.current = data.stockCashFlow;
   302→        setCurrentIntervalCashFlow(intervalDelta);
   303→
   304→        // Accumulate only the deltas
   305→        cumulativeCash.current.nse += nseDelta;
   306→        cumulativeCash.current.bse += bseDelta;
   307→        cumulativeCash.current.net += intervalDelta;
   308→        cumulativeCash.current.weighted += weightedDelta;
   309→
   310→        const CR = 10000000;
   311→        const point: CashFlowTrendPoint = {
   312→          time,
   313→          nse: Math.round((cumulativeCash.current.nse / CR) * 10) / 10,
   314→          bse: Math.round((cumulativeCash.current.bse / CR) * 10) / 10,
   315→          net: Math.round((cumulativeCash.current.net / CR) * 10) / 10,
   316→          weighted: Math.round((cumulativeCash.current.weighted / CR) * 10) / 10,
   317→          interval: Math.round((intervalDelta / CR) * 10) / 10,
   318→        };
   319→
   320→        setCashFlowTrend((prev) => {
   321→          const next = [...prev, point];
   322→          return next.length > 200 ? next.slice(-200) : next;
   323→        });
   324→      }
   325→    } catch (err) {
   326→      console.error('[TrendsTab] fetchTrends error:', err);
   327→    }
   328→  }, []);
   329→
   330→  // ─── Fetch highest-bet for options flow ───
   331→
   332→  const fetchOptionsFlow = useCallback(async () => {
   333→    try {
   334→      let url = '/api/kite/highest-bet';
   335→      url = withCreds(url);
   336→      const res = await fetch(url);
   337→      const data: HighestBetResponse = await res.json();
   338→
   339→      if (!data.symbols || data.symbols.length === 0) return;
   340→
   341→      const time = new Date().toLocaleTimeString('en-IN', {
   342→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   343→      });
   344→
   345→      // Compute per-symbol flow from consecutive snapshots
   346→      const intervalFlows: Record<string, number> = {};
   347→      const indexSymbols = ['NIFTY', 'BANKNIFTY', 'FINNIFTY', 'SENSEX'];
   348→      let stockAgg = 0;
   349→      const newCurrentIdx: Record<string, number> = {};
   350→
   351→      for (const sym of data.symbols) {
   352→        if (sym.strikes.length === 0) continue;
   353→
   354→        const prevStrikes = prevSnapshot.current.get(sym.symbol);
   355→        if (!prevStrikes || prevStrikes.length === 0) {
   356→          // First snapshot — store and skip
   357→          prevSnapshot.current.set(sym.symbol, sym.strikes);
   358→          continue;
   359→        }
   360→
   361→        const flow = computeSymbolFlow(prevStrikes, sym.strikes, sym.lotSize);
   362→        intervalFlows[sym.symbol] = flow.net;
   363→
   364→        if (sym.type === 'index' && indexSymbols.includes(sym.symbol)) {
   365→          cumulativeFlow.current[sym.symbol] += flow.net;
   366→          newCurrentIdx[sym.symbol] = flow.net;
   367→        } else if (sym.type === 'stock') {
   368→          stockAgg += flow.net;
   369→        }
   370→
   371→        // Update previous snapshot
   372→        prevSnapshot.current.set(sym.symbol, sym.strikes);
   373→      }
   374→
   375→      cumulativeFlow.current.stockAggregate += stockAgg;
   376→      setCurrentStockFlow(stockAgg);
   377→
   378→      // Update current index flows
   379→      for (const idx of indexSymbols) {
   380→        newCurrentIdx[idx] = newCurrentIdx[idx] || 0;
   381→      }
   382→      setCurrentIdxFlows(newCurrentIdx);
   383→
   384→      // Append to trend (keep last 200 points)
   385→      const point: FlowTrendPoint = {
   386→        time,
   387→        NIFTY: Math.round(cumulativeFlow.current.NIFTY * 10) / 10,
   388→        BANKNIFTY: Math.round(cumulativeFlow.current.BANKNIFTY * 10) / 10,
   389→        FINNIFTY: Math.round(cumulativeFlow.current.FINNIFTY * 10) / 10,
   390→        SENSEX: Math.round(cumulativeFlow.current.SENSEX * 10) / 10,
   391→        stockAggregate: Math.round(cumulativeFlow.current.stockAggregate * 10) / 10,
   392→      };
   393→
   394→      setFlowTrend((prev) => {
   395→        const next = [...prev, point];
   396→        return next.length > 200 ? next.slice(-200) : next;
   397→      });
   398→    } catch (err) {
   399→      console.error('[TrendsTab] fetchOptionsFlow error:', err);
   400→    }
   401→  }, []);
   402→
   403→  // ─── Polling ───
   404→
   405→  useEffect(() => {
   406→    fetchTrends();
   407→    fetchOptionsFlow();
   408→
   409→    const trendsInterval = setInterval(fetchTrends, 15000);
   410→    const flowInterval = setInterval(fetchOptionsFlow, 15000);
   411→
   412→    return () => {
   413→      clearInterval(trendsInterval);
   414→      clearInterval(flowInterval);
   415→    };
   416→  }, [fetchTrends, fetchOptionsFlow]);
   417→
   418→  // ─── Derived values ───
   419→
   420→  const niftyCurrentPrice = niftyCandles.length > 0 ? niftyCandles[niftyCandles.length - 1].close : 0;
   421→  const niftyOpenPrice = niftyCandles.length > 0 ? niftyCandles[0].close : 0;
   422→  const niftyChange = niftyOpenPrice > 0 ? niftyCurrentPrice - niftyOpenPrice : 0;
   423→  const niftyChangePct = niftyOpenPrice > 0 ? (niftyChange / niftyOpenPrice) * 100 : 0;
   424→
   425→  const totalNseFlow = stockCashFlow.reduce((s, v) => s + v.nseCashFlow, 0);
   426→  const totalBseFlow = stockCashFlow.reduce((s, v) => s + v.bseCashFlow, 0);
   427→  const totalCombinedFlow = totalNseFlow + totalBseFlow;
   428→  const totalWeightedFlow = stockCashFlow.reduce((s, v) => s + v.weightedFlow, 0);
   429→
   430→  // ─── Format helpers ───
   431→
   432→  const fmtCr = (v: number) => {
   433→    const cr = v / 10000000;
   434→    if (Math.abs(cr) >= 100) return `${cr.toFixed(0)}`;
   435→    if (Math.abs(cr) >= 1) return `${cr.toFixed(1)}`;
   436→    return `${cr.toFixed(2)}`;
   437→  };
   438→
   439→  // ─── Render ───
   440→
   441→  return (
   442→    <div className="space-y-4">
   443→      {/* Header with mode badge + summary cards */}
   444→      <div className="flex items-center justify-between flex-wrap gap-2">
   445→        <div className="flex items-center gap-2">
   446→          <Badge variant="outline" className={trendMode === 'live' ? 'border-emerald-500/40 text-emerald-300' : 'border-orange-500/40 text-orange-300'}>
   447→            {trendMode === 'live' ? 'LIVE' : 'Demo'}
   448→          </Badge>
   449→          <span className="text-xs text-muted-foreground">Trend Analysis — Price + Cash + Options Flow</span>
   450→        </div>
   451→        <div className="flex items-center gap-3 text-xs">
   452→          <span className="text-muted-foreground">Auto-refresh: 15s</span>
   453→          <span className="text-muted-foreground">|</span>
   454→          <span className="text-muted-foreground">Flow pts: {flowTrend.length}</span>
   455→        </div>
   456→      </div>
   457→
   458→      {/* Section 1: Nifty 50 Spot Price + Cash Flow Trend (stacked) */}
   459→      <div className="rounded-xl border border-border/50 bg-card/50 p-4">
   460→        <div className="flex items-center justify-between mb-3">
   461→          <div className="flex items-center gap-2">
   462→            <Activity className="h-4 w-4 text-emerald-400" />
   463→            <h3 className="text-sm font-semibold">Nifty 50 — Intraday Price Trend</h3>
   464→          </div>
   465→          <div className="flex items-center gap-3 text-xs">
   466→            {niftyCurrentPrice > 0 && (
   467→              <>
   468→                <span className="font-mono font-semibold text-emerald-400">
   469→                  {niftyCurrentPrice.toLocaleString('en-IN')}
   470→                </span>
   471→                <span className={`flex items-center gap-0.5 font-mono ${niftyChange >= 0 ? 'text-emerald-400' : 'text-red-400'}`}>
   472→                  {niftyChange >= 0 ? <ArrowUpRight className="h-3 w-3" /> : <ArrowDownRight className="h-3 w-3" />}
   473→                  {niftyChange >= 0 ? '+' : ''}{niftyChange.toFixed(1)} ({niftyChangePct >= 0 ? '+' : ''}{niftyChangePct.toFixed(2)}%)
   474→                </span>
   475→              </>
   476→            )}
   477→          </div>
   478→        </div>
   479→        <div className="h-[200px]">
   480→          {niftyCandles.length > 0 ? (
   481→            <ResponsiveContainer width="100%" height="100%">
   482→              <ComposedChart data={niftyCandles} margin={{ top: 5, right: 10, left: 10, bottom: 0 }}>
   483→                <defs>
   484→                  <linearGradient id="niftyGrad" x1="0" y1="0" x2="0" y2="1">
   485→                    <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
   486→                    <stop offset="100%" stopColor="#10b981" stopOpacity={0.02} />
   487→                  </linearGradient>

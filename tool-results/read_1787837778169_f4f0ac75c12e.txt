     1→/**
     2→ * Zerodha Kite API Integration
     3→ *
     4→ * Connects to Kite Connect REST API for real NSE/BSE market data.
     5→ * When KITE_API_KEY + KITE_ACCESS_TOKEN are set in .env, real data flows.
     6→ * Otherwise, falls back to demo data.
     7→ *
     8→ * Setup:
     9→ * 1. Go to https://developers.kite.trade
    10→ * 2. Create an app → get API Key + Secret
    11→ * 3. Login: https://kite.zerodha.com/connect/login?api_key=YOUR_KEY
    12→ * 4. After login, capture request_token from redirect URL
    13→ * 5. Call /api/kite/auth?request_token=xxx to generate access_token
    14→ * 6. Put access_token in .env → KITE_ACCESS_TOKEN=xxx
    15→ * 7. Redeploy
    16→ *
    17→ * Access tokens expire daily at midnight. Re-login each morning.
    18→ */

   100→/**
   101→ * Generate access_token from request_token (one-time, after login redirect)
   102→ */
   103→export async function generateSession(requestToken: string): Promise<{
   104→  accessToken: string;
   105→  refreshToken: string;
   106→} | null> {
   107→  const apiKey = process.env.KITE_API_KEY;
   108→  const apiSecret = process.env.KITE_API_SECRET;
   109→  if (!apiKey || !apiSecret) return null;
   110→
   111→  try {
   112→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   113→    const { createHash } = await import('crypto');
   114→    const checksum = createHash('sha256')
   115→      .update(apiKey + requestToken + apiSecret)
   116→      .digest('hex');
   117→
   118→    const res = await fetch(`${KITE_BASE}/session/token`, {
   119→      method: 'POST',
   120→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   121→      body: new URLSearchParams({
   122→        api_key: apiKey,
   123→        request_token: requestToken,
   124→        checksum,
   125→      }).toString(),
   126→    });
   127→
   128→    const data = await res.json();
   129→    if (data.status === 'success' && data.data) {
   130→      return {
   131→        accessToken: data.data.access_token,
   132→        refreshToken: data.data.refresh_token || '',
   133→      };
   134→    }
   135→    console.error('[Kite] Session error:', data.message);
   136→    return null;
   137→  } catch (err) {
   138→    console.error('[Kite] generateSession failed:', err);
   139→    return null;
   140→  }
   141→}

   154→export function invalidateInstrumentsCache(): void {

   159→/**
   160→ * Fetch all instruments from Kite (cached for 1 hour)
   161→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   162→ * We filter by exchange after caching
   163→ *
   164→ * @param exchange Optional exchange filter (e.g. 'NFO' returns only NFO F&O instruments)
   165→ * @param forceRefresh Bypass the 1-hour cache and re-download. Useful when creds change.
   166→ */
   167→export async function getInstruments(exchange?: string, forceRefresh?: boolean): Promise<KiteInstrument[]> {
   168→  // Return from cache if fresh (and not forced to refresh)
   169→  if (!forceRefresh && instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   170→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   171→  }
   172→
   173→  try {
   174→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   175→    if (!res.ok) {
   176→      const errText = await res.text();
   177→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   178→      return [];
   179→    }
   180→    const text = await res.text();
   181→
   182→    // Kite returns CSV format
   183→    const lines = text.trim().split('\n');
   184→    const instruments: KiteInstrument[] = [];
   185→
   186→    // Kite CSV column order (2025+ format):
   187→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   188→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   189→    //
   190→    // Kite changed the CSV format in 2025:
   191→    //   OLD: instrument_type was 'OPTIDX'/'OPTSTK'/'FUTIDX'/'FUTSTK'/'EQ'/'INDEX'
   192→    //        segment was 'NFO'/'BFO'/'NSE'/'BSE'
   193→    //   NEW: instrument_type is 'CE'/'PE'/'FUT'/'EQ'
   194→    //        segment is 'NFO-OPT'/'NFO-FUT'/'NFO'/'BFO-OPT'/'BFO-FUT'/'NSE'/'BSE'
   195→    // To avoid breaking the rest of the code (which filters by OPTIDX/OPTSTK/FUTIDX/FUTSTK),
   196→    // we normalize the new format back to the legacy types.
   197→    //
   198→    // Disambiguation: in the new CSV, ALL options (index + stock) are CE/PE — there is no
   199→    // index-vs-stock hint in the row itself. We use the `name` column to decide:
   200→    //   - If the underlying name matches a known F&O index (NIFTY, BANKNIFTY, FINNIFTY,
   201→    //     SENSEX, BANKEX, MIDCPNIFTY, etc.) → OPTIDX / FUTIDX
   202→    //   - Otherwise (RELIANCE, TCS, HDFCBANK, ...) → OPTSTK / FUTSTK
   203→    // The set below is checked at parse time (cheap Set.has lookup).
   204→    const KNOWN_FNO_INDICES = new Set([
   205→      'NIFTY', 'NIFTY 50',
   206→      'BANKNIFTY', 'NIFTY BANK',
   207→      'FINNIFTY', 'NIFTY FIN SERVICE',
   208→      'SENSEX', 'BANKEX',
   209→      'MIDCPNIFTY', 'NIFTY MID SELECT', 'NIFTY PSE',
   210→    ]);
   211→    const normalizeInstrumentType = (rawType: string, segment: string, name: string): string => {
   212→      // Keep legacy values as-is (in case Kite reverts or some entries still use old format)
   213→      if (rawType === 'OPTIDX' || rawType === 'OPTSTK' || rawType === 'FUTIDX' || rawType === 'FUTSTK' ||
   214→          rawType === 'INDEX') {
   215→        return rawType;
   216→      }
   217→      // CRITICAL FIX: In the 2025+ Kite CSV, cash index instruments (NIFTY 50, SENSEX, etc.)
   218→      // have instrument_type='EQ' and segment='INDICES'. We must normalize these to 'INDEX'
   219→      // so that downstream lookups (e.g. highest-bet finding cash instruments) can filter by
   220→      // instrumentType === 'INDEX' instead of the ambiguous 'EQ' (which also matches stocks).
   221→      if (rawType === 'EQ' && segment === 'INDICES') {
   222→        return 'INDEX';
   223→      }
   224→      // Regular EQ stocks stay as 'EQ'
   225→      if (rawType === 'EQ') {
   226→        return rawType;
   227→      }
   228→      const nameUpper = name.toUpperCase();
   229→      const isIndex = KNOWN_FNO_INDICES.has(nameUpper);
   230→      if (rawType === 'CE' || rawType === 'PE') {
   231→        // BFO always has index options only (SENSEX/BANKEX). NFO has both index + stock.
   232→        return isIndex ? 'OPTIDX' : 'OPTSTK';
   233→      }
   234→      if (rawType === 'FUT') {
   235→        return isIndex ? 'FUTIDX' : 'FUTSTK';
   236→      }
   237→      return rawType;
   238→    };
   239→
   240→    for (let i = 1; i < lines.length; i++) { // skip header
   241→      const cols = lines[i].split(',');
   242→      if (cols.length < 12) continue;
   243→      const rawType = cols[9] || '';
   244→      const segment = cols[10] || '';
   245→      // Kite wraps the `name` field in double quotes (e.g. "NIFTY").
   246→      // Strip them so name-based filters (e.g. .includes('NIFTY')) work correctly.
   247→      const rawName = cols[3] || '';
   248→      const name = rawName.startsWith('"') && rawName.endsWith('"')
   249→        ? rawName.slice(1, -1)
   250→        : rawName;
   251→      instruments.push({
   252→        instrumentToken: parseInt(cols[0]) || 0,
   253→        exchangeToken: parseInt(cols[1]) || 0,
   254→        tradingSymbol: cols[2],
   255→        name,
   256→        exchange: cols[11] || '',
   257→        segment,
   258→        instrumentType: normalizeInstrumentType(rawType, segment, name),
   259→        strike: parseFloat(cols[6]) || 0,
   260→        lotSize: parseInt(cols[8]) || 1,
   261→        expiry: cols[5] || '',
   262→      });
   263→    }
   264→
   265→    instrumentsCache = instruments;
   266→    instrumentsCacheTime = Date.now();
   267→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   268→  } catch (err) {
   269→    const errMsg = err instanceof Error ? err.message : String(err);
   270→    console.error('[Kite] instruments fetch failed:', errMsg);
   271→    // Return empty but set error flag for callers to detect
   272→    return [];
   273→  }
   274→}

   278→/**
   279→ * Get real-time quotes for instruments
   280→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   281→ */
   282→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   283→  if (!isKiteConfigured()) return {};
   284→
   285→  try {
   286→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   287→    // Tokens have no encoding issues on any platform.
   288→    // Build a map from token → original instrument key for response mapping.
   289→    const tokenToKey: Record<string, string> = {};
   290→    const tokenList: string[] = [];
   291→
   292→    for (const inst of instruments) {
   293→      // If it's a numeric token, use directly
   294→      if (/^\d+$/.test(inst)) {
   295→        tokenToKey[inst] = inst;
   296→        tokenList.push(inst);
   297→      } else {
   298→        // Look up token from KITE_INDEX_INSTRUMENTS
   299→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   300→          ([, v]) => v.symbol === inst
   301→        );
   302→        if (found) {
   303→          const tok = String(found[1].token);
   304→          tokenToKey[tok] = inst; // map token back to symbol name
   305→          tokenList.push(tok);
   306→        } else {
   307→          // Fallback: use the instrument string as-is with encodeURI
   308→          tokenToKey[inst] = inst;
   309→          tokenList.push(inst);
   310→        }
   311→      }
   312→    }
   313→
   314→    const iList = tokenList.join(',');
   315→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   316→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   317→    // Chunk into batches of 100 to avoid URL length limits and API throttling.
   318→    const CHUNK_SIZE = 100;
   319→    const allQuotes: Record<string, KiteQuote> = {};
   320→
   321→    for (let chunkStart = 0; chunkStart < tokenList.length; chunkStart += CHUNK_SIZE) {
   322→      const chunk = tokenList.slice(chunkStart, chunkStart + CHUNK_SIZE);
   323→      const params = chunk.map(t => `i=${t}`).join('&');
   324→      const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   325→        headers: kiteHeaders(),
   326→      });
   327→
   328→      // Log non-success for debugging
   329→      if (!res.ok) {
   330→        const errText = await res.text();
   331→        console.error(`[Kite] quote API ${res.status}: ${errText}`);
   332→        if (Object.keys(allQuotes).length === 0) {
   333→          return { _error: `HTTP ${res.status}: ${errText}` } as any;
   334→        }
   335→        continue; // partial success — return what we have
   336→      }
   337→
   338→      const data = await res.json();
   339→      if (data.status !== 'success' || !data.data) {
   340→        console.error('[Kite] quote API error:', JSON.stringify(data));
   341→        if (Object.keys(allQuotes).length === 0) {
   342→          return { _error: data.message || 'Unknown error' } as any;
   343→        }
   344→        continue;
   345→      }
   346→
   347→      for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   348→        const displayKey = tokenToKey[key] || key;
   349→        allQuotes[displayKey] = {
   350→          instrumentToken: q.instrument_token || 0,
   351→          lastPrice: q.last_price || 0,
   352→          open: q.ohlc?.open || 0,
   353→          high: q.ohlc?.high || 0,
   354→          low: q.ohlc?.low || 0,
   355→          close: q.ohlc?.close || 0,
   356→          volume: q.volume || 0,
   357→          netChange: q.net_change || 0,
   358→          averagePrice: q.average_price || 0,
   359→          oi: q.oi || 0,
   360→          oiDayHigh: q.oi_day_high || 0,
   361→          oiDayLow: q.oi_day_low || 0,
   362→          dayHigh: q.ohlc?.high || 0,
   363→          dayLow: q.ohlc?.low || 0,
   364→        };
   365→      }
   366→    }
   367→
   368→    return allQuotes;
   369→  } catch (err) {
   370→    const errMsg = err instanceof Error ? err.message : String(err);
   371→    console.error('[Kite] quote error:', errMsg);
   372→    return { _error: `Exception: ${errMsg}` } as any;
   373→  }
   374→}

   378→/**
   379→ * Get historical candle data for price line chart
   380→ * instrumentToken: numeric token from instruments list
   381→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   382→ */
   383→export async function getCandles(
   384→  instrumentToken: number,
   385→  interval: string = '15minute',
   386→  days: number = 1,
   387→): Promise<KiteHistoricalCandle[]> {
   388→  if (!isKiteConfigured()) return [];
   389→
   390→  // Kite historical API expects IST dates.
   391→  // On Vercel (UTC), we must convert to IST before formatting.
   392→  const IST_OFFSET = 5.5 * 60 * 60 * 1000;
   393→  const toIST = (d: Date) => new Date(d.getTime() + IST_OFFSET);
   394→
   395→  const toDate = toIST(new Date());
   396→  const fromDate = toIST(new Date());
   397→  fromDate.setDate(fromDate.getDate() - days);
   398→
   399→  // Kite historical API requires format "YYYY-MM-DD HH:MM:SS" (WITH seconds).
   400→  // Sending "YYYY-MM-DD HH:MM" (no seconds) returns HTTP 400 "invalid from date".
   401→  const fmt = (d: Date) =>
   402→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}:00`;
   403→
   404→  try {
   405→    const fromStr = fmt(fromDate);
   406→    const toStr = fmt(toDate);
   407→    // URL-encode the date strings (they contain a space: "2026-08-25 09:15")
   408→    const url = `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${encodeURIComponent(fromStr)}&to=${encodeURIComponent(toStr)}&continuous=0`;
   409→
   410→    const res = await fetch(url, { headers: kiteHeaders() });
   411→
   412→    // Surface HTTP-level errors (403 expired token, 429 rate limit, etc.)
   413→    if (!res.ok) {
   414→      const errText = await res.text();
   415→      console.error(`[Kite] candles HTTP ${res.status} for token ${instrumentToken}: ${errText.substring(0, 300)}`);
   416→      return [];
   417→    }
   418→
   419→    const data = await res.json();
   420→    if (data.status !== 'success' || !data.data?.candles) {
   421→      console.error('[Kite] candles non-success: status=%s, message=%s, from=%s, to=%s',
   422→        data.status, data.message || 'no message', fromStr, toStr);
   423→      return [];
   424→    }
   425→
   426→    return (data.data.candles as any[][]).map(c => ({
   427→      timestamp: c[0],
   428→      open: c[1],
   429→      high: c[2],
   430→      low: c[3],
   431→      close: c[4],
   432→      volume: c[5],
   433→      oi: c[6] || 0,  // c[6] = OI (F&O only), undefined for cash
   434→    }));
   435→  } catch (err) {
   436→    const errMsg = err instanceof Error ? err.message : String(err);
   437→    console.error('[Kite] candles error:', errMsg);
   438→    return [];
   439→  }
   440→}

   444→/**
   445→ * Get option chain for an index or stock
   446→ * Uses InstrumentSpec for correct exchange, segment
   447→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   448→ * Returns all CE + PE instruments for the current expiry near ATM
   449→ */
   450→export async function getOptionInstruments(
   451→  symbol: string,
   452→  spotPrice: number,
   453→  strikesAroundOverride?: number,
   454→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   455→  const spec = getInstrumentSpec(symbol);
   456→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   457→
   458→  // IMPORTANT: F&O options/futures are NOT on the cash exchange.
   459→  //   - NIFTY index spot is on exchange='NSE', but NIFTY OPTIONS are on exchange='NFO'
   460→  //   - SENSEX spot is on exchange='BSE', but SENSEX OPTIONS are on exchange='BFO'
   461→  //   - Stock (RELIANCE/TCS/...) spot is on exchange='NSE', but their OPTIONS are on exchange='NFO'
   462→  // So we use spec.segment ('NFO' or 'BFO') — which equals the F&O exchange column in
   463→  // Kite's CSV — NOT spec.exchange ('NSE'/'BSE', which is the cash exchange).
   464→  // If you call getInstruments(spec.exchange) you'll only get cash/index instruments,
   465→  // and the option filter below will return ZERO matches.
   466→  const instruments = await getInstruments(spec.segment);
   467→
   468→  // Filter options using spec.
   469→  //
   470→  // Kite changed their CSV format in 2025:
   471→  //   OLD: segment='NFO'/'BFO', instrument_type='OPTIDX'/'OPTSTK'
   472→  //   NEW: segment='NFO-OPT'/'BFO-OPT', instrument_type='CE'/'PE' (we normalize to OPTIDX)
   473→  //
   474→  // So we use a "starts with" match on segment (so 'NFO' matches both 'NFO' and 'NFO-OPT'),
   475→  // and an equality match on instrumentType (we've already normalized new → legacy).
   476→  const symUpper = symbol.toUpperCase();
   477→  const indexOptions = instruments.filter(i =>
   478→    i.segment.startsWith(spec.segment) &&
   479→    i.instrumentType === spec.instrumentType &&
   480→    (i.name.toUpperCase().includes(symUpper) ||
   481→     i.tradingSymbol.toUpperCase().includes(symUpper))
   482→  );
   483→
   484→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   485→
   486→  // Get unique expiries, sort by nearest
   487→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   488→
   489→  // Use nearest expiry that's not expired
   490→  const today = new Date();
   491→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   492→
   493→  // Filter to nearest expiry
   494→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   495→
   496→  // Get lot size from first option (all same for same underlying)
   497→  const lotSize = expiryOptions[0]?.lotSize || 1;
   498→
   499→  // Derive strike step DYNAMICALLY from actual strike gaps
   500→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   501→  let strikeStep = 50;
   502→  if (strikes.length >= 2) {
   503→    const gaps: Record<number, number> = {};
   504→    for (let i = 1; i < strikes.length; i++) {
   505→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   506→      gaps[gap] = (gaps[gap] || 0) + 1;
   507→    }
   508→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   509→  }
   510→
   511→  // Round ATM strike using the DYNAMIC strike step
   512→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   513→
   514→  // Get strikes around ATM
   515→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   516→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   517→    atmStrike - strikesAround * strikeStep + i * strikeStep
   518→  );
   519→
   520→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   521→
   522→  return {
   523→    instruments: filtered,
   524→    meta: { lotSize, strikeStep },
   525→  };
   526→}

   528→/**
   529→ * Build options flow data from quotes + OI changes
   530→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   531→ */
   532→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   533→  const spec = getInstrumentSpec(symbol);
   534→  if (!spec) return null;
   535→
   536→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   537→  if (optInstruments.length === 0) return null;
   538→
   539→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   540→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   541→
   542→  // Batch fetch quotes
   543→  const quotes = await getQuotes(iKeys);
   544→  if (Object.keys(quotes).length === 0) return null;
   545→
   546→  // Group by strike → build flow data
   547→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   548→
   549→  // Use DYNAMIC strike step and lot size from CSV
   550→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   551→
   552→  for (const inst of optInstruments) {
   553→    // Quotes are keyed by instrument token (string)
   554→    const key = String(inst.instrumentToken);
   555→    const q = quotes[key];
   556→    if (!q) continue;
   557→
   558→    if (!strikes.has(inst.strike)) {
   559→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   560→    }
   561→    const s = strikes.get(inst.strike)!;
   562→
   563→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   564→    const flow = q.volume * meta.lotSize * q.averagePrice;
   565→
   566→    if (inst.tradingSymbol.endsWith('CE')) {
   567→      s.callBuy = flow * 0.6;
   568→      s.callWrite = flow * 0.4;
   569→      s.callOI = q.oi;
   570→      s.callVol = q.volume;
   571→    } else if (inst.tradingSymbol.endsWith('PE')) {
   572→      s.putBuy = flow * 0.6;
   573→      s.putWrite = flow * 0.4;
   574→      s.putOI = q.oi;
   575→      s.putVol = q.volume;
   576→    }
   577→  }
   578→
   579→  // Build result
   580→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   581→    strike,
   582→    isATM: strike === atmStrike,
   583→    ...flow,
   584→    bullishFlow: flow.callBuy + flow.putWrite,
   585→    bearishFlow: flow.putBuy + flow.callWrite,
   586→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   587→  })).sort((a, b) => a.strike - b.strike);
   588→}

   669→/**
   670→ * Derive lot size and strike step from Kite instruments CSV
   671→ * - lotSize: from the option instrument's lot_size field
   672→ * - strikeStep: derived from the gap between adjacent strikes
   673→ */
   674→export async function getInstrumentMeta(
   675→  symbol: string,
   676→  spotPrice: number,
   677→): Promise<InstrumentMeta> {
   678→  const spec = getInstrumentSpec(symbol);
   679→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   680→
   681→  // Fetch instruments from the F&O exchange (NFO/BFO), NOT the cash exchange (NSE/BSE).
   682→  // See getOptionInstruments() for the full explanation.
   683→  const allInstruments = await getInstruments(spec.segment);
   684→
   685→  // Filter to this symbol's options for current/near expiry.
   686→  // Use startsWith on segment to match both legacy 'NFO' and new 'NFO-OPT' formats.
   687→  const symbolOpts = allInstruments.filter(i =>
   688→    i.segment.startsWith(spec.segment) &&
   689→    i.instrumentType === spec.instrumentType &&
   690→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   691→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   692→  );
   693→
   694→  if (symbolOpts.length === 0) {
   695→    return { lotSize: 1, strikeStep: 50 }; // fallback
   696→  }
   697→
   698→  // Get lot size from the first matching option (all have same lot size for same underlying)
   699→  const lotSize = symbolOpts[0].lotSize || 1;
   700→
   701→  // Derive strike step from unique sorted strikes near ATM
   702→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   703→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   704→
   705→  const expiryOpts = nearestExpiry
   706→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   707→    : symbolOpts;
   708→
   709→  // Get unique strikes, sorted
   710→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   711→
   712→  // Find strikes near ATM and compute step from adjacent gaps
   713→  let strikeStep = 50; // default fallback
   714→  if (strikes.length >= 2) {
   715→    // Use the most common gap between adjacent strikes
   716→    const gaps: Record<number, number> = {};
   717→    for (let i = 1; i < strikes.length; i++) {
   718→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   719→      gaps[gap] = (gaps[gap] || 0) + 1;
   720→    }
   721→    // Most frequent gap = the strike step
   722→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   723→  }
   724→
   725→  return { lotSize, strikeStep };
   726→}

   737→// ─── Black-Scholes Delta Calculator ───
   738→// Approximate delta for options when Kite doesn't provide Greeks.
   739→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   740→
   741→function normCDF(x: number): number {
   742→  // Rational approximation (max error ~7.5e-8)
   743→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   744→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   745→  const sign = x < 0 ? -1 : 1;
   746→  const absX = Math.abs(x);
   747→  const t = 1.0 / (1.0 + p * absX);
   748→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   749→  return 0.5 * (1.0 + sign * y);
   750→}
   751→
   752→/**
   753→ * Calculate Black-Scholes delta for an option
   754→ * @param isCall - true for CE, false for PE
   755→ * @param S - Spot price
   756→ * @param K - Strike price
   757→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   758→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   759→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   760→ */
   761→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   762→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   763→  // If no IV provided, estimate from moneyness
   764→  if (sigma <= 0) {
   765→    const moneyness = (S - K) / S;
   766→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   767→  }
   768→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   769→  const delta = normCDF(d1);
   770→  return isCall ? delta : delta - 1;
   771→}

   778→export interface StrikeFlowSnapshot {
   779→  timestamp: string;
   780→  symbol: string;

   785→  expiry: string;

   790→  strike: number;
   791→  isATM: boolean;
   792→  ceLTP: number;
   793→  peLTP: number;
   794→  ceOI: number;
   795→  peOI: number;
   796→  ceVol: number;
   797→  peVol: number;
   798→  ceDelta: number;
   799→  peDelta: number;
   800→  ceToken: number;
   801→  peToken: number;

   804→/**
   805→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   806→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   807→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   808→ */
   809→export async function getStrikeFlowSnapshot(
   810→  symbol: string,
   811→  spotPrice: number,
   812→): Promise<StrikeFlowSnapshot | null> {
   813→  const spec = getInstrumentSpec(symbol);
   814→  if (!spec) return null;
   815→
   816→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   817→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   818→  if (optInstruments.length === 0) return null;
   819→
   820→  // Batch fetch quotes using instrument tokens
   821→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   822→  const quotes = await getQuotes(iKeys);
   823→  if ('_error' in quotes) return null;
   824→
   825→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   826→
   827→  // Group by strike
   828→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   829→  for (const inst of optInstruments) {
   830→    if (!strikeMap.has(inst.strike)) {
   831→      strikeMap.set(inst.strike, { ce: null, pe: null });
   832→    }
   833→    const entry = strikeMap.get(inst.strike)!;
   834→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   835→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   836→  }
   837→
   838→  // Calculate time to expiry
   839→  const expiry = optInstruments[0]?.expiry || '';
   840→  const now = new Date();
   841→  const expiryDate = new Date(expiry);
   842→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   843→  const T = daysToExpiry / 365;
   844→
   845→  // Build strike data
   846→  const strikes: StrikeFlowData[] = [];
   847→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   848→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   849→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   850→
   851→    const ceLTP = ceQuote?.lastPrice || 0;
   852→    const peLTP = peQuote?.lastPrice || 0;
   853→    const ceOI = ceQuote?.oi || 0;
   854→    const peOI = peQuote?.oi || 0;
   855→    const ceVol = ceQuote?.volume || 0;
   856→    const peVol = peQuote?.volume || 0;
   857→
   858→    // Compute delta via Black-Scholes
   859→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   860→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   861→
   862→    strikes.push({
   863→      strike,
   864→      isATM: strike === atmStrike,
   865→      ceLTP,
   866→      peLTP,
   867→      ceOI,
   868→      peOI,
   869→      ceVol,
   870→      peVol,
   871→      ceDelta,
   872→      peDelta,
   873→      ceToken: ce?.instrumentToken || 0,
   874→      peToken: pe?.instrumentToken || 0,
   875→    });
   876→  }
   877→
   878→  strikes.sort((a, b) => a.strike - b.strike);
   879→
   880→  return {
   881→    timestamp: new Date().toISOString(),
   882→    symbol,
   883→    spotPrice,
   884→    atmStrike,
   885→    lotSize: meta.lotSize,
   886→    strikeStep: meta.strikeStep,
   887→    expiry,
   888→    strikes,
   889→  };
   890→}

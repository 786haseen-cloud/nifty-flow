    37→    37→// OI change + Price change → 4 cases
    38→    38→// ═══════════════════════════════════════════
    39→    39→
    40→    40→function computeStrikeFlow(
    41→    41→  prev: StrikeFlowData,
    42→    42→  curr: StrikeFlowData,
    43→    43→  lotSize: number,
    44→    44→): Omit<StrikeFlowCell, 'strike' | 'isATM'> {
    45→    45→  const DIVISOR = 10000000; // Convert to Crores
    46→    46→
    47→    47→  // CE side
    48→    48→  const dceOI = curr.ceOI - prev.ceOI;
    49→    49→  const dceP = curr.ceLTP - prev.ceLTP;
    50→    50→  let ceBuy = 0, ceWrite = 0;
    51→    51→
    52→    52→  if (dceOI > 0 && dceP >= 0) {
    53→    53→    // Case 1: OI up + Price up → CE BUY (writers losing, buyers aggressive)
    54→    54→    ceBuy = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    55→    55→  } else if (dceOI > 0 && dceP < 0) {
    56→    56→    // Case 2: OI up + Price down → CE WRITE (writing at higher premium, price falling)
    57→    57→    ceWrite = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    58→    58→  } else if (dceOI < 0 && dceP < 0) {
    59→    59→    // Case 3: OI down + Price down → CE WRITE (short covering at lower price = write)
    60→    60→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    61→    61→  } else if (dceOI < 0 && dceP >= 0) {
    62→    62→    // Case 4: OI down + Price up → CE WRITE (long unwinding at higher price = write)
    63→    63→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    64→    64→  }
    65→    65→
    66→    66→  // Volume fallback when OI unchanged (from your script v5)
    67→    67→  if (ceBuy === 0 && ceWrite === 0 && curr.ceVol > 0) {
    68→    68→    const vf = 0.4;
    69→    69→    if (dceP >= 0) ceBuy = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    70→    70→    else ceWrite = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    71→    71→  }
    72→    72→
    73→    73→  // PE side (same 4 cases, mirrored)
    74→    74→  const dpeOI = curr.peOI - prev.peOI;
    75→    75→  const dpeP = curr.peLTP - prev.peLTP;
    76→    76→  let peBuy = 0, peWrite = 0;
    77→    77→
    78→    78→  if (dpeOI > 0 && dpeP >= 0) {
    79→    79→    peBuy = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    80→    80→  } else if (dpeOI > 0 && dpeP < 0) {
    81→    81→    peWrite = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    82→    82→  } else if (dpeOI < 0 && dpeP < 0) {
    83→    83→    peWrite = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    84→    84→  } else if (dpeOI < 0 && dpeP >= 0) {
    85→    85→    peBuy = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    86→    86→  }
    87→    87→
    88→    88→  // Volume fallback for PE
    89→    89→  if (peBuy === 0 && peWrite === 0 && curr.peVol > 0) {
    90→    90→    const vf = 0.4;
    91→    91→    if (dpeP >= 0) peBuy = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    92→    92→    else peWrite = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    93→    93→  }
    94→    94→
    95→    95→  // Determine display values (show dominant flow per side)
    96→    96→  const ceDisplay = ceBuy >= ceWrite ? ceBuy : ceWrite;
    97→    97→  const peDisplay = peWrite >= peBuy ? peWrite : peBuy;
    98→    98→  const ceType = ceBuy > 0 && ceBuy >= ceWrite ? 'buy' : ceWrite > 0 ? 'write' : 'none';
    99→    99→  const peType = peWrite > 0 && peWrite >= peBuy ? 'write' : peBuy > 0 ? 'buy' : 'none';
   100→   100→
   101→   101→  return { ceBuy, ceWrite, peBuy, peWrite, ceDisplay, peDisplay, ceType, peType };
   102→   102→}

   206→   206→  // Compute flow cells from two snapshots
   207→   207→  function computeFlowCells(prev: StrikeFlowSnapshot, curr: StrikeFlowSnapshot): StrikeFlowCell[] {

   219→   219→  // Compute totals

   221→   221→    let ceBuy = 0, ceWrite = 0, peBuy = 0, peWrite = 0;
   222→   222→    for (const c of cells) {
   223→   223→      ceBuy += c.ceBuy;
   224→   224→      ceWrite += c.ceWrite;
   225→   225→      peBuy += c.peBuy;
   226→   226→      peWrite += c.peWrite;
   227→   227→    }
   228→   228→    return {
   229→   229→      ceBuy: Math.round(ceBuy * 100) / 100,
   230→   230→      ceWrite: Math.round(ceWrite * 100) / 100,
   231→   231→      peBuy: Math.round(peBuy * 100) / 100,
   232→   232→      peWrite: Math.round(peWrite * 100) / 100,
   233→   233→      netFlow: Math.round((ceBuy + peWrite - ceWrite - peBuy) * 100) / 100,
   234→   234→    };

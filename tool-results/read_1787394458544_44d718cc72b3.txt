    26→interface FlowTotals {

    34→// ═══════════════════════════════════════════
    35→// 4-COLOR FLOW ENGINE (from your Google Script)
    36→// OI change + Price change → 4 cases
    37→// ═══════════════════════════════════════════
    38→
    39→function computeStrikeFlow(
    40→  prev: StrikeFlowData,
    41→  curr: StrikeFlowData,
    42→  lotSize: number,
    43→): Omit<StrikeFlowCell, 'strike' | 'isATM'> {
    44→  const DIVISOR = 10000000; // Convert to Crores
    45→
    46→  // CE side
    47→  const dceOI = curr.ceOI - prev.ceOI;
    48→  const dceP = curr.ceLTP - prev.ceLTP;
    49→  let ceBuy = 0, ceWrite = 0;
    50→
    51→  if (dceOI > 0 && dceP >= 0) {
    52→    // Case 1: OI up + Price up → CE BUY (writers losing, buyers aggressive)
    53→    ceBuy = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    54→  } else if (dceOI > 0 && dceP < 0) {
    55→    // Case 2: OI up + Price down → CE WRITE (writing at higher premium, price falling)
    56→    ceWrite = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    57→  } else if (dceOI < 0 && dceP < 0) {
    58→    // Case 3: OI down + Price down → CE WRITE (short covering at lower price = write)
    59→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    60→  } else if (dceOI < 0 && dceP >= 0) {
    61→    // Case 4: OI down + Price up → CE WRITE (long unwinding at higher price = write)
    62→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    63→  }
    64→
    65→  // Volume fallback when OI unchanged (from your script v5)
    66→  if (ceBuy === 0 && ceWrite === 0 && curr.ceVol > 0) {
    67→    const vf = 0.4;
    68→    if (dceP >= 0) ceBuy = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    69→    else ceWrite = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    70→  }
    71→
    72→  // PE side (same 4 cases, mirrored)
    73→  const dpeOI = curr.peOI - prev.peOI;
    74→  const dpeP = curr.peLTP - prev.peLTP;
    75→  let peBuy = 0, peWrite = 0;
    76→
    77→  if (dpeOI > 0 && dpeP >= 0) {
    78→    peBuy = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    79→  } else if (dpeOI > 0 && dpeP < 0) {
    80→    peWrite = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    81→  } else if (dpeOI < 0 && dpeP < 0) {
    82→    peWrite = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    83→  } else if (dpeOI < 0 && dpeP >= 0) {
    84→    peBuy = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    85→  }
    86→
    87→  // Volume fallback for PE
    88→  if (peBuy === 0 && peWrite === 0 && curr.peVol > 0) {
    89→    const vf = 0.4;
    90→    if (dpeP >= 0) peBuy = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    91→    else peWrite = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    92→  }
    93→
    94→  // Determine display values (show dominant flow per side)
    95→  const ceDisplay = ceBuy >= ceWrite ? ceBuy : ceWrite;
    96→  const peDisplay = peWrite >= peBuy ? peWrite : peBuy;
    97→  const ceType = ceBuy > 0 && ceBuy >= ceWrite ? 'buy' : ceWrite > 0 ? 'write' : 'none';
    98→  const peType = peWrite > 0 && peWrite >= peBuy ? 'write' : peBuy > 0 ? 'buy' : 'none';
    99→
   100→  return { ceBuy, ceWrite, peBuy, peWrite, ceDisplay, peDisplay, ceType, peType };
   101→}

   133→const REFRESH_INTERVAL = 30000; // 30 seconds (options data doesn't change every second)
   134→const ALL_SYMBOLS = [
   135→  ...INDEX_SPECS.map(s => ({ symbol: s.symbol, name: s.name, type: 'index' as const })),

   138→
   139→export default function StrikeFlowMap() {
   140→  const [symbol, setSymbol] = useState('NIFTY');
   141→  const [prevSnapshot, setPrevSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   142→  const [currSnapshot, setCurrSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   143→  const [loading, setLoading] = useState(false);
   144→  const [error, setError] = useState<string | null>(null);
   145→  const [isLive, setIsLive] = useState(false);
   146→  const [lastUpdate, setLastUpdate] = useState<string>('');
   147→  const [flowHistory, setFlowHistory] = useState<FlowTotals[]>([]);
   148→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   149→
   150→  const fetchData = useCallback(async () => {
   151→    setLoading(true);
   152→    setError(null);
   153→    try {
   154→      const res = await fetch(`/api/kite/strike-flow?symbol=${symbol}`);
   155→      const data = await res.json();
   156→
   157→      if (data.mode === 'demo') {
   158→        setIsLive(false);
   159→        setError('Kite API not configured. Set KITE_API_KEY + KITE_ACCESS_TOKEN in Settings.');
   160→        setLoading(false);
   161→        return;
   162→      }
   163→
   164→      if (data.mode === 'error') {
   165→        setError(data.error || 'Unknown error');
   166→        setLoading(false);
   167→        return;
   168→      }
   169→
   170→      setIsLive(true);
   171→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   172→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   173→      }));
   174→
   175→      // Shift current to prev, set new current
   176→      setPrevSnapshot(prev => {
   177→        if (prev && prev.symbol === data.symbol && data.strikes?.length >= 11) {
   178→          // We have both snapshots — compute flow
   179→          const cells = computeFlowCells(prev, data);
   180→          const totals = computeTotals(cells);
   181→          setFlowHistory(h => {
   182→            const updated = [...h, { ...totals, netFlow: totals.netFlow }];
   183→            return updated.slice(-60); // keep last 60 intervals
   184→          });
   185→        }
   186→        return currSnapshot;
   187→      });
   188→      setCurrSnapshot(data);
   189→    } catch (err) {
   190→      setError(err instanceof Error ? err.message : 'Fetch failed');
   191→    } finally {
   192→      setLoading(false);
   193→    }
   194→  }, [symbol, currSnapshot]);
   195→
   196→  // Initial fetch + interval
   197→  useEffect(() => {
   198→    fetchData();
   199→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   200→    return () => {
   201→      if (intervalRef.current) clearInterval(intervalRef.current);
   202→    };
   203→  }, [fetchData]);
   204→
   205→  // Compute flow cells from two snapshots
   206→  function computeFlowCells(prev: StrikeFlowSnapshot, curr: StrikeFlowSnapshot): StrikeFlowCell[] {
   207→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   208→    return curr.strikes
   209→      .filter(s => prevMap.has(s.strike))
   210→      .map(s => ({
   211→        strike: s.strike,
   212→        isATM: s.isATM,
   213→        ...computeStrikeFlow(prevMap.get(s.strike)!, s, curr.lotSize),
   214→      }))
   215→      .sort((a, b) => a.strike - b.strike);
   216→  }
   217→
   218→  // Compute totals
   219→  function computeTotals(cells: StrikeFlowCell[]): FlowTotals {
   220→    let ceBuy = 0, ceWrite = 0, peBuy = 0, peWrite = 0;
   221→    for (const c of cells) {
   222→      ceBuy += c.ceBuy;
   223→      ceWrite += c.ceWrite;
   224→      peBuy += c.peBuy;
   225→      peWrite += c.peWrite;
   226→    }
   227→    return {
   228→      ceBuy: Math.round(ceBuy * 100) / 100,
   229→      ceWrite: Math.round(ceWrite * 100) / 100,
   230→      peBuy: Math.round(peBuy * 100) / 100,
   231→      peWrite: Math.round(peWrite * 100) / 100,
   232→      netFlow: Math.round((ceBuy + peWrite - ceWrite - peBuy) * 100) / 100,
   233→    };
   234→  }
   235→
   236→  // Compute current flow from stored prev/curr
   237→  const flowCells: StrikeFlowCell[] = (() => {
   238→    if (!prevSnapshot || !currSnapshot) return [];
   239→    if (prevSnapshot.symbol !== currSnapshot.symbol) return [];
   240→    if (currSnapshot.strikes.length < 11) return [];
   241→    return computeFlowCells(prevSnapshot, currSnapshot);
   242→  })();
   243→
   244→  const totals: FlowTotals = flowCells.length > 0 ? computeTotals(flowCells) : { ceBuy: 0, ceWrite: 0, peBuy: 0, peWrite: 0, netFlow: 0 };
   245→
   246→  // Find max single bet for thick border
   247→  let maxBetVal = 0, maxBetSide: 'ce' | 'pe' | null = null, maxBetIdx = -1;
   248→  for (let i = 0; i < flowCells.length; i++) {
   249→    const c = flowCells[i];
   250→    if (c.ceDisplay > maxBetVal) { maxBetVal = c.ceDisplay; maxBetSide = 'ce'; maxBetIdx = i; }
   251→    if (c.peDisplay > maxBetVal) { maxBetVal = c.peDisplay; maxBetSide = 'pe'; maxBetIdx = i; }
   252→  }
   253→
   254→  // Dynamic offsets based on strikeStep (works for Nifty=50, HDFCBANK=10, etc.)
   255→  const step = currSnapshot ? currSnapshot.strikeStep : 50;
   256→  const n = 5;
   257→  const CE_OFFSETS = Array.from({ length: n * 2 + 1 }, function(_, i) { return (n - i) * step; });
   258→  const PE_OFFSETS = Array.from({ length: n * 2 + 1 }, function(_, i) { return (i - n) * step; });
   259→  const CE_CATS = Array.from({ length: n * 2 + 1 }, function(_, i) { return i < n ? 'OTM' : i === n ? 'ATM' : 'ITM'; });
   260→  const PE_CATS = Array.from({ length: n * 2 + 1 }, function(_, i) { return i < n ? 'ITM' : i === n ? 'ATM' : 'OTM'; });
   261→
   262→  function getCellByOffset(cells: StrikeFlowCell[], offset: number): StrikeFlowCell | undefined {
   263→    if (!currSnapshot) return undefined;
   264→    const strike = currSnapshot.atmStrike + offset;
   265→    return cells.find(c => c.strike === strike);
   266→  }
   267→
   268→  function cellBg(type: 'buy' | 'write' | 'none', side: 'ce' | 'pe'): string {
   269→    if (type === 'none') return COLORS.dark;
   270→    if (side === 'ce') return type === 'buy' ? COLORS.ceBuy : COLORS.ceWrite;
   271→    return type === 'write' ? COLORS.peWrite : COLORS.peBuy;
   272→  }
   273→
   274→  function cellText(type: 'buy' | 'write' | 'none', side: 'ce' | 'pe'): string {
   275→    if (type === 'none') return COLORS.zero;
   276→    if (side === 'ce') return type === 'buy' ? COLORS.ceBuyText : COLORS.ceWriteText;
   277→    return type === 'write' ? COLORS.peWriteText : COLORS.peBuyText;
   278→  }
   279→
   280→  function catStyle(cat: string) {
   281→    const c = cat === 'OTM' ? COLORS.otm : cat === 'ATM' ? COLORS.atm : COLORS.itm;
   282→    return { background: c.bg, color: c.fg };
   283→  }
   284→
   285→  function fmtVal(v: number): string {
   286→    if (v === 0) return '—';
   287→    return v >= 0 ? `+${v.toFixed(2)}` : v.toFixed(2);
   288→  }
   289→
   290→  // ═══════════════════════════════════════════
   291→  // NET FLOW MINI CHART (last 60 intervals)
   292→  // ═══════════════════════════════════════════
   293→  function NetFlowMiniChart() {
   294→    if (flowHistory.length < 2) return null;
   295→    const vals = flowHistory.map(h => h.netFlow);
   296→    const maxAbs = Math.max(...vals.map(Math.abs), 0.01);
   297→    const w = 300, h = 40;
   298→    const step = w / (vals.length - 1);
   299→
   300→    const points = vals.map((v, i) => `${i * step},${h / 2 - (v / maxAbs) * (h / 2 - 2)}`).join(' ');
   301→    const zeroLine = `M0,${h / 2} L${w},${h / 2}`;
   302→
   303→    // Fill area
   304→    const areaPoints = `0,${h / 2} ${points} ${w},${h / 2}`;
   305→    const lastVal = vals[vals.length - 1];
   306→    const fill = lastVal >= 0 ? 'rgba(0,176,80,0.15)' : 'rgba(189,33,48,0.15)';
   307→    const stroke = lastVal >= 0 ? '#00B050' : '#BD2130';
   308→
   309→    return (
   310→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   311→        <line x1="0" y1={h/2} x2={w} y2={h/2} stroke="#333" strokeWidth="0.5" />
   312→        <polygon points={areaPoints} fill={fill} />
   313→        <polyline points={points} fill="none" stroke={stroke} strokeWidth="1.5" />
   314→        {/* Current value dot */}
   315→        <circle cx={(vals.length - 1) * step} cy={h / 2 - (lastVal / maxAbs) * (h / 2 - 2)} r="2.5" fill={stroke} />
   316→      </svg>
   317→    );
   318→  }
   319→
   320→  // ═══════════════════════════════════════════
   321→  // TOTALS STACKED BAR (matching your SF_TOTAL sheet)
   322→  // ═══════════════════════════════════════════
   323→  function TotalFlowBar() {
   324→    const maxTotal = Math.max(Math.abs(totals.ceBuy), Math.abs(totals.ceWrite),
   325→      Math.abs(totals.peWrite), Math.abs(totals.peBuy), 0.01);
   326→    const barH = 20;
   327→
   328→    return (
   329→      <div className="space-y-1">
   330→        <div className="flex items-center gap-2 text-[10px]">
   331→          <span className="text-muted-foreground w-16">CE Buy</span>
   332→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   333→            <div className="h-full rounded-sm transition-all duration-500" style={{
   334→              width: `${Math.min(100, (totals.ceBuy / maxTotal) * 100)}%`,
   335→              background: COLORS.ceBuy,
   336→              minWidth: totals.ceBuy > 0 ? '2px' : '0',
   337→            }} />
   338→          </div>
   339→          <span className="w-16 text-right font-mono" style={{ color: COLORS.ceBuy }}>{totals.ceBuy.toFixed(2)} Cr</span>
   340→        </div>
   341→        <div className="flex items-center gap-2 text-[10px]">
   342→          <span className="text-muted-foreground w-16">CE Write</span>
   343→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   344→            <div className="h-full rounded-sm transition-all duration-500" style={{
   345→              width: `${Math.min(100, (totals.ceWrite / maxTotal) * 100)}%`,
   346→              background: COLORS.ceWrite,
   347→              minWidth: totals.ceWrite > 0 ? '2px' : '0',
   348→            }} />
   349→          </div>
   350→          <span className="w-16 text-right font-mono" style={{ color: COLORS.ceWriteText }}>{totals.ceWrite.toFixed(2)} Cr</span>
   351→        </div>
   352→        <div className="flex items-center gap-2 text-[10px]">
   353→          <span className="text-muted-foreground w-16">PE Write</span>
   354→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   355→            <div className="h-full rounded-sm transition-all duration-500" style={{
   356→              width: `${Math.min(100, (totals.peWrite / maxTotal) * 100)}%`,
   357→              background: COLORS.peWrite,
   358→              minWidth: totals.peWrite > 0 ? '2px' : '0',
   359→            }} />
   360→          </div>
   361→          <span className="w-16 text-right font-mono" style={{ color: COLORS.peWriteText }}>{totals.peWrite.toFixed(2)} Cr</span>
   362→        </div>
   363→        <div className="flex items-center gap-2 text-[10px]">
   364→          <span className="text-muted-foreground w-16">PE Buy</span>
   365→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   366→            <div className="h-full rounded-sm transition-all duration-500" style={{
   367→              width: `${Math.min(100, (totals.peBuy / maxTotal) * 100)}%`,
   368→              background: COLORS.peBuy,
   369→              minWidth: totals.peBuy > 0 ? '2px' : '0',
   370→            }} />
   371→          </div>
   372→          <span className="w-16 text-right font-mono" style={{ color: COLORS.peBuyText }}>{totals.peBuy.toFixed(2)} Cr</span>
   373→        </div>
   374→      </div>
   375→    );
   376→  }
   377→
   378→  // ═══════════════════════════════════════════
   379→  // RENDER
   380→  // ═══════════════════════════════════════════
   381→
   382→  return (
   383→    <div className="space-y-3">
   384→      {/* Header Bar */}
   385→      <div className="flex items-center justify-between flex-wrap gap-2">
   386→        <div className="flex items-center gap-2">
   387→          <h2 className="text-sm font-bold" style={{ color: COLORS.accent }}>
   388→            STRIKE FLOW MAP
   389→          </h2>
   390→          {isLive ? (
   391→            <Badge variant="outline" className="text-[10px] border-emerald-500/40 text-emerald-300">
   392→              <Wifi className="mr-1 h-3 w-3" />Kite LIVE
   393→            </Badge>
   394→          ) : (
   395→            <Badge variant="outline" className="text-[10px] border-red-500/40 text-red-300">
   396→              <WifiOff className="mr-1 h-3 w-3" />Offline
   397→            </Badge>
   398→          )}
   399→          {loading && <RefreshCw className="h-3 w-3 animate-spin text-muted-foreground" />}
   400→          {lastUpdate && (
   401→            <span className="text-[10px] font-mono text-muted-foreground">Updated: {lastUpdate}</span>
   402→          )}
   403→        </div>
   404→
   405→        {/* Symbol Selector — dropdown for 19 symbols */}
   406→        <div className="flex items-center gap-2">
   407→          <select
   408→            value={symbol}
   409→            onChange={function(e) { setSymbol(e.target.value); setPrevSnapshot(null); setCurrSnapshot(null); setFlowHistory([]); }}
   410→            className="bg-muted/50 border border-border/50 rounded-md px-2 py-1 text-xs font-mono focus:outline-none focus:ring-1 focus:ring-purple-500/50"
   411→          >
   412→            <optgroup label="Indices">
   413→              {ALL_SYMBOLS.filter(function(s) { return s.type === 'index'; }).map(function(s) {
   414→                return <option key={s.symbol} value={s.symbol}>{s.name}</option>;
   415→              })}
   416→            </optgroup>
   417→            <optgroup label="Stocks">
   418→              {ALL_SYMBOLS.filter(function(s) { return s.type === 'stock'; }).map(function(s) {
   419→                return <option key={s.symbol} value={s.symbol}>{s.name}</option>;
   420→              })}
   421→            </optgroup>
   422→          </select>
   423→          <Badge variant="outline" className="text-[9px] border-purple-500/30 text-purple-300">
   424→            {ALL_SYMBOLS.find(function(s) { return s.symbol === symbol; }) ? 'INDEX/STOCK' : ''}
   425→          </Badge>
   426→        </div>
   427→      </div>
   428→
   429→      {error && (
   430→        <div className="rounded-md p-3 text-xs" style={{ background: '#2e0a0a', color: '#ff6666', border: '1px solid #5a1a1a' }}>
   431→          {error}
   432→        </div>
   433→      )}
   434→
   435→      {/* Waiting for 2 snapshots */}
   436→      {isLive && !error && flowCells.length === 0 && (
   437→        <div className="rounded-md p-4 text-center text-xs text-muted-foreground" style={{ background: COLORS.dark }}>
   438→          <Crosshair className="h-5 w-5 mx-auto mb-2 animate-pulse" style={{ color: COLORS.info }} />
   439→          <p className="font-semibold" style={{ color: COLORS.info }}>Collecting snapshots...</p>

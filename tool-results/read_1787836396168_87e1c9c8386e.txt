    39→    39→function NiftyTooltip({ active, payload, label }: any) {
    40→    40→  if (!active || !payload?.length) return null;
    41→    41→  const d = payload[0]?.payload as NiftyCandle;
    42→    42→  return (
    43→    43→    <div className="bg-card border border-border rounded-lg p-2 shadow-xl text-xs">
    44→    44→      <div className="font-mono text-muted-foreground mb-1">{label}</div>
    45→    45→      <div className="text-emerald-400 font-semibold">Close: {d?.close?.toLocaleString('en-IN')}</div>
    46→    46→      {d?.high ? (
    47→    47→        <div className="text-muted-foreground">
    48→    48→          H: {d.high.toLocaleString('en-IN')} L: {d.low.toLocaleString('en-IN')}
    49→    49→        </div>
    50→    50→      ) : null}
    51→    51→      {d?.volume ? (
    52→    52→        <div className="text-muted-foreground">Vol: {(d.volume / 1000000).toFixed(1)}M</div>
    53→    53→      ) : null}
    54→    54→    </div>

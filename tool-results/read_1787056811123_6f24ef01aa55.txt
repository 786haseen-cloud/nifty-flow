    69→    if (Math.abs(changePercent) > 0.5 && changePercent < 0) impactOnNifty = 'negative';

    73→      country: idx.country,
    74→      value,
    75→      change,

   108→// --- Expiry Info ---
   109→function getNextExpiry(expiryDay: number, expiryType: 'weekly' | 'monthly'): { date: Date; daysToExpiry: number } {
   110→  const now = new Date();
   111→  const istOffset = 5.5 * 60 * 60 * 1000;
   112→  const istNow = new Date(now.getTime() + istOffset);
   113→
   114→  if (expiryType === 'weekly') {
   115→    // Find next expiryDay (0=Sun, 1=Mon, 2=Tue, ..., 6=Sat)
   116→    const currentDay = istNow.getUTCDay();
   117→    let daysUntil = expiryDay - currentDay;
   118→    if (daysUntil <= 0) daysUntil += 7;
   119→    const expiryDate = new Date(istNow);
   120→    expiryDate.setUTCDate(expiryDate.getUTCDate() + daysUntil);
   121→    expiryDate.setUTCHours(15, 30, 0, 0);
   122→    const diff = expiryDate.getTime() - istNow.getTime();
   123→    return { date: expiryDate, daysToExpiry: Math.max(1, Math.ceil(diff / (24 * 60 * 60 * 1000))) };
   124→  } else {
   125→    // Last specific weekday of current or next month
   126→    const currentMonth = istNow.getUTCMonth();
   127→    const currentYear = istNow.getUTCFullYear();
   128→    const targetDay = expiryDay; // reuse for monthly: -1 means last Tuesday
   129→
   130→    const actualDay = expiryDay === -1 ? 2 : expiryDay; // Tuesday = 2 for BankNifty/Finnifty
   131→    // For Bankex: Thursday = 4 (handled separately in caller)
   132→
   133→    let month = currentMonth;
   134→    let year = currentYear;
   135→
   136→    // Find last Tuesday/Thursday of the month
   137→    const lastDay = new Date(Date.UTC(year, month + 1, 0));
   138→    let lastTarget = lastDay.getUTCDay();
   139→    let diff2 = lastTarget - actualDay;
   140→    if (diff2 < 0) diff2 += 7;
   141→    if (diff2 > 6) diff2 -= 7;
   142→    let expiryDayNum = lastDay.getUTCDate() - diff2;
   143→    let expiryDate = new Date(Date.UTC(year, month, expiryDayNum, 15, 30, 0));
   144→
   145→    if (expiryDate <= istNow) {
   146→      month++;
   147→      if (month > 11) { month = 0; year++; }
   148→      const lastDay2 = new Date(Date.UTC(year, month + 1, 0));
   149→      let lastTarget2 = lastDay2.getUTCDay();
   150→      let diff3 = lastTarget2 - actualDay;
   151→      if (diff3 < 0) diff3 += 7;
   152→      expiryDayNum = lastDay2.getUTCDate() - diff3;
   153→      expiryDate = new Date(Date.UTC(year, month, expiryDayNum, 15, 30, 0));
   154→    }
   155→
   156→    const diffMs = expiryDate.getTime() - istNow.getTime();
   157→    return { date: expiryDate, daysToExpiry: Math.max(1, Math.ceil(diffMs / (24 * 60 * 60 * 1000))) };
   158→  }
   159→}

   193→function getLastWeekdayOfMonth(weekday: number): { dateStr: string; daysToExpiry: number } {
   194→  const now = new Date();
   195→  const istOffset = 5.5 * 60 * 60 * 1000;
   196→  const istNow = new Date(now.getTime() + istOffset);
   197→  let month = istNow.getUTCMonth();
   198→  let year = istNow.getUTCFullYear();
   199→
   200→  const lastDay = new Date(Date.UTC(year, month + 1, 0));
   201→  let diff = lastDay.getUTCDay() - weekday;
   202→  if (diff < 0) diff += 7;
   203→  let dayNum = lastDay.getUTCDate() - diff;
   204→  let expiryDate = new Date(Date.UTC(year, month, dayNum, 15, 30, 0));
   205→
   206→  if (expiryDate <= istNow) {
   207→    month++;
   208→    if (month > 11) { month = 0; year++; }
   209→    const lastDay2 = new Date(Date.UTC(year, month + 1, 0));
   210→    let diff2 = lastDay2.getUTCDay() - weekday;
   211→    if (diff2 < 0) diff2 += 7;
   212→    dayNum = lastDay2.getUTCDate() - diff2;
   213→    expiryDate = new Date(Date.UTC(year, month, dayNum, 15, 30, 0));
   214→  }
   215→
   216→  const diffMs = expiryDate.getTime() - istNow.getTime();
   217→  const daysToExpiry = Math.max(1, Math.ceil(diffMs / (24 * 60 * 60 * 1000)));
   218→  const dateStr = expiryDate.toLocaleDateString('en-IN', { weekday: 'short', month: 'short', day: 'numeric' });
   219→  return { dateStr, daysToExpiry };
   220→}

   286→// --- Instrument Data ---
   287→export function generateDemoInstrument(
   288→  symbol: string = 'NIFTY',
   289→  name: string = 'Nifty 50',
   290→  type: InstrumentType = 'index',
   291→  basePrice: number = 24350
   292→): InstrumentData {
   293→  const cashChange = round2(rand(-200, 200));
   294→  const cashLTP = round2(basePrice + cashChange);
   295→  const cashChangePercent = round2((cashChange / basePrice) * 100);
   296→  const futureLTP = round2(cashLTP + rand(-20, 40));
   297→  const futureBasis = round2(futureLTP - cashLTP);
   298→
   299→  const step = symbol === 'NIFTY' || symbol === 'FINNIFTY' ? 50 : 100;
   300→  const numStrikes = 11;
   301→  const atmStrike = Math.round(cashLTP / step) * step;
   302→
   303→  const strikes: OptionStrike[] = [];
   304→  const halfStrikes = Math.floor(numStrikes / 2);
   305→
   306→  for (let i = -halfStrikes; i <= halfStrikes; i++) {
   307→    const strike = atmStrike + i * step;
   308→    const isATM = i === 0;
   309→    const distFromATM = Math.abs(i);
   310→
   311→    const callITM = strike < cashLTP;
   312→    const putITM = strike > cashLTP;
   313→
   314→    const callBase = Math.max(1, cashLTP - strike + rand(-20, 20));
   315→    const callLTP = round2(Math.max(0.5, callBase * Math.exp(-distFromATM * 0.15)));
   316→    const putBase = Math.max(1, strike - cashLTP + rand(-20, 20));
   317→    const putLTP = round2(Math.max(0.5, putBase * Math.exp(-distFromATM * 0.15)));
   318→
   319→    const callOI = roundN(rand(50000, 5000000) * (1 + distFromATM * 0.3), 0);
   320→    const putOI = roundN(rand(50000, 5000000) * (1 + distFromATM * 0.3), 0);
   321→
   322→    const callIV = round2(rand(10, 25) + distFromATM * 1.5);
   323→    const putIV = round2(rand(10, 25) + distFromATM * 1.5);
   324→
   325→    const callDelta = round2(callITM ? rand(0.5, 0.99) : rand(0.01, 0.49));
   326→    const putDelta = round2(-(putITM ? rand(0.5, 0.99) : rand(0.01, 0.49)));
   327→
   328→    const callGamma = round2(rand(0.0001, 0.002) * (1 - distFromATM * 0.08));
   329→    const putGamma = round2(rand(0.0001, 0.002) * (1 - distFromATM * 0.08));
   330→
   331→    const callTheta = round2(-rand(1, 20) * (1 + distFromATM * 0.05));
   332→    const putTheta = round2(-rand(1, 20) * (1 + distFromATM * 0.05));
   333→
   334→    const callVega = round2(rand(5, 30) * (1 - distFromATM * 0.06));
   335→    const putVega = round2(rand(5, 30) * (1 - distFromATM * 0.06));
   336→
   337→    let builtUp: BuiltUpType = 'None';
   338→    const callOIChg = roundN(rand(-200000, 500000), 0);
   339→    const putOIChg = roundN(rand(-200000, 500000), 0);
   340→
   341→    if (callOIChg > 100000 && callLTP < putLTP) builtUp = 'Call Writing';
   342→    else if (callOIChg > 100000 && callLTP > putLTP) builtUp = 'Call Long Cvr.';
   343→    else if (putOIChg > 100000 && putLTP < callLTP) builtUp = 'Put Writing';
   344→    else if (putOIChg > 100000 && putLTP > callLTP) builtUp = 'Put Long Cvr.';
   345→    else if (putOIChg < -100000) builtUp = 'Put Short Cvr.';
   346→
   347→    strikes.push({
   348→      strike,
   349→      callLTP,
   350→      callOI,
   351→      callOIChg,
   352→      callVolume: randInt(1000, 500000),
   353→      callIV,
   354→      callDelta,
   355→      callGamma,
   356→      callTheta,
   357→      callVega,
   358→      callChg: round2(rand(-10, 10)),
   359→      putLTP,
   360→      putOI,
   361→      putOIChg,
   362→      putVolume: randInt(1000, 500000),
   363→      putIV,
   364→      putDelta,
   365→      putGamma,
   366→      putTheta,
   367→      putVega,
   368→      putChg: round2(rand(-10, 10)),
   369→      isATM,
   370→      builtUp,
   371→      callITM,
   372→      putITM,
   373→    });
   374→  }
   375→
   376→  const totalCallOI = strikes.reduce((s, k) => s + k.callOI, 0);
   377→  const totalPutOI = strikes.reduce((s, k) => s + k.putOI, 0);
   378→  const pcr = round2(totalPutOI / totalCallOI);
   379→
   380→  return {
   381→    symbol,
   382→    name,
   383→    type,
   384→    cashLTP,
   385→    cashChange,
   386→    cashChangePercent,
   387→    futureLTP,
   388→    futureBasis,
   389→    nextMonthFutLTP: round2(futureLTP + rand(30, 80)),
   390→    nextMonthFutBasis: round2(futureLTP + rand(30, 80) - cashLTP),
   391→    atmStrike,
   392→    strikes,
   393→    totalCallOI,
   394→    totalPutOI,
   395→    pcr,
   396→    chgOiPCR: round2(rand(0.5, 1.5)),
   397→    volumePCR: round2(rand(0.4, 1.2)),
   398→    maxPainStrike: atmStrike + (Math.random() > 0.5 ? step : -step),
   399→  };
   400→}

   751→// --- Weighted Cash Flow Bars ---
   752→// Based on Pine Script: CashFlow = (Close-Open) × Volume × Weight%
   753→// 4 bars per minute (every 15 seconds)
   754→export function generateDemoWeightedCashFlowBars(numBars: number = 60): WeightedCashFlowBar[] {
   755→  const bars: WeightedCashFlowBar[] = [];
   756→  const now = new Date();
   757→
   758→  for (let i = numBars - 1; i >= 0; i--) {
   759→    const ts = new Date(now.getTime() - i * 15000); // 15-second intervals
   760→    const istH = ((ts.getUTCHours() + 5) % 24 + 24) % 24;
   761→    const istM = ts.getUTCMinutes();
   762→    const istS = ts.getUTCSeconds();
   763→    const timeLabel = `${String(istH).padStart(2, '0')}:${String(istM).padStart(2, '0')}:${String(istS).padStart(2, '0')}`;
   764→
   765→    const stockFlows: WeightedCashFlowBar['stockFlows'] = [];
   766→    let niftyWeightedCF = 0;
   767→    let sensexWeightedCF = 0;
   768→    let totalMoneyIn = 0;
   769→    let totalMoneyOut = 0;
   770→
   771→    for (const stock of STOCK_BASE_PRICES) {
   772→      const base = stock.base;
   773→      // Simulate intra-bar price movement
   774→      const openPrice = base + rand(-base * 0.002, base * 0.002);
   775→      const closePrice = openPrice + rand(-base * 0.005, base * 0.005);
   776→      const volume = randInt(50000, 2000000);
   777→
   778→      // Cash Flow = (Close - Open) × Volume  (Pine Script logic)
   779→      const cashFlow = (closePrice - openPrice) * volume;
   780→
   781→      // Weighted Cash Flow = CashFlow × Weight%
   782→      const niftyWeighted = cashFlow * (stock.niftyWeight / 100);
   783→      const sensexWeighted = cashFlow * (stock.sensexWeight / 100);
   784→
   785→      // Money In / Money Out
   786→      const moneyIn = cashFlow >= 0 ? cashFlow : 0;
   787→      const moneyOut = cashFlow < 0 ? Math.abs(cashFlow) : 0;
   788→
   789→      stockFlows.push({
   790→        symbol: stock.symbol,
   791→        cashFlow: roundN(cashFlow, 0),
   792→        niftyWeighted: roundN(niftyWeighted, 0),
   793→        sensexWeighted: roundN(sensexWeighted, 0),
   794→        weight: stock.niftyWeight,
   795→        moneyIn: roundN(moneyIn, 0),
   796→        moneyOut: roundN(moneyOut, 0),
   797→      });
   798→
   799→      niftyWeightedCF += niftyWeighted;
   800→      sensexWeightedCF += sensexWeighted;
   801→      totalMoneyIn += moneyIn;
   802→      totalMoneyOut += moneyOut;
   803→    }
   804→
   805→    bars.push({
   806→      timestamp: timeLabel,
   807→      stockFlows,
   808→      niftyWeightedCF: roundN(niftyWeightedCF, 0),
   809→      sensexWeightedCF: roundN(sensexWeightedCF, 0),
   810→      totalMoneyIn: roundN(totalMoneyIn, 0),
   811→      totalMoneyOut: roundN(totalMoneyOut, 0),
   812→      netFlow: roundN(totalMoneyIn - totalMoneyOut, 0),
   813→      isStrongInflow: niftyWeightedCF > 0 && (totalMoneyIn / (totalMoneyOut || 1)) > 2,
   814→      isStrongOutflow: niftyWeightedCF < 0 && (totalMoneyOut / (totalMoneyIn || 1)) > 2,
   815→    });
   816→  }
   817→
   818→  return bars;
   819→}

   843→// --- Market Status ---
   844→export function getMarketStatus(): 'pre-open' | 'open' | 'closing' | 'closed' {
   845→  const now = new Date();
   846→  const istH = ((now.getUTCHours() + 5) % 24 + 24) % 24;
   847→  const istM = now.getUTCMinutes() + 30;
   848→  const totalMin = istH * 60 + (istM % 60);
   849→  const day = now.getUTCDay();
   850→
   851→  if (day === 0 || day === 6) return 'closed';
   852→  if (totalMin >= 570 && totalMin < 585) return 'pre-open';
   853→  if (totalMin >= 585 && totalMin < 930) return 'open';
   854→  if (totalMin >= 930 && totalMin < 935) return 'closing';
   855→  return 'closed';
   856→}

   406→  function calculateRealIV(marketPrice, strike, T, r, optionPrice, type) {
   407→    if (optionPrice <= 0 || marketPrice <= 0 || T <= 0.001) return 0.18;
   408→    
   409→    var sigma = 0.20;
   410→    var tolerance = 0.0001;
   411→    var maxIterations = 30;
   412→    
   413→    for (var iter = 0; iter < maxIterations; iter++) {
   414→      var d1 = (Math.log(marketPrice / strike) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   415→      var d2 = d1 - sigma * Math.sqrt(T);
   416→      
   417→      var priceDiff;
   418→      if (type === "CE") {
   419→        priceDiff = marketPrice * normCDF(d1) - strike * Math.exp(-r * T) * normCDF(d2) - optionPrice;
   420→      } else {
   421→        priceDiff = strike * Math.exp(-r * T) * normCDF(-d2) - marketPrice * normCDF(-d1) - optionPrice;
   422→      }
   423→      
   424→      if (Math.abs(priceDiff) < tolerance) break;
   425→      
   426→      var vega = marketPrice * Math.sqrt(T) * normPDF(d1);
   427→      if (vega < 0.0001) vega = 0.0001;
   428→      
   429→      var newSigma = sigma - priceDiff / vega;
   430→      if (newSigma <= 0.05) newSigma = 0.05;
   431→      if (newSigma > 1.5) newSigma = 1.5;
   432→      
   433→      sigma = newSigma;
   434→    }
   435→    
   436→    return sigma;
   437→  }

   439→  function calculateCompleteGreeks(S, K, T, r, sigma, type) {
   440→    if (T <= 0.001) T = 0.001;
   441→    if (sigma <= 0.01) sigma = 0.15;
   442→    
   443→    var d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   444→    var d2 = d1 - sigma * Math.sqrt(T);
   445→    
   446→    var delta, gamma, theta, vega, rho;
   447→    var pdfD1 = normPDF(d1);
   448→    
   449→    if (type === "CE") {
   450→      delta = normCDF(d1);
   451→      theta = (-S * pdfD1 * sigma / (2 * Math.sqrt(T))) - (r * K * Math.exp(-r * T) * normCDF(d2));
   452→      rho = K * T * Math.exp(-r * T) * normCDF(d2);
   453→    } else {
   454→      delta = normCDF(d1) - 1;
   455→      theta = (-S * pdfD1 * sigma / (2 * Math.sqrt(T))) + (r * K * Math.exp(-r * T) * normCDF(-d2));
   456→      rho = -K * T * Math.exp(-r * T) * normCDF(-d2);
   457→    }
   458→    
   459→    gamma = pdfD1 / (S * sigma * Math.sqrt(T));
   460→    vega = S * pdfD1 * Math.sqrt(T);
   461→    
   462→    var tradingDays = 252;
   463→    theta = theta / tradingDays;
   464→    vega = vega / 100;
   465→    
   466→    return {
   467→      delta: delta,
   468→      gamma: gamma,
   469→      theta: theta,
   470→      vega: vega,
   471→      rho: rho,
   472→      iv: sigma
   473→    };
   474→  }

   962→function normCDF(x) {
   963→  return (1.0 + erf(x / Math.sqrt(2.0))) / 2.0;

   966→function erf(x) {
   967→  var sign = (x >= 0) ? 1 : -1;
   968→  x = Math.abs(x);
   969→
   970→  var a1 = 0.254829592;
   971→  var a2 = -0.284496736;
   972→  var a3 = 1.421413741;
   973→  var a4 = -1.453152027;
   974→  var a5 = 1.061405429;
   975→  var p  = 0.3275911;
   976→
   977→  var t = 1.0 / (1.0 + p * x);

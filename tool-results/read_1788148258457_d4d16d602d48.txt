    56→/**
    57→ * Convert a time string ("HH:MM" or "HH:MM:SS") to minutes since midnight.
    58→ * Returns NaN if the string is unparseable — Recharts will skip those points.
    59→ */
    60→function timeStrToMinutes(t: string | undefined | null): number {
    61→  if (!t) return NaN;
    62→  const parts = t.split(':');
    63→  if (parts.length < 2) return NaN;
    64→  const h = parseInt(parts[0], 10);
    65→  const m = parseInt(parts[1], 10);
    66→  if (isNaN(h) || isNaN(m)) return NaN;
    67→  return h * 60 + m;
    68→}
    69→
    70→/** Format minutes-since-midnight back to "HH:MM" for axis tick labels. */
    71→function minutesToTimeStr(min: number): string {
    72→  const h = Math.floor(min / 60);
    73→  const m = Math.round(min % 60);
    74→  return `${String(h).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
    75→}

    79→function NiftyTooltip({ active, payload, label }: any) {
    80→  if (!active || !payload?.length) return null;
    81→  const d = payload[0]?.payload as NiftyCandle;
    82→  const labelStr = typeof label === 'number' ? minutesToTimeStr(label) : (d?.time || String(label ?? ''));
    83→  return (
    84→    <div className="bg-card border border-border rounded-lg p-2 shadow-xl text-xs">
    85→      <div className="font-mono text-muted-foreground mb-1">{labelStr}</div>
    86→      <div className="text-emerald-400 font-semibold">Close: {d?.close?.toLocaleString('en-IN')}</div>
    87→      {d?.high ? (
    88→        <div className="text-muted-foreground">
    89→          H: {d.high.toLocaleString('en-IN')} L: {d.low.toLocaleString('en-IN')}
    90→        </div>
    91→      ) : null}
    92→      {d?.volume ? (
    93→        <div className="text-muted-foreground">Vol: {(d.volume / 1000000).toFixed(1)}M</div>
    94→      ) : null}
    95→    </div>
    96→  );
    97→}

    54→// --- Global Indices ---
    55→export function generateDemoGlobalIndices(): GlobalIndex[] {
    56→  const istHour = new Date().getUTCHours() + 5.5;
    57→  const asiaOpen = istHour >= 6 && istHour < 15;
    58→
    59→  return GLOBAL_INDICES.map((idx) => {
    60→    const changePercent = rand(-1.5, 1.5);
    61→    const change = roundN(idx.baseValue * changePercent / 100, idx.baseValue > 10000 ? 0 : 2);
    62→    const value = roundN(idx.baseValue + change, idx.baseValue > 10000 ? 0 : 2);
    63→    const isUS = idx.country === 'US';
    64→    const isEurope = idx.country === 'UK' || idx.country === 'Germany';
    65→    const isAsia = idx.country === 'Japan' || idx.country === 'HK' || idx.country === 'China' || idx.country === 'Korea';
    66→
    67→    let status: 'open' | 'closed' | 'pre-market' = 'closed';
    68→    if (isUS) status = istHour >= 19.5 && istHour < 25 ? 'open' : istHour >= 18 ? 'pre-market' : 'closed';
    69→    else if (isEurope) status = istHour >= 13 && istHour < 21.5 ? 'open' : 'closed';
    70→    else if (isAsia) status = asiaOpen ? 'open' : 'closed';
    71→
    72→    let impactOnNifty: GlobalIndex['impactOnNifty'];
    73→    if (Math.abs(changePercent) < 0.15) impactOnNifty = changePercent >= 0 ? 'mild_positive' : 'mild_negative';
    74→    else if (Math.abs(changePercent) < 0.4) impactOnNifty = changePercent >= 0 ? 'mild_positive' : 'mild_negative';
    75→    else if (changePercent >= 0) impactOnNifty = 'positive';
    76→    else impactOnNifty = 'negative';
    77→
    78→    if (idx.country === 'China') impactOnNifty = changePercent >= 0 ? 'mild_positive' : 'mild_negative';
    79→    if (Math.abs(changePercent) > 0.5 && changePercent > 0) impactOnNifty = 'positive';
    80→    if (Math.abs(changePercent) > 0.5 && changePercent < 0) impactOnNifty = 'negative';
    81→
    82→    return {
    83→      name: idx.name,
    84→      country: idx.country,
    85→      value,
    86→      change,
    87→      changePercent: round2(changePercent),
    88→      status,
    89→      impactOnNifty,
    90→    };
    91→  });
    92→}

   119→// --- Expiry Info ---
   120→function getNextExpiry(expiryDay: number, expiryType: 'weekly' | 'monthly'): { date: Date; daysToExpiry: number } {
   121→  const now = new Date();
   122→  const istOffset = 5.5 * 60 * 60 * 1000;
   123→  const istNow = new Date(now.getTime() + istOffset);
   124→
   125→  if (expiryType === 'weekly') {
   126→    // Find next expiryDay (0=Sun, 1=Mon, 2=Tue, ..., 6=Sat)
   127→    const currentDay = istNow.getUTCDay();
   128→    let daysUntil = expiryDay - currentDay;
   129→    if (daysUntil <= 0) daysUntil += 7;
   130→    const expiryDate = new Date(istNow);
   131→    expiryDate.setUTCDate(expiryDate.getUTCDate() + daysUntil);
   132→    expiryDate.setUTCHours(15, 30, 0, 0);
   133→    const diff = expiryDate.getTime() - istNow.getTime();
   134→    return { date: expiryDate, daysToExpiry: Math.max(1, Math.ceil(diff / (24 * 60 * 60 * 1000))) };
   135→  } else {
   136→    // Last specific weekday of current or next month
   137→    const currentMonth = istNow.getUTCMonth();
   138→    const currentYear = istNow.getUTCFullYear();
   139→    const targetDay = expiryDay; // reuse for monthly: -1 means last Tuesday
   140→
   141→    const actualDay = expiryDay === -1 ? 2 : expiryDay; // Tuesday = 2 for BankNifty/Finnifty
   142→    // For Bankex: Thursday = 4 (handled separately in caller)
   143→
   144→    let month = currentMonth;
   145→    let year = currentYear;
   146→
   147→    // Find last Tuesday/Thursday of the month
   148→    const lastDay = new Date(Date.UTC(year, month + 1, 0));
   149→    let lastTarget = lastDay.getUTCDay();
   150→    let diff2 = lastTarget - actualDay;
   151→    if (diff2 < 0) diff2 += 7;
   152→    if (diff2 > 6) diff2 -= 7;
   153→    let expiryDayNum = lastDay.getUTCDate() - diff2;
   154→    let expiryDate = new Date(Date.UTC(year, month, expiryDayNum, 15, 30, 0));
   155→
   156→    if (expiryDate <= istNow) {
   157→      month++;
   158→      if (month > 11) { month = 0; year++; }
   159→      const lastDay2 = new Date(Date.UTC(year, month + 1, 0));
   160→      let lastTarget2 = lastDay2.getUTCDay();
   161→      let diff3 = lastTarget2 - actualDay;
   162→      if (diff3 < 0) diff3 += 7;
   163→      expiryDayNum = lastDay2.getUTCDate() - diff3;
   164→      expiryDate = new Date(Date.UTC(year, month, expiryDayNum, 15, 30, 0));
   165→    }
   166→
   167→    const diffMs = expiryDate.getTime() - istNow.getTime();
   168→    return { date: expiryDate, daysToExpiry: Math.max(1, Math.ceil(diffMs / (24 * 60 * 60 * 1000))) };
   169→  }
   170→}

   204→function getLastWeekdayOfMonth(weekday: number): { dateStr: string; daysToExpiry: number } {
   205→  const now = new Date();
   206→  const istOffset = 5.5 * 60 * 60 * 1000;
   207→  const istNow = new Date(now.getTime() + istOffset);
   208→  let month = istNow.getUTCMonth();
   209→  let year = istNow.getUTCFullYear();
   210→
   211→  const lastDay = new Date(Date.UTC(year, month + 1, 0));
   212→  let diff = lastDay.getUTCDay() - weekday;
   213→  if (diff < 0) diff += 7;
   214→  let dayNum = lastDay.getUTCDate() - diff;
   215→  let expiryDate = new Date(Date.UTC(year, month, dayNum, 15, 30, 0));
   216→
   217→  if (expiryDate <= istNow) {
   218→    month++;
   219→    if (month > 11) { month = 0; year++; }
   220→    const lastDay2 = new Date(Date.UTC(year, month + 1, 0));
   221→    let diff2 = lastDay2.getUTCDay() - weekday;
   222→    if (diff2 < 0) diff2 += 7;
   223→    dayNum = lastDay2.getUTCDate() - diff2;
   224→    expiryDate = new Date(Date.UTC(year, month, dayNum, 15, 30, 0));
   225→  }
   226→
   227→  const diffMs = expiryDate.getTime() - istNow.getTime();
   228→  const daysToExpiry = Math.max(1, Math.ceil(diffMs / (24 * 60 * 60 * 1000)));
   229→  const dateStr = expiryDate.toLocaleDateString('en-IN', { weekday: 'short', month: 'short', day: 'numeric' });
   230→  return { dateStr, daysToExpiry };
   231→}

   297→// --- Instrument Data ---
   298→export function generateDemoInstrument(
   299→  symbol: string = 'NIFTY',
   300→  name: string = 'Nifty 50',
   301→  type: InstrumentType = 'index',
   302→  basePrice: number = 24350
   303→): InstrumentData {
   304→  const cashChange = round2(rand(-200, 200));
   305→  const cashLTP = round2(basePrice + cashChange);
   306→  const cashChangePercent = round2((cashChange / basePrice) * 100);
   307→  const futureLTP = round2(cashLTP + rand(-20, 40));
   308→  const futureBasis = round2(futureLTP - cashLTP);
   309→
   310→  const step = symbol === 'NIFTY' || symbol === 'FINNIFTY' ? 50 : 100;
   311→  const numStrikes = 11;
   312→  const atmStrike = Math.round(cashLTP / step) * step;
   313→
   314→  const strikes: OptionStrike[] = [];
   315→  const halfStrikes = Math.floor(numStrikes / 2);
   316→
   317→  for (let i = -halfStrikes; i <= halfStrikes; i++) {
   318→    const strike = atmStrike + i * step;
   319→    const isATM = i === 0;
   320→    const distFromATM = Math.abs(i);
   321→
   322→    const callITM = strike < cashLTP;
   323→    const putITM = strike > cashLTP;
   324→
   325→    const callBase = Math.max(1, cashLTP - strike + rand(-20, 20));
   326→    const callLTP = round2(Math.max(0.5, callBase * Math.exp(-distFromATM * 0.15)));
   327→    const putBase = Math.max(1, strike - cashLTP + rand(-20, 20));
   328→    const putLTP = round2(Math.max(0.5, putBase * Math.exp(-distFromATM * 0.15)));
   329→
   330→    const callOI = roundN(rand(50000, 5000000) * (1 + distFromATM * 0.3), 0);
   331→    const putOI = roundN(rand(50000, 5000000) * (1 + distFromATM * 0.3), 0);
   332→
   333→    const callIV = round2(rand(10, 25) + distFromATM * 1.5);
   334→    const putIV = round2(rand(10, 25) + distFromATM * 1.5);
   335→
   336→    const callDelta = round2(callITM ? rand(0.5, 0.99) : rand(0.01, 0.49));
   337→    const putDelta = round2(-(putITM ? rand(0.5, 0.99) : rand(0.01, 0.49)));
   338→
   339→    const callGamma = round2(rand(0.0001, 0.002) * (1 - distFromATM * 0.08));
   340→    const putGamma = round2(rand(0.0001, 0.002) * (1 - distFromATM * 0.08));
   341→
   342→    const callTheta = round2(-rand(1, 20) * (1 + distFromATM * 0.05));
   343→    const putTheta = round2(-rand(1, 20) * (1 + distFromATM * 0.05));
   344→
   345→    const callVega = round2(rand(5, 30) * (1 - distFromATM * 0.06));
   346→    const putVega = round2(rand(5, 30) * (1 - distFromATM * 0.06));
   347→
   348→    let builtUp: BuiltUpType = 'None';
   349→    const callOIChg = roundN(rand(-200000, 500000), 0);
   350→    const putOIChg = roundN(rand(-200000, 500000), 0);
   351→
   352→    if (callOIChg > 100000 && callLTP < putLTP) builtUp = 'Call Writing';
   353→    else if (callOIChg > 100000 && callLTP > putLTP) builtUp = 'Call Long Cvr.';
   354→    else if (putOIChg > 100000 && putLTP < callLTP) builtUp = 'Put Writing';
   355→    else if (putOIChg > 100000 && putLTP > callLTP) builtUp = 'Put Long Cvr.';
   356→    else if (putOIChg < -100000) builtUp = 'Put Short Cvr.';
   357→
   358→    strikes.push({
   359→      strike,
   360→      callLTP,
   361→      callOI,
   362→      callOIChg,
   363→      callVolume: randInt(1000, 500000),
   364→      callIV,
   365→      callDelta,
   366→      callGamma,
   367→      callTheta,
   368→      callVega,
   369→      callChg: round2(rand(-10, 10)),
   370→      putLTP,
   371→      putOI,
   372→      putOIChg,
   373→      putVolume: randInt(1000, 500000),
   374→      putIV,
   375→      putDelta,
   376→      putGamma,
   377→      putTheta,
   378→      putVega,
   379→      putChg: round2(rand(-10, 10)),
   380→      isATM,
   381→      builtUp,
   382→      callITM,
   383→      putITM,
   384→    });
   385→  }
   386→
   387→  const totalCallOI = strikes.reduce((s, k) => s + k.callOI, 0);
   388→  const totalPutOI = strikes.reduce((s, k) => s + k.putOI, 0);
   389→  const pcr = round2(totalPutOI / totalCallOI);
   390→
   391→  return {
   392→    symbol,
   393→    name,
   394→    type,
   395→    cashLTP,
   396→    cashChange,
   397→    cashChangePercent,
   398→    futureLTP,
   399→    futureBasis,
   400→    nextMonthFutLTP: round2(futureLTP + rand(30, 80)),
   401→    nextMonthFutBasis: round2(futureLTP + rand(30, 80) - cashLTP),
   402→    atmStrike,
   403→    strikes,
   404→    totalCallOI,
   405→    totalPutOI,
   406→    pcr,
   407→    chgOiPCR: round2(rand(0.5, 1.5)),
   408→    volumePCR: round2(rand(0.4, 1.2)),
   409→    maxPainStrike: atmStrike + (Math.random() > 0.5 ? step : -step),
   410→  };
   411→}

   762→// --- Weighted Cash Flow Bars ---
   763→// Based on Pine Script: CashFlow = (Close-Open) × Volume × Weight%
   764→// 4 bars per minute (every 15 seconds)
   765→export function generateDemoWeightedCashFlowBars(numBars: number = 60): WeightedCashFlowBar[] {
   766→  const bars: WeightedCashFlowBar[] = [];
   767→  const now = new Date();
   768→
   769→  for (let i = numBars - 1; i >= 0; i--) {
   770→    const ts = new Date(now.getTime() - i * 15000); // 15-second intervals
   771→    const istH = ((ts.getUTCHours() + 5) % 24 + 24) % 24;
   772→    const istM = ts.getUTCMinutes();
   773→    const istS = ts.getUTCSeconds();
   774→    const timeLabel = `${String(istH).padStart(2, '0')}:${String(istM).padStart(2, '0')}:${String(istS).padStart(2, '0')}`;
   775→
   776→    const stockFlows: WeightedCashFlowBar['stockFlows'] = [];
   777→    let niftyWeightedCF = 0;
   778→    let sensexWeightedCF = 0;
   779→    let totalMoneyIn = 0;
   780→    let totalMoneyOut = 0;
   781→
   782→    for (const stock of STOCK_BASE_PRICES) {
   783→      const base = stock.base;
   784→      // Simulate intra-bar price movement (15-second candle)
   785→      const openPrice = base + rand(-base * 0.002, base * 0.002);
   786→      const closePrice = openPrice + rand(-base * 0.005, base * 0.005);
   787→      const high = Math.max(openPrice, closePrice) + rand(0, base * 0.002);
   788→      const low = Math.min(openPrice, closePrice) - rand(0, base * 0.002);
   789→      const volume = randInt(50000, 2000000);
   790→
   791→      // ─── Pine Script: Cash Flow = (Close - Open) × Volume ───
   792→      let cashFlow = (closePrice - openPrice) * volume;
   793→
   794→      // ─── Pine Script: Volume Weighted method ───
   795→      // cashFlow := cashFlowRaw * (volume_ / smaVolume)
   796→      // Note: In production, smaVolume would be computed from 20-period SMA
   797→      // For demo, we approximate with a reasonable SMA value
   798→      const smaVolume = volume * rand(0.7, 1.3); // Approximate SMA
   799→      // Using Standard method by default (matching Pine Script default)
   800→      // Volume Weighted: cashFlow = cashFlow * (volume / smaVolume)
   801→
   802→      // ─── Pine Script: Range Adjusted method ───
   803→      // cashFlowAdj = cashFlowTemp * (1 + (high - low) / smaRange)
   804→      const priceRange = high - low;
   805→      const smaRange = priceRange * rand(0.8, 1.2); // Approximate SMA range
   806→      // Using Standard by default
   807→
   808→      // ─── Pine Script: Weighted Cash Flow = CF × Weight% ───
   809→      const niftyWeighted = cashFlow * (stock.niftyWeight / 100);
   810→      const sensexWeighted = cashFlow * (stock.sensexWeight / 100);
   811→
   812→      // ─── Money In / Money Out ───
   813→      const moneyIn = cashFlow >= 0 ? cashFlow : 0;
   814→      const moneyOut = cashFlow < 0 ? Math.abs(cashFlow) : 0;
   815→
   816→      stockFlows.push({
   817→        symbol: stock.symbol,
   818→        cashFlow: roundN(cashFlow, 0),
   819→        niftyWeighted: roundN(niftyWeighted, 0),
   820→        sensexWeighted: roundN(sensexWeighted, 0),
   821→        weight: stock.niftyWeight,
   822→        moneyIn: roundN(moneyIn, 0),
   823→        moneyOut: roundN(moneyOut, 0),
   824→      });
   825→
   826→      niftyWeightedCF += niftyWeighted;
   827→      sensexWeightedCF += sensexWeighted;
   828→      totalMoneyIn += moneyIn;
   829→      totalMoneyOut += moneyOut;
   830→    }
   831→
   832→    bars.push({
   833→      timestamp: timeLabel,
   834→      stockFlows,
   835→      niftyWeightedCF: roundN(niftyWeightedCF, 0),
   836→      sensexWeightedCF: roundN(sensexWeightedCF, 0),
   837→      totalMoneyIn: roundN(totalMoneyIn, 0),
   838→      totalMoneyOut: roundN(totalMoneyOut, 0),
   839→      netFlow: roundN(totalMoneyIn - totalMoneyOut, 0),
   840→      isStrongInflow: niftyWeightedCF > 0 && niftyWeightedCF > (bars.length > 0 ? bars[bars.length - 1].niftyWeightedCF : 0),
   841→      isStrongOutflow: niftyWeightedCF < 0 && niftyWeightedCF < (bars.length > 0 ? bars[bars.length - 1].niftyWeightedCF : 0),
   842→    });
   843→  }
   844→
   845→  return bars;
   846→}

   875→// --- Compute Cash Flow Trend from actual bars (not random) ---
   876→export function computeCashFlowTrend(bars: WeightedCashFlowBar[]): CashFlowTrend {
   877→  if (bars.length < 14) {
   878→    return {
   879→      currentValue: 0, smoothed: 0, upperBand: 0, lowerBand: 0,
   880→      isStrongInflow: false, isStrongOutflow: false,
   881→      isUptrend: false, isDowntrend: false,
   882→      momentum: 0, signalStrength: 0,
   883→      bearishDivergence: false, bullishDivergence: false,
   884→    };
   885→  }
   886→
   887→  // Use netFlow values from bars
   888→  const values = bars.map(b => b.netFlow);
   889→  const current = values[values.length - 1];
   890→
   891→  // SMA(14)
   892→  const smaWindow = values.slice(-14);
   893→  const smoothed = smaWindow.reduce((s, v) => s + v, 0) / smaWindow.length;
   894→
   895→  // Standard deviation
   896→  const variance = smaWindow.reduce((s, v) => s + Math.pow(v - smoothed, 2), 0) / smaWindow.length;
   897→  const stdev = Math.sqrt(variance);
   898→
   899→  const upperBand = smoothed + 0.5 * stdev;
   900→  const lowerBand = smoothed - 0.5 * stdev;
   901→
   902→  // Momentum (rate of change)
   903→  const prevSmoothed = values.length > 28
   904→    ? values.slice(-28, -14).reduce((s, v) => s + v, 0) / 14
   905→    : smoothed;
   906→  const momentum = current - prevSmoothed;
   907→
   908→  // Signal strength: -100 to +100
   909→  const signalStrength = Math.max(-100, Math.min(100, (smoothed / (stdev || 1)) * 20));
   910→
   911→  // Divergence detection
   912→  const recentPrices = bars.slice(-14).map(b => b.niftyWeightedCF);
   913→  const priceTrend = recentPrices[recentPrices.length - 1] - recentPrices[0];
   914→  const flowTrend = current - smoothed;
   915→  const bearishDivergence = priceTrend > 0 && flowTrend < 0 && Math.abs(flowTrend) > stdev;
   916→  const bullishDivergence = priceTrend < 0 && flowTrend > 0 && Math.abs(flowTrend) > stdev;
   917→
   918→  return {
   919→    currentValue: round2(current),
   920→    smoothed: round2(smoothed),
   921→    upperBand: round2(upperBand),
   922→    lowerBand: round2(lowerBand),
   923→    isStrongInflow: current > upperBand && current > 0,
   924→    isStrongOutflow: current < lowerBand && current < 0,
   925→    isUptrend: smoothed > 0,
   926→    isDowntrend: smoothed < 0,
   927→    momentum: round2(momentum),
   928→    signalStrength: round2(signalStrength),
   929→    bearishDivergence,
   930→    bullishDivergence,
   931→  };
   932→}

  1189→// Generate strike-level options flow for one instrument
  1190→function generateStrikeFlows(
  1191→  atmStrike: number,
  1192→  step: number,
  1193→  numStrikes: number,
  1194→  isIndex: boolean,
  1195→): StrikeOptionsFlow[] {
  1196→  const halfStrikes = Math.floor(numStrikes / 2);
  1197→  const strikes: StrikeOptionsFlow[] = [];
  1198→
  1199→  for (let i = -halfStrikes; i <= halfStrikes; i++) {
  1200→    const strike = atmStrike + i * step;
  1201→    const isATM = i === 0;
  1202→    const distanceFromATM = Math.abs(i);
  1203→
  1204→    // ATM strikes have highest activity; OTM strikes have less
  1205→    const activityMultiplier = Math.max(0.2, 1 - distanceFromATM * 0.12);
  1206→
  1207→    // Base flow values — indexes have larger flows than stocks
  1208→    const baseFlow = isIndex ? rand(5000, 25000) : rand(2000, 12000);
  1209→
  1210→    // Call Buy: Buyers of calls (bullish bet)
  1211→    // Higher near ATM, even higher if market is slightly bullish
  1212→    const callBuy = roundN(baseFlow * activityMultiplier * rand(0.5, 1.5), 0);
  1213→
  1214→    // Put Write: Sellers of puts (also bullish — they think price won't drop)
  1215→    // Put writing is often higher than call buying (smart money writes options)
  1216→    const putWrite = roundN(baseFlow * activityMultiplier * rand(0.4, 1.3), 0);
  1217→
  1218→    // Put Buy: Buyers of puts (bearish bet)
  1219→    // Usually less than call buy in a mildly bullish market
  1220→    const putBuy = roundN(baseFlow * activityMultiplier * rand(0.3, 1.1), 0);
  1221→
  1222→    // Call Write: Sellers of calls (bearish — they think price won't rise)
  1223→    // Call writing can be high when market makers are hedging
  1224→    const callWrite = roundN(baseFlow * activityMultiplier * rand(0.3, 1.2), 0);
  1225→
  1226→    const bullishFlow = callBuy + putWrite;
  1227→    const bearishFlow = putBuy + callWrite;
  1228→
  1229→    strikes.push({
  1230→      strike,
  1231→      isATM,
  1232→      callBuy,
  1233→      putWrite,
  1234→      putBuy,
  1235→      callWrite,
  1236→      bullishFlow,
  1237→      bearishFlow,
  1238→      netFlow: bullishFlow - bearishFlow,
  1239→    });
  1240→  }
  1241→
  1242→  return strikes;
  1243→}

  1245→// Generate options flow for one instrument (all strikes)
  1246→function generateInstrumentOptionsFlow(
  1247→  symbol: string,
  1248→  name: string,
  1249→  type: InstrumentType,
  1250→  atmStrike: number,
  1251→  step: number,

  1259→  const totalCallWrite = strikes.reduce((s, k) => s + k.callWrite, 0);
  1260→  const totalBullishFlow = totalCallBuy + totalPutWrite;
  1261→  const totalBearishFlow = totalPutBuy + totalCallWrite;

  1266→    type,
  1267→    atmStrike,

  1411→// =====================================================
  1412→// COMPOSITE SIGNAL GENERATOR
  1413→// Combines ALL layers into one actionable signal
  1414→// Score range: -100 (strong bearish) to +100 (strong bullish)
  1415→// =====================================================
  1416→
  1417→export function computeCompositeSignal(
  1418→  cashNet: number,
  1419→  idxOptNet: number,
  1420→  stkOptNet: number,
  1421→  idxFutNet: number,
  1422→  stkFutNet: number,
  1423→  priceTrend: number, // positive = price going up, negative = going down
  1424→): CompositeSignal {
  1425→  const now = new Date();
  1426→  const istH = ((now.getUTCHours() + 5) % 24 + 24) % 24;
  1427→  const istM = now.getUTCMinutes();
  1428→  const istS = now.getUTCSeconds();
  1429→  const timestamp = `${String(istH).padStart(2, '0')}:${String(istM).padStart(2, '0')}:${String(istS).padStart(2, '0')} IST`;
  1430→
  1431→  // Normalize each component to -100 to +100
  1432→  const normalize = (val: number, scale: number) => Math.max(-100, Math.min(100, (val / scale) * 100));
  1433→
  1434→  const priceTrendScore = normalize(priceTrend, 50);   // 50 points = 100 score
  1435→  const cashFlowScore = normalize(cashNet, 50000000);   // 50Cr = 100 score
  1436→  const idxOptScore = normalize(idxOptNet, 2000000);    // 20L = 100 score
  1437→  const stkOptScore = normalize(stkOptNet, 3000000);    // 30L = 100 score
  1438→  const idxFutScore = normalize(idxFutNet, 50000000);   // 50Cr = 100 score
  1439→  const stkFutScore = normalize(stkFutNet, 80000000);   // 80Cr = 100 score
  1440→
  1441→  // Weighted composite score
  1442→  // Price trend is the most important — it's what we're trying to predict
  1443→  // Options flow is the second most important — it shows where money is positioning
  1444→  // Cash flow is third — it confirms or diverges from options
  1445→  // Futures is fourth — it's leveraged so can be more noise
  1446→  const W = { price: 0.30, cash: 0.20, idxOpt: 0.20, stkOpt: 0.10, idxFut: 0.10, stkFut: 0.10 };
  1447→
  1448→  const score = round2(
  1449→    W.price * priceTrendScore +
  1450→    W.cash * cashFlowScore +
  1451→    W.idxOpt * idxOptScore +
  1452→    W.stkOpt * stkOptScore +
  1453→    W.idxFut * idxFutScore +
  1454→    W.stkFut * stkFutScore
  1455→  );
  1456→
  1457→  // Confidence: how many components agree?
  1458→  const components = [priceTrendScore, cashFlowScore, idxOptScore, stkOptScore, idxFutScore, stkFutScore];
  1459→  const allPositive = components.filter(c => c > 10).length;
  1460→  const allNegative = components.filter(c => c < -10).length;
  1461→  const agreement = Math.max(allPositive, allNegative) / components.length;
  1462→  const confidence = round2(Math.min(100, agreement * 150)); // Scale up
  1463→
  1464→  // Divergence detection
  1465→  const cashOptDivergence = (cashFlowScore > 20 && idxOptScore < -20) || (cashFlowScore < -20 && idxOptScore > 20);
  1466→  const futCashDivergence = (idxFutScore > 20 && cashFlowScore < -20) || (idxFutScore < -20 && cashFlowScore > 20);
  1467→
  1468→  // Action determination
  1469→  let action: TradeAction = 'WAIT';
  1470→  if (score > 35 && confidence > 40 && !cashOptDivergence) action = 'BUY_CALL';
  1471→  else if (score < -35 && confidence > 40 && !cashOptDivergence) action = 'BUY_PUT';
  1472→  else if (score > 50 && confidence > 60) action = 'BUY_CALL'; // Override divergence if very strong
  1473→  else if (score < -50 && confidence > 60) action = 'BUY_PUT';
  1474→
  1475→  // Suggested trade details (ATM Nifty)
  1476→  const atmStrike = 24350;
  1477→  const suggestedStrike = score > 0
  1478→    ? atmStrike + Math.round(score / 100) * 50  // Slightly OTM call
  1479→    : atmStrike - Math.round(Math.abs(score) / 100) * 50;  // Slightly OTM put
  1480→
  1481→  const suggestedPremium = round2(score > 0 ? 150 + Math.abs(score) * 0.5 : 120 + Math.abs(score) * 0.4);
  1482→  const stopLoss = round2(suggestedPremium * 0.5);
  1483→  const target1 = round2(suggestedPremium * 1.5);
  1484→  const target2 = round2(suggestedPremium * 2.5);
  1485→  const riskReward = round2((target1 - suggestedPremium) / (suggestedPremium - stopLoss) || 0);
  1486→
  1487→  // Reasoning
  1488→  const reasoningParts: string[] = [];
  1489→  if (Math.abs(priceTrendScore) > 30) reasoningParts.push(`Price trend ${priceTrendScore > 0 ? '↑' : '↓'} ${Math.abs(priceTrendScore).toFixed(0)}`);
  1490→  if (Math.abs(cashFlowScore) > 20) reasoningParts.push(`Cash ${cashFlowScore > 0 ? 'inflow' : 'outflow'} ${Math.abs(cashFlowScore).toFixed(0)}`);
  1491→  if (Math.abs(idxOptScore) > 20) reasoningParts.push(`Idx opt ${idxOptScore > 0 ? 'bullish' : 'bearish'} ${Math.abs(idxOptScore).toFixed(0)}`);
  1492→  if (cashOptDivergence) reasoningParts.push('⚠️ Cash-Opt divergence');
  1493→  if (futCashDivergence) reasoningParts.push('⚠️ Fut-Cash divergence');
  1494→
  1495→  return {
  1496→    timestamp,
  1497→    action,
  1498→    confidence,
  1499→    score,
  1500→    priceTrendScore: round2(priceTrendScore),
  1501→    cashFlowScore: round2(cashFlowScore),
  1502→    idxOptScore: round2(idxOptScore),
  1503→    stkOptScore: round2(stkOptScore),
  1504→    idxFutScore: round2(idxFutScore),
  1505→    stkFutScore: round2(stkFutScore),
  1506→    suggestedStrike,
  1507→    suggestedPremium,
  1508→    stopLoss,
  1509→    target1,
  1510→    target2,
  1511→    riskReward,
  1512→    reasoning: reasoningParts.join(' | '),
  1513→    cashOptDivergence,
  1514→    futCashDivergence,
  1515→  };
  1516→}

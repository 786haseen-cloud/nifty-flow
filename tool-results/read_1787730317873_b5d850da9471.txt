   100→   100→/**
   101→   101→ * Generate access_token from request_token (one-time, after login redirect)
   102→   102→ */
   103→   103→export async function generateSession(requestToken: string): Promise<{
   104→   104→  accessToken: string;
   105→   105→  refreshToken: string;
   106→   106→} | null> {
   107→   107→  const apiKey = process.env.KITE_API_KEY;
   108→   108→  const apiSecret = process.env.KITE_API_SECRET;
   109→   109→  if (!apiKey || !apiSecret) return null;
   110→   110→
   111→   111→  try {
   112→   112→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   113→   113→    const { createHash } = await import('crypto');
   114→   114→    const checksum = createHash('sha256')
   115→   115→      .update(apiKey + requestToken + apiSecret)
   116→   116→      .digest('hex');
   117→   117→
   118→   118→    const res = await fetch(`${KITE_BASE}/session/token`, {
   119→   119→      method: 'POST',
   120→   120→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   121→   121→      body: new URLSearchParams({
   122→   122→        api_key: apiKey,
   123→   123→        request_token: requestToken,
   124→   124→        checksum,
   125→   125→      }).toString(),
   126→   126→    });
   127→   127→
   128→   128→    const data = await res.json();
   129→   129→    if (data.status === 'success' && data.data) {
   130→   130→      return {
   131→   131→        accessToken: data.data.access_token,
   132→   132→        refreshToken: data.data.refresh_token || '',
   133→   133→      };
   134→   134→    }
   135→   135→    console.error('[Kite] Session error:', data.message);
   136→   136→    return null;
   137→   137→  } catch (err) {
   138→   138→    console.error('[Kite] generateSession failed:', err);
   139→   139→    return null;
   140→   140→  }
   141→   141→}

   146→   146→let instrumentsCache: KiteInstrument[] | null = null;
   147→   147→let instrumentsCacheTime = 0;
   148→   148→
   149→   149→/**

   151→   151→ * Use after credential changes (e.g. user switched from expired → valid token)
   152→   152→ * to force a fresh CSV download on the next getInstruments() call.
   153→   153→ */
   154→   154→export function invalidateInstrumentsCache(): void {
   155→   155→  instrumentsCache = null;
   156→   156→  instrumentsCacheTime = 0;
   157→   157→}
   158→   158→
   159→   159→/**
   160→   160→ * Fetch all instruments from Kite (cached for 1 hour)
   161→   161→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   162→   162→ * We filter by exchange after caching
   163→   163→ *
   164→   164→ * @param exchange Optional exchange filter (e.g. 'NFO' returns only NFO F&O instruments)
   165→   165→ * @param forceRefresh Bypass the 1-hour cache and re-download. Useful when creds change.
   166→   166→ */
   167→   167→export async function getInstruments(exchange?: string, forceRefresh?: boolean): Promise<KiteInstrument[]> {
   168→   168→  // Return from cache if fresh (and not forced to refresh)
   169→   169→  if (!forceRefresh && instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   170→   170→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   171→   171→  }
   172→   172→
   173→   173→  try {
   174→   174→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   175→   175→    if (!res.ok) {
   176→   176→      const errText = await res.text();
   177→   177→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   178→   178→      return [];
   179→   179→    }
   180→   180→    const text = await res.text();
   181→   181→
   182→   182→    // Kite returns CSV format
   183→   183→    const lines = text.trim().split('\n');
   184→   184→    const instruments: KiteInstrument[] = [];
   185→   185→
   186→   186→    // Kite CSV column order (2025+ format):
   187→   187→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   188→   188→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   189→   189→    //
   190→   190→    // Kite changed the CSV format in 2025:
   191→   191→    //   OLD: instrument_type was 'OPTIDX'/'OPTSTK'/'FUTIDX'/'FUTSTK'/'EQ'/'INDEX'
   192→   192→    //        segment was 'NFO'/'BFO'/'NSE'/'BSE'
   193→   193→    //   NEW: instrument_type is 'CE'/'PE'/'FUT'/'EQ'
   194→   194→    //        segment is 'NFO-OPT'/'NFO-FUT'/'NFO'/'BFO-OPT'/'BFO-FUT'/'NSE'/'BSE'
   195→   195→    // To avoid breaking the rest of the code (which filters by OPTIDX/OPTSTK/FUTIDX/FUTSTK),
   196→   196→    // we normalize the new format back to the legacy types.
   197→   197→    //
   198→   198→    // Disambiguation: in the new CSV, ALL options (index + stock) are CE/PE — there is no
   199→   199→    // index-vs-stock hint in the row itself. We use the `name` column to decide:
   200→   200→    //   - If the underlying name matches a known F&O index (NIFTY, BANKNIFTY, FINNIFTY,
   201→   201→    //     SENSEX, BANKEX, MIDCPNIFTY, etc.) → OPTIDX / FUTIDX
   202→   202→    //   - Otherwise (RELIANCE, TCS, HDFCBANK, ...) → OPTSTK / FUTSTK
   203→   203→    // The set below is checked at parse time (cheap Set.has lookup).
   204→   204→    const KNOWN_FNO_INDICES = new Set([
   205→   205→      'NIFTY', 'BANKNIFTY', 'FINNIFTY', 'SENSEX', 'BANKEX',
   206→   206→      'MIDCPNIFTY', 'NIFTY MID SELECT', 'NIFTY PSE',
   207→   207→    ]);
   208→   208→    const normalizeInstrumentType = (rawType: string, segment: string, name: string): string => {
   209→   209→      // Keep legacy values as-is (in case Kite reverts or some entries still use old format)
   210→   210→      if (rawType === 'OPTIDX' || rawType === 'OPTSTK' || rawType === 'FUTIDX' || rawType === 'FUTSTK' ||
   211→   211→          rawType === 'EQ' || rawType === 'INDEX') {
   212→   212→        return rawType;
   213→   213→      }
   214→   214→      const nameUpper = name.toUpperCase();
   215→   215→      const isIndex = KNOWN_FNO_INDICES.has(nameUpper);
   216→   216→      if (rawType === 'CE' || rawType === 'PE') {
   217→   217→        // BFO always has index options only (SENSEX/BANKEX). NFO has both index + stock.
   218→   218→        return isIndex ? 'OPTIDX' : 'OPTSTK';
   219→   219→      }
   220→   220→      if (rawType === 'FUT') {
   221→   221→        return isIndex ? 'FUTIDX' : 'FUTSTK';
   222→   222→      }
   223→   223→      return rawType;
   224→   224→    };
   225→   225→
   226→   226→    for (let i = 1; i < lines.length; i++) { // skip header
   227→   227→      const cols = lines[i].split(',');
   228→   228→      if (cols.length < 12) continue;
   229→   229→      const rawType = cols[9] || '';
   230→   230→      const segment = cols[10] || '';
   231→   231→      // Kite wraps the `name` field in double quotes (e.g. "NIFTY").
   232→   232→      // Strip them so name-based filters (e.g. .includes('NIFTY')) work correctly.
   233→   233→      const rawName = cols[3] || '';
   234→   234→      const name = rawName.startsWith('"') && rawName.endsWith('"')
   235→   235→        ? rawName.slice(1, -1)
   236→   236→        : rawName;
   237→   237→      instruments.push({
   238→   238→        instrumentToken: parseInt(cols[0]) || 0,
   239→   239→        exchangeToken: parseInt(cols[1]) || 0,
   240→   240→        tradingSymbol: cols[2],
   241→   241→        name,
   242→   242→        exchange: cols[11] || '',
   243→   243→        segment,
   244→   244→        instrumentType: normalizeInstrumentType(rawType, segment, name),
   245→   245→        strike: parseFloat(cols[6]) || 0,
   246→   246→        lotSize: parseInt(cols[8]) || 1,
   247→   247→        expiry: cols[5] || '',
   248→   248→      });
   249→   249→    }
   250→   250→
   251→   251→    instrumentsCache = instruments;
   252→   252→    instrumentsCacheTime = Date.now();
   253→   253→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   254→   254→  } catch (err) {
   255→   255→    const errMsg = err instanceof Error ? err.message : String(err);
   256→   256→    console.error('[Kite] instruments fetch failed:', errMsg);
   257→   257→    // Return empty but set error flag for callers to detect
   258→   258→    return [];
   259→   259→  }
   260→   260→}

   262→   262→// ─── Quotes (Real-time) ───
   263→   263→
   264→   264→/**
   265→   265→ * Get real-time quotes for instruments
   266→   266→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   267→   267→ */
   268→   268→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   269→   269→  if (!isKiteConfigured()) return {};
   270→   270→
   271→   271→  try {
   272→   272→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   273→   273→    // Tokens have no encoding issues on any platform.
   274→   274→    // Build a map from token → original instrument key for response mapping.
   275→   275→    const tokenToKey: Record<string, string> = {};
   276→   276→    const tokenList: string[] = [];
   277→   277→
   278→   278→    for (const inst of instruments) {
   279→   279→      // If it's a numeric token, use directly
   280→   280→      if (/^\d+$/.test(inst)) {
   281→   281→        tokenToKey[inst] = inst;
   282→   282→        tokenList.push(inst);
   283→   283→      } else {
   284→   284→        // Look up token from KITE_INDEX_INSTRUMENTS
   285→   285→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   286→   286→          ([, v]) => v.symbol === inst
   287→   287→        );
   288→   288→        if (found) {
   289→   289→          const tok = String(found[1].token);
   290→   290→          tokenToKey[tok] = inst; // map token back to symbol name
   291→   291→          tokenList.push(tok);
   292→   292→        } else {
   293→   293→          // Fallback: use the instrument string as-is with encodeURI
   294→   294→          tokenToKey[inst] = inst;
   295→   295→          tokenList.push(inst);
   296→   296→        }
   297→   297→      }
   298→   298→    }
   299→   299→
   300→   300→    const iList = tokenList.join(',');
   301→   301→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   302→   302→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   303→   303→    const params = tokenList.map(t => `i=${t}`).join('&');
   304→   304→    const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   305→   305→      headers: kiteHeaders(),
   306→   306→    });
   307→   307→
   308→   308→    // Log non-success for debugging
   309→   309→    if (!res.ok) {
   310→   310→      const errText = await res.text();
   311→   311→      console.error(`[Kite] quote API ${res.status}: ${errText}`);
   312→   312→      return { _error: `HTTP ${res.status}: ${errText}` } as any;
   313→   313→    }
   314→   314→
   315→   315→    const data = await res.json();
   316→   316→    if (data.status !== 'success' || !data.data) {
   317→   317→      console.error('[Kite] quote API error:', JSON.stringify(data));
   318→   318→      return { _error: data.message || 'Unknown error' } as any;
   319→   319→    }
   320→   320→
   321→   321→    const quotes: Record<string, KiteQuote> = {};
   322→   322→    for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   323→   323→      // Kite returns the key as the token string (e.g. "256265")
   324→   324→      // Map it back to the original instrument name (e.g. "NSE:NIFTY 50")
   325→   325→      const displayKey = tokenToKey[key] || key;
   326→   326→      quotes[displayKey] = {
   327→   327→        instrumentToken: q.instrument_token || 0,
   328→   328→        lastPrice: q.last_price || 0,
   329→   329→        open: q.ohlc?.open || 0,
   330→   330→        high: q.ohlc?.high || 0,
   331→   331→        low: q.ohlc?.low || 0,
   332→   332→        close: q.ohlc?.close || 0,
   333→   333→        volume: q.volume || 0,
   334→   334→        netChange: q.net_change || 0,
   335→   335→        averagePrice: q.average_price || 0,
   336→   336→        oi: q.oi || 0,
   337→   337→        oiDayHigh: q.oi_day_high || 0,
   338→   338→        oiDayLow: q.oi_day_low || 0,
   339→   339→        dayHigh: q.ohlc?.high || 0,
   340→   340→        dayLow: q.ohlc?.low || 0,
   341→   341→      };
   342→   342→    }
   343→   343→    return quotes;
   344→   344→  } catch (err) {
   345→   345→    const errMsg = err instanceof Error ? err.message : String(err);
   346→   346→    console.error('[Kite] quote error:', errMsg);
   347→   347→    return { _error: `Exception: ${errMsg}` } as any;
   348→   348→  }
   349→   349→}
   350→   350→
   351→   351→// ─── Historical Candles ───
   352→   352→
   353→   353→/**
   354→   354→ * Get historical candle data for price line chart
   355→   355→ * instrumentToken: numeric token from instruments list
   356→   356→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   357→   357→ */
   358→   358→export async function getCandles(
   359→   359→  instrumentToken: number,
   360→   360→  interval: string = '15minute',
   361→   361→  days: number = 1,
   362→   362→): Promise<KiteHistoricalCandle[]> {
   363→   363→  if (!isKiteConfigured()) return [];
   364→   364→
   365→   365→  // Kite historical API expects IST dates.
   366→   366→  // On Vercel (UTC), we must convert to IST before formatting.
   367→   367→  const IST_OFFSET = 5.5 * 60 * 60 * 1000;
   368→   368→  const toIST = (d: Date) => new Date(d.getTime() + IST_OFFSET);
   369→   369→
   370→   370→  const toDate = toIST(new Date());
   371→   371→  const fromDate = toIST(new Date());
   372→   372→  fromDate.setDate(fromDate.getDate() - days);
   373→   373→
   374→   374→  // Kite historical API requires format "YYYY-MM-DD HH:MM:SS" (WITH seconds).
   375→   375→  // Sending "YYYY-MM-DD HH:MM" (no seconds) returns HTTP 400 "invalid from date".
   376→   376→  const fmt = (d: Date) =>
   377→   377→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}:00`;
   378→   378→
   379→   379→  try {
   380→   380→    const fromStr = fmt(fromDate);
   381→   381→    const toStr = fmt(toDate);
   382→   382→    // URL-encode the date strings (they contain a space: "2026-08-25 09:15")
   383→   383→    const url = `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${encodeURIComponent(fromStr)}&to=${encodeURIComponent(toStr)}&continuous=0`;
   384→   384→
   385→   385→    const res = await fetch(url, { headers: kiteHeaders() });
   386→   386→
   387→   387→    // Surface HTTP-level errors (403 expired token, 429 rate limit, etc.)
   388→   388→    if (!res.ok) {
   389→   389→      const errText = await res.text();
   390→   390→      console.error(`[Kite] candles HTTP ${res.status} for token ${instrumentToken}: ${errText.substring(0, 300)}`);
   391→   391→      return [];
   392→   392→    }
   393→   393→
   394→   394→    const data = await res.json();
   395→   395→    if (data.status !== 'success' || !data.data?.candles) {
   396→   396→      console.error('[Kite] candles non-success: status=%s, message=%s, from=%s, to=%s',
   397→   397→        data.status, data.message || 'no message', fromStr, toStr);
   398→   398→      return [];
   399→   399→    }
   400→   400→
   401→   401→    return (data.data.candles as any[][]).map(c => ({
   402→   402→      timestamp: c[0],
   403→   403→      open: c[1],
   404→   404→      high: c[2],
   405→   405→      low: c[3],
   406→   406→      close: c[4],
   407→   407→      volume: c[5],
   408→   408→      oi: c[6] || 0,  // c[6] = OI (F&O only), undefined for cash
   409→   409→    }));
   410→   410→  } catch (err) {
   411→   411→    const errMsg = err instanceof Error ? err.message : String(err);
   412→   412→    console.error('[Kite] candles error:', errMsg);
   413→   413→    return [];
   414→   414→  }
   415→   415→}
   416→   416→
   417→   417→// ─── Option Chain ───
   418→   418→
   419→   419→/**
   420→   420→ * Get option chain for an index or stock
   421→   421→ * Uses InstrumentSpec for correct exchange, segment
   422→   422→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   423→   423→ * Returns all CE + PE instruments for the current expiry near ATM
   424→   424→ */
   425→   425→export async function getOptionInstruments(
   426→   426→  symbol: string,
   427→   427→  spotPrice: number,
   428→   428→  strikesAroundOverride?: number,
   429→   429→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   430→   430→  const spec = getInstrumentSpec(symbol);
   431→   431→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   432→   432→
   433→   433→  // IMPORTANT: F&O options/futures are NOT on the cash exchange.
   434→   434→  //   - NIFTY index spot is on exchange='NSE', but NIFTY OPTIONS are on exchange='NFO'
   435→   435→  //   - SENSEX spot is on exchange='BSE', but SENSEX OPTIONS are on exchange='BFO'
   436→   436→  //   - Stock (RELIANCE/TCS/...) spot is on exchange='NSE', but their OPTIONS are on exchange='NFO'
   437→   437→  // So we use spec.segment ('NFO' or 'BFO') — which equals the F&O exchange column in
   438→   438→  // Kite's CSV — NOT spec.exchange ('NSE'/'BSE', which is the cash exchange).
   439→   439→  // If you call getInstruments(spec.exchange) you'll only get cash/index instruments,
   440→   440→  // and the option filter below will return ZERO matches.
   441→   441→  const instruments = await getInstruments(spec.segment);
   442→   442→
   443→   443→  // Filter options using spec.
   444→   444→  //
   445→   445→  // Kite changed their CSV format in 2025:
   446→   446→  //   OLD: segment='NFO'/'BFO', instrument_type='OPTIDX'/'OPTSTK'
   447→   447→  //   NEW: segment='NFO-OPT'/'BFO-OPT', instrument_type='CE'/'PE' (we normalize to OPTIDX)
   448→   448→  //
   449→   449→  // So we use a "starts with" match on segment (so 'NFO' matches both 'NFO' and 'NFO-OPT'),
   450→   450→  // and an equality match on instrumentType (we've already normalized new → legacy).
   451→   451→  const symUpper = symbol.toUpperCase();
   452→   452→  const indexOptions = instruments.filter(i =>
   453→   453→    i.segment.startsWith(spec.segment) &&
   454→   454→    i.instrumentType === spec.instrumentType &&
   455→   455→    (i.name.toUpperCase().includes(symUpper) ||
   456→   456→     i.tradingSymbol.toUpperCase().includes(symUpper))
   457→   457→  );
   458→   458→
   459→   459→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   460→   460→
   461→   461→  // Get unique expiries, sort by nearest
   462→   462→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   463→   463→
   464→   464→  // Use nearest expiry that's not expired
   465→   465→  const today = new Date();
   466→   466→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   467→   467→
   468→   468→  // Filter to nearest expiry
   469→   469→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   470→   470→
   471→   471→  // Get lot size from first option (all same for same underlying)
   472→   472→  const lotSize = expiryOptions[0]?.lotSize || 1;
   473→   473→
   474→   474→  // Derive strike step DYNAMICALLY from actual strike gaps
   475→   475→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   476→   476→  let strikeStep = 50;
   477→   477→  if (strikes.length >= 2) {
   478→   478→    const gaps: Record<number, number> = {};
   479→   479→    for (let i = 1; i < strikes.length; i++) {
   480→   480→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   481→   481→      gaps[gap] = (gaps[gap] || 0) + 1;
   482→   482→    }
   483→   483→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   484→   484→  }
   485→   485→
   486→   486→  // Round ATM strike using the DYNAMIC strike step
   487→   487→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   488→   488→
   489→   489→  // Get strikes around ATM
   490→   490→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   491→   491→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   492→   492→    atmStrike - strikesAround * strikeStep + i * strikeStep
   493→   493→  );
   494→   494→
   495→   495→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   496→   496→
   497→   497→  return {
   498→   498→    instruments: filtered,
   499→   499→    meta: { lotSize, strikeStep },
   500→   500→  };
   501→   501→}

   504→   504→ * Build options flow data from quotes + OI changes
   505→   505→ * Lot size and strike step come DYNAMICALLY from Kite's CSV

   507→   507→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   508→   508→  const spec = getInstrumentSpec(symbol);
   509→   509→  if (!spec) return null;
   510→   510→
   511→   511→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   512→   512→  if (optInstruments.length === 0) return null;
   513→   513→
   514→   514→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   515→   515→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   516→   516→
   517→   517→  // Batch fetch quotes
   518→   518→  const quotes = await getQuotes(iKeys);
   519→   519→  if (Object.keys(quotes).length === 0) return null;
   520→   520→
   521→   521→  // Group by strike → build flow data
   522→   522→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   523→   523→
   524→   524→  // Use DYNAMIC strike step and lot size from CSV
   525→   525→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   526→   526→
   527→   527→  for (const inst of optInstruments) {
   528→   528→    // Quotes are keyed by instrument token (string)
   529→   529→    const key = String(inst.instrumentToken);
   530→   530→    const q = quotes[key];
   531→   531→    if (!q) continue;
   532→   532→
   533→   533→    if (!strikes.has(inst.strike)) {
   534→   534→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   535→   535→    }
   536→   536→    const s = strikes.get(inst.strike)!;
   537→   537→
   538→   538→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   539→   539→    const flow = q.volume * meta.lotSize * q.averagePrice;
   540→   540→
   541→   541→    if (inst.tradingSymbol.endsWith('CE')) {
   542→   542→      s.callBuy = flow * 0.6;
   543→   543→      s.callWrite = flow * 0.4;
   544→   544→      s.callOI = q.oi;
   545→   545→      s.callVol = q.volume;
   546→   546→    } else if (inst.tradingSymbol.endsWith('PE')) {
   547→   547→      s.putBuy = flow * 0.6;
   548→   548→      s.putWrite = flow * 0.4;
   549→   549→      s.putOI = q.oi;
   550→   550→      s.putVol = q.volume;
   551→   551→    }
   552→   552→  }
   553→   553→
   554→   554→  // Build result
   555→   555→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   556→   556→    strike,
   557→   557→    isATM: strike === atmStrike,
   558→   558→    ...flow,
   559→   559→    bullishFlow: flow.callBuy + flow.putWrite,
   560→   560→    bearishFlow: flow.putBuy + flow.callWrite,
   561→   561→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   562→   562→  })).sort((a, b) => a.strike - b.strike);
   563→   563→}

   663→   663→    i.segment.startsWith(spec.segment) &&
   664→   664→    i.instrumentType === spec.instrumentType &&

   666→   666→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   667→   667→  );
   668→   668→

   670→   670→    return { lotSize: 1, strikeStep: 50 }; // fallback
   671→   671→  }

   674→   674→  const lotSize = symbolOpts[0].lotSize || 1;
   675→   675→

   677→   677→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()

   712→   712→// ─── Black-Scholes Delta Calculator ───
   713→   713→// Approximate delta for options when Kite doesn't provide Greeks.
   714→   714→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   715→   715→
   716→   716→function normCDF(x: number): number {
   717→   717→  // Rational approximation (max error ~7.5e-8)
   718→   718→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   719→   719→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   720→   720→  const sign = x < 0 ? -1 : 1;
   721→   721→  const absX = Math.abs(x);
   722→   722→  const t = 1.0 / (1.0 + p * absX);
   723→   723→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   724→   724→  return 0.5 * (1.0 + sign * y);
   725→   725→}
   726→   726→
   727→   727→/**
   728→   728→ * Calculate Black-Scholes delta for an option
   729→   729→ * @param isCall - true for CE, false for PE
   730→   730→ * @param S - Spot price
   731→   731→ * @param K - Strike price
   732→   732→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   733→   733→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   734→   734→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   735→   735→ */
   736→   736→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   737→   737→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   738→   738→  // If no IV provided, estimate from moneyness
   739→   739→  if (sigma <= 0) {
   740→   740→    const moneyness = (S - K) / S;
   741→   741→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   742→   742→  }
   743→   743→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   744→   744→  const delta = normCDF(d1);
   745→   745→  return isCall ? delta : delta - 1;
   746→   746→}

   782→   782→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   783→   783→ */
   784→   784→export async function getStrikeFlowSnapshot(
   785→   785→  symbol: string,
   786→   786→  spotPrice: number,
   787→   787→): Promise<StrikeFlowSnapshot | null> {
   788→   788→  const spec = getInstrumentSpec(symbol);
   789→   789→  if (!spec) return null;
   790→   790→
   791→   791→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   792→   792→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   793→   793→  if (optInstruments.length === 0) return null;
   794→   794→
   795→   795→  // Batch fetch quotes using instrument tokens
   796→   796→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   797→   797→  const quotes = await getQuotes(iKeys);
   798→   798→  if ('_error' in quotes) return null;
   799→   799→
   800→   800→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   801→   801→
   802→   802→  // Group by strike
   803→   803→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   804→   804→  for (const inst of optInstruments) {
   805→   805→    if (!strikeMap.has(inst.strike)) {
   806→   806→      strikeMap.set(inst.strike, { ce: null, pe: null });
   807→   807→    }
   808→   808→    const entry = strikeMap.get(inst.strike)!;
   809→   809→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   810→   810→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   811→   811→  }
   812→   812→
   813→   813→  // Calculate time to expiry
   814→   814→  const expiry = optInstruments[0]?.expiry || '';
   815→   815→  const now = new Date();
   816→   816→  const expiryDate = new Date(expiry);
   817→   817→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   818→   818→  const T = daysToExpiry / 365;
   819→   819→
   820→   820→  // Build strike data
   821→   821→  const strikes: StrikeFlowData[] = [];
   822→   822→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   823→   823→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   824→   824→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   825→   825→
   826→   826→    const ceLTP = ceQuote?.lastPrice || 0;
   827→   827→    const peLTP = peQuote?.lastPrice || 0;
   828→   828→    const ceOI = ceQuote?.oi || 0;
   829→   829→    const peOI = peQuote?.oi || 0;
   830→   830→    const ceVol = ceQuote?.volume || 0;
   831→   831→    const peVol = peQuote?.volume || 0;
   832→   832→
   833→   833→    // Compute delta via Black-Scholes
   834→   834→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   835→   835→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   836→   836→
   837→   837→    strikes.push({
   838→   838→      strike,
   839→   839→      isATM: strike === atmStrike,
   840→   840→      ceLTP,
   841→   841→      peLTP,
   842→   842→      ceOI,
   843→   843→      peOI,
   844→   844→      ceVol,
   845→   845→      peVol,
   846→   846→      ceDelta,
   847→   847→      peDelta,
   848→   848→      ceToken: ce?.instrumentToken || 0,
   849→   849→      peToken: pe?.instrumentToken || 0,
   850→   850→    });
   851→   851→  }
   852→   852→
   853→   853→  strikes.sort((a, b) => a.strike - b.strike);
   854→   854→
   855→   855→  return {
   856→   856→    timestamp: new Date().toISOString(),
   857→   857→    symbol,
   858→   858→    spotPrice,
   859→   859→    atmStrike,
   860→   860→    lotSize: meta.lotSize,
   861→   861→    strikeStep: meta.strikeStep,
   862→   862→    expiry,
   863→   863→    strikes,
   864→   864→  };
   865→   865→}

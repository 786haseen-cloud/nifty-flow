   896→// 4-color options bar (Call Buy + Put Write up, Put Buy + Call Write down)
   897→function FourColorBar({
   898→  buy1, buy2, sell1, sell2, maxAbs, height,
   899→}: {
   900→  buy1: number; buy2: number; sell1: number; sell2: number;
   901→  maxAbs: number; height: number;
   902→}) {
   903→  const b1H = Math.max(0.5, (buy1 / maxAbs) * (height * 0.25));
   904→  const b2H = Math.max(0.5, (buy2 / maxAbs) * (height * 0.25));
   905→  const s1H = Math.max(0.5, (sell1 / maxAbs) * (height * 0.25));
   906→  const s2H = Math.max(0.5, (sell2 / maxAbs) * (height * 0.25));
   907→
   908→  return (
   909→    <div className="flex flex-col items-center justify-center" style={{ width: BAR_WIDTH, height: '100%' }}>
   910→      {/* Bullish (up) */}
   911→      <div className="w-full flex-1 flex flex-col items-end justify-end">
   912→        <div className="w-full rounded-t-sm" style={{ height: b1H, background: C.callBuy }} />
   913→        <div className="w-full" style={{ height: b2H, background: C.putWrite }} />
   914→      </div>
   915→      {/* Bearish (down) */}
   916→      <div className="w-full flex-1 flex flex-col items-start">
   917→        <div className="w-full" style={{ height: s1H, background: C.putBuy }} />
   918→        <div className="w-full rounded-b-sm" style={{ height: s2H, background: C.callWrite }} />
   919→      </div>
   920→    </div>
   921→  );

   924→// Score bar component
   925→function ScoreBar({ label, score, bold = false }: { label: string; score: number; bold?: boolean }) {
   926→  const isPos = score > 0;
   927→  const pct = Math.abs(score) / 100 * 50; // max 50% from center
   928→
   929→  return (
   930→    <div className="flex items-center gap-1.5">
   931→      <span className={`w-16 text-right text-[9px] ${bold ? 'font-bold' : ''}`}>{label}</span>
   932→      <div className="flex-1 h-2 rounded-full bg-muted overflow-hidden relative">
   933→        <div className="absolute inset-y-0 left-1/2 w-px bg-border z-10" />
   934→        <div
   935→          className={`absolute inset-y-0 rounded-full ${isPos ? 'bg-emerald-500' : 'bg-red-500'}`}
   936→          style={{
   937→            left: isPos ? '50%' : `${50 - pct}%`,
   938→            width: `${pct}%`,
   939→          }}
   940→        />
   941→      </div>
   942→      <span className={`w-8 text-[9px] font-mono ${isPos ? 'text-emerald-400' : 'text-red-400'} ${bold ? 'font-bold' : ''}`}>
   943→        {isPos ? '+' : ''}{score.toFixed(0)}
   944→      </span>
   945→    </div>
   946→  );
   947→}

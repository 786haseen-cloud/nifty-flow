     1→/**
     2→ * Trend Store — global persistent state for the Trend Analysis tab.
     3→ *
     4→ * PROBLEM SOLVED
     5→ * --------------
     6→ * Previously, all trend state lived inside the <TrendAnalysisTab> component as
     7→ * useState/useRef. The shadcn Tabs component unmounts inactive tab content, so
     8→ * switching tabs destroyed all state — when the user came back, the chart was
     9→ * empty and had to re-accumulate from zero.
    10→ *
    11→ * This store fixes that by:
    12→ *   1. Holding all trend state outside any component, so tab switches don't
    13→ *      destroy it.
    14→ *   2. Persisting to localStorage via zustand/middleware/persist, so page
    15→ *      reloads (and even browser restarts on the same day) restore the
    16→ *      morning-to-now trend.
    17→ *   3. Running polling as a singleton — started once at app boot, never
    18→ *      stopped by tab changes. Only one poller instance runs regardless of
    19→ *      how many components subscribe.
    20→ *
    21→ * DATE BOUNDARY
    22→ * -------------
    23→ * All accumulated trend data is keyed to the IST trading date. When a new
    24→ * trading day starts, the store auto-clears stale data so the chart starts
    25→ * fresh each morning.
    26→ */
    27→
    28→'use client';
    29→
    30→import { create } from 'zustand';
    31→import { persist, createJSONStorage } from 'zustand/middleware';
    32→import { withCreds } from './kite-creds';
    33→import {
    34→  INDEX_SYMBOLS,
    35→  computeSymbolFlow,
    36→  type NiftyCandle,
    37→  type StockCashFlow,
    38→  type StrikeData,
    39→  type SymbolSnapshot,
    40→  type CashFlowTrendPoint,
    41→  type FlowTrendPoint,
    42→  type HighestBetResponse,
    43→} from './trend-types';
    44→
    45→// ─── IST date helper ───
    46→
    47→const IST_OFFSET_MS = 5.5 * 60 * 60 * 1000;
    48→
    49→function getISTDate(): string {
    50→  // Format YYYY-MM-DD in IST
    51→  const ist = new Date(Date.now() + IST_OFFSET_MS);
    52→  const y = ist.getUTCFullYear();
    53→  const m = String(ist.getUTCMonth() + 1).padStart(2, '0');
    54→  const d = String(ist.getUTCDate()).padStart(2, '0');
    55→  return `${y}-${m}-${d}`;
    56→}
    57→
    58→function getISTTime(): string {
    59→  return new Date().toLocaleTimeString('en-IN', {
    60→    hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
    61→  });
    62→}
    63→
    64→// ─── Store interface ───
    65→
    66→interface TrendState {
    67→  // ─── Persisted (survives reload) ───
    68→  istDate: string;                              // YYYY-MM-DD IST — date boundary check
    69→  lastPollAt: number;                           // epoch ms of last successful poll
    70→  cashFlowTrend: CashFlowTrendPoint[];          // appended each poll (max 600 pts = 2.5h @ 15s)
    71→  flowTrend: FlowTrendPoint[];                  // appended each poll (max 600 pts)
    72→  prevSnapshots: Record<string, StrikeData[]>;  // last OI snapshot per symbol (for delta computation)
    73→  cumulativeFlow: Record<string, number>;       // running totals per index + stockAggregate
    74→  prevStockTotals: { nse: number; bse: number; weighted: number }; // for computing 15s interval delta
    75→
    76→  // ─── Latest snapshot (always fresh, also persisted for snappy reload) ───
    77→  niftyCandles: NiftyCandle[];
    78→  stockCashFlow: StockCashFlow[];
    79→  trendMode: 'live' | 'demo' | 'error';
    80→
    81→  // ─── Current interval display values ───
    82→  currentIdxFlows: Record<string, number>;
    83→  currentStockFlow: number;
    84→  currentIntervalCashFlow: number;
    85→
    86→  // ─── Polling control (NOT persisted — runtime only) ───
    87→  _pollingStarted: boolean;
    88→  _pollTimer: ReturnType<typeof setInterval> | null;
    89→  _historicalBackfillDone: boolean;  // true after first successful backfill today
    90→
    91→  // ─── Actions ───
    92→  startPolling: () => void;
    93→  stopPolling: () => void;
    94→  pollOnce: () => Promise<void>;
    95→  clearTrendData: () => void;
    96→  backfillHistoricalFlow: () => Promise<void>;
    97→}
    98→
    99→// ─── Constants ───
   100→
   101→const POLL_INTERVAL_MS = 15000;
   102→// 9:15 → 15:40 IST = 6h25m = 385min = 1540 × 15s polls. Round up to 1600 for headroom.
   103→// Previously 600 (2.5h) was too small — the chart would lose its morning data
   104→// once the user kept the page open past ~11:45. Now it holds the full session.
   105→const MAX_TREND_POINTS = 1600;
   106→const STALE_GAP_MS = 4 * 60 * 60 * 1000;  // 4h gap = treat as new session
   107→
   108→const INITIAL_FLOW: Record<string, number> = {
   109→  NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0, stockAggregate: 0,
   110→};
   111→
   112→// ─── Store implementation ───
   113→
   114→export const useTrendStore = create<TrendState>()(
   115→  persist(
   116→    (set, get) => ({
   117→      // ─── Initial state ───
   118→      istDate: getISTDate(),
   119→      lastPollAt: 0,
   120→      cashFlowTrend: [],
   121→      flowTrend: [],
   122→      prevSnapshots: {},
   123→      cumulativeFlow: { ...INITIAL_FLOW },
   124→      prevStockTotals: { nse: 0, bse: 0, weighted: 0 },
   125→
   126→      niftyCandles: [],
   127→      stockCashFlow: [],
   128→      trendMode: 'demo',
   129→
   130→      currentIdxFlows: { NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0 },
   131→      currentStockFlow: 0,
   132→      currentIntervalCashFlow: 0,
   133→
   134→      _pollingStarted: false,
   135→      _pollTimer: null,
   136→      _historicalBackfillDone: false,
   137→
   138→      // ─── Actions ───
   139→
   140→      /**
   141→       * Start the singleton poller. Idempotent — safe to call from any
   142→       * component's useEffect. The first caller starts the interval;
   143→       * subsequent calls are no-ops.
   144→       *
   145→       * Also runs an immediate poll so data shows up right away on first
   146→       * app boot (instead of waiting up to 15s).
   147→       */
   148→      startPolling: () => {
   149→        const state = get();
   150→        if (state._pollingStarted) return;
   151→
   152→        // Date boundary check — clear stale trend data if the trading day changed
   153→        const today = getISTDate();
   154→        if (state.istDate !== today) {
   155→          console.log(`[TrendStore] New trading day (${state.istDate} → ${today}), clearing accumulated data`);
   156→          get().clearTrendData();
   157→        }
   158→
   159→        // Stale gap check — if last poll was >4h ago, also clear (likely a fresh session next morning)
   160→        if (state.lastPollAt > 0 && (Date.now() - state.lastPollAt) > STALE_GAP_MS) {
   161→          console.log('[TrendStore] Long gap since last poll (>4h), clearing accumulated data');
   162→          get().clearTrendData();
   163→        }
   164→
   165→        set({ _pollingStarted: true });
   166→        console.log('[TrendStore] Starting singleton poller');
   167→
   168→        // Immediate first poll
   169→        get().pollOnce().catch((e) => console.error('[TrendStore] initial poll error:', e));
   170→
   171→        // Trigger historical backfill in background (only once per day)
   172→        // This fetches today's OI history from Kite and reconstructs the
   173→        // morning-to-now options flow trend.
   174→        if (!get()._historicalBackfillDone && get().flowTrend.length === 0) {
   175→          // Wait a few seconds for the first poll to complete and set trendMode
   176→          setTimeout(() => {
   177→            const state = get();
   178→            if (state.trendMode === 'live' && !state._historicalBackfillDone) {
   179→              console.log('[TrendStore] Triggering historical flow backfill...');
   180→              get().backfillHistoricalFlow().catch((e) =>
   181→                console.error('[TrendStore] backfill error:', e)
   182→              );
   183→            }
   184→          }, 3000);
   185→        }
   186→
   187→        // Start interval
   188→        const timer = setInterval(() => {
   189→          try {
   190→            get().pollOnce().catch((e) => console.error('[TrendStore] poll error:', e));
   191→          } catch (e) {
   192→            console.error('[TrendStore] poll sync error:', e);
   193→          }
   194→        }, POLL_INTERVAL_MS);
   195→
   196→        // Self-healing watchdog: if lastPollAt becomes stale (> 90s),
   197→        // the main poller likely died (browser throttled background tab,
   198→        // unhandled edge case, etc.). Detect and restart.
   199→        const watchdog = setInterval(() => {
   200→          const s = get();
   201→          const gap = Date.now() - s.lastPollAt;
   202→          if (s.lastPollAt > 0 && gap > 90_000 && s.trendMode === 'live') {
   203→            console.warn(`[TrendStore] Watchdog: no poll for ${Math.round(gap / 1000)}s, restarting`);
   204→            if (s._pollTimer) clearInterval(s._pollTimer);
   205→            const newTimer = setInterval(() => {
   206→              try {
   207→                get().pollOnce().catch((e) => console.error('[TrendStore] poll error:', e));
   208→              } catch (e) {
   209→                console.error('[TrendStore] poll sync error:', e);
   210→              }
   211→            }, POLL_INTERVAL_MS);
   212→            set({ _pollTimer: newTimer });
   213→            get().pollOnce().catch((e) => console.error('[TrendStore] watchdog poll error:', e));
   214→          }
   215→        }, 60_000);
   216→
   217→        // Store timers — _pollTimer is in state (not persisted),
   218→        // watchdog is stored off-band to avoid extending the interface.
   219→        set({ _pollTimer: timer });
   220→        (get() as any)._watchdogTimer = watchdog;
   221→      },
   222→
   223→      /**
   224→       * Stop polling. Should only be called on app unmount — never on tab
   225→       * switch. In practice, this almost never fires for a single-page app.
   226→       */
   227→      stopPolling: () => {
   228→        const state = get();
   229→        if (state._pollTimer) clearInterval(state._pollTimer);
   230→        const wd = (get() as any)._watchdogTimer;
   231→        if (wd) clearInterval(wd);
   232→        set({ _pollTimer: null, _pollingStarted: false });
   233→      },
   234→
   235→      /**
   236→       * Clear all accumulated trend data (called on date boundary or stale gap).
   237→       * Keeps polling config but resets the trend arrays + cumulative counters.
   238→       *
   239→       * BUG FIX: Previously did not clear `niftyCandles` or `stockCashFlow`,
   240→       * so stale demo data from a previous session could persist into the new
   241→       * day and confuse the user (e.g. showing yesterday's fake 24350 price
   242→       * instead of today's real market data).
   243→       */
   244→      clearTrendData: () => {
   245→        set({
   246→          istDate: getISTDate(),
   247→          lastPollAt: 0,
   248→          cashFlowTrend: [],
   249→          flowTrend: [],
   250→          prevSnapshots: {},
   251→          cumulativeFlow: { ...INITIAL_FLOW },
   252→          prevStockTotals: { nse: 0, bse: 0, weighted: 0 },
   253→          niftyCandles: [],
   254→          stockCashFlow: [],
   255→          trendMode: 'demo',
   256→          currentIdxFlows: { NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0 },
   257→          currentStockFlow: 0,
   258→          currentIntervalCashFlow: 0,
   259→          _historicalBackfillDone: false,
   260→        });
   261→      },
   262→
   263→      /**
   264→       * Historical Flow Backfill
   265→       * --------------------------
   266→       * Called once per day (on first poll when flowTrend is empty and mode is live).
   267→       * Fetches /api/kite/historical-flow which reconstructs morning-to-now
   268→       * options flow from Kite's historical 5-min OI candles.
   269→       *
   270→       * When the backfill returns:
   271→       * 1. Replaces flowTrend with the historical data
   272→       * 2. Sets cumulativeFlow to the server-computed totals
   273→       * 3. Stores prevSnapshots so the next live poll computes correct deltas
   274→       *    from the last historical candle (no double-counting)
   275→       *
   276→       * The backfill takes ~2 minutes (rate-limited API calls), so it runs in
   277→       * the background. Live polls continue appending 15s points. When the
   278→       * backfill completes, it merges — keeping any live points that arrived
   279→       * during the backfill, but replacing the cumulative totals to match.
   280→       */
   281→      backfillHistoricalFlow: async () => {
   282→        try {
   283→          const res = await fetch(withCreds('/api/kite/historical-flow'));
   284→          const data = await res.json();
   285→
   286→          if (data.mode !== 'live' || !data.flowTrend || data.flowTrend.length === 0) {
   287→            console.log('[TrendStore] Backfill returned no data, will retry later if creds are refreshed');
   288→            // Do NOT set _historicalBackfillDone = true here. This allows retry
   289→            // when the user refreshes their Kite credentials in the Settings tab
   290→            // mid-session. The backfill is rate-limited by the in-memory 60s
   291→            // server-side cache anyway, so this won't hammer the API.
   292→            // Instead, schedule a single retry after 5 minutes.
   293→            setTimeout(() => {
   294→              const s = get();
   295→              if (!s._historicalBackfillDone && s.trendMode === 'live') {
   296→                console.log('[TrendStore] Retrying historical flow backfill...');
   297→                get().backfillHistoricalFlow().catch((e) =>
   298→                  console.error('[TrendStore] backfill retry error:', e)
   299→                );
   300→              }
   301→            }, 5 * 60 * 1000);
   302→            return;
   303→          }
   304→
   305→          const state = get();
   306→          const histFlow = data.flowTrend as FlowTrendPoint[];
   307→          const histCumulative = data.cumulativeFlow as Record<string, number>;
   308→          const histPrevSnapshots = data.prevSnapshots as Record<string, StrikeData[]>;
   309→
   310→          // If live polls arrived during the backfill, we need to offset them.
   311→          // The live polls accumulated from zero (or from a previous session).
   312→          // We keep any live points that arrived AFTER the last historical point,
   313→          // but adjust their cumulative values to continue from the historical totals.
   314→          const liveFlowTrend = state.flowTrend;
   315→          const liveCumulative = { ...state.cumulativeFlow };
   316→
   317→          let mergedFlow: FlowTrendPoint[];
   318→          let mergedCumulative: Record<string, number>;
   319→          let mergedPrevSnapshots: Record<string, StrikeData[]>;
   320→
   321→          if (liveFlowTrend.length === 0) {
   322→            // Simple case: no live data arrived during backfill
   323→            mergedFlow = histFlow;
   324→            mergedCumulative = { ...INITIAL_FLOW, ...histCumulative };
   325→            mergedPrevSnapshots = { ...histPrevSnapshots };
   326→          } else {
   327→            // Live data arrived during the ~2min backfill.
   328→            // Keep historical data + append live data with adjusted cumulative values.
   329→            // The offset = historical total - live total at the overlap point.
   330→            mergedFlow = [...histFlow];
   331→            mergedCumulative = { ...INITIAL_FLOW, ...histCumulative };
   332→
   333→            // Compute offsets for each symbol
   334→            const offsets: Record<string, number> = {};
   335→            for (const sym of Object.keys(mergedCumulative)) {
   336→              const histVal = mergedCumulative[sym] || 0;
   337→              const liveVal = liveCumulative[sym] || 0;
   338→              offsets[sym] = histVal - liveVal;
   339→            }
   340→
   341→            // Re-compute live points with offset applied
   342→            // Only append live points that are newer than the last historical point
   343→            const lastHistTime = histFlow[histFlow.length - 1]?.time || '';
   344→            for (const pt of liveFlowTrend) {
   345→              if (pt.time > lastHistTime) {
   346→                mergedFlow.push({
   347→                  time: pt.time,
   348→                  NIFTY: Math.round(((pt.NIFTY || 0) + (offsets.NIFTY || 0)) * 10) / 10,
   349→                  BANKNIFTY: Math.round(((pt.BANKNIFTY || 0) + (offsets.BANKNIFTY || 0)) * 10) / 10,
   350→                  FINNIFTY: Math.round(((pt.FINNIFTY || 0) + (offsets.FINNIFTY || 0)) * 10) / 10,
   351→                  SENSEX: Math.round(((pt.SENSEX || 0) + (offsets.SENSEX || 0)) * 10) / 10,
   352→                  stockAggregate: Math.round(((pt.stockAggregate || 0) + (offsets.stockAggregate || 0)) * 10) / 10,
   353→                });
   354→              }
   355→            }
   356→
   357→            // Update cumulative to be the latest merged value
   358→            if (mergedFlow.length > 0) {
   359→              const last = mergedFlow[mergedFlow.length - 1];
   360→              mergedCumulative.NIFTY = last.NIFTY;
   361→              mergedCumulative.BANKNIFTY = last.BANKNIFTY;
   362→              mergedCumulative.FINNIFTY = last.FINNIFTY;
   363→              mergedCumulative.SENSEX = last.SENSEX;
   364→              mergedCumulative.stockAggregate = last.stockAggregate;
   365→            }
   366→
   367→            // Keep the more recent prevSnapshots (live > historical)
   368→            mergedPrevSnapshots = {
   369→              ...histPrevSnapshots,
   370→              ...state.prevSnapshots,
   371→            };
   372→          }
   373→
   374→          // Trim to max points
   375→          const trimmed = mergedFlow.length > MAX_TREND_POINTS
   376→            ? mergedFlow.slice(mergedFlow.length - MAX_TREND_POINTS)
   377→            : mergedFlow;
   378→
   379→          set({
   380→            flowTrend: trimmed,
   381→            cumulativeFlow: mergedCumulative,
   382→            prevSnapshots: mergedPrevSnapshots,
   383→            _historicalBackfillDone: true,
   384→          });
   385→
   386→          console.log(
   387→            `[TrendStore] Backfill complete: ${histFlow.length} historical + ${liveFlowTrend.length} live points, ` +
   388→            `${Object.keys(mergedPrevSnapshots).length} symbols with snapshots`
   389→          );
   390→        } catch (err) {
   391→          console.error('[TrendStore] backfillHistoricalFlow error:', err);
   392→          // Don't set _historicalBackfillDone on error — allow retry.
   393→          // Schedule a single retry in 5 minutes (same as empty-response case).
   394→          setTimeout(() => {
   395→            const s = get();
   396→            if (!s._historicalBackfillDone && s.trendMode === 'live') {
   397→              console.log('[TrendStore] Retrying historical flow backfill after error...');
   398→              get().backfillHistoricalFlow().catch((e) =>
   399→                console.error('[TrendStore] backfill retry error:', e)
   400→              );
   401→            }
   402→          }, 5 * 60 * 1000);
   403→        }
   404→      },
   405→
   406→      /**
   407→       * Single polling iteration. Fetches both the trends API (Nifty candles
   408→       * + stock cash flow) and the highest-bet API (options OI snapshots),
   409→       * computes deltas, and appends to the trend arrays.
   410→       *
   411→       * Safe to call even if a previous poll is still in-flight — Zustand
   412→       * merges updates atomically.
   413→       *
   414→       * DATE BOUNDARY (in-poll check): If the IST date has changed since the
   415→       * last poll (e.g. user kept the page open overnight), clear all
   416→       * accumulated data before appending the new poll. This guarantees the
   417→       * chart always shows ONLY the current trading day's data — yesterday's
   418→       * flow is wiped automatically the moment the new IST day begins.
   419→       */
   420→      pollOnce: async () => {
   421→        // ─── In-poll date boundary check ───
   422→        // If the IST date has changed since the last poll, clear the trend
   423→        // data so the new day starts fresh. Without this, a user who leaves
   424→        // the page open overnight would see yesterday's flow bleed into the
   425→        // new day's chart.
   426→        const prevDate = get().istDate;
   427→        const todayDate = getISTDate();
   428→        if (prevDate !== todayDate) {
   429→          console.log(`[TrendStore] Date rollover detected during poll (${prevDate} → ${todayDate}), clearing accumulated data`);
   430→          get().clearTrendData();
   431→        }
   432→
   433→        // Run both fetches in parallel
   434→        const [trendsRes, betRes] = await Promise.allSettled([
   435→          fetch(withCreds('/api/kite/trends')).then((r) => r.json()),
   436→          fetch(withCreds('/api/kite/highest-bet')).then((r) => r.json()),
   437→        ]);
   438→
   439→        const now = Date.now();
   440→        const time = getISTTime();
   441→        const CR = 10000000;
   442→
   443→        // ─── Process trends (Nifty candles + stock cash flow) ───
   444→
   445→        if (trendsRes.status === 'fulfilled') {
   446→          const data = trendsRes.value;
   447→          const mode = (data.mode || 'demo') as 'live' | 'demo' | 'error';
   448→
   449→          const niftyCandles: NiftyCandle[] = data.niftyCandles || [];
   450→          const stockCashFlow: StockCashFlow[] = data.stockCashFlow || [];
   451→
   452→          // DEMO → LIVE TRANSITION: If we were in demo mode (e.g. user had no
   453→          // creds or expired creds) and now we're in live mode (user just
   454→          // refreshed their token in Settings), clear all accumulated demo
   455→          // data so the charts start fresh from real data. Otherwise the user
   456→          // would see a confusing mix of fake + real data points.
   457→          const prevMode = get().trendMode;
   458→          if (mode === 'live' && prevMode === 'demo' &&
   459→              (get().cashFlowTrend.length > 0 || get().niftyCandles.length > 0)) {
   460→            console.log('[TrendStore] Demo → Live transition detected, clearing stale demo data');
   461→            // Clear everything except istDate + _historicalBackfillDone
   462→            set({
   463→              lastPollAt: 0,
   464→              cashFlowTrend: [],
   465→              flowTrend: [],
   466→              prevSnapshots: {},
   467→              cumulativeFlow: { ...INITIAL_FLOW },
   468→              prevStockTotals: { nse: 0, bse: 0, weighted: 0 },
   469→              currentIdxFlows: { NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0 },
   470→              currentStockFlow: 0,
   471→              currentIntervalCashFlow: 0,
   472→              _historicalBackfillDone: false,
   473→            });
   474→          }
   475→
   476→          // Compute cumulative-since-market-open totals (these come directly
   477→          // from the API — no client-side accumulation needed).
   478→          if (stockCashFlow.length > 0) {
   479→            const nseTotal = stockCashFlow.reduce((s, v) => s + v.nseCashFlow, 0);
   480→            const bseTotal = stockCashFlow.reduce((s, v) => s + v.bseCashFlow, 0);
   481→            const netTotal = nseTotal + bseTotal;
   482→            const weightedTotal = stockCashFlow.reduce((s, v) => s + v.weightedFlow, 0);
   483→
   484→            // 15s interval delta = current cumulative − previous cumulative
   485→            const prev = get().prevStockTotals;
   486→            const intervalDelta = (prev.nse === 0 && prev.bse === 0)
   487→              ? 0  // first poll of the day — no delta to compute
   488→              : netTotal - (prev.nse + prev.bse);
   489→
   490→            const point: CashFlowTrendPoint = {
   491→              time,
   492→              nse: Math.round((nseTotal / CR) * 10) / 10,
   493→              bse: Math.round((bseTotal / CR) * 10) / 10,
   494→              net: Math.round((netTotal / CR) * 10) / 10,
   495→              weighted: Math.round((weightedTotal / CR) * 10) / 10,
   496→              interval: Math.round((intervalDelta / CR) * 10) / 10,
   497→            };
   498→
   499→            const prevTrend = get().cashFlowTrend;
   500→            const newTrend = [...prevTrend, point];
   501→            const trimmed = newTrend.length > MAX_TREND_POINTS
   502→              ? newTrend.slice(newTrend.length - MAX_TREND_POINTS)
   503→              : newTrend;
   504→
   505→            set({
   506→              niftyCandles,
   507→              stockCashFlow,
   508→              trendMode: mode,
   509→              cashFlowTrend: trimmed,
   510→              prevStockTotals: { nse: nseTotal, bse: weightedTotal, weighted: weightedTotal },
   511→              currentIntervalCashFlow: intervalDelta,
   512→              lastPollAt: now,
   513→              istDate: getISTDate(),
   514→            });
   515→          } else {
   516→            // No stock data — still update candles + mode
   517→            set({ niftyCandles, trendMode: mode, lastPollAt: now, istDate: getISTDate() });
   518→          }
   519→        }
   520→
   521→        // ─── Process highest-bet (options OI snapshots → delta-weighted flow) ───
   522→
   523→        if (betRes.status === 'fulfilled') {
   524→          const data = betRes.value as HighestBetResponse;
   525→          if (data.symbols && data.symbols.length > 0) {
   526→            const prevSnapshots = get().prevSnapshots;
   527→            const cumulativeFlow = { ...get().cumulativeFlow };
   528→            const newCurrentIdx: Record<string, number> = {
   529→              NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0,
   530→            };
   531→            const newPrevSnapshots: Record<string, StrikeData[]> = { ...prevSnapshots };
   532→            let stockAgg = 0;
   533→
   534→            for (const sym of data.symbols) {
   535→              if (sym.strikes.length === 0) continue;
   536→
   537→              const prevStrikes = prevSnapshots[sym.symbol];
   538→              if (!prevStrikes || prevStrikes.length === 0) {
   539→                // First snapshot — store and skip delta computation
   540→                newPrevSnapshots[sym.symbol] = sym.strikes;
   541→                continue;
   542→              }
   543→
   544→              const flow = computeSymbolFlow(prevStrikes, sym.strikes, sym.lotSize);
   545→
   546→              if (sym.type === 'index' && (INDEX_SYMBOLS as readonly string[]).includes(sym.symbol)) {
   547→                cumulativeFlow[sym.symbol] = (cumulativeFlow[sym.symbol] || 0) + flow.net;
   548→                newCurrentIdx[sym.symbol] = flow.net;
   549→              } else if (sym.type === 'stock') {
   550→                stockAgg += flow.net;
   551→              }
   552→
   553→              newPrevSnapshots[sym.symbol] = sym.strikes;
   554→            }
   555→
   556→            cumulativeFlow.stockAggregate = (cumulativeFlow.stockAggregate || 0) + stockAgg;
   557→
   558→            const flowPoint: FlowTrendPoint = {
   559→              time,
   560→              NIFTY: Math.round((cumulativeFlow.NIFTY || 0) * 10) / 10,
   561→              BANKNIFTY: Math.round((cumulativeFlow.BANKNIFTY || 0) * 10) / 10,
   562→              FINNIFTY: Math.round((cumulativeFlow.FINNIFTY || 0) * 10) / 10,
   563→              SENSEX: Math.round((cumulativeFlow.SENSEX || 0) * 10) / 10,
   564→              stockAggregate: Math.round((cumulativeFlow.stockAggregate || 0) * 10) / 10,
   565→            };
   566→
   567→            const prevFlowTrend = get().flowTrend;
   568→            const newFlowTrend = [...prevFlowTrend, flowPoint];
   569→            const trimmedFlow = newFlowTrend.length > MAX_TREND_POINTS
   570→              ? newFlowTrend.slice(newFlowTrend.length - MAX_TREND_POINTS)
   571→              : newFlowTrend;
   572→
   573→            set({
   574→              flowTrend: trimmedFlow,
   575→              prevSnapshots: newPrevSnapshots,
   576→              cumulativeFlow,
   577→              currentIdxFlows: newCurrentIdx,
   578→              currentStockFlow: stockAgg,
   579→              lastPollAt: now,
   580→            });
   581→          }
   582→        }
   583→      },
   584→    }),
   585→    {
   586→      name: 'trend-store-v2',  // v2: added historical OI backfill
   587→      storage: createJSONStorage(() => {
   588→        // Guard for SSR — localStorage is only available in the browser
   589→        if (typeof window === 'undefined') {
   590→          return {
   591→            getItem: () => null,
   592→            setItem: () => {},
   593→            removeItem: () => {},
   594→          };
   595→        }
   596→        return localStorage;
   597→      }),
   598→      // Don't persist runtime-only fields (timers, polling flag)
   599→      partialize: (state) => ({
   600→        istDate: state.istDate,
   601→        lastPollAt: state.lastPollAt,
   602→        cashFlowTrend: state.cashFlowTrend,
   603→        flowTrend: state.flowTrend,
   604→        prevSnapshots: state.prevSnapshots,
   605→        cumulativeFlow: state.cumulativeFlow,
   606→        prevStockTotals: state.prevStockTotals,
   607→        niftyCandles: state.niftyCandles,
   608→        stockCashFlow: state.stockCashFlow,
   609→        trendMode: state.trendMode,
   610→        currentIdxFlows: state.currentIdxFlows,
   611→        currentStockFlow: state.currentStockFlow,
   612→        currentIntervalCashFlow: state.currentIntervalCashFlow,
   613→      }),
   614→    }
   615→  )

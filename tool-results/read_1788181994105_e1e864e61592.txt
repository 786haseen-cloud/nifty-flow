    15→    15→ * 4-Color OI classification (matches Strike Flow Map convention):
    16→    16→ *
    17→    17→ *  CE  Buy   = ΔOI>0 & ΔLTP≥0  → bright green  #00B050
    18→    18→ *  CE  Write = ΔOI>0 & ΔLTP<0  → light red     #FFCCCC (dark red border)
    19→    19→ *  CE  Close = ΔOI<0           → grey          #6b7280
    20→    20→ *  PE  Write = ΔOI>0 & ΔLTP≥0  → light green   #C6EFCE (dark green border)
    21→    21→ *  PE  Buy   = ΔOI>0 & ΔLTP<0  → bright red    #BD2130
    22→    22→ *  PE  Close = ΔOI<0           → grey          #6b7280
    23→    23→ */
    24→    24→type OIColor = {
    25→    25→  bg: string;        // inline background color
    26→    26→  border?: string;   // inline border color (for Write classes)
    27→    27→  text: string;      // tailwind text class for the OI label
    28→    28→  label?: string;    // short label: "Buy", "Write", "Close"
    29→    29→};

    31→    31→function getOIColor(oiChange: number, ltpChange: number, isPE: boolean): OIColor {
    32→    32→  // No previous data — neutral
    33→    33→  if (oiChange === 0 && ltpChange === 0) {
    34→    34→    return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    35→    35→  }
    36→    36→  // Unwinding / short covering (ΔOI < 0)
    37→    37→  if (oiChange < 0) {
    38→    38→    return { bg: '#6b7280', text: 'text-zinc-400', label: 'Close' };
    39→    39→  }
    40→    40→  // oiChange >= 0 — new positions opened (or no change)
    41→    41→  if (isPE) {
    42→    42→    // PE: Write = bullish (price up/stable + OI up), Buy = bearish (price down + OI up)
    43→    43→    if (ltpChange >= 0) return { bg: '#C6EFCE', border: '#16a34a', text: 'text-emerald-700', label: 'Write' };
    44→    44→    if (ltpChange < 0)  return { bg: '#BD2130', text: 'text-red-100', label: 'Buy' };
    45→    45→  } else {
    46→    46→    // CE: Buy = bullish (price up/stable + OI up), Write = bearish (price down + OI up)
    47→    47→    if (ltpChange >= 0) return { bg: '#00B050', text: 'text-green-100', label: 'Buy' };
    48→    48→    if (ltpChange < 0)  return { bg: '#FFCCCC', border: '#dc2626', text: 'text-red-700', label: 'Write' };
    49→    49→  }
    50→    50→  // Fallback (shouldn't reach here)
    51→    51→  return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    52→    52→}

   138→   138→function formatLakhs(n: number): string {
   139→   139→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
   140→   140→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
   141→   141→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
   142→   142→  return n.toLocaleString('en-IN');
   143→   143→}

   149→   149→// Compute Max Pain: strike where option buyers lose the most
   150→   150→function computeMaxPain(strikes: StrikeFlowData[]): number {
   151→   151→  if (strikes.length === 0) return 0;
   152→   152→  let minLoss = Infinity;
   153→   153→  let maxPainStrike = strikes[0].strike;
   154→   154→
   155→   155→  for (const k of strikes) {
   156→   156→    let totalLoss = 0;
   157→   157→    for (const s of strikes) {
   158→   158→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
   159→   159→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
   160→   160→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
   161→   161→      // But we only have OI, not premium. Standard Max Pain uses:
   162→   162→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
   163→   163→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
   164→   164→      // Simplified: use intrinsic value approach
   165→   165→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
   166→   166→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
   167→   167→    }
   168→   168→    if (totalLoss < minLoss) {
   169→   169→      minLoss = totalLoss;
   170→   170→      maxPainStrike = k.strike;
   171→   171→    }
   172→   172→  }
   173→   173→  return maxPainStrike;
   174→   174→}

   176→   176→// Generate demo data for 11 strikes
   177→   177→function generateDemoData(symbol: string, prev?: StrikeFlowSnapshot | null): StrikeFlowSnapshot {
   178→   178→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   179→   179→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   180→   180→  const atmStrike = Math.round(spot / step) * step;
   181→   181→
   182→   182→  // If we have a previous snapshot, apply small perturbations to it.
   183→   183→  // This produces realistic per-strike ΔOI and ΔLTP values that
   184→   184→  // feed into the 4-color OI coding (Buy/Write/Close).
   185→   185→  if (prev && prev.strikes.length === 11) {
   186→   186→    const spotJitter = Math.round((Math.random() - 0.5) * step * 0.3);
   187→   187→    const newSpot = prev.spotPrice + spotJitter;
   188→   188→    const newAtm = Math.round(newSpot / step) * step;
   189→   189→
   190→   190→    const strikes: StrikeFlowData[] = prev.strikes.map((s, idx) => {
   191→   191→      // OI change: random walk — some strikes gain, some lose
   192→   192→      const oiDelta = Math.round((Math.random() - 0.45) * s.ceOI * 0.04);
   193→   193→      const peOiDelta = Math.round((Math.random() - 0.45) * s.peOI * 0.04);
   194→   194→
   195→   195→      // LTP change based on spot movement + moneyness
   196→   196→      const ceMoneyness = (newSpot - s.strike) / newSpot;
   197→   197→      const peMoneyness = (s.strike - newSpot) / newSpot;
   198→   198→      const iv = 0.12 + Math.abs(ceMoneyness) * 0.3;
   199→   199→      const ceLTPDelta = spotJitter > 0
   200→   200→        ? iv * Math.abs(spotJitter) * (0.3 + Math.random() * 0.3)
   201→   201→        : -iv * Math.abs(spotJitter) * (0.2 + Math.random() * 0.3);
   202→   202→      const peLTPDelta = -ceLTPDelta * 0.8;
   203→   203→
   204→   204→      const ceITM = newSpot > s.strike;
   205→   205→      const peITM = newSpot < s.strike;
   206→   206→
   207→   207→      return {
   208→   208→        ...s,
   209→   209→        ceLTP: Math.max(0.5, s.ceLTP + ceLTPDelta),
   210→   210→        peLTP: Math.max(0.5, s.peLTP + peLTPDelta),
   211→   211→        ceOI: Math.max(1000, s.ceOI + oiDelta),
   212→   212→        peOI: Math.max(1000, s.peOI + peOiDelta),
   213→   213→        isATM: s.strike === newAtm,
   214→   214→        ceVol: Math.round(Math.abs(oiDelta) * (1 + Math.random())),
   215→   215→        peVol: Math.round(Math.abs(peOiDelta) * (1 + Math.random())),
   216→   216→      };
   217→   217→    });
   218→   218→
   219→   219→    return {
   220→   220→      timestamp: new Date().toISOString(),
   221→   221→      symbol,
   222→   222→      spotPrice: newSpot,
   223→   223→      atmStrike: newAtm,
   224→   224→      lotSize: prev.lotSize,
   225→   225→      strikeStep: step,
   226→   226→      expiry: prev.expiry,
   227→   227→      strikes,
   228→   228→    };
   229→   229→  }
   230→   230→
   231→   231→  // First call (no prev) — generate from scratch with random initial values
   232→   232→  const strikes: StrikeFlowData[] = [];
   233→   233→
   234→   234→  for (let i = -5; i <= 5; i++) {
   235→   235→    const strike = atmStrike + i * step;
   236→   236→    const dist = Math.abs(i);
   237→   237→    // Heavier OI near ATM, with some randomness
   238→   238→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   239→   239→    const noise = 0.6 + Math.random() * 0.8;
   240→   240→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   241→   241→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   242→   242→    const moneyness = (spot - strike) / spot;
   243→   243→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   244→   244→    const ceITM = spot > strike;
   245→   245→    const peITM = spot < strike;
   246→   246→
   247→   247→    strikes.push({
   248→   248→      strike,
   249→   249→      isATM: i === 0,
   250→   250→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   251→   251→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   252→   252→      ceOI,
   253→   253→      peOI,
   254→   254→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   255→   255→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   256→   256→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   257→   257→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   258→   258→      ceToken: 100 + i,
   259→   259→      peToken: 200 + i,
   260→   260→    });
   261→   261→  }
   262→   262→
   263→   263→  return {
   264→   264→    timestamp: new Date().toISOString(),
   265→   265→    symbol,
   266→   266→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   267→   267→    atmStrike,
   268→   268→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   269→   269→    strikeStep: step,
   270→   270→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   271→   271→    strikes,
   272→   272→  };
   273→   273→}

   284→   284→function loadPCRHistory(symbol: string): number[] {
   285→   285→  if (typeof window === 'undefined') return [];
   286→   286→  try {
   287→   287→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   288→   288→    return raw ? JSON.parse(raw) : [];
   289→   289→  } catch {
   290→   290→    return [];
   291→   291→  }
   292→   292→}

   301→   301→}

   343→   343→        const demo = generateDemoData(symbol, prev);
   344→   344→        // Compute per-strike deltas BEFORE updating the ref

   348→   348→          for (const s of demo.strikes) {

   367→   367→      setDemoData(null);
   368→   368→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   369→   369→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,

   371→   371→

   373→   373→      const prev = prevSnapshotRef.current;
   374→   374→      if (prev && prev.symbol === data.symbol && prev.strikes) {

   380→   380→        }
   381→   381→        setStrikeDeltas(d);
   382→   382→      } else {
   383→   383→        setStrikeDeltas(new Map());
   384→   384→      }
   385→   385→      prevSnapshotRef.current = data;
   386→   386→      setSnapshot(data);
   387→   387→    } catch (err) {
   388→   388→      setError(err instanceof Error ? err.message : 'Fetch failed');
   389→   389→    } finally {
   390→   390→      setLoading(false);
   391→   391→    }
   392→   392→  }, [symbol]);

   394→   394→  // Initial fetch + polling for per-symbol OI walls

   398→   398→    return () => {
   399→   399→      if (intervalRef.current) clearInterval(intervalRef.current);
   400→   400→    };

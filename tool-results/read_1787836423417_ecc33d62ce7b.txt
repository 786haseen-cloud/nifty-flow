     1→/**
     2→ * Zerodha Kite API Integration
     3→ *
     4→ * Connects to Kite Connect REST API for real NSE/BSE market data.
     5→ * When KITE_API_KEY + KITE_ACCESS_TOKEN are set in .env, real data flows.
     6→ * Otherwise, falls back to demo data.
     7→ *
     8→ * Setup:
     9→ * 1. Go to https://developers.kite.trade
    10→ * 2. Create an app → get API Key + Secret
    11→ * 3. Login: https://kite.zerodha.com/connect/login?api_key=YOUR_KEY
    12→ * 4. After login, capture request_token from redirect URL
    13→ * 5. Call /api/kite/auth?request_token=xxx to generate access_token
    14→ * 6. Put access_token in .env → KITE_ACCESS_TOKEN=xxx
    15→ * 7. Redeploy
    16→ *
    17→ * Access tokens expire daily at midnight. Re-login each morning.
    18→ */

    36→export function isKiteConfigured(): boolean {
    37→  return !!(
    38→    (process.env.KITE_API_KEY && process.env.KITE_ACCESS_TOKEN) ||
    39→    (_overrideApiKey && _overrideAccessToken)
    40→  );

    43→function kiteHeaders() {

    46→  return {
    47→    'Authorization': `token ${apiKey}:${accessToken}`,
    48→    'X-Kite-Version': '3',
    49→    'Content-Type': 'application/x-www-form-urlencoded',
    50→  };
    51→}

   100→/**
   101→ * Generate access_token from request_token (one-time, after login redirect)
   102→ */
   103→export async function generateSession(requestToken: string): Promise<{
   104→  accessToken: string;
   105→  refreshToken: string;
   106→} | null> {
   107→  const apiKey = process.env.KITE_API_KEY;
   108→  const apiSecret = process.env.KITE_API_SECRET;
   109→  if (!apiKey || !apiSecret) return null;
   110→
   111→  try {
   112→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   113→    const { createHash } = await import('crypto');
   114→    const checksum = createHash('sha256')
   115→      .update(apiKey + requestToken + apiSecret)
   116→      .digest('hex');
   117→
   118→    const res = await fetch(`${KITE_BASE}/session/token`, {
   119→      method: 'POST',
   120→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   121→      body: new URLSearchParams({
   122→        api_key: apiKey,
   123→        request_token: requestToken,
   124→        checksum,
   125→      }).toString(),
   126→    });
   127→
   128→    const data = await res.json();
   129→    if (data.status === 'success' && data.data) {
   130→      return {
   131→        accessToken: data.data.access_token,
   132→        refreshToken: data.data.refresh_token || '',
   133→      };
   134→    }
   135→    console.error('[Kite] Session error:', data.message);
   136→    return null;
   137→  } catch (err) {
   138→    console.error('[Kite] generateSession failed:', err);
   139→    return null;
   140→  }
   141→}

   145→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   146→let instrumentsCache: KiteInstrument[] | null = null;
   147→let instrumentsCacheTime = 0;
   148→
   149→/**
   150→ * Invalidate the in-memory instruments cache.
   151→ * Use after credential changes (e.g. user switched from expired → valid token)
   152→ * to force a fresh CSV download on the next getInstruments() call.
   153→ */
   154→export function invalidateInstrumentsCache(): void {
   155→  instrumentsCache = null;
   156→  instrumentsCacheTime = 0;
   157→}
   158→
   159→/**
   160→ * Fetch all instruments from Kite (cached for 1 hour)
   161→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   162→ * We filter by exchange after caching
   163→ *
   164→ * @param exchange Optional exchange filter (e.g. 'NFO' returns only NFO F&O instruments)
   165→ * @param forceRefresh Bypass the 1-hour cache and re-download. Useful when creds change.
   166→ */
   167→export async function getInstruments(exchange?: string, forceRefresh?: boolean): Promise<KiteInstrument[]> {
   168→  // Return from cache if fresh (and not forced to refresh)
   169→  if (!forceRefresh && instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   170→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   171→  }
   172→
   173→  try {
   174→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   175→    if (!res.ok) {
   176→      const errText = await res.text();
   177→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   178→      return [];
   179→    }
   180→    const text = await res.text();
   181→
   182→    // Kite returns CSV format
   183→    const lines = text.trim().split('\n');
   184→    const instruments: KiteInstrument[] = [];
   185→
   186→    // Kite CSV column order (2025+ format):
   187→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   188→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   189→    //
   190→    // Kite changed the CSV format in 2025:
   191→    //   OLD: instrument_type was 'OPTIDX'/'OPTSTK'/'FUTIDX'/'FUTSTK'/'EQ'/'INDEX'
   192→    //        segment was 'NFO'/'BFO'/'NSE'/'BSE'
   193→    //   NEW: instrument_type is 'CE'/'PE'/'FUT'/'EQ'
   194→    //        segment is 'NFO-OPT'/'NFO-FUT'/'NFO'/'BFO-OPT'/'BFO-FUT'/'NSE'/'BSE'
   195→    // To avoid breaking the rest of the code (which filters by OPTIDX/OPTSTK/FUTIDX/FUTSTK),
   196→    // we normalize the new format back to the legacy types.
   197→    //
   198→    // Disambiguation: in the new CSV, ALL options (index + stock) are CE/PE — there is no
   199→    // index-vs-stock hint in the row itself. We use the `name` column to decide:
   200→    //   - If the underlying name matches a known F&O index (NIFTY, BANKNIFTY, FINNIFTY,
   201→    //     SENSEX, BANKEX, MIDCPNIFTY, etc.) → OPTIDX / FUTIDX
   202→    //   - Otherwise (RELIANCE, TCS, HDFCBANK, ...) → OPTSTK / FUTSTK
   203→    // The set below is checked at parse time (cheap Set.has lookup).
   204→    const KNOWN_FNO_INDICES = new Set([
   205→      'NIFTY', 'BANKNIFTY', 'FINNIFTY', 'SENSEX', 'BANKEX',
   206→      'MIDCPNIFTY', 'NIFTY MID SELECT', 'NIFTY PSE',
   207→    ]);
   208→    const normalizeInstrumentType = (rawType: string, segment: string, name: string): string => {
   209→      // Keep legacy values as-is (in case Kite reverts or some entries still use old format)
   210→      if (rawType === 'OPTIDX' || rawType === 'OPTSTK' || rawType === 'FUTIDX' || rawType === 'FUTSTK' ||
   211→          rawType === 'INDEX') {
   212→        return rawType;
   213→      }
   214→      // CRITICAL FIX: In the 2025+ Kite CSV, cash index instruments (NIFTY 50, SENSEX, etc.)
   215→      // have instrument_type='EQ' and segment='INDICES'. We must normalize these to 'INDEX'
   216→      // so that downstream lookups (e.g. highest-bet finding cash instruments) can filter by
   217→      // instrumentType === 'INDEX' instead of the ambiguous 'EQ' (which also matches stocks).
   218→      if (rawType === 'EQ' && segment === 'INDICES') {
   219→        return 'INDEX';
   220→      }
   221→      // Regular EQ stocks stay as 'EQ'
   222→      if (rawType === 'EQ') {
   223→        return rawType;
   224→      }
   225→      const nameUpper = name.toUpperCase();
   226→      const isIndex = KNOWN_FNO_INDICES.has(nameUpper);
   227→      if (rawType === 'CE' || rawType === 'PE') {
   228→        // BFO always has index options only (SENSEX/BANKEX). NFO has both index + stock.
   229→        return isIndex ? 'OPTIDX' : 'OPTSTK';
   230→      }
   231→      if (rawType === 'FUT') {
   232→        return isIndex ? 'FUTIDX' : 'FUTSTK';
   233→      }
   234→      return rawType;
   235→    };
   236→
   237→    for (let i = 1; i < lines.length; i++) { // skip header
   238→      const cols = lines[i].split(',');
   239→      if (cols.length < 12) continue;
   240→      const rawType = cols[9] || '';
   241→      const segment = cols[10] || '';
   242→      // Kite wraps the `name` field in double quotes (e.g. "NIFTY").
   243→      // Strip them so name-based filters (e.g. .includes('NIFTY')) work correctly.
   244→      const rawName = cols[3] || '';
   245→      const name = rawName.startsWith('"') && rawName.endsWith('"')
   246→        ? rawName.slice(1, -1)
   247→        : rawName;
   248→      instruments.push({
   249→        instrumentToken: parseInt(cols[0]) || 0,
   250→        exchangeToken: parseInt(cols[1]) || 0,
   251→        tradingSymbol: cols[2],
   252→        name,
   253→        exchange: cols[11] || '',
   254→        segment,
   255→        instrumentType: normalizeInstrumentType(rawType, segment, name),
   256→        strike: parseFloat(cols[6]) || 0,
   257→        lotSize: parseInt(cols[8]) || 1,
   258→        expiry: cols[5] || '',
   259→      });
   260→    }
   261→
   262→    instrumentsCache = instruments;
   263→    instrumentsCacheTime = Date.now();
   264→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   265→  } catch (err) {
   266→    const errMsg = err instanceof Error ? err.message : String(err);
   267→    console.error('[Kite] instruments fetch failed:', errMsg);
   268→    // Return empty but set error flag for callers to detect
   269→    return [];
   270→  }
   271→}

   275→/**
   276→ * Get real-time quotes for instruments
   277→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   278→ */
   279→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   280→  if (!isKiteConfigured()) return {};
   281→
   282→  try {
   283→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   284→    // Tokens have no encoding issues on any platform.
   285→    // Build a map from token → original instrument key for response mapping.
   286→    const tokenToKey: Record<string, string> = {};
   287→    const tokenList: string[] = [];
   288→
   289→    for (const inst of instruments) {
   290→      // If it's a numeric token, use directly
   291→      if (/^\d+$/.test(inst)) {
   292→        tokenToKey[inst] = inst;
   293→        tokenList.push(inst);
   294→      } else {
   295→        // Look up token from KITE_INDEX_INSTRUMENTS
   296→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   297→          ([, v]) => v.symbol === inst
   298→        );
   299→        if (found) {
   300→          const tok = String(found[1].token);
   301→          tokenToKey[tok] = inst; // map token back to symbol name
   302→          tokenList.push(tok);
   303→        } else {
   304→          // Fallback: use the instrument string as-is with encodeURI
   305→          tokenToKey[inst] = inst;
   306→          tokenList.push(inst);
   307→        }
   308→      }
   309→    }
   310→
   311→    const iList = tokenList.join(',');
   312→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   313→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   314→    // Chunk into batches of 100 to avoid URL length limits and API throttling.
   315→    const CHUNK_SIZE = 100;
   316→    const allQuotes: Record<string, KiteQuote> = {};
   317→
   318→    for (let chunkStart = 0; chunkStart < tokenList.length; chunkStart += CHUNK_SIZE) {
   319→      const chunk = tokenList.slice(chunkStart, chunkStart + CHUNK_SIZE);
   320→      const params = chunk.map(t => `i=${t}`).join('&');
   321→      const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   322→        headers: kiteHeaders(),
   323→      });
   324→
   325→      // Log non-success for debugging
   326→      if (!res.ok) {
   327→        const errText = await res.text();
   328→        console.error(`[Kite] quote API ${res.status}: ${errText}`);
   329→        if (Object.keys(allQuotes).length === 0) {
   330→          return { _error: `HTTP ${res.status}: ${errText}` } as any;
   331→        }
   332→        continue; // partial success — return what we have
   333→      }
   334→
   335→      const data = await res.json();
   336→      if (data.status !== 'success' || !data.data) {
   337→        console.error('[Kite] quote API error:', JSON.stringify(data));
   338→        if (Object.keys(allQuotes).length === 0) {
   339→          return { _error: data.message || 'Unknown error' } as any;
   340→        }
   341→        continue;
   342→      }
   343→
   344→      for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   345→        const displayKey = tokenToKey[key] || key;
   346→        allQuotes[displayKey] = {
   347→          instrumentToken: q.instrument_token || 0,
   348→          lastPrice: q.last_price || 0,
   349→          open: q.ohlc?.open || 0,
   350→          high: q.ohlc?.high || 0,
   351→          low: q.ohlc?.low || 0,
   352→          close: q.ohlc?.close || 0,
   353→          volume: q.volume || 0,
   354→          netChange: q.net_change || 0,
   355→          averagePrice: q.average_price || 0,
   356→          oi: q.oi || 0,
   357→          oiDayHigh: q.oi_day_high || 0,
   358→          oiDayLow: q.oi_day_low || 0,
   359→          dayHigh: q.ohlc?.high || 0,
   360→          dayLow: q.ohlc?.low || 0,
   361→        };
   362→      }
   363→    }
   364→
   365→    return allQuotes;
   366→  } catch (err) {
   367→    const errMsg = err instanceof Error ? err.message : String(err);
   368→    console.error('[Kite] quote error:', errMsg);
   369→    return { _error: `Exception: ${errMsg}` } as any;
   370→  }
   371→}

   375→/**
   376→ * Get historical candle data for price line chart
   377→ * instrumentToken: numeric token from instruments list
   378→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   379→ */
   380→export async function getCandles(
   381→  instrumentToken: number,
   382→  interval: string = '15minute',
   383→  days: number = 1,
   384→): Promise<KiteHistoricalCandle[]> {
   385→  if (!isKiteConfigured()) return [];
   386→
   387→  // Kite historical API expects IST dates.
   388→  // On Vercel (UTC), we must convert to IST before formatting.
   389→  const IST_OFFSET = 5.5 * 60 * 60 * 1000;
   390→  const toIST = (d: Date) => new Date(d.getTime() + IST_OFFSET);
   391→
   392→  const toDate = toIST(new Date());
   393→  const fromDate = toIST(new Date());
   394→  fromDate.setDate(fromDate.getDate() - days);
   395→
   396→  // Kite historical API requires format "YYYY-MM-DD HH:MM:SS" (WITH seconds).
   397→  // Sending "YYYY-MM-DD HH:MM" (no seconds) returns HTTP 400 "invalid from date".
   398→  const fmt = (d: Date) =>
   399→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}:00`;
   400→
   401→  try {
   402→    const fromStr = fmt(fromDate);
   403→    const toStr = fmt(toDate);
   404→    // URL-encode the date strings (they contain a space: "2026-08-25 09:15")
   405→    const url = `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${encodeURIComponent(fromStr)}&to=${encodeURIComponent(toStr)}&continuous=0`;
   406→
   407→    const res = await fetch(url, { headers: kiteHeaders() });
   408→
   409→    // Surface HTTP-level errors (403 expired token, 429 rate limit, etc.)
   410→    if (!res.ok) {
   411→      const errText = await res.text();
   412→      console.error(`[Kite] candles HTTP ${res.status} for token ${instrumentToken}: ${errText.substring(0, 300)}`);
   413→      return [];
   414→    }
   415→
   416→    const data = await res.json();
   417→    if (data.status !== 'success' || !data.data?.candles) {
   418→      console.error('[Kite] candles non-success: status=%s, message=%s, from=%s, to=%s',
   419→        data.status, data.message || 'no message', fromStr, toStr);
   420→      return [];
   421→    }
   422→
   423→    return (data.data.candles as any[][]).map(c => ({
   424→      timestamp: c[0],
   425→      open: c[1],
   426→      high: c[2],
   427→      low: c[3],
   428→      close: c[4],
   429→      volume: c[5],
   430→      oi: c[6] || 0,  // c[6] = OI (F&O only), undefined for cash
   431→    }));
   432→  } catch (err) {
   433→    const errMsg = err instanceof Error ? err.message : String(err);
   434→    console.error('[Kite] candles error:', errMsg);
   435→    return [];
   436→  }
   437→}

   439→// ─── Option Chain ───

   441→/**
   442→ * Get option chain for an index or stock
   443→ * Uses InstrumentSpec for correct exchange, segment
   444→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   445→ * Returns all CE + PE instruments for the current expiry near ATM
   446→ */
   447→export async function getOptionInstruments(
   448→  symbol: string,
   449→  spotPrice: number,
   450→  strikesAroundOverride?: number,
   451→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   452→  const spec = getInstrumentSpec(symbol);
   453→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   454→
   455→  // IMPORTANT: F&O options/futures are NOT on the cash exchange.
   456→  //   - NIFTY index spot is on exchange='NSE', but NIFTY OPTIONS are on exchange='NFO'
   457→  //   - SENSEX spot is on exchange='BSE', but SENSEX OPTIONS are on exchange='BFO'
   458→  //   - Stock (RELIANCE/TCS/...) spot is on exchange='NSE', but their OPTIONS are on exchange='NFO'
   459→  // So we use spec.segment ('NFO' or 'BFO') — which equals the F&O exchange column in
   460→  // Kite's CSV — NOT spec.exchange ('NSE'/'BSE', which is the cash exchange).
   461→  // If you call getInstruments(spec.exchange) you'll only get cash/index instruments,
   462→  // and the option filter below will return ZERO matches.
   463→  const instruments = await getInstruments(spec.segment);
   464→
   465→  // Filter options using spec.
   466→  //
   467→  // Kite changed their CSV format in 2025:
   468→  //   OLD: segment='NFO'/'BFO', instrument_type='OPTIDX'/'OPTSTK'
   469→  //   NEW: segment='NFO-OPT'/'BFO-OPT', instrument_type='CE'/'PE' (we normalize to OPTIDX)
   470→  //
   471→  // So we use a "starts with" match on segment (so 'NFO' matches both 'NFO' and 'NFO-OPT'),
   472→  // and an equality match on instrumentType (we've already normalized new → legacy).
   473→  const symUpper = symbol.toUpperCase();
   474→  const indexOptions = instruments.filter(i =>
   475→    i.segment.startsWith(spec.segment) &&
   476→    i.instrumentType === spec.instrumentType &&
   477→    (i.name.toUpperCase().includes(symUpper) ||
   478→     i.tradingSymbol.toUpperCase().includes(symUpper))
   479→  );
   480→
   481→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   482→
   483→  // Get unique expiries, sort by nearest
   484→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   485→
   486→  // Use nearest expiry that's not expired
   487→  const today = new Date();
   488→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   489→
   490→  // Filter to nearest expiry
   491→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   492→
   493→  // Get lot size from first option (all same for same underlying)
   494→  const lotSize = expiryOptions[0]?.lotSize || 1;
   495→
   496→  // Derive strike step DYNAMICALLY from actual strike gaps
   497→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   498→  let strikeStep = 50;
   499→  if (strikes.length >= 2) {
   500→    const gaps: Record<number, number> = {};
   501→    for (let i = 1; i < strikes.length; i++) {
   502→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   503→      gaps[gap] = (gaps[gap] || 0) + 1;
   504→    }
   505→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   506→  }
   507→
   508→  // Round ATM strike using the DYNAMIC strike step
   509→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   510→
   511→  // Get strikes around ATM
   512→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   513→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   514→    atmStrike - strikesAround * strikeStep + i * strikeStep
   515→  );
   516→
   517→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   518→
   519→  return {
   520→    instruments: filtered,
   521→    meta: { lotSize, strikeStep },
   522→  };
   523→}

   525→/**
   526→ * Build options flow data from quotes + OI changes
   527→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   528→ */
   529→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   530→  const spec = getInstrumentSpec(symbol);
   531→  if (!spec) return null;
   532→
   533→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   534→  if (optInstruments.length === 0) return null;
   535→
   536→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   537→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   538→
   539→  // Batch fetch quotes
   540→  const quotes = await getQuotes(iKeys);
   541→  if (Object.keys(quotes).length === 0) return null;
   542→
   543→  // Group by strike → build flow data
   544→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   545→
   546→  // Use DYNAMIC strike step and lot size from CSV
   547→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   548→
   549→  for (const inst of optInstruments) {
   550→    // Quotes are keyed by instrument token (string)
   551→    const key = String(inst.instrumentToken);
   552→    const q = quotes[key];
   553→    if (!q) continue;
   554→
   555→    if (!strikes.has(inst.strike)) {
   556→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   557→    }
   558→    const s = strikes.get(inst.strike)!;
   559→
   560→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   561→    const flow = q.volume * meta.lotSize * q.averagePrice;
   562→
   563→    if (inst.tradingSymbol.endsWith('CE')) {
   564→      s.callBuy = flow * 0.6;
   565→      s.callWrite = flow * 0.4;
   566→      s.callOI = q.oi;
   567→      s.callVol = q.volume;
   568→    } else if (inst.tradingSymbol.endsWith('PE')) {
   569→      s.putBuy = flow * 0.6;
   570→      s.putWrite = flow * 0.4;
   571→      s.putOI = q.oi;
   572→      s.putVol = q.volume;
   573→    }
   574→  }
   575→
   576→  // Build result
   577→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   578→    strike,
   579→    isATM: strike === atmStrike,
   580→    ...flow,
   581→    bullishFlow: flow.callBuy + flow.putWrite,
   582→    bearishFlow: flow.putBuy + flow.callWrite,
   583→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   584→  })).sort((a, b) => a.strike - b.strike);
   585→}

   666→/**
   667→ * Derive lot size and strike step from Kite instruments CSV
   668→ * - lotSize: from the option instrument's lot_size field
   669→ * - strikeStep: derived from the gap between adjacent strikes
   670→ */
   671→export async function getInstrumentMeta(
   672→  symbol: string,
   673→  spotPrice: number,
   674→): Promise<InstrumentMeta> {
   675→  const spec = getInstrumentSpec(symbol);
   676→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   677→
   678→  // Fetch instruments from the F&O exchange (NFO/BFO), NOT the cash exchange (NSE/BSE).
   679→  // See getOptionInstruments() for the full explanation.
   680→  const allInstruments = await getInstruments(spec.segment);
   681→
   682→  // Filter to this symbol's options for current/near expiry.
   683→  // Use startsWith on segment to match both legacy 'NFO' and new 'NFO-OPT' formats.
   684→  const symbolOpts = allInstruments.filter(i =>
   685→    i.segment.startsWith(spec.segment) &&
   686→    i.instrumentType === spec.instrumentType &&
   687→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   688→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   689→  );
   690→
   691→  if (symbolOpts.length === 0) {
   692→    return { lotSize: 1, strikeStep: 50 }; // fallback
   693→  }
   694→
   695→  // Get lot size from the first matching option (all have same lot size for same underlying)
   696→  const lotSize = symbolOpts[0].lotSize || 1;
   697→
   698→  // Derive strike step from unique sorted strikes near ATM
   699→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   700→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   701→
   702→  const expiryOpts = nearestExpiry
   703→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   704→    : symbolOpts;
   705→
   706→  // Get unique strikes, sorted
   707→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   708→
   709→  // Find strikes near ATM and compute step from adjacent gaps
   710→  let strikeStep = 50; // default fallback
   711→  if (strikes.length >= 2) {
   712→    // Use the most common gap between adjacent strikes
   713→    const gaps: Record<number, number> = {};
   714→    for (let i = 1; i < strikes.length; i++) {
   715→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   716→      gaps[gap] = (gaps[gap] || 0) + 1;
   717→    }
   718→    // Most frequent gap = the strike step
   719→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   720→  }
   721→
   722→  return { lotSize, strikeStep };
   723→}

   734→// ─── Black-Scholes Delta Calculator ───
   735→// Approximate delta for options when Kite doesn't provide Greeks.
   736→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   737→
   738→function normCDF(x: number): number {
   739→  // Rational approximation (max error ~7.5e-8)
   740→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   741→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   742→  const sign = x < 0 ? -1 : 1;
   743→  const absX = Math.abs(x);
   744→  const t = 1.0 / (1.0 + p * absX);
   745→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   746→  return 0.5 * (1.0 + sign * y);
   747→}
   748→
   749→/**
   750→ * Calculate Black-Scholes delta for an option
   751→ * @param isCall - true for CE, false for PE
   752→ * @param S - Spot price
   753→ * @param K - Strike price
   754→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   755→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   756→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   757→ */
   758→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   759→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   760→  // If no IV provided, estimate from moneyness
   761→  if (sigma <= 0) {
   762→    const moneyness = (S - K) / S;
   763→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   764→  }
   765→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   766→  const delta = normCDF(d1);
   767→  return isCall ? delta : delta - 1;
   768→}

   801→/**
   802→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   803→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   804→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   805→ */
   806→export async function getStrikeFlowSnapshot(
   807→  symbol: string,
   808→  spotPrice: number,
   809→): Promise<StrikeFlowSnapshot | null> {
   810→  const spec = getInstrumentSpec(symbol);
   811→  if (!spec) return null;
   812→
   813→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   814→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   815→  if (optInstruments.length === 0) return null;
   816→
   817→  // Batch fetch quotes using instrument tokens
   818→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   819→  const quotes = await getQuotes(iKeys);
   820→  if ('_error' in quotes) return null;
   821→
   822→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   823→
   824→  // Group by strike
   825→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   826→  for (const inst of optInstruments) {
   827→    if (!strikeMap.has(inst.strike)) {
   828→      strikeMap.set(inst.strike, { ce: null, pe: null });
   829→    }
   830→    const entry = strikeMap.get(inst.strike)!;
   831→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   832→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   833→  }
   834→
   835→  // Calculate time to expiry
   836→  const expiry = optInstruments[0]?.expiry || '';
   837→  const now = new Date();
   838→  const expiryDate = new Date(expiry);
   839→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   840→  const T = daysToExpiry / 365;
   841→
   842→  // Build strike data
   843→  const strikes: StrikeFlowData[] = [];
   844→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   845→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   846→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   847→
   848→    const ceLTP = ceQuote?.lastPrice || 0;
   849→    const peLTP = peQuote?.lastPrice || 0;
   850→    const ceOI = ceQuote?.oi || 0;
   851→    const peOI = peQuote?.oi || 0;
   852→    const ceVol = ceQuote?.volume || 0;
   853→    const peVol = peQuote?.volume || 0;
   854→
   855→    // Compute delta via Black-Scholes
   856→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   857→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   858→
   859→    strikes.push({
   860→      strike,
   861→      isATM: strike === atmStrike,
   862→      ceLTP,
   863→      peLTP,
   864→      ceOI,
   865→      peOI,
   866→      ceVol,
   867→      peVol,
   868→      ceDelta,
   869→      peDelta,
   870→      ceToken: ce?.instrumentToken || 0,
   871→      peToken: pe?.instrumentToken || 0,
   872→    });
   873→  }
   874→
   875→  strikes.sort((a, b) => a.strike - b.strike);
   876→
   877→  return {
   878→    timestamp: new Date().toISOString(),
   879→    symbol,
   880→    spotPrice,
   881→    atmStrike,
   882→    lotSize: meta.lotSize,
   883→    strikeStep: meta.strikeStep,
   884→    expiry,
   885→    strikes,
   886→  };
   887→}

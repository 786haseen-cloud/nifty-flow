     1→/**
     2→ * Zerodha Kite API Integration
     3→ *
     4→ * Connects to Kite Connect REST API for real NSE/BSE market data.
     5→ * When KITE_API_KEY + KITE_ACCESS_TOKEN are set in .env, real data flows.
     6→ * Otherwise, falls back to demo data.
     7→ *
     8→ * Setup:
     9→ * 1. Go to https://developers.kite.trade
    10→ * 2. Create an app → get API Key + Secret
    11→ * 3. Login: https://kite.zerodha.com/connect/login?api_key=YOUR_KEY
    12→ * 4. After login, capture request_token from redirect URL
    13→ * 5. Call /api/kite/auth?request_token=xxx to generate access_token
    14→ * 6. Put access_token in .env → KITE_ACCESS_TOKEN=xxx
    15→ * 7. Redeploy
    16→ *
    17→ * Access tokens expire daily at midnight. Re-login each morning.
    18→ */

    31→export function setKiteOverride(apiKey?: string, accessToken?: string): void {

    36→export function isKiteConfigured(): boolean {
    37→  return !!(
    38→    (process.env.KITE_API_KEY && process.env.KITE_ACCESS_TOKEN) ||
    39→    (_overrideApiKey && _overrideAccessToken)
    40→  );
    41→}

    43→function kiteHeaders() {
    44→  const apiKey = _overrideApiKey || process.env.KITE_API_KEY || '';
    45→  const accessToken = _overrideAccessToken || process.env.KITE_ACCESS_TOKEN || '';
    46→  return {
    47→    'Authorization': `token ${apiKey}:${accessToken}`,
    48→    'X-Kite-Version': '3',
    49→    'Content-Type': 'application/x-www-form-urlencoded',
    50→  };
    51→}

    96→/**
    97→ * Generate access_token from request_token (one-time, after login redirect)
    98→ */
    99→export async function generateSession(requestToken: string): Promise<{
   100→  accessToken: string;
   101→  refreshToken: string;
   102→} | null> {
   103→  const apiKey = process.env.KITE_API_KEY;
   104→  const apiSecret = process.env.KITE_API_SECRET;
   105→  if (!apiKey || !apiSecret) return null;
   106→
   107→  try {
   108→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   109→    const { createHash } = await import('crypto');
   110→    const checksum = createHash('sha256')
   111→      .update(apiKey + requestToken + apiSecret)
   112→      .digest('hex');
   113→
   114→    const res = await fetch(`${KITE_BASE}/session/token`, {
   115→      method: 'POST',
   116→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   117→      body: new URLSearchParams({
   118→        api_key: apiKey,
   119→        request_token: requestToken,
   120→        checksum,
   121→      }).toString(),
   122→    });
   123→
   124→    const data = await res.json();
   125→    if (data.status === 'success' && data.data) {
   126→      return {
   127→        accessToken: data.data.access_token,
   128→        refreshToken: data.data.refresh_token || '',
   129→      };
   130→    }
   131→    console.error('[Kite] Session error:', data.message);
   132→    return null;
   133→  } catch (err) {
   134→    console.error('[Kite] generateSession failed:', err);
   135→    return null;
   136→  }
   137→}

   139→// ─── Instruments ───
   140→
   141→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   142→let instrumentsCache: KiteInstrument[] | null = null;
   143→let instrumentsCacheTime = 0;
   144→
   145→/**
   146→ * Fetch all instruments from Kite (cached for 1 hour)
   147→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   148→ * We filter by exchange after caching
   149→ */
   150→export async function getInstruments(exchange?: string): Promise<KiteInstrument[]> {
   151→  // Return from cache if fresh
   152→  if (instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   153→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   154→  }
   155→
   156→  try {
   157→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   158→    if (!res.ok) {
   159→      const errText = await res.text();
   160→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   161→      return [];
   162→    }
   163→    const text = await res.text();
   164→
   165→    // Kite returns CSV format
   166→    const lines = text.trim().split('\n');
   167→    const instruments: KiteInstrument[] = [];
   168→
   169→    // Kite CSV column order (2025+ format):
   170→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   171→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   172→    for (let i = 1; i < lines.length; i++) { // skip header
   173→      const cols = lines[i].split(',');
   174→      if (cols.length < 12) continue;
   175→      instruments.push({
   176→        instrumentToken: parseInt(cols[0]) || 0,
   177→        exchangeToken: parseInt(cols[1]) || 0,
   178→        tradingSymbol: cols[2],
   179→        name: cols[3] || '',
   180→        exchange: cols[11] || '',
   181→        segment: cols[10] || '',
   182→        instrumentType: cols[9] || '',
   183→        strike: parseFloat(cols[6]) || 0,
   184→        lotSize: parseInt(cols[8]) || 1,
   185→        expiry: cols[5] || '',
   186→      });
   187→    }
   188→
   189→    instrumentsCache = instruments;
   190→    instrumentsCacheTime = Date.now();
   191→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   192→  } catch (err) {
   193→    const errMsg = err instanceof Error ? err.message : String(err);
   194→    console.error('[Kite] instruments fetch failed:', errMsg);
   195→    // Return empty but set error flag for callers to detect
   196→    return [];
   197→  }
   198→}

   200→// ─── Quotes (Real-time) ───
   201→
   202→/**
   203→ * Get real-time quotes for instruments
   204→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   205→ */
   206→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   207→  if (!isKiteConfigured()) return {};
   208→
   209→  try {
   210→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   211→    // Tokens have no encoding issues on any platform.
   212→    // Build a map from token → original instrument key for response mapping.
   213→    const tokenToKey: Record<string, string> = {};
   214→    const tokenList: string[] = [];
   215→
   216→    for (const inst of instruments) {
   217→      // If it's a numeric token, use directly
   218→      if (/^\d+$/.test(inst)) {
   219→        tokenToKey[inst] = inst;
   220→        tokenList.push(inst);
   221→      } else {
   222→        // Look up token from KITE_INDEX_INSTRUMENTS
   223→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   224→          ([, v]) => v.symbol === inst
   225→        );
   226→        if (found) {
   227→          const tok = String(found[1].token);
   228→          tokenToKey[tok] = inst; // map token back to symbol name
   229→          tokenList.push(tok);
   230→        } else {
   231→          // Fallback: use the instrument string as-is with encodeURI
   232→          tokenToKey[inst] = inst;
   233→          tokenList.push(inst);
   234→        }
   235→      }
   236→    }
   237→
   238→    const iList = tokenList.join(',');
   239→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   240→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   241→    const params = tokenList.map(t => `i=${t}`).join('&');
   242→    const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   243→      headers: kiteHeaders(),
   244→    });
   245→
   246→    // Log non-success for debugging
   247→    if (!res.ok) {
   248→      const errText = await res.text();
   249→      console.error(`[Kite] quote API ${res.status}: ${errText}`);
   250→      return { _error: `HTTP ${res.status}: ${errText}` } as any;
   251→    }
   252→
   253→    const data = await res.json();
   254→    if (data.status !== 'success' || !data.data) {
   255→      console.error('[Kite] quote API error:', JSON.stringify(data));
   256→      return { _error: data.message || 'Unknown error' } as any;
   257→    }
   258→
   259→    const quotes: Record<string, KiteQuote> = {};
   260→    for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   261→      // Kite returns the key as the token string (e.g. "256265")
   262→      // Map it back to the original instrument name (e.g. "NSE:NIFTY 50")
   263→      const displayKey = tokenToKey[key] || key;
   264→      quotes[displayKey] = {
   265→        instrumentToken: q.instrument_token || 0,
   266→        lastPrice: q.last_price || 0,
   267→        open: q.ohlc?.open || 0,
   268→        high: q.ohlc?.high || 0,
   269→        low: q.ohlc?.low || 0,
   270→        close: q.ohlc?.close || 0,
   271→        volume: q.volume || 0,
   272→        netChange: q.net_change || 0,
   273→        averagePrice: q.average_price || 0,
   274→        oi: q.oi || 0,
   275→        oiDayHigh: q.oi_day_high || 0,
   276→        oiDayLow: q.oi_day_low || 0,
   277→      };
   278→    }
   279→    return quotes;
   280→  } catch (err) {
   281→    const errMsg = err instanceof Error ? err.message : String(err);
   282→    console.error('[Kite] quote error:', errMsg);
   283→    return { _error: `Exception: ${errMsg}` } as any;
   284→  }
   285→}

   289→/**
   290→ * Get historical candle data for price line chart
   291→ * instrumentToken: numeric token from instruments list
   292→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   293→ */
   294→export async function getCandles(
   295→  instrumentToken: number,
   296→  interval: string = '15minute',
   297→  days: number = 1,
   298→): Promise<KiteHistoricalCandle[]> {
   299→  if (!isKiteConfigured()) return [];
   300→
   301→  const toDate = new Date();
   302→  const fromDate = new Date();
   303→  fromDate.setDate(fromDate.getDate() - days);
   304→
   305→  const fmt = (d: Date) =>
   306→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`;
   307→
   308→  try {
   309→    const res = await fetch(
   310→      `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${fmt(fromDate)}&to=${fmt(toDate)}&continuous=0`,
   311→      { headers: kiteHeaders() }
   312→    );
   313→
   314→    const data = await res.json();
   315→    if (data.status !== 'success' || !data.data?.candles) return [];
   316→
   317→    return (data.data.candles as any[][]).map(c => ({
   318→      timestamp: c[0],
   319→      open: c[1],
   320→      high: c[2],
   321→      low: c[3],
   322→      close: c[4],
   323→      volume: c[5],
   324→    }));
   325→  } catch (err) {
   326→    const errMsg = err instanceof Error ? err.message : String(err);
   327→    console.error('[Kite] candles error:', errMsg);
   328→    return [];
   329→  }
   330→}

   332→// ─── Option Chain ───

   334→/**
   335→ * Get option chain for an index or stock
   336→ * Uses InstrumentSpec for correct exchange, segment
   337→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   338→ * Returns all CE + PE instruments for the current expiry near ATM
   339→ */
   340→export async function getOptionInstruments(
   341→  symbol: string,
   342→  spotPrice: number,
   343→  strikesAroundOverride?: number,
   344→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   345→  const spec = getInstrumentSpec(symbol);
   346→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   347→
   348→  // Fetch instruments from the correct exchange (NSE or BSE)
   349→  const instruments = await getInstruments(spec.exchange);
   350→
   351→  // Filter options using spec
   352→  const indexOptions = instruments.filter(i =>
   353→    i.segment === spec.segment &&
   354→    i.instrumentType === spec.instrumentType &&
   355→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   356→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   357→  );
   358→
   359→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   360→
   361→  // Get unique expiries, sort by nearest
   362→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   363→
   364→  // Use nearest expiry that's not expired
   365→  const today = new Date();
   366→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   367→
   368→  // Filter to nearest expiry
   369→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   370→
   371→  // Get lot size from first option (all same for same underlying)
   372→  const lotSize = expiryOptions[0]?.lotSize || 1;
   373→
   374→  // Derive strike step DYNAMICALLY from actual strike gaps
   375→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   376→  let strikeStep = 50;
   377→  if (strikes.length >= 2) {
   378→    const gaps: Record<number, number> = {};
   379→    for (let i = 1; i < strikes.length; i++) {
   380→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   381→      gaps[gap] = (gaps[gap] || 0) + 1;
   382→    }
   383→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   384→  }
   385→
   386→  // Round ATM strike using the DYNAMIC strike step
   387→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   388→
   389→  // Get strikes around ATM
   390→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   391→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   392→    atmStrike - strikesAround * strikeStep + i * strikeStep
   393→  );
   394→
   395→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   396→
   397→  return {
   398→    instruments: filtered,
   399→    meta: { lotSize, strikeStep },
   400→  };
   401→}

   403→/**
   404→ * Build options flow data from quotes + OI changes
   405→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   406→ */
   407→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   408→  const spec = getInstrumentSpec(symbol);
   409→  if (!spec) return null;
   410→
   411→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   412→  if (optInstruments.length === 0) return null;
   413→
   414→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   415→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   416→
   417→  // Batch fetch quotes
   418→  const quotes = await getQuotes(iKeys);
   419→  if (Object.keys(quotes).length === 0) return null;
   420→
   421→  // Group by strike → build flow data
   422→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   423→
   424→  // Use DYNAMIC strike step and lot size from CSV
   425→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   426→
   427→  for (const inst of optInstruments) {
   428→    // Quotes are keyed by instrument token (string)
   429→    const key = String(inst.instrumentToken);
   430→    const q = quotes[key];
   431→    if (!q) continue;
   432→
   433→    if (!strikes.has(inst.strike)) {
   434→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   435→    }
   436→    const s = strikes.get(inst.strike)!;
   437→
   438→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   439→    const flow = q.volume * meta.lotSize * q.averagePrice;
   440→
   441→    if (inst.tradingSymbol.endsWith('CE')) {
   442→      s.callBuy = flow * 0.6;
   443→      s.callWrite = flow * 0.4;
   444→      s.callOI = q.oi;
   445→      s.callVol = q.volume;
   446→    } else if (inst.tradingSymbol.endsWith('PE')) {
   447→      s.putBuy = flow * 0.6;
   448→      s.putWrite = flow * 0.4;
   449→      s.putOI = q.oi;
   450→      s.putVol = q.volume;
   451→    }
   452→  }
   453→
   454→  // Build result
   455→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   456→    strike,
   457→    isATM: strike === atmStrike,
   458→    ...flow,
   459→    bullishFlow: flow.callBuy + flow.putWrite,
   460→    bearishFlow: flow.putBuy + flow.callWrite,
   461→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   462→  })).sort((a, b) => a.strike - b.strike);
   463→}

   535→// Lot sizes change with NSE revisions (e.g., Nifty went from 50→25→75→65 etc.)
   536→// Strike steps can also change.
   537→// We derive BOTH from Kite's instrument master CSV at runtime.

   540→  lotSize: number;
   541→  strikeStep: number;
   542→}

   544→/**
   545→ * Derive lot size and strike step from Kite instruments CSV
   546→ * - lotSize: from the option instrument's lot_size field
   547→ * - strikeStep: derived from the gap between adjacent strikes
   548→ */
   549→export async function getInstrumentMeta(
   550→  symbol: string,
   551→  spotPrice: number,
   552→): Promise<InstrumentMeta> {
   553→  const spec = getInstrumentSpec(symbol);
   554→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   555→
   556→  // Fetch all instruments for this symbol's exchange
   557→  const allInstruments = await getInstruments(spec.exchange);
   558→
   559→  // Filter to this symbol's options for current/near expiry
   560→  const symbolOpts = allInstruments.filter(i =>
   561→    i.segment === spec.segment &&
   562→    i.instrumentType === spec.instrumentType &&
   563→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   564→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   565→  );
   566→
   567→  if (symbolOpts.length === 0) {
   568→    return { lotSize: 1, strikeStep: 50 }; // fallback
   569→  }
   570→
   571→  // Get lot size from the first matching option (all have same lot size for same underlying)
   572→  const lotSize = symbolOpts[0].lotSize || 1;
   573→
   574→  // Derive strike step from unique sorted strikes near ATM
   575→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   576→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   577→
   578→  const expiryOpts = nearestExpiry
   579→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   580→    : symbolOpts;
   581→
   582→  // Get unique strikes, sorted
   583→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   584→
   585→  // Find strikes near ATM and compute step from adjacent gaps
   586→  let strikeStep = 50; // default fallback
   587→  if (strikes.length >= 2) {
   588→    // Use the most common gap between adjacent strikes
   589→    const gaps: Record<number, number> = {};
   590→    for (let i = 1; i < strikes.length; i++) {
   591→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   592→      gaps[gap] = (gaps[gap] || 0) + 1;
   593→    }
   594→    // Most frequent gap = the strike step
   595→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   596→  }
   597→
   598→  return { lotSize, strikeStep };
   599→}

   610→// ─── Black-Scholes Delta Calculator ───
   611→// Approximate delta for options when Kite doesn't provide Greeks.
   612→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   613→
   614→function normCDF(x: number): number {
   615→  // Rational approximation (max error ~7.5e-8)
   616→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   617→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   618→  const sign = x < 0 ? -1 : 1;
   619→  const absX = Math.abs(x);
   620→  const t = 1.0 / (1.0 + p * absX);
   621→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   622→  return 0.5 * (1.0 + sign * y);
   623→}
   624→
   625→/**
   626→ * Calculate Black-Scholes delta for an option
   627→ * @param isCall - true for CE, false for PE
   628→ * @param S - Spot price
   629→ * @param K - Strike price
   630→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   631→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   632→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   633→ */
   634→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   635→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   636→  // If no IV provided, estimate from moneyness
   637→  if (sigma <= 0) {
   638→    const moneyness = (S - K) / S;
   639→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   640→  }
   641→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   642→  const delta = normCDF(d1);
   643→  return isCall ? delta : delta - 1;
   644→}

   664→  isATM: boolean;

   668→  peOI: number;

   677→/**
   678→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   679→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   680→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   681→ */
   682→export async function getStrikeFlowSnapshot(
   683→  symbol: string,
   684→  spotPrice: number,
   685→): Promise<StrikeFlowSnapshot | null> {
   686→  const spec = getInstrumentSpec(symbol);
   687→  if (!spec) return null;
   688→
   689→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   690→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   691→  if (optInstruments.length === 0) return null;
   692→
   693→  // Batch fetch quotes using instrument tokens
   694→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   695→  const quotes = await getQuotes(iKeys);
   696→  if ('_error' in quotes) return null;
   697→
   698→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   699→
   700→  // Group by strike
   701→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   702→  for (const inst of optInstruments) {
   703→    if (!strikeMap.has(inst.strike)) {
   704→      strikeMap.set(inst.strike, { ce: null, pe: null });
   705→    }
   706→    const entry = strikeMap.get(inst.strike)!;
   707→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   708→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   709→  }
   710→
   711→  // Calculate time to expiry
   712→  const expiry = optInstruments[0]?.expiry || '';
   713→  const now = new Date();
   714→  const expiryDate = new Date(expiry);
   715→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   716→  const T = daysToExpiry / 365;
   717→
   718→  // Build strike data
   719→  const strikes: StrikeFlowData[] = [];
   720→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   721→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   722→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   723→
   724→    const ceLTP = ceQuote?.lastPrice || 0;
   725→    const peLTP = peQuote?.lastPrice || 0;
   726→    const ceOI = ceQuote?.oi || 0;
   727→    const peOI = peQuote?.oi || 0;
   728→    const ceVol = ceQuote?.volume || 0;
   729→    const peVol = peQuote?.volume || 0;
   730→
   731→    // Compute delta via Black-Scholes
   732→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   733→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   734→
   735→    strikes.push({
   736→      strike,
   737→      isATM: strike === atmStrike,
   738→      ceLTP,
   739→      peLTP,
   740→      ceOI,
   741→      peOI,
   742→      ceVol,
   743→      peVol,
   744→      ceDelta,
   745→      peDelta,
   746→      ceToken: ce?.instrumentToken || 0,
   747→      peToken: pe?.instrumentToken || 0,
   748→    });
   749→  }
   750→
   751→  strikes.sort((a, b) => a.strike - b.strike);
   752→
   753→  return {
   754→    timestamp: new Date().toISOString(),
   755→    symbol,
   756→    spotPrice,
   757→    atmStrike,
   758→    lotSize: meta.lotSize,
   759→    strikeStep: meta.strikeStep,
   760→    expiry,
   761→    strikes,
   762→  };
   763→}

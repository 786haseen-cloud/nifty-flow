    40→// 4-COLOR FLOW ENGINE — Computes real flow from snapshot diffs
    41→// ═══════════════════════════════════════════════════════════
    42→
    43→interface FourColorFlow {
    44→  callBuy: number;
    45→  callWrite: number;
    46→  putBuy: number;
    47→  putWrite: number;
    48→}
    49→
    50→function computeStrike4ColorFlow(
    51→  prev: SnapshotSymbol | undefined,
    52→  curr: SnapshotSymbol,
    53→  strike: number,
    54→): FourColorFlow {
    55→  const currStrike = curr.strikes.find(s => s.strike === strike);
    56→  const prevStrike = prev?.strikes.find(s => s.strike === strike);
    57→  if (!currStrike) return { callBuy: 0, callWrite: 0, putBuy: 0, putWrite: 0 };
    58→
    59→  const dceOI = currStrike.ceOI - (prevStrike?.ceOI ?? 0);
    60→  const dceP = currStrike.ceLTP - (prevStrike?.ceLTP ?? 0);
    61→  const dpeOI = currStrike.peOI - (prevStrike?.peOI ?? 0);
    62→  const dpeP = currStrike.peLTP - (prevStrike?.peLTP ?? 0);
    63→
    64→  const lotSize = curr.lotSize || curr.futLotSize || 25;
    65→  const ceDelta = Math.abs(currStrike.ceDelta) || 0.5;
    66→  const peDelta = Math.abs(currStrike.peDelta) || 0.5;
    67→
    68→  // OI decrease = unwinding → 0.3 factor
    69→  const ceFactor = dceOI > 0 ? 1.0 : 0.3;
    70→  const peFactor = dpeOI > 0 ? 1.0 : 0.3;
    71→
    72→  const result: FourColorFlow = { callBuy: 0, callWrite: 0, putBuy: 0, putWrite: 0 };
    73→
    74→  // CE flow: (dOI * delta * lotSize) / 1Cr
    75→  if (dceOI !== 0) {
    76→    const ceValue = Math.abs(dceOI) * ceDelta * lotSize;
    77→    if (dceP >= 0) {
    78→      result.callBuy = ceValue * ceFactor;  // CE Buy: OI up + price up
    79→    } else {
    80→      result.callWrite = ceValue * ceFactor; // CE Write: OI up + price down
    81→    }
    82→  }
    83→
    84→  // PE flow: (dOI * delta * lotSize) / 1Cr
    85→  if (dpeOI !== 0) {
    86→    const peValue = Math.abs(dpeOI) * peDelta * lotSize;
    87→    if (dpeP <= 0) {
    88→      result.putBuy = peValue * peFactor;   // PE Buy: OI up + price down
    89→    } else {
    90→      result.putWrite = peValue * peFactor;  // PE Write: OI up + price up
    91→    }
    92→  }
    93→
    94→  return result;
    95→}

    97→// Aggregate 4-color flow across all strikes of a symbol
    98→function aggregateSymbolFlow(
    99→  prev: SnapshotSymbol | undefined,
   100→  curr: SnapshotSymbol,
   101→): FourColorFlow {
   102→  let callBuy = 0, callWrite = 0, putBuy = 0, putWrite = 0;
   103→  for (const s of curr.strikes) {
   104→    const f = computeStrike4ColorFlow(prev, curr, s.strike);
   105→    callBuy += f.callBuy;
   106→    callWrite += f.callWrite;
   107→    putBuy += f.putBuy;
   108→    putWrite += f.putWrite;
   109→  }
   110→  return { callBuy, callWrite, putBuy, putWrite };
   111→}

   113→// Compute index options flow bar from snapshot diffs
   114→function computeOptionsFlowBar(
   115→  prevSnapshot: SnapshotSymbol[] | undefined,
   116→  currSnapshot: SnapshotSymbol[],
   117→  timestamp: string,
   118→): OptionsFlowBar {
   119→  const prevMap = new Map((prevSnapshot ?? []).map(s => [s.symbol, s]));
   120→
   121→  let idxCallBuy = 0, idxPutWrite = 0, idxPutBuy = 0, idxCallWrite = 0;
   122→  let stkCallBuy = 0, stkPutWrite = 0, stkPutBuy = 0, stkCallWrite = 0;
   123→
   124→  for (const sym of currSnapshot) {
   125→    const prev = prevMap.get(sym.symbol);
   126→    const flow = aggregateSymbolFlow(prev, sym);
   127→
   128→    if (sym.type === 'index') {
   129→      idxCallBuy += flow.callBuy;
   130→      idxPutWrite += flow.putWrite;
   131→      idxPutBuy += flow.putBuy;
   132→      idxCallWrite += flow.callWrite;
   133→    } else {
   134→      stkCallBuy += flow.callBuy;
   135→      stkPutWrite += flow.putWrite;
   136→      stkPutBuy += flow.putBuy;
   137→      stkCallWrite += flow.callWrite;
   138→    }
   139→  }
   140→
   141→  return {
   142→    timestamp,
   143→    indexFlows: [],
   144→    indexTotalCallBuy: idxCallBuy,
   145→    indexTotalPutWrite: idxPutWrite,
   146→    indexTotalPutBuy: idxPutBuy,
   147→    indexTotalCallWrite: idxCallWrite,
   148→    indexBullishFlow: idxCallBuy + idxPutWrite,
   149→    indexBearishFlow: idxPutBuy + idxCallWrite,
   150→    indexNetFlow: (idxCallBuy + idxPutWrite) - (idxPutBuy + idxCallWrite),
   151→    stockFlows: [],
   152→    stockTotalCallBuy: stkCallBuy,
   153→    stockTotalPutWrite: stkPutWrite,
   154→    stockTotalPutBuy: stkPutBuy,
   155→    stockTotalCallWrite: stkCallWrite,
   156→    stockBullishFlow: stkCallBuy + stkPutWrite,
   157→    stockBearishFlow: stkPutBuy + stkCallWrite,
   158→    stockNetFlow: (stkCallBuy + stkPutWrite) - (stkPutBuy + stkCallWrite),
   159→  };
   160→}

   162→// Compute futures flow bar from snapshot diffs
   163→function computeFuturesFlowBar(
   164→  prevSnapshot: SnapshotSymbol[] | undefined,
   165→  currSnapshot: SnapshotSymbol[],
   166→  timestamp: string,
   167→): FuturesFlowBar {
   168→  const prevMap = new Map((prevSnapshot ?? []).map(s => [s.symbol, s]));
   169→
   170→  let idxFutBuy = 0, idxFutSell = 0, idxFutOI = 0;
   171→  let stkFutBuy = 0, stkFutSell = 0, stkFutOI = 0;
   172→  let idxBasis = 0, idxCount = 0;
   173→  let stkBasis = 0, stkCount = 0;
   174→
   175→  const indexBreakdown: FuturesFlowBar['indexBreakdown'] = [];
   176→
   177→  for (const sym of currSnapshot) {
   178→    const prev = prevMap.get(sym.symbol);
   179→    const dFutOI = sym.futOI - (prev?.futOI ?? 0);
   180→    const basis = sym.futPrice - sym.spotPrice;
   181→    const flowValue = Math.abs(dFutOI) * sym.futPrice;
   182→
   183→    if (sym.type === 'index') {
   184→      if (dFutOI > 0) idxFutBuy += flowValue;
   185→      else idxFutSell += flowValue;
   186→      idxFutOI += dFutOI;
   187→      idxBasis += basis;
   188→      idxCount++;
   189→      indexBreakdown.push({
   190→        symbol: sym.symbol,
   191→        futBuy: dFutOI > 0 ? flowValue : 0,
   192→        futSell: dFutOI < 0 ? flowValue : 0,
   193→        futNet: dFutOI > 0 ? flowValue : -flowValue,
   194→        basis,
   195→        oiChg: dFutOI,
   196→      });
   197→    } else {
   198→      if (dFutOI > 0) stkFutBuy += flowValue;
   199→      else stkFutSell += flowValue;
   200→      stkFutOI += dFutOI;
   201→      stkBasis += basis;
   202→      stkCount++;
   203→    }
   204→  }
   205→
   206→  return {
   207→    timestamp,
   208→    indexFutBuy,
   209→    indexFutSell,
   210→    indexFutNet: idxFutBuy - idxFutSell,
   211→    indexFutOI,
   212→    indexFutBasis: idxCount > 0 ? idxBasis / idxCount : 0,
   213→    stockFutBuy,
   214→    stockFutSell,
   215→    stockFutNet: stkFutBuy - stkFutSell,
   216→    stockFutOI,
   217→    stockFutBasis: stkCount > 0 ? stkBasis / stkCount : 0,
   218→    indexBreakdown,
   219→  };
   220→}

   344→  const priceMax = visPrices.length > 0 ? Math.max(...visPrices) : 1;

  1010→// 4-color options bar (Call Buy + Put Write up, Put Buy + Call Write down)
  1011→function FourColorBar({
  1012→  buy1, buy2, sell1, sell2, maxAbs, height,
  1013→}: {
  1014→  buy1: number; buy2: number; sell1: number; sell2: number;
  1015→  maxAbs: number; height: number;
  1016→}) {
  1017→  const b1H = Math.max(0.5, (buy1 / maxAbs) * (height * 0.25));
  1018→  const b2H = Math.max(0.5, (buy2 / maxAbs) * (height * 0.25));
  1019→  const s1H = Math.max(0.5, (sell1 / maxAbs) * (height * 0.25));
  1020→  const s2H = Math.max(0.5, (sell2 / maxAbs) * (height * 0.25));
  1021→
  1022→  return (
  1023→    <div className="flex flex-col items-center justify-center" style={{ width: BAR_WIDTH, height: '100%' }}>
  1024→      {/* Bullish (up) */}
  1025→      <div className="w-full flex-1 flex flex-col items-end justify-end">
  1026→        <div className="w-full rounded-t-sm" style={{ height: b1H, background: C.callBuy }} />
  1027→        <div className="w-full" style={{ height: b2H, background: C.putWrite }} />
  1028→      </div>
  1029→      {/* Bearish (down) */}
  1030→      <div className="w-full flex-1 flex flex-col items-start">
  1031→        <div className="w-full" style={{ height: s1H, background: C.putBuy }} />
  1032→        <div className="w-full rounded-b-sm" style={{ height: s2H, background: C.callWrite }} />
  1033→      </div>
  1034→    </div>
  1035→  );

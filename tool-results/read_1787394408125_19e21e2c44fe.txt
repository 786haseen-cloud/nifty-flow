    82→    82→/**
    83→    83→ * Generate access_token from request_token (one-time, after login redirect)
    84→    84→ */
    85→    85→export async function generateSession(requestToken: string): Promise<{
    86→    86→  accessToken: string;
    87→    87→  refreshToken: string;
    88→    88→} | null> {
    89→    89→  const apiKey = process.env.KITE_API_KEY;
    90→    90→  const apiSecret = process.env.KITE_API_SECRET;
    91→    91→  if (!apiKey || !apiSecret) return null;
    92→    92→
    93→    93→  try {
    94→    94→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
    95→    95→    const { createHash } = await import('crypto');
    96→    96→    const checksum = createHash('sha256')
    97→    97→      .update(apiKey + requestToken + apiSecret)
    98→    98→      .digest('hex');
    99→    99→
   100→   100→    const res = await fetch(`${KITE_BASE}/session/token`, {
   101→   101→      method: 'POST',
   102→   102→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   103→   103→      body: new URLSearchParams({
   104→   104→        api_key: apiKey,
   105→   105→        request_token: requestToken,
   106→   106→        checksum,
   107→   107→      }).toString(),
   108→   108→    });
   109→   109→
   110→   110→    const data = await res.json();
   111→   111→    if (data.status === 'success' && data.data) {
   112→   112→      return {
   113→   113→        accessToken: data.data.access_token,
   114→   114→        refreshToken: data.data.refresh_token || '',
   115→   115→      };
   116→   116→    }
   117→   117→    console.error('[Kite] Session error:', data.message);
   118→   118→    return null;
   119→   119→  } catch (err) {
   120→   120→    console.error('[Kite] generateSession failed:', err);
   121→   121→    return null;
   122→   122→  }
   123→   123→}

   125→   125→// ─── Instruments ───
   126→   126→
   127→   127→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   128→   128→let instrumentsCache: KiteInstrument[] | null = null;
   129→   129→let instrumentsCacheTime = 0;
   130→   130→
   131→   131→/**
   132→   132→ * Fetch all instruments from Kite (cached for 1 hour)
   133→   133→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   134→   134→ * We filter by exchange after caching
   135→   135→ */
   136→   136→export async function getInstruments(exchange?: string): Promise<KiteInstrument[]> {
   137→   137→  // Return from cache if fresh
   138→   138→  if (instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   139→   139→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   140→   140→  }
   141→   141→
   142→   142→  try {
   143→   143→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   144→   144→    if (!res.ok) {
   145→   145→      const errText = await res.text();
   146→   146→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   147→   147→      return [];
   148→   148→    }
   149→   149→    const text = await res.text();
   150→   150→
   151→   151→    // Kite returns CSV format
   152→   152→    const lines = text.trim().split('\n');
   153→   153→    const instruments: KiteInstrument[] = [];
   154→   154→
   155→   155→    // Kite CSV column order (2025+ format):
   156→   156→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   157→   157→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   158→   158→    for (let i = 1; i < lines.length; i++) { // skip header
   159→   159→      const cols = lines[i].split(',');
   160→   160→      if (cols.length < 12) continue;
   161→   161→      instruments.push({
   162→   162→        instrumentToken: parseInt(cols[0]) || 0,
   163→   163→        exchangeToken: parseInt(cols[1]) || 0,
   164→   164→        tradingSymbol: cols[2],
   165→   165→        name: cols[3] || '',
   166→   166→        exchange: cols[11] || '',
   167→   167→        segment: cols[10] || '',
   168→   168→        instrumentType: cols[9] || '',
   169→   169→        strike: parseFloat(cols[6]) || 0,
   170→   170→        lotSize: parseInt(cols[8]) || 1,
   171→   171→        expiry: cols[5] || '',
   172→   172→      });
   173→   173→    }
   174→   174→
   175→   175→    instrumentsCache = instruments;
   176→   176→    instrumentsCacheTime = Date.now();
   177→   177→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   178→   178→  } catch (err) {
   179→   179→    const errMsg = err instanceof Error ? err.message : String(err);
   180→   180→    console.error('[Kite] instruments fetch failed:', errMsg);
   181→   181→    // Return empty but set error flag for callers to detect
   182→   182→    return [];
   183→   183→  }
   184→   184→}

   186→   186→// ─── Quotes (Real-time) ───
   187→   187→
   188→   188→/**
   189→   189→ * Get real-time quotes for instruments
   190→   190→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   191→   191→ */
   192→   192→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   193→   193→  if (!isKiteConfigured()) return {};
   194→   194→
   195→   195→  try {
   196→   196→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   197→   197→    // Tokens have no encoding issues on any platform.
   198→   198→    // Build a map from token → original instrument key for response mapping.
   199→   199→    const tokenToKey: Record<string, string> = {};
   200→   200→    const tokenList: string[] = [];
   201→   201→
   202→   202→    for (const inst of instruments) {
   203→   203→      // If it's a numeric token, use directly
   204→   204→      if (/^\d+$/.test(inst)) {
   205→   205→        tokenToKey[inst] = inst;
   206→   206→        tokenList.push(inst);
   207→   207→      } else {
   208→   208→        // Look up token from KITE_INDEX_INSTRUMENTS
   209→   209→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   210→   210→          ([, v]) => v.symbol === inst
   211→   211→        );
   212→   212→        if (found) {
   213→   213→          const tok = String(found[1].token);
   214→   214→          tokenToKey[tok] = inst; // map token back to symbol name
   215→   215→          tokenList.push(tok);
   216→   216→        } else {
   217→   217→          // Fallback: use the instrument string as-is with encodeURI
   218→   218→          tokenToKey[inst] = inst;
   219→   219→          tokenList.push(inst);
   220→   220→        }
   221→   221→      }
   222→   222→    }
   223→   223→
   224→   224→    const iList = tokenList.join(',');
   225→   225→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   226→   226→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   227→   227→    const params = tokenList.map(t => `i=${t}`).join('&');
   228→   228→    const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   229→   229→      headers: kiteHeaders(),
   230→   230→    });
   231→   231→
   232→   232→    // Log non-success for debugging
   233→   233→    if (!res.ok) {
   234→   234→      const errText = await res.text();
   235→   235→      console.error(`[Kite] quote API ${res.status}: ${errText}`);
   236→   236→      return { _error: `HTTP ${res.status}: ${errText}` } as any;
   237→   237→    }
   238→   238→
   239→   239→    const data = await res.json();
   240→   240→    if (data.status !== 'success' || !data.data) {
   241→   241→      console.error('[Kite] quote API error:', JSON.stringify(data));
   242→   242→      return { _error: data.message || 'Unknown error' } as any;
   243→   243→    }
   244→   244→
   245→   245→    const quotes: Record<string, KiteQuote> = {};
   246→   246→    for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   247→   247→      // Kite returns the key as the token string (e.g. "256265")
   248→   248→      // Map it back to the original instrument name (e.g. "NSE:NIFTY 50")
   249→   249→      const displayKey = tokenToKey[key] || key;
   250→   250→      quotes[displayKey] = {
   251→   251→        instrumentToken: q.instrument_token || 0,
   252→   252→        lastPrice: q.last_price || 0,
   253→   253→        open: q.ohlc?.open || 0,
   254→   254→        high: q.ohlc?.high || 0,
   255→   255→        low: q.ohlc?.low || 0,
   256→   256→        close: q.ohlc?.close || 0,
   257→   257→        volume: q.volume || 0,
   258→   258→        netChange: q.net_change || 0,
   259→   259→        averagePrice: q.average_price || 0,
   260→   260→        oi: q.oi || 0,
   261→   261→        oiDayHigh: q.oi_day_high || 0,
   262→   262→        oiDayLow: q.oi_day_low || 0,
   263→   263→      };
   264→   264→    }
   265→   265→    return quotes;
   266→   266→  } catch (err) {
   267→   267→    const errMsg = err instanceof Error ? err.message : String(err);
   268→   268→    console.error('[Kite] quote error:', errMsg);
   269→   269→    return { _error: `Exception: ${errMsg}` } as any;
   270→   270→  }
   271→   271→}

   275→   275→/**
   276→   276→ * Get historical candle data for price line chart
   277→   277→ * instrumentToken: numeric token from instruments list
   278→   278→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"

   280→   280→export async function getCandles(
   281→   281→  instrumentToken: number,
   282→   282→  interval: string = '15minute',
   283→   283→  days: number = 1,
   284→   284→): Promise<KiteHistoricalCandle[]> {

   286→   286→

   292→   292→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`;
   293→   293→
   294→   294→  try {
   295→   295→    const res = await fetch(
   296→   296→      `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${fmt(fromDate)}&to=${fmt(toDate)}&continuous=0`,
   297→   297→      { headers: kiteHeaders() }
   298→   298→    );
   299→   299→
   300→   300→    const data = await res.json();
   301→   301→    if (data.status !== 'success' || !data.data?.candles) return [];
   302→   302→
   303→   303→    return (data.data.candles as any[][]).map(c => ({
   304→   304→      timestamp: c[0],
   305→   305→      open: c[1],
   306→   306→      high: c[2],
   307→   307→      low: c[3],
   308→   308→      close: c[4],
   309→   309→      volume: c[5],
   310→   310→    }));
   311→   311→  } catch (err) {
   312→   312→    const errMsg = err instanceof Error ? err.message : String(err);
   313→   313→    console.error('[Kite] candles error:', errMsg);
   314→   314→    return [];
   315→   315→  }
   316→   316→}

   318→   318→// ─── Option Chain ───
   319→   319→
   320→   320→/**
   321→   321→ * Get option chain for an index or stock
   322→   322→ * Uses InstrumentSpec for correct exchange, segment
   323→   323→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   324→   324→ * Returns all CE + PE instruments for the current expiry near ATM
   325→   325→ */
   326→   326→export async function getOptionInstruments(
   327→   327→  symbol: string,
   328→   328→  spotPrice: number,
   329→   329→  strikesAroundOverride?: number,
   330→   330→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   331→   331→  const spec = getInstrumentSpec(symbol);
   332→   332→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   333→   333→
   334→   334→  // Fetch instruments from the correct exchange (NSE or BSE)
   335→   335→  const instruments = await getInstruments(spec.exchange);
   336→   336→
   337→   337→  // Filter options using spec
   338→   338→  const indexOptions = instruments.filter(i =>
   339→   339→    i.segment === spec.segment &&
   340→   340→    i.instrumentType === spec.instrumentType &&
   341→   341→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   342→   342→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   343→   343→  );
   344→   344→
   345→   345→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   346→   346→
   347→   347→  // Get unique expiries, sort by nearest
   348→   348→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   349→   349→
   350→   350→  // Use nearest expiry that's not expired
   351→   351→  const today = new Date();
   352→   352→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   353→   353→
   354→   354→  // Filter to nearest expiry
   355→   355→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   356→   356→
   357→   357→  // Get lot size from first option (all same for same underlying)
   358→   358→  const lotSize = expiryOptions[0]?.lotSize || 1;
   359→   359→
   360→   360→  // Derive strike step DYNAMICALLY from actual strike gaps
   361→   361→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   362→   362→  let strikeStep = 50;
   363→   363→  if (strikes.length >= 2) {
   364→   364→    const gaps: Record<number, number> = {};
   365→   365→    for (let i = 1; i < strikes.length; i++) {
   366→   366→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   367→   367→      gaps[gap] = (gaps[gap] || 0) + 1;
   368→   368→    }
   369→   369→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   370→   370→  }
   371→   371→
   372→   372→  // Round ATM strike using the DYNAMIC strike step
   373→   373→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   374→   374→
   375→   375→  // Get strikes around ATM
   376→   376→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   377→   377→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   378→   378→    atmStrike - strikesAround * strikeStep + i * strikeStep
   379→   379→  );
   380→   380→
   381→   381→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   382→   382→
   383→   383→  return {
   384→   384→    instruments: filtered,
   385→   385→    meta: { lotSize, strikeStep },
   386→   386→  };
   387→   387→}
   388→   388→
   389→   389→/**
   390→   390→ * Build options flow data from quotes + OI changes
   391→   391→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   392→   392→ */
   393→   393→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   394→   394→  const spec = getInstrumentSpec(symbol);
   395→   395→  if (!spec) return null;
   396→   396→
   397→   397→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   398→   398→  if (optInstruments.length === 0) return null;
   399→   399→
   400→   400→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   401→   401→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   402→   402→
   403→   403→  // Batch fetch quotes
   404→   404→  const quotes = await getQuotes(iKeys);
   405→   405→  if (Object.keys(quotes).length === 0) return null;
   406→   406→
   407→   407→  // Group by strike → build flow data
   408→   408→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   409→   409→
   410→   410→  // Use DYNAMIC strike step and lot size from CSV
   411→   411→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   412→   412→
   413→   413→  for (const inst of optInstruments) {
   414→   414→    // Quotes are keyed by instrument token (string)
   415→   415→    const key = String(inst.instrumentToken);
   416→   416→    const q = quotes[key];
   417→   417→    if (!q) continue;
   418→   418→
   419→   419→    if (!strikes.has(inst.strike)) {
   420→   420→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   421→   421→    }
   422→   422→    const s = strikes.get(inst.strike)!;
   423→   423→
   424→   424→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   425→   425→    const flow = q.volume * meta.lotSize * q.averagePrice;
   426→   426→
   427→   427→    if (inst.tradingSymbol.endsWith('CE')) {
   428→   428→      s.callBuy = flow * 0.6;
   429→   429→      s.callWrite = flow * 0.4;
   430→   430→      s.callOI = q.oi;
   431→   431→      s.callVol = q.volume;
   432→   432→    } else if (inst.tradingSymbol.endsWith('PE')) {
   433→   433→      s.putBuy = flow * 0.6;
   434→   434→      s.putWrite = flow * 0.4;
   435→   435→      s.putOI = q.oi;
   436→   436→      s.putVol = q.volume;
   437→   437→    }
   438→   438→  }
   439→   439→
   440→   440→  // Build result
   441→   441→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   442→   442→    strike,
   443→   443→    isATM: strike === atmStrike,
   444→   444→    ...flow,
   445→   445→    bullishFlow: flow.callBuy + flow.putWrite,
   446→   446→    bearishFlow: flow.putBuy + flow.callWrite,
   447→   447→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   448→   448→  })).sort((a, b) => a.strike - b.strike);
   449→   449→}

   451→   451→// ─── NSE Index Token Map ───
   452→   452→// We store BOTH the trading symbol (for display) and instrument token (for API calls).
   453→   453→// Instrument tokens are numeric — no encoding issues on any platform.
   454→   454→
   455→   455→export const KITE_INDEX_INSTRUMENTS: Record<string, { symbol: string; token: number }> = {
   456→   456→  NIFTY:      { symbol: 'NSE:NIFTY 50',           token: 256265 },

   530→   530→/**
   531→   531→ * Derive lot size and strike step from Kite instruments CSV
   532→   532→ * - lotSize: from the option instrument's lot_size field
   533→   533→ * - strikeStep: derived from the gap between adjacent strikes
   534→   534→ */
   535→   535→export async function getInstrumentMeta(
   536→   536→  symbol: string,
   537→   537→  spotPrice: number,
   538→   538→): Promise<InstrumentMeta> {
   539→   539→  const spec = getInstrumentSpec(symbol);
   540→   540→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   541→   541→
   542→   542→  // Fetch all instruments for this symbol's exchange
   543→   543→  const allInstruments = await getInstruments(spec.exchange);
   544→   544→
   545→   545→  // Filter to this symbol's options for current/near expiry
   546→   546→  const symbolOpts = allInstruments.filter(i =>
   547→   547→    i.segment === spec.segment &&
   548→   548→    i.instrumentType === spec.instrumentType &&
   549→   549→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   550→   550→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   551→   551→  );
   552→   552→
   553→   553→  if (symbolOpts.length === 0) {
   554→   554→    return { lotSize: 1, strikeStep: 50 }; // fallback
   555→   555→  }
   556→   556→
   557→   557→  // Get lot size from the first matching option (all have same lot size for same underlying)
   558→   558→  const lotSize = symbolOpts[0].lotSize || 1;
   559→   559→
   560→   560→  // Derive strike step from unique sorted strikes near ATM
   561→   561→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   562→   562→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   563→   563→
   564→   564→  const expiryOpts = nearestExpiry
   565→   565→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   566→   566→    : symbolOpts;
   567→   567→
   568→   568→  // Get unique strikes, sorted
   569→   569→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   570→   570→
   571→   571→  // Find strikes near ATM and compute step from adjacent gaps
   572→   572→  let strikeStep = 50; // default fallback
   573→   573→  if (strikes.length >= 2) {
   574→   574→    // Use the most common gap between adjacent strikes
   575→   575→    const gaps: Record<number, number> = {};
   576→   576→    for (let i = 1; i < strikes.length; i++) {
   577→   577→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   578→   578→      gaps[gap] = (gaps[gap] || 0) + 1;
   579→   579→    }
   580→   580→    // Most frequent gap = the strike step
   581→   581→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   582→   582→  }
   583→   583→
   584→   584→  return { lotSize, strikeStep };
   585→   585→}

   596→   596→// ─── Black-Scholes Delta Calculator ───
   597→   597→// Approximate delta for options when Kite doesn't provide Greeks.
   598→   598→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   599→   599→
   600→   600→function normCDF(x: number): number {
   601→   601→  // Rational approximation (max error ~7.5e-8)
   602→   602→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   603→   603→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   604→   604→  const sign = x < 0 ? -1 : 1;
   605→   605→  const absX = Math.abs(x);
   606→   606→  const t = 1.0 / (1.0 + p * absX);
   607→   607→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   608→   608→  return 0.5 * (1.0 + sign * y);
   609→   609→}
   610→   610→
   611→   611→/**
   612→   612→ * Calculate Black-Scholes delta for an option
   613→   613→ * @param isCall - true for CE, false for PE
   614→   614→ * @param S - Spot price
   615→   615→ * @param K - Strike price
   616→   616→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   617→   617→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   618→   618→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   619→   619→ */
   620→   620→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   621→   621→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   622→   622→  // If no IV provided, estimate from moneyness
   623→   623→  if (sigma <= 0) {
   624→   624→    const moneyness = (S - K) / S;
   625→   625→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   626→   626→  }
   627→   627→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   628→   628→  const delta = normCDF(d1);
   629→   629→  return isCall ? delta : delta - 1;
   630→   630→}

   653→   653→  ceOI: number;

   663→   663→/**
   664→   664→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   665→   665→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   666→   666→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   667→   667→ */
   668→   668→export async function getStrikeFlowSnapshot(
   669→   669→  symbol: string,
   670→   670→  spotPrice: number,
   671→   671→): Promise<StrikeFlowSnapshot | null> {
   672→   672→  const spec = getInstrumentSpec(symbol);
   673→   673→  if (!spec) return null;
   674→   674→
   675→   675→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   676→   676→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   677→   677→  if (optInstruments.length === 0) return null;
   678→   678→
   679→   679→  // Batch fetch quotes using instrument tokens
   680→   680→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   681→   681→  const quotes = await getQuotes(iKeys);
   682→   682→  if ('_error' in quotes) return null;
   683→   683→
   684→   684→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   685→   685→
   686→   686→  // Group by strike
   687→   687→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   688→   688→  for (const inst of optInstruments) {
   689→   689→    if (!strikeMap.has(inst.strike)) {
   690→   690→      strikeMap.set(inst.strike, { ce: null, pe: null });
   691→   691→    }
   692→   692→    const entry = strikeMap.get(inst.strike)!;
   693→   693→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   694→   694→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   695→   695→  }
   696→   696→
   697→   697→  // Calculate time to expiry
   698→   698→  const expiry = optInstruments[0]?.expiry || '';
   699→   699→  const now = new Date();
   700→   700→  const expiryDate = new Date(expiry);
   701→   701→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   702→   702→  const T = daysToExpiry / 365;
   703→   703→
   704→   704→  // Build strike data
   705→   705→  const strikes: StrikeFlowData[] = [];
   706→   706→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   707→   707→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   708→   708→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   709→   709→
   710→   710→    const ceLTP = ceQuote?.lastPrice || 0;
   711→   711→    const peLTP = peQuote?.lastPrice || 0;
   712→   712→    const ceOI = ceQuote?.oi || 0;
   713→   713→    const peOI = peQuote?.oi || 0;
   714→   714→    const ceVol = ceQuote?.volume || 0;
   715→   715→    const peVol = peQuote?.volume || 0;
   716→   716→
   717→   717→    // Compute delta via Black-Scholes
   718→   718→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   719→   719→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   720→   720→
   721→   721→    strikes.push({
   722→   722→      strike,
   723→   723→      isATM: strike === atmStrike,
   724→   724→      ceLTP,
   725→   725→      peLTP,
   726→   726→      ceOI,
   727→   727→      peOI,
   728→   728→      ceVol,
   729→   729→      peVol,
   730→   730→      ceDelta,
   731→   731→      peDelta,
   732→   732→      ceToken: ce?.instrumentToken || 0,
   733→   733→      peToken: pe?.instrumentToken || 0,
   734→   734→    });
   735→   735→  }
   736→   736→
   737→   737→  strikes.sort((a, b) => a.strike - b.strike);
   738→   738→
   739→   739→  return {
   740→   740→    timestamp: new Date().toISOString(),
   741→   741→    symbol,
   742→   742→    spotPrice,
   743→   743→    atmStrike,
   744→   744→    lotSize: meta.lotSize,
   745→   745→    strikeStep: meta.strikeStep,
   746→   746→    expiry,
   747→   747→    strikes,
   748→   748→  };
   749→   749→}

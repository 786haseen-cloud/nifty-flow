     3→/**
     4→ * Trend Analysis Tab — Price + Cash + Options Flow visualization.
     5→ *
     6→ * ARCHITECTURE
     7→ * ------------
     8→ * This component is a PURE VIEW — it contains no state and no polling logic.
     9→ * All state is held in the global `useTrendStore` Zustand store, which:
    10→ *   - Persists to localStorage (survives page reload + browser restart)
    11→ *   - Survives tab switches (because it lives outside the component tree)
    12→ *   - Is updated by a singleton poller started once at app boot
    13→ *
    14→ * This fixes the original bug where switching away from the Trends tab
    15→ * would unmount the component and destroy all accumulated chart data.
    16→ *
    17→ * To add new visualizations: read additional fields from the store, or
    18→ * extend the store's `pollOnce()` action to fetch new data.
    19→ */

    37→// ─── Custom Tooltips ───
    38→
    39→function NiftyTooltip({ active, payload, label }: any) {
    40→  if (!active || !payload?.length) return null;
    41→  const d = payload[0]?.payload as NiftyCandle;
    42→  return (
    43→    <div className="bg-card border border-border rounded-lg p-2 shadow-xl text-xs">
    44→      <div className="font-mono text-muted-foreground mb-1">{label}</div>
    45→      <div className="text-emerald-400 font-semibold">Close: {d?.close?.toLocaleString('en-IN')}</div>
    46→      {d?.high ? (
    47→        <div className="text-muted-foreground">
    48→          H: {d.high.toLocaleString('en-IN')} L: {d.low.toLocaleString('en-IN')}
    49→        </div>
    50→      ) : null}
    51→      {d?.volume ? (
    52→        <div className="text-muted-foreground">Vol: {(d.volume / 1000000).toFixed(1)}M</div>
    53→      ) : null}
    54→    </div>
    55→  );
    56→}
    57→
    58→function CashFlowTrendTooltip({ active, payload, label }: any) {
    59→  if (!active || !payload?.length) return null;
    60→  const d = payload[0]?.payload;
    61→  return (
    62→    <div className="bg-card border border-border rounded-lg p-2 shadow-xl text-xs">
    63→      <div className="font-mono text-muted-foreground mb-1">{label}</div>
    64→      <div className="text-emerald-400">NSE Cum: {d?.nse?.toFixed(1)} Cr</div>
    65→      <div className="text-sky-400">BSE Cum: {d?.bse?.toFixed(1)} Cr</div>
    66→      <div className={d?.net >= 0 ? 'text-amber-400 font-semibold' : 'text-red-400 font-semibold'}>
    67→        Net Cum: {d?.net?.toFixed(1)} Cr
    68→      </div>
    69→      <div className="text-muted-foreground mt-1">
    70→        This 15s:{' '}
    71→        <span className={d?.interval >= 0 ? 'text-emerald-300' : 'text-red-300'}>
    72→          {d?.interval >= 0 ? '+' : ''}{d?.interval?.toFixed(1)} Cr
    73→        </span>
    74→      </div>
    75→    </div>
    76→  );
    77→}
    78→
    79→function FlowTooltip({ active, payload, label }: any) {
    80→  if (!active || !payload?.length) return null;
    81→  return (
    82→    <div className="bg-card border border-border rounded-lg p-2 shadow-xl text-xs">
    83→      <div className="font-mono text-muted-foreground mb-1">{label}</div>
    84→      {payload.map((p: any) => (
    85→        <div key={p.dataKey} style={{ color: p.color }} className="flex items-center gap-1">
    86→          <span className="inline-block w-2 h-2 rounded-full" style={{ backgroundColor: p.color }} />
    87→          {p.dataKey}: {typeof p.value === 'number' ? p.value.toFixed(1) : p.value} Cr
    88→        </div>
    89→      ))}
    90→    </div>
    91→  );
    92→}
    93→
    94→function CashTooltip({ active, payload, label }: any) {
    95→  if (!active || !payload?.length) return null;
    96→  const d = payload[0]?.payload as StockCashFlow;
    97→  if (!d) return null;
    98→  return (
    99→    <div className="bg-card border border-border rounded-lg p-2 shadow-xl text-xs">
   100→      <div className="font-semibold text-foreground mb-1">{d.symbol}</div>
   101→      <div className="text-muted-foreground">{d.name}</div>
   102→      <div className="mt-1 text-emerald-400">NSE: {(d.nseCashFlow / 10000000).toFixed(1)} Cr</div>
   103→      <div className="text-sky-400">BSE: {(d.bseCashFlow / 10000000).toFixed(1)} Cr</div>
   104→      <div className={d.combinedFlow >= 0 ? 'text-emerald-300 font-semibold mt-1' : 'text-red-400 font-semibold mt-1'}>
   105→        Net: {(d.combinedFlow / 10000000).toFixed(1)} Cr
   106→      </div>
   107→      <div className="text-muted-foreground">Wt: {d.niftyWeight}%</div>
   108→    </div>
   109→  );
   110→}

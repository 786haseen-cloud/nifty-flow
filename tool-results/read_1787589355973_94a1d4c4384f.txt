   102→   102→// ─── 4-Color Flow Engine ───
   103→   103→// Computes net options flow from consecutive snapshots
   104→   104→// Delta-weighted: (deltaOI * delta * lotSize) / 1Cr
   105→   105→// OI decrease: multiply by 0.3 (short covering)
   106→   106→
   107→   107→function computeSymbolFlow(
   108→   108→  prev: StrikeData[],
   109→   109→  curr: StrikeData[],
   110→   110→  lotSize: number
   111→   111→): { bullish: number; bearish: number; net: number } {
   112→   112→  let bullish = 0;
   113→   113→  let bearish = 0;
   114→   114→
   115→   115→  const CR = 10000000; // 1 Crore
   116→   116→
   117→   117→  for (const currStrike of curr) {
   118→   118→    const prevStrike = prev.find((s) => s.strike === currStrike.strike);
   119→   119→    if (!prevStrike) continue;
   120→   120→
   121→   121→    const ceDeltaOI = currStrike.ceOI - prevStrike.ceOI;
   122→   122→    const peDeltaOI = currStrike.peOI - prevStrike.peOI;
   123→   123→    const ceDeltaPrice = currStrike.ceLTP - prevStrike.ceLTP;
   124→   124→    const peDeltaPrice = currStrike.peLTP - prevStrike.peLTP;
   125→   125→
   126→   126→    // CE Flow
   127→   127→    if (ceDeltaOI > 0) {
   128→   128→      const val = (Math.abs(ceDeltaOI) * currStrike.ceDelta * lotSize) / CR;
   129→   129→      if (ceDeltaPrice > 0) {
   130→   130→        bullish += val; // CE Buy (bullish)
   131→   131→      } else {
   132→   132→        bearish += val; // CE Write (bearish)
   133→   133→      }
   134→   134→    } else if (ceDeltaOI < 0) {
   135→   135→      // Short covering: 0.3 factor
   136→   136→      const val = (Math.abs(ceDeltaOI) * 0.3 * currStrike.ceDelta * lotSize) / CR;
   137→   137→      if (ceDeltaPrice > 0) {
   138→   138→        bullish += val; // CE Buy (short cover)
   139→   139→      } else {
   140→   140→        bearish += val; // CE Write (short cover)
   141→   141→      }
   142→   142→    }
   143→   143→
   144→   144→    // PE Flow
   145→   145→    if (peDeltaOI > 0) {
   146→   146→      const val = (Math.abs(peDeltaOI) * currStrike.peDelta * lotSize) / CR;
   147→   147→      if (peDeltaPrice > 0) {
   148→   148→        bullish += val; // PE Write (bullish)
   149→   149→      } else {
   150→   150→        bearish += val; // PE Buy (bearish)
   151→   151→      }
   152→   152→    } else if (peDeltaOI < 0) {
   153→   153→      const val = (Math.abs(peDeltaOI) * 0.3 * currStrike.peDelta * lotSize) / CR;
   154→   154→      if (peDeltaPrice > 0) {
   155→   155→        bullish += val; // PE Write (short cover)
   156→   156→      } else {
   157→   157→        bearish += val; // PE Buy (short cover)
   158→   158→      }
   159→   159→    }
   160→   160→  }
   161→   161→
   162→   162→  return { bullish, bearish, net: bullish - bearish };
   163→   163→}

   233→   233→export default function TrendAnalysisTab() {
   234→   234→  const isLive = hasKiteCreds();
   235→   235→
   236→   236→  // Trends API data (candles + cash flow)
   237→   237→  const [niftyCandles, setNiftyCandles] = useState<NiftyCandle[]>([]);
   238→   238→  const [stockCashFlow, setStockCashFlow] = useState<StockCashFlow[]>([]);
   239→   239→  const [trendMode, setTrendMode] = useState<string>('demo');
   240→   240→
   241→   241→  // Highest-bet snapshots for options flow computation
   242→   242→  const prevSnapshot = useRef<Map<string, StrikeData[]>>(new Map());
   243→   243→  const [flowTrend, setFlowTrend] = useState<FlowTrendPoint[]>([]);
   244→   244→  const cumulativeFlow = useRef<Record<string, number>>({
   245→   245→    NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0, stockAggregate: 0,
   246→   246→  });
   247→   247→
   248→   248→  // Cash flow trend accumulation
   249→   249→  const [cashFlowTrend, setCashFlowTrend] = useState<CashFlowTrendPoint[]>([]);
   250→   250→  const cumulativeCash = useRef({ nse: 0, bse: 0, net: 0, weighted: 0 });
   251→   251→  const prevStockCashFlow = useRef<StockCashFlow[]>([]);
   252→   252→
   253→   253→  // Current flow values (for header cards)
   254→   254→  const [currentIdxFlows, setCurrentIdxFlows] = useState<Record<string, number>>({
   255→   255→    NIFTY: 0, BANKNIFTY: 0, FINNIFTY: 0, SENSEX: 0,
   256→   256→  });
   257→   257→  const [currentStockFlow, setCurrentStockFlow] = useState(0);
   258→   258→  const [currentIntervalCashFlow, setCurrentIntervalCashFlow] = useState(0);
   259→   259→
   260→   260→  // ─── Fetch trends data (candles + cash) ───
   261→   261→
   262→   262→  const fetchTrends = useCallback(async () => {
   263→   263→    try {
   264→   264→      let url = '/api/kite/trends';
   265→   265→      url = withCreds(url);
   266→   266→      const res = await fetch(url);
   267→   267→      const data = await res.json();
   268→   268→
   269→   269→      setTrendMode(data.mode || 'demo');
   270→   270→      if (data.niftyCandles) setNiftyCandles(data.niftyCandles);
   271→   271→
   272→   272→      // Build cash flow trend from consecutive snapshots
   273→   273→      if (data.stockCashFlow && data.stockCashFlow.length > 0) {
   274→   274→        setStockCashFlow(data.stockCashFlow);
   275→   275→
   276→   276→        const time = new Date().toLocaleTimeString('en-IN', {
   277→   277→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   278→   278→        });
   279→   279→
   280→   280→        const nseTotal = data.stockCashFlow.reduce((s, v) => s + v.nseCashFlow, 0);
   281→   281→        const bseTotal = data.stockCashFlow.reduce((s, v) => s + v.bseCashFlow, 0);
   282→   282→        const netTotal = nseTotal + bseTotal;
   283→   283→        const weightedTotal = data.stockCashFlow.reduce((s, v) => s + v.weightedFlow, 0);
   284→   284→
   285→   285→        // Compute interval deltas (this poll minus previous poll)
   286→   286→        // cashFlow = (ltp - open) * volume is CUMULATIVE since market open,
   287→   287→        // so we must diff consecutive polls to get the 15s delta.
   288→   288→        let nseDelta = 0;
   289→   289→        let bseDelta = 0;
   290→   290→        let weightedDelta = 0;
   291→   291→        if (prevStockCashFlow.current.length > 0) {
   292→   292→          const prevNse = prevStockCashFlow.current.reduce((s, v) => s + v.nseCashFlow, 0);
   293→   293→          const prevBse = prevStockCashFlow.current.reduce((s, v) => s + v.bseCashFlow, 0);
   294→   294→          const prevWeighted = prevStockCashFlow.current.reduce((s, v) => s + v.weightedFlow, 0);
   295→   295→          nseDelta = nseTotal - prevNse;
   296→   296→          bseDelta = bseTotal - prevBse;
   297→   297→          weightedDelta = weightedTotal - prevWeighted;
   298→   298→        }
   299→   299→        const intervalDelta = nseDelta + bseDelta;
   300→   300→
   301→   301→        prevStockCashFlow.current = data.stockCashFlow;
   302→   302→        setCurrentIntervalCashFlow(intervalDelta);
   303→   303→
   304→   304→        // Accumulate only the deltas
   305→   305→        cumulativeCash.current.nse += nseDelta;
   306→   306→        cumulativeCash.current.bse += bseDelta;
   307→   307→        cumulativeCash.current.net += intervalDelta;
   308→   308→        cumulativeCash.current.weighted += weightedDelta;
   309→   309→
   310→   310→        const CR = 10000000;
   311→   311→        const point: CashFlowTrendPoint = {
   312→   312→          time,
   313→   313→          nse: Math.round((cumulativeCash.current.nse / CR) * 10) / 10,
   314→   314→          bse: Math.round((cumulativeCash.current.bse / CR) * 10) / 10,
   315→   315→          net: Math.round((cumulativeCash.current.net / CR) * 10) / 10,
   316→   316→          weighted: Math.round((cumulativeCash.current.weighted / CR) * 10) / 10,
   317→   317→          interval: Math.round((intervalDelta / CR) * 10) / 10,
   318→   318→        };
   319→   319→
   320→   320→        setCashFlowTrend((prev) => {
   321→   321→          const next = [...prev, point];
   322→   322→          return next.length > 200 ? next.slice(-200) : next;
   323→   323→        });
   324→   324→      }
   325→   325→    } catch (err) {
   326→   326→      console.error('[TrendsTab] fetchTrends error:', err);
   327→   327→    }
   328→   328→  }, []);
   329→   329→
   330→   330→  // ─── Fetch highest-bet for options flow ───
   331→   331→
   332→   332→  const fetchOptionsFlow = useCallback(async () => {
   333→   333→    try {
   334→   334→      let url = '/api/kite/highest-bet';
   335→   335→      url = withCreds(url);
   336→   336→      const res = await fetch(url);
   337→   337→      const data: HighestBetResponse = await res.json();
   338→   338→
   339→   339→      if (!data.symbols || data.symbols.length === 0) return;
   340→   340→
   341→   341→      const time = new Date().toLocaleTimeString('en-IN', {
   342→   342→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   343→   343→      });
   344→   344→
   345→   345→      // Compute per-symbol flow from consecutive snapshots
   346→   346→      const intervalFlows: Record<string, number> = {};
   347→   347→      const indexSymbols = ['NIFTY', 'BANKNIFTY', 'FINNIFTY', 'SENSEX'];
   348→   348→      let stockAgg = 0;
   349→   349→      const newCurrentIdx: Record<string, number> = {};
   350→   350→
   351→   351→      for (const sym of data.symbols) {
   352→   352→        if (sym.strikes.length === 0) continue;
   353→   353→
   354→   354→        const prevStrikes = prevSnapshot.current.get(sym.symbol);
   355→   355→        if (!prevStrikes || prevStrikes.length === 0) {
   356→   356→          // First snapshot — store and skip
   357→   357→          prevSnapshot.current.set(sym.symbol, sym.strikes);
   358→   358→          continue;
   359→   359→        }
   360→   360→
   361→   361→        const flow = computeSymbolFlow(prevStrikes, sym.strikes, sym.lotSize);
   362→   362→        intervalFlows[sym.symbol] = flow.net;
   363→   363→
   364→   364→        if (sym.type === 'index' && indexSymbols.includes(sym.symbol)) {
   365→   365→          cumulativeFlow.current[sym.symbol] += flow.net;
   366→   366→          newCurrentIdx[sym.symbol] = flow.net;
   367→   367→        } else if (sym.type === 'stock') {
   368→   368→          stockAgg += flow.net;
   369→   369→        }
   370→   370→
   371→   371→        // Update previous snapshot
   372→   372→        prevSnapshot.current.set(sym.symbol, sym.strikes);
   373→   373→      }
   374→   374→
   375→   375→      cumulativeFlow.current.stockAggregate += stockAgg;
   376→   376→      setCurrentStockFlow(stockAgg);
   377→   377→
   378→   378→      // Update current index flows
   379→   379→      for (const idx of indexSymbols) {
   380→   380→        newCurrentIdx[idx] = newCurrentIdx[idx] || 0;
   381→   381→      }
   382→   382→      setCurrentIdxFlows(newCurrentIdx);
   383→   383→
   384→   384→      // Append to trend (keep last 200 points)
   385→   385→      const point: FlowTrendPoint = {
   386→   386→        time,
   387→   387→        NIFTY: Math.round(cumulativeFlow.current.NIFTY * 10) / 10,
   388→   388→        BANKNIFTY: Math.round(cumulativeFlow.current.BANKNIFTY * 10) / 10,
   389→   389→        FINNIFTY: Math.round(cumulativeFlow.current.FINNIFTY * 10) / 10,
   390→   390→        SENSEX: Math.round(cumulativeFlow.current.SENSEX * 10) / 10,
   391→   391→        stockAggregate: Math.round(cumulativeFlow.current.stockAggregate * 10) / 10,
   392→   392→      };
   393→   393→
   394→   394→      setFlowTrend((prev) => {
   395→   395→        const next = [...prev, point];
   396→   396→        return next.length > 200 ? next.slice(-200) : next;
   397→   397→      });
   398→   398→    } catch (err) {
   399→   399→      console.error('[TrendsTab] fetchOptionsFlow error:', err);
   400→   400→    }
   401→   401→  }, []);
   402→   402→
   403→   403→  // ─── Polling ───
   404→   404→
   405→   405→  useEffect(() => {
   406→   406→    fetchTrends();
   407→   407→    fetchOptionsFlow();
   408→   408→
   409→   409→    const trendsInterval = setInterval(fetchTrends, 15000);
   410→   410→    const flowInterval = setInterval(fetchOptionsFlow, 15000);
   411→   411→
   412→   412→    return () => {
   413→   413→      clearInterval(trendsInterval);
   414→   414→      clearInterval(flowInterval);
   415→   415→    };
   416→   416→  }, [fetchTrends, fetchOptionsFlow]);
   417→   417→
   418→   418→  // ─── Derived values ───
   419→   419→
   420→   420→  const niftyCurrentPrice = niftyCandles.length > 0 ? niftyCandles[niftyCandles.length - 1].close : 0;
   421→   421→  const niftyOpenPrice = niftyCandles.length > 0 ? niftyCandles[0].close : 0;
   422→   422→  const niftyChange = niftyOpenPrice > 0 ? niftyCurrentPrice - niftyOpenPrice : 0;
   423→   423→  const niftyChangePct = niftyOpenPrice > 0 ? (niftyChange / niftyOpenPrice) * 100 : 0;

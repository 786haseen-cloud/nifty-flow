    14→/**
    15→ * 4-Color OI classification (matches Strike Flow Map convention):
    16→ *
    17→ *  CE  Buy   = ΔOI>0 & ΔLTP≥0  → bright green  #00B050
    18→ *  CE  Write = ΔOI>0 & ΔLTP<0  → light red     #FFCCCC (dark red border)
    19→ *  CE  Close = ΔOI<0           → grey          #6b7280
    20→ *  PE  Write = ΔOI>0 & ΔLTP≥0  → light green   #C6EFCE (dark green border)
    21→ *  PE  Buy   = ΔOI>0 & ΔLTP<0  → bright red    #BD2130
    22→ *  PE  Close = ΔOI<0           → grey          #6b7280
    23→ */
    24→type OIColor = {
    25→  bg: string;        // inline background color
    26→  border?: string;   // inline border color (for Write classes)
    27→  text: string;      // tailwind text class for the OI label
    28→  label?: string;    // short label: "Buy", "Write", "Close"
    29→};
    30→
    31→function getOIColor(oiChange: number, ltpChange: number, isPE: boolean): OIColor {
    32→  // No previous data — neutral
    33→  if (oiChange === 0 && ltpChange === 0) {
    34→    return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    35→  }
    36→  // Unwinding / short covering (ΔOI < 0)
    37→  if (oiChange < 0) {
    38→    return { bg: '#6b7280', text: 'text-zinc-400', label: 'Close' };
    39→  }
    40→  // oiChange >= 0 — new positions opened (or no change)
    41→  if (isPE) {
    42→    // PE: Write = bullish (price up/stable + OI up), Buy = bearish (price down + OI up)
    43→    if (ltpChange >= 0) return { bg: '#C6EFCE', border: '#16a34a', text: 'text-emerald-700', label: 'Write' };
    44→    if (ltpChange < 0)  return { bg: '#BD2130', text: 'text-red-100', label: 'Buy' };
    45→  } else {
    46→    // CE: Buy = bullish (price up/stable + OI up), Write = bearish (price down + OI up)
    47→    if (ltpChange >= 0) return { bg: '#00B050', text: 'text-green-100', label: 'Buy' };
    48→    if (ltpChange < 0)  return { bg: '#FFCCCC', border: '#dc2626', text: 'text-red-700', label: 'Write' };
    49→  }
    50→  // Fallback (shouldn't reach here)
    51→  return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    52→}

   138→function formatLakhs(n: number): string {
   139→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
   140→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
   141→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
   142→  return n.toLocaleString('en-IN');
   143→}

   149→// Compute Max Pain: strike where option buyers lose the most
   150→function computeMaxPain(strikes: StrikeFlowData[]): number {
   151→  if (strikes.length === 0) return 0;
   152→  let minLoss = Infinity;
   153→  let maxPainStrike = strikes[0].strike;
   154→
   155→  for (const k of strikes) {
   156→    let totalLoss = 0;
   157→    for (const s of strikes) {
   158→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
   159→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
   160→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
   161→      // But we only have OI, not premium. Standard Max Pain uses:
   162→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
   163→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
   164→      // Simplified: use intrinsic value approach
   165→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
   166→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
   167→    }
   168→    if (totalLoss < minLoss) {
   169→      minLoss = totalLoss;
   170→      maxPainStrike = k.strike;
   171→    }
   172→  }
   173→  return maxPainStrike;
   174→}

   176→// Generate demo data for 11 strikes
   177→function generateDemoData(symbol: string, prev?: StrikeFlowSnapshot | null): StrikeFlowSnapshot {
   178→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   179→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   180→  const atmStrike = Math.round(spot / step) * step;
   181→
   182→  // If we have a previous snapshot, apply small perturbations to it.
   183→  // This produces realistic per-strike ΔOI and ΔLTP values that
   184→  // feed into the 4-color OI coding (Buy/Write/Close).
   185→  if (prev && prev.strikes.length === 11) {
   186→    const spotJitter = Math.round((Math.random() - 0.5) * step * 0.3);
   187→    const newSpot = prev.spotPrice + spotJitter;
   188→    const newAtm = Math.round(newSpot / step) * step;
   189→
   190→    const strikes: StrikeFlowData[] = prev.strikes.map((s, idx) => {
   191→      // OI change: random walk — some strikes gain, some lose
   192→      const oiDelta = Math.round((Math.random() - 0.45) * s.ceOI * 0.04);
   193→      const peOiDelta = Math.round((Math.random() - 0.45) * s.peOI * 0.04);
   194→
   195→      // LTP change based on spot movement + moneyness
   196→      const ceMoneyness = (newSpot - s.strike) / newSpot;
   197→      const peMoneyness = (s.strike - newSpot) / newSpot;
   198→      const iv = 0.12 + Math.abs(ceMoneyness) * 0.3;
   199→      const ceLTPDelta = spotJitter > 0
   200→        ? iv * Math.abs(spotJitter) * (0.3 + Math.random() * 0.3)
   201→        : -iv * Math.abs(spotJitter) * (0.2 + Math.random() * 0.3);
   202→      const peLTPDelta = -ceLTPDelta * 0.8;
   203→
   204→      const ceITM = newSpot > s.strike;
   205→      const peITM = newSpot < s.strike;
   206→
   207→      return {
   208→        ...s,
   209→        ceLTP: Math.max(0.5, s.ceLTP + ceLTPDelta),
   210→        peLTP: Math.max(0.5, s.peLTP + peLTPDelta),
   211→        ceOI: Math.max(1000, s.ceOI + oiDelta),
   212→        peOI: Math.max(1000, s.peOI + peOiDelta),
   213→        isATM: s.strike === newAtm,
   214→        ceVol: Math.round(Math.abs(oiDelta) * (1 + Math.random())),
   215→        peVol: Math.round(Math.abs(peOiDelta) * (1 + Math.random())),
   216→      };
   217→    });
   218→
   219→    return {
   220→      timestamp: new Date().toISOString(),
   221→      symbol,
   222→      spotPrice: newSpot,
   223→      atmStrike: newAtm,
   224→      lotSize: prev.lotSize,
   225→      strikeStep: step,
   226→      expiry: prev.expiry,
   227→      strikes,
   228→    };
   229→  }
   230→
   231→  // First call (no prev) — generate from scratch with random initial values
   232→  const strikes: StrikeFlowData[] = [];
   233→
   234→  for (let i = -5; i <= 5; i++) {
   235→    const strike = atmStrike + i * step;
   236→    const dist = Math.abs(i);
   237→    // Heavier OI near ATM, with some randomness
   238→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   239→    const noise = 0.6 + Math.random() * 0.8;
   240→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   241→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   242→    const moneyness = (spot - strike) / spot;
   243→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   244→    const ceITM = spot > strike;
   245→    const peITM = spot < strike;
   246→
   247→    strikes.push({
   248→      strike,
   249→      isATM: i === 0,
   250→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   251→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   252→      ceOI,
   253→      peOI,
   254→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   255→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   256→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   257→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   258→      ceToken: 100 + i,
   259→      peToken: 200 + i,
   260→    });
   261→  }
   262→
   263→  return {
   264→    timestamp: new Date().toISOString(),
   265→    symbol,
   266→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   267→    atmStrike,
   268→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   269→    strikeStep: step,
   270→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   271→    strikes,
   272→  };
   273→}

   276→// PCR HISTORY (localStorage)
   277→// ═══════════════════════════════════════════
   278→
   279→function getPCRHistoryKey(symbol: string): string {
   280→  const today = new Date().toISOString().split('T')[0];
   281→  return `oi-walls-pcr-history-${today}-${symbol}`;
   282→}
   283→
   284→function loadPCRHistory(symbol: string): number[] {
   285→  if (typeof window === 'undefined') return [];
   286→  try {
   287→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   288→    return raw ? JSON.parse(raw) : [];
   289→  } catch {
   290→    return [];
   291→  }
   292→}
   293→
   294→function savePCRHistory(symbol: string, history: number[]) {
   295→  if (typeof window === 'undefined') return;
   296→  try {
   297→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   298→  } catch {
   299→    // localStorage full or unavailable
   300→  }
   301→}

   307→export default function OIWallsTab() {
   308→  const [symbol, setSymbol] = useState('NIFTY');
   309→  const [snapshot, setSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   310→  const [loading, setLoading] = useState(false);
   311→  const [error, setError] = useState<string | null>(null);
   312→  const [isLive, setIsLive] = useState(false);
   313→  const [lastUpdate, setLastUpdate] = useState<string>('');
   314→  const [pcrHistory, setPcrHistory] = useState<number[]>([]);
   315→  const [demoData, setDemoData] = useState<StrikeFlowSnapshot | null>(null);
   316→  const [scanData, setScanData] = useState<MaxPainScanItem[]>([]);
   317→  const [scanLoading, setScanLoading] = useState(false);
   318→  const [scanError, setScanError] = useState<string | null>(null);
   319→  const [gravitySignal, setGravitySignal] = useState<GravitySignal | null>(null);
   320→
   321→  const prevSnapshotRef = useRef<StrikeFlowSnapshot | null>(null);
   322→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   323→
   324→  // Load PCR history on mount / symbol change
   325→  useEffect(() => {
   326→    setPcrHistory(loadPCRHistory(symbol));
   327→  }, [symbol]);
   328→
   329→  const fetchData = useCallback(async () => {
   330→    setLoading(true);
   331→    setError(null);
   332→    try {
   333→      const res = await fetch(withCreds(`/api/kite/strike-flow?symbol=${symbol}`));
   334→      const data = await res.json();
   335→
   336→      if (data.mode !== 'live') {
   337→        // Demo mode — store prev snapshot for delta computation,
   338→        // then generate fresh demo data with small perturbations.
   339→        // This enables 4-color OI coding even in demo mode.
   340→        const prev = prevSnapshotRef.current;
   341→        const demo = generateDemoData(symbol, prev);
   342→        // NOTE: do NOT update prevSnapshotRef here — it must remain
   343→        // pointing to the PREVIOUS data so the walls computation
   344→        // (which runs during render) can compute real ΔOI/ΔLTP.
   345→        // The ref is updated in a post-render useEffect instead.
   346→        setIsLive(false);
   347→        setDemoData(demo);
   348→        setLastUpdate(new Date().toLocaleTimeString('en-IN', {
   349→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   350→        }));
   351→        setLoading(false);
   352→        return;
   353→      }
   354→
   355→      setIsLive(true);
   356→      setDemoData(null);
   357→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   358→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   359→      }));
   360→
   361→      // Compute OI change before updating ref
   362→      const prev = prevSnapshotRef.current;
   363→      if (prev && prev.symbol === data.symbol) {
   364→        // OI change is computed in metrics; we just need to store prev for diff
   365→      }
   366→      // NOTE: do NOT update prevSnapshotRef here (same reason as demo branch)
   367→      setSnapshot(data);
   368→    } catch (err) {
   369→      setError(err instanceof Error ? err.message : 'Fetch failed');
   370→    } finally {
   371→      setLoading(false);
   372→    }
   373→  }, [symbol]);
   374→
   375→  // Initial fetch + polling for per-symbol OI walls
   376→  useEffect(() => {
   377→    fetchData();
   378→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   379→    return () => {
   380→      if (intervalRef.current) clearInterval(intervalRef.current);
   381→    };
   382→  }, [fetchData]);
   383→
   384→  // Fetch max pain scan for all symbols (longer interval — 120s)
   385→  const scanIntervalRef = useRef<NodeJS.Timeout | null>(null);
   386→  const fetchScan = useCallback(async () => {
   387→    setScanLoading(true);
   388→    setScanError(null);
   389→    try {
   390→      const res = await fetch(withCreds('/api/kite/max-pain-scan'));
   391→      const json = await res.json();
   392→
   393→      if (json.mode === 'live' && json.symbols?.length > 0) {
   394→        const items: MaxPainScanItem[] = json.symbols;
   395→        setScanData(items);
   396→
   397→        // Compute gravity signal — works BOTH directions
   398→        const indices = items.filter(s => s.type === 'index');
   399→        const stocks = items.filter(s => s.type === 'stock');
   400→        const indicesAbove = indices.filter(s => s.dist > 0).length;
   401→        const indicesBelow = indices.filter(s => s.dist < 0).length;
   402→        const stocksAbove = stocks.filter(s => s.dist > 0).length;
   403→        const stocksBelow = stocks.filter(s => s.dist < 0).length;
   404→        const avgIdxDist = indices.length > 0 ? indices.reduce((s, i) => s + i.distPct, 0) / indices.length : 0;
   405→        const avgStkDist = stocks.length > 0 ? stocks.reduce((s, i) => s + i.distPct, 0) / stocks.length : 0;
   406→
   407→        // Determine direction: which side has more index alignment?
   408→        let direction: GravitySignal['direction'] = 'mixed';
   409→        if (indicesAbove > indicesBelow) direction = 'down';   // spot above MP → gravity pulls DOWN
   410→        else if (indicesBelow > indicesAbove) direction = 'up'; // spot below MP → gravity pulls UP
   411→
   412→        // Determine aligned count (whichever direction is dominant)
   413→        const alignedIndices = Math.max(indicesAbove, indicesBelow);
   414→        const alignedStocks = direction === 'down' ? stocksAbove : stocksBelow;
   415→        const stockAvgConfirms = direction === 'down' ? avgStkDist > 0.1 : avgStkDist < -0.1;
   416→
   417→        let strength: GravitySignal['strength'] = 'divergent';
   418→        if (alignedIndices === 4 && stockAvgConfirms) {
   419→          strength = 'strong';
   420→        } else if (alignedIndices >= 3 && stockAvgConfirms) {
   421→          strength = 'moderate';
   422→        } else if (alignedIndices >= 2 && direction !== 'mixed') {
   423→          strength = 'weak';
   424→        }
   425→
   426→        setGravitySignal({
   427→          indicesAbove,
   428→          indicesBelow,
   429→          indicesTotal: indices.length,
   430→          stocksAbove,
   431→          stocksBelow,
   432→          stocksTotal: stocks.length,
   433→          avgIndexDistPct: avgIdxDist,
   434→          avgStockDistPct: avgStkDist,
   435→          direction,
   436→          strength,
   437→        });
   438→      } else if (json.mode === 'demo') {
   439→        setScanData([]);
   440→        setGravitySignal(null);
   441→      } else if (json.error) {
   442→        setScanError(json.error);
   443→      }
   444→    } catch (err) {
   445→      setScanError(err instanceof Error ? err.message : 'Scan failed');
   446→    } finally {
   447→      setScanLoading(false);
   448→    }
   449→  }, []);
   450→
   451→  useEffect(() => {
   452→    fetchScan();
   453→    scanIntervalRef.current = setInterval(fetchScan, 120000); // every 2 min
   454→    return () => {
   455→      if (scanIntervalRef.current) clearInterval(scanIntervalRef.current);
   456→    };
   457→  }, [fetchScan]);
   458→
   459→  // Compute OI change using ref to prev snapshot
   460→  const oiChange = (() => {
   461→    const prev = prevSnapshotRef.current;
   462→    const curr = snapshot;
   463→    if (!prev || !curr || prev.symbol !== curr.symbol) {
   464→      // Check demo data diff
   465→      if (demoData) return { ce: 0, pe: 0 };
   466→      return { ce: 0, pe: 0 };
   467→    }
   468→    let ceChange = 0, peChange = 0;
   469→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   470→    for (const s of curr.strikes) {
   471→      const p = prevMap.get(s.strike);
   472→      if (p) {
   473→        ceChange += s.ceOI - p.ceOI;
   474→        peChange += s.peOI - p.peOI;
   475→      }
   476→    }
   477→    return { ce: ceChange, pe: peChange };
   478→  })();
   479→
   480→  // Use demo or live data
   481→  const data = isLive ? snapshot : demoData;
   482→
   483→  // Update prevSnapshotRef AFTER render so the walls computation
   484→  // (above, during render) always compares against the PREVIOUS data.
   485→  // This ensures ΔOI and ΔLTP are non-zero between polls.
   486→  useEffect(() => {
   487→    if (data) {
   488→      prevSnapshotRef.current = data;
   489→    }
   490→  }, [data?.timestamp]);
   491→
   492→  // Compute walls and metrics
   493→  const { walls, metrics } = (() => {
   494→    if (!data || !data.strikes?.length) {
   495→      return { walls: [] as OIWallsStrike[], metrics: null as ComputedMetrics | null };
   496→    }

   516→      let peOiChg = p ? s.peOI - p.peOI : 0;

   521→      // infer activity from OI concentration patterns.
   522→      // Strikes with highest OI relative to neighbors are likely

   525→      if (!p) {

   527→        const avgPeOI = data.strikes.reduce((sum, x) => sum + x.peOI, 0) / data.strikes.length;

   529→        // Below-average = likely closed or never opened (negative OI change)

   536→        peOiChg = peRatio > 1.1 ? Math.round(s.peOI * 0.02)

   538→                   : Math.round(s.peOI * 0.005);
   539→        // LTP heuristic: ITM options have positive intrinsic value movement
   540→        ceLtpChg = s.ceLTP * 0.002 * (Math.random() - 0.3);  // slight upward bias for CE
   541→        peLtpChg = s.peLTP * 0.002 * (Math.random() - 0.7);  // slight downward bias for PE
   542→      }

   545→        strike: s.strike,
   546→        isATM: s.isATM,
   547→        isCeITM: !s.isATM && s.strike < atmStrike,
   548→        isPeITM: !s.isATM && s.strike > atmStrike,

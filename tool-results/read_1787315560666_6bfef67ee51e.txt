     1→// =============================================================================
     2→// STRIKE FLOW MAP — Per-Strike Smart Money Tracker  (v5 — 4-case + vol fallback)
     3→// =============================================================================
     4→// Standalone script — paste into a NEW .gs file in Apps Script editor
     5→// Does NOT touch your existing NIFTY_SMART_MONEY_TERMINAL script
     6→//
     7→// HOW IT WORKS:
     8→//   1. Reads RAW_OPTIONS sheet (populated by your main script every minute)
     9→//   2. Compares last 2 snapshots to calculate per-strike OI + price changes
    10→//   3. Applies the same Greek-powered 4-color flow logic
    11→//   4. Writes 1 row/min to STRIKE_FLOW tab (24-col color table + NET FLOW in col M)
    12→//   5. Writes 4 raw values per strike to SF_RAWFLOW tab (for Google Sheets charts)
    13→//   6. Colors: CE Write=Light Red, CE Buy=Bright Green, PE Write=Light Green, PE Buy=Bright Red
    14→//   7. Thick black border on highest single bet per minute
    15→//
    16→// SHEETS CREATED:
    17→//   STRIKE_FLOW  — minute-by-minute color table with NET FLOW (rows grow all day)
    18→//   SF_RAWFLOW   — raw 4 values per strike (for native Google Sheets stacked bar chart)
    19→//   SF_TOTAL     — total flow across all strikes (4 cols: CE Buy, CE Write, PE Write, PE Buy)
    20→//                  CE Write & PE Buy stored negative → select all data → Insert stacked bar chart
    21→//
    22→// SETUP (one-time):
    23→//   1. In Apps Script editor, click + next to Files > New Script
    24→//   2. Name it "STRIKE_FLOW_MAP"
    25→//   3. Select ALL > Delete > Paste this ENTIRE file > Save
    26→//   4. Set trigger: Triggers > Add Trigger > runStrikeFlowMap > Time-driven > Every minute
    27→//      (offset by 15-20 sec so it runs AFTER your main script)
    28→//

    31→var LOT_SIZE = 65;
    32→var STRIKE_COUNT = 11;
    33→var STRIKE_STEP = 50;
    34→var START_ROW = 5;
    35→var SF_DATA_START = 7;
    36→
    37→// Expiry auto-rollover: on expiry day after ROLLOVER_HOUR, switch to next expiry
    38→var EXPIRY_DAY = 2;        // 0=Sun, 1=Mon, 2=Tue, 3=Wed, 4=Thu, 5=Fri, 6=Sat
    39→var ROLLOVER_HOUR = 12;    // Switch to next expiry after 12:00 IST on expiry day

    49→function _sfNormTime(rawTime) {
    50→  if (!rawTime) return "";
    51→  if (rawTime instanceof Date)
    52→    return Utilities.formatDate(rawTime, "Asia/Kolkata", "yyyy-MM-dd HH:mm:ss");
    53→  return String(rawTime).trim();
    54→}
    55→
    56→function _sfTimeOnly(rawTime) {
    57→  if (!rawTime) return "";
    58→  if (rawTime instanceof Date)
    59→    return Utilities.formatDate(rawTime, "Asia/Kolkata", "HH:mm:ss");
    60→  var s = String(rawTime).trim();
    61→  var p = s.split(" ");
    62→  return p.length > 1 ? p[p.length - 1] : s;
    63→}
    64→
    65→function _getExtWb() {
    66→  return SpreadsheetApp.openById(EXTERNAL_WB_ID);
    67→}

    69→// =============================================================================
    70→// MAIN FUNCTION
    71→// =============================================================================
    72→function runStrikeFlowMap() {
    73→
    74→  // --- Market hours filter ---
    75→  var now = new Date();
    76→  var ist = new Date(now.toLocaleString("en-US", {timeZone: "Asia/Kolkata"}));
    77→  var day = ist.getDay();
    78→  if (day === 0 || day === 6) { Logger.log("[SF] Weekend"); return; }
    79→  var mins = ist.getHours() * 60 + ist.getMinutes();
    80→  if (mins < 555 || mins > 931) { Logger.log("[SF] Outside Market Hours"); return; }
    81→
    82→  // --- Read RAW_OPTIONS ---
    83→  var ss = SpreadsheetApp.getActiveSpreadsheet();
    84→  var rawSheet = ss.getSheetByName("RAW_OPTIONS");
    85→  if (!rawSheet) { Logger.log("[SF] ERROR: RAW_OPTIONS not found"); return; }
    86→
    87→  var lastRow = rawSheet.getLastRow();
    88→  if (lastRow < START_ROW + STRIKE_COUNT) {
    89→    Logger.log("[SF] Not enough data. Rows: " + lastRow); return;
    90→  }
    91→
    92→  var scanRows = Math.min(100, lastRow - START_ROW + 1);
    93→  var scanStart = lastRow - scanRows + 1;
    94→  var rawData = rawSheet.getRange(scanStart, 1, scanRows, 13).getValues();
    95→
    96→  // --- Find 2 unique timestamps ---
    97→  var timestamps = [];
    98→  for (var i = 0; i < rawData.length; i++) {
    99→    var t = _sfNormTime(rawData[i][0]);
   100→    if (t === "") continue;
   101→    if (timestamps.length === 0 || timestamps[timestamps.length - 1] !== t)
   102→      timestamps.push(t);
   103→  }
   104→  if (timestamps.length < 2) { Logger.log("[SF] Only 1 snapshot"); return; }
   105→
   106→  var prevTs = timestamps[timestamps.length - 2];
   107→  var currTs = timestamps[timestamps.length - 1];
   108→
   109→  // --- Validate timestamps are within 2 minutes (skip if gap/spike) ---
   110→  var prevTime = new Date(prevTs);
   111→  var currTime = new Date(currTs);
   112→  var tsGapMs = currTime.getTime() - prevTime.getTime();
   113→  if (tsGapMs > 2 * 60 * 1000 || tsGapMs <= 0) {
   114→    Logger.log("[SF] SKIP: Timestamp gap " + (tsGapMs / 1000).toFixed(0) + "s (>2 min). Prev: " + prevTs + " | Curr: " + currTs); return;
   115→  }
   116→
   117→  var currTimeOnly = _sfTimeOnly(rawData[rawData.length - 1][0]);
   118→  Logger.log("[SF] Prev: " + prevTs + " | Curr: " + currTs + " | Gap: " + (tsGapMs / 1000).toFixed(0) + "s");
   119→
   120→  // --- Smart expiry selection (auto-rollover on expiry day) ---
   121→  var allExpiries = [];
   122→  for (var i = 0; i < rawData.length; i++) {
   123→    if (_sfNormTime(rawData[i][0]) === currTs) {
   124→      var e = String(rawData[i][1]).trim();
   125→      if (e !== "" && allExpiries.indexOf(e) === -1) allExpiries.push(e);
   126→    }
   127→  }
   128→  if (allExpiries.length === 0) { Logger.log("[SF] ERROR: No expiry found"); return; }
   129→
   130→  // On expiry day after ROLLOVER_HOUR, use next expiry (index 1)
   131→  var expiryIndex = 0;
   132→  if (day === EXPIRY_DAY && ist.getHours() >= ROLLOVER_HOUR && allExpiries.length > 1) {
   133→    expiryIndex = 1;
   134→  }
   135→  var expiry = allExpiries[expiryIndex];
   136→  Logger.log("[SF] Expiries: " + allExpiries.join(", ") + " | Using: " + expiry +
   137→    (expiryIndex === 1 ? " (ROLLOVER)" : " (normal)"));
   138→
   139→  // --- Parse prev & curr data ---
   140→  var prevData = {}, currData = {};
   141→  for (var i = 0; i < rawData.length; i++) {
   142→    var t = _sfNormTime(rawData[i][0]);
   143→    var exp = String(rawData[i][1]).trim();
   144→    var strike = Number(rawData[i][2]);
   145→    if (exp !== expiry || strike <= 0) continue;
   146→
   147→    var entry = {
   148→      ceLTP: Number(rawData[i][3]) || 0,
   149→      peLTP: Number(rawData[i][4]) || 0,
   150→      ceOI:  Number(rawData[i][5]) || 0,
   151→      peOI:  Number(rawData[i][6]) || 0,
   152→      ceVol: Number(rawData[i][7]) || 0,
   153→      peVol: Number(rawData[i][8]) || 0,
   154→      ceDelta: Math.abs(Number(rawData[i][11]) || 0.5),
   155→      peDelta: Math.abs(Number(rawData[i][12]) || 0.5)
   156→    };
   157→
   158→    if (t === prevTs) prevData[strike] = entry;
   159→    else if (t === currTs) currData[strike] = entry;
   160→  }
   161→
   162→  var strikes = Object.keys(currData).map(Number).sort(function(a,b){return a-b;});
   163→  if (strikes.length < STRIKE_COUNT) {
   164→    Logger.log("[SF] Only " + strikes.length + " strikes"); return;
   165→  }
   166→  if (Object.keys(prevData).length < STRIKE_COUNT) {
   167→    Logger.log("[SF] Prev snapshot incomplete"); return;
   168→  }
   169→
   170→  // --- Get Nifty spot ---
   171→  var niftySpot = 0;
   172→  try {
   173→    var extWb = _getExtWb();
   174→    var dyn = extWb.getSheetByName("NIFTY50 DYNAMICS");
   175→    if (dyn) {
   176→      var dr = dyn.getLastRow();
   177→      if (dr >= 5) {
   178→        var spotVal = Number(dyn.getRange(dr, 2).getValue()) || 0;
   179→        if (spotVal > 10000) niftySpot = spotVal;  // Nifty50 is 22000+, reject bad values
   180→      }
   181→    }
   182→  } catch(e) { niftySpot = 0; }
   183→  // Fallback — use midpoint of actual strikes from data
   184→  if (niftySpot === 0 || niftySpot < 10000)
   185→    niftySpot = strikes[Math.floor(strikes.length / 2)];
   186→
   187→  var liveATM = Math.round(niftySpot / 50) * 50;
   188→  Logger.log("[SF] Spot: " + niftySpot + " | ATM: " + liveATM);
   189→
   190→  // --- Calculate 4-color flows ---
   191→  var ceBuy = {}, ceWrite = {}, peBuy = {}, peWrite = {};
   192→
   193→  for (var si = 0; si < strikes.length; si++) {
   194→    var stk = strikes[si];
   195→    if (!prevData[stk]) continue;
   196→    var c = currData[stk], p = prevData[stk];
   197→    var dceOI = c.ceOI - p.ceOI;
   198→    var dpeOI = c.peOI - p.peOI;
   199→    var dceP = c.ceLTP - p.ceLTP;
   200→    var dpeP = c.peLTP - p.peLTP;
   201→
   202→    // === CE Flow (v5: all 4 cases) ===
   203→    if (dceOI > 0 && dceP >= 0)
   204→      ceBuy[stk] = (dceOI * c.ceDelta * LOT_SIZE) / 10000000;
   205→    else if (dceOI > 0 && dceP < 0)
   206→      ceWrite[stk] = (dceOI * c.ceDelta * LOT_SIZE) / 10000000;
   207→    else if (dceOI < 0 && dceP < 0)
   208→      ceWrite[stk] = (Math.abs(dceOI) * c.ceDelta * LOT_SIZE * 0.3) / 10000000;
   209→    else if (dceOI < 0 && dceP >= 0)
   210→      ceWrite[stk] = (Math.abs(dceOI) * c.ceDelta * LOT_SIZE * 0.3) / 10000000;
   211→
   212→    // === PE Flow (v5: all 4 cases) ===
   213→    if (dpeOI > 0 && dpeP >= 0)
   214→      peBuy[stk] = (dpeOI * c.peDelta * LOT_SIZE) / 10000000;
   215→    else if (dpeOI > 0 && dpeP < 0)
   216→      peWrite[stk] = (dpeOI * c.peDelta * LOT_SIZE) / 10000000;
   217→    else if (dpeOI < 0 && dpeP < 0)
   218→      peWrite[stk] = (Math.abs(dpeOI) * c.peDelta * LOT_SIZE * 0.3) / 10000000;
   219→    else if (dpeOI < 0 && dpeP >= 0)
   220→      peBuy[stk] = (Math.abs(dpeOI) * c.peDelta * LOT_SIZE * 0.3) / 10000000;
   221→
   222→    // === v5: Volume fallback when OI unchanged ===
   223→    if (!ceBuy[stk] && !ceWrite[stk] && (c.ceVol > 0)) {
   224→      var vf = 0.4;
   225→      if (dceP >= 0) ceBuy[stk] = (c.ceVol * c.ceDelta * LOT_SIZE * vf) / 10000000;
   226→      else ceWrite[stk] = (c.ceVol * c.ceDelta * LOT_SIZE * vf) / 10000000;
   227→    }
   228→    if (!peBuy[stk] && !peWrite[stk] && (c.peVol > 0)) {
   229→      var vf = 0.4;
   230→      if (dpeP >= 0) peBuy[stk] = (c.peVol * c.peDelta * LOT_SIZE * vf) / 10000000;
   231→      else peWrite[stk] = (c.peVol * c.peDelta * LOT_SIZE * vf) / 10000000;
   232→    }
   233→  }
   234→
   235→  // =========================
   236→  // WRITE TO STRIKE_FLOW TAB
   237→  // =========================
   238→  var extWb = _getExtWb();
   239→  var sfSheet = extWb.getSheetByName("STRIKE_FLOW");
   240→  if (!sfSheet) {
   241→    sfSheet = extWb.insertSheet("STRIKE_FLOW");
   242→    _buildHeaders(sfSheet);
   243→  }
   244→
   245→  var ceOff = [250,200,150,100,50,0,-50,-100,-150,-200,-250];
   246→  var peOff = [-250,-200,-150,-100,-50,0,50,100,150,200,250];
   247→
   248→  // Update strike headers
   249→  for (var ci = 0; ci < 11; ci++) {
   250→    sfSheet.getRange(4, 2+ci).setValue((liveATM + ceOff[ci]) + " CE");
   251→    sfSheet.getRange(4, 14+ci).setValue((liveATM + peOff[ci]) + " PE");
   252→  }
   253→
   254→  // Build 24-col row
   255→  var sfRow = new Array(24);
   256→  sfRow[0] = currTimeOnly;
   257→
   258→  // Calculate NET FLOW for column M (index 12)
   259→  // Positive = CE Buy + PE Write (bullish) | Negative = CE Write + PE Buy (bearish)
   260→  var totalCEBuy = 0, totalCEWrite = 0, totalPEBuy = 0, totalPEWrite = 0;
   261→  var allFlowKeys = Object.keys(ceBuy).concat(Object.keys(ceWrite))
   262→    .concat(Object.keys(peBuy)).concat(Object.keys(peWrite));
   263→  var seenStk = {};
   264→  for (var k = 0; k < allFlowKeys.length; k++) {
   265→    if (!seenStk[allFlowKeys[k]]) {
   266→      seenStk[allFlowKeys[k]] = true;
   267→      var s = Number(allFlowKeys[k]);
   268→      totalCEBuy   += Number(ceBuy[s]   || 0);
   269→      totalCEWrite += Number(ceWrite[s] || 0);
   270→      totalPEBuy   += Number(peBuy[s]   || 0);
   271→      totalPEWrite += Number(peWrite[s] || 0);
   272→    }
   273→  }
   274→  var netFlow = (totalCEBuy + totalPEWrite) - (totalCEWrite + totalPEBuy);
   275→  sfRow[12] = Number(netFlow.toFixed(2));
   276→
   277→  for (var ci = 0; ci < 11; ci++) {
   278→    var stk = liveATM + ceOff[ci];
   279→    var bv = Number(ceBuy[stk]||0), wv = Number(ceWrite[stk]||0);
   280→    sfRow[ci+1] = (bv > 0 || wv > 0) ? (bv >= wv ? Number(bv.toFixed(2)) : Number(wv.toFixed(2))) : 0;
   281→  }
   282→  for (var pi = 0; pi < 11; pi++) {
   283→    var stk = liveATM + peOff[pi];
   284→    var bv = Number(peBuy[stk]||0), wv = Number(peWrite[stk]||0);
   285→    sfRow[pi+13] = (bv > 0 || wv > 0) ? (wv >= bv ? Number(wv.toFixed(2)) : Number(bv.toFixed(2))) : 0;
   286→  }
   287→
   288→  // De-dup same minute
   289→  var sfLast = sfSheet.getLastRow();
   290→  var target = (sfLast < SF_DATA_START) ? SF_DATA_START : sfLast + 1;
   291→  if (sfLast >= SF_DATA_START) {
   292→    var times = sfSheet.getRange(SF_DATA_START, 1, sfLast - SF_DATA_START + 1, 1).getValues();
   293→    for (var si = 0; si < times.length; si++) {
   294→      if (times[si][0]) {
   295→        var ft = (times[si][0] instanceof Date) ?
   296→          Utilities.formatDate(times[si][0], "Asia/Kolkata", "HH:mm:ss") : times[si][0].toString().trim();
   297→        if (ft === currTimeOnly) { target = si + SF_DATA_START; break; }
   298→      }
   299→    }
   300→  }
   301→
   302→  sfSheet.getRange(target, 1, 1, 24).setValues([sfRow]);
   303→
   304→  // --- Color coding ---
   305→  var CE_W = "#FFCCCC", CE_B = "#00B050", PE_W = "#C6EFCE", PE_B = "#BD2130";
   306→  var DARK = "#1a1a2e", ZERO = "#555555";
   307→
   308→  sfSheet.getRange(target, 1).setBackground(DARK).setFontColor("#a8d8ea").setFontSize(9);
   309→
   310→  // Column M: NET FLOW color — green if bullish, red if bearish
   311→  var netCell = sfSheet.getRange(target, 13);
   312→  if (netFlow > 0)
   313→    netCell.setBackground("#00B050").setFontColor("#ffffff").setFontWeight("bold");
   314→  else if (netFlow < 0)
   315→    netCell.setBackground("#BD2130").setFontColor("#ffffff").setFontWeight("bold");
   316→  else
   317→    netCell.setBackground(DARK).setFontColor(ZERO).setFontWeight("normal");
   318→  netCell.setHorizontalAlignment("center").setFontSize(9);
   319→
   320→  var maxVal = 0, maxCol = 0;
   321→
   322→  // CE columns B-L
   323→  for (var ci = 0; ci < 11; ci++) {
   324→    var stk = liveATM + ceOff[ci];
   325→    var bv = Number(ceBuy[stk]||0), wv = Number(ceWrite[stk]||0);
   326→    var cell = sfSheet.getRange(target, 2+ci);
   327→    if (bv <= 0 && wv <= 0)
   328→      cell.setBackground(DARK).setFontColor(ZERO).setFontWeight("normal");
   329→    else if (bv >= wv)
   330→      cell.setBackground(CE_B).setFontColor("#ffffff").setFontWeight("bold");
   331→    else
   332→      cell.setBackground(CE_W).setFontColor("#9c0006").setFontWeight("bold");
   333→    cell.setHorizontalAlignment("center").setFontSize(9);
   334→    var dom = Math.max(bv, wv);
   335→    if (dom > maxVal) { maxVal = dom; maxCol = 2+ci; }
   336→  }
   337→
   338→  // PE columns N-X
   339→  for (var pi = 0; pi < 11; pi++) {
   340→    var stk = liveATM + peOff[pi];
   341→    var bv = Number(peBuy[stk]||0), wv = Number(peWrite[stk]||0);
   342→    var cell = sfSheet.getRange(target, 14+pi);
   343→    if (bv <= 0 && wv <= 0)
   344→      cell.setBackground(DARK).setFontColor(ZERO).setFontWeight("normal");
   345→    else if (wv >= bv)
   346→      cell.setBackground(PE_W).setFontColor("#006100").setFontWeight("bold");
   347→    else
   348→      cell.setBackground(PE_B).setFontColor("#ffffff").setFontWeight("bold");
   349→    cell.setHorizontalAlignment("center").setFontSize(9);
   350→    var dom = Math.max(bv, wv);
   351→    if (dom > maxVal) { maxVal = dom; maxCol = 14+pi; }
   352→  }
   353→
   354→  // Thick black border on highest bet
   355→  if (maxVal > 0) {
   356→    sfSheet.getRange(target, maxCol).setBorder(
   357→      true, true, true, true, null, null, "#000000", SpreadsheetApp.BorderStyle.SOLID_THICK
   358→    );
   359→  }
   360→
   361→  Logger.log("[SF] Row " + target + " | Net: " + netFlow.toFixed(2) + " Cr | Max: " + maxVal.toFixed(2) + " Cr | Spot: " + niftySpot);
   362→
   363→  // =========================
   364→  // WRITE TO SF_RAWFLOW TAB
   365→  // =========================
   366→  _writeRawFlow(ceBuy, ceWrite, peBuy, peWrite, liveATM, currTimeOnly);
   367→
   368→  // =========================
   369→  // WRITE TO SF_TOTAL TAB
   370→  // =========================
   371→  _writeTotalFlow(ceBuy, ceWrite, peBuy, peWrite, currTimeOnly);
   372→}

   375→// =============================================================================
   376→// SF_RAWFLOW — All 4 raw flow values per strike, minute by minute
   377→// =============================================================================
   378→// Layout: 45 columns total
   379→//   Col A: TIME
   380→//   Cols B-E: ATM-250 -> CE Buy | CE Write | PE Write | PE Buy
   381→//   Cols F-I: ATM-200 -> CE Buy | CE Write | PE Write | PE Buy
   382→//   ... (11 strikes x 4 sub-cols = 44 data cols)
   383→//   CE Write and PE Buy stored as NEGATIVE numbers for charting.
   384→//   User can select all data -> Insert -> Stacked bar chart in Google Sheets.
   385→//   Green bars go UP (CE Buy + PE Write), Red bars go DOWN (CE Write + PE Buy).
   386→// =============================================================================
   387→var RF_DATA_START = 7;
   388→
   389→function _writeRawFlow(ceBuy, ceWrite, peBuy, peWrite, liveATM, currTimeOnly) {
   390→  var extWb = _getExtWb();
   391→  var off = [-250,-200,-150,-100,-50,0,50,100,150,200,250];
   392→  var TOTAL_COLS = 1 + 11 * 4; // 45
   393→
   394→  var rf = extWb.getSheetByName("SF_RAWFLOW");
   395→  if (!rf) {
   396→    rf = extWb.insertSheet("SF_RAWFLOW");
   397→    _buildRawFlowHeaders(rf, liveATM, off);
   398→  }
   399→
   400→  // Update strike labels in row 1 (batch — no merge, avoids timeout on external sheet)
   401→  var hdrRow = ["TIME"];
   402→  for (var i = 0; i < 11; i++) {
   403→    var stk = liveATM + off[i];
   404→    var label = off[i] === 0 ? "ATM " + stk : String(stk);
   405→    hdrRow.push(label, label, label, label);
   406→  }
   407→  rf.getRange(1, 1, 1, 45).setValues([hdrRow]);
   408→
   409→  // Build 45-col row
   410→  var row = new Array(TOTAL_COLS);
   411→  row[0] = currTimeOnly;
   412→
   413→  for (var i = 0; i < 11; i++) {
   414→    var stk = liveATM + off[i];
   415→    var b = 1 + i * 4; // 0-indexed position in row array
   416→    row[b]     =  Number((ceBuy[stk]   || 0).toFixed(2)); // CE Buy   (positive)
   417→    row[b + 1] = -Number((ceWrite[stk] || 0).toFixed(2)); // CE Write (negative)
   418→    row[b + 2] =  Number((peWrite[stk] || 0).toFixed(2)); // PE Write (positive)
   419→    row[b + 3] = -Number((peBuy[stk]   || 0).toFixed(2)); // PE Buy   (negative)
   420→  }
   421→
   422→  // De-dup same minute
   423→  var lastR = rf.getLastRow();
   424→  var target = (lastR < RF_DATA_START) ? RF_DATA_START : lastR + 1;
   425→  if (lastR >= RF_DATA_START) {
   426→    var times = rf.getRange(RF_DATA_START, 1, lastR - RF_DATA_START + 1, 1).getValues();
   427→    for (var si = 0; si < times.length; si++) {
   428→      if (times[si][0]) {
   429→        var ft = (times[si][0] instanceof Date) ?
   430→          Utilities.formatDate(times[si][0], "Asia/Kolkata", "HH:mm:ss") : times[si][0].toString().trim();
   431→        if (ft === currTimeOnly) { target = si + RF_DATA_START; break; }
   432→      }
   433→    }
   434→  }
   435→
   436→  rf.getRange(target, 1, 1, TOTAL_COLS).setValues([row]);

   508→// =============================================================================
   509→// SF_TOTAL — Total flow across all 11 strikes, 4 values per minute
   510→// =============================================================================
   511→// Layout: 5 columns: TIME | CE Buy (+) | CE Write (-) | PE Write (+) | PE Buy (-)
   512→// CE Write & PE Buy stored as NEGATIVE so stacked bar chart shows red going DOWN.
   513→// Select rows 7 onwards → Insert → Chart → Stacked column chart.
   514→// =============================================================================
   515→var FT_DATA_START = 7;
   516→
   517→function _writeTotalFlow(ceBuy, ceWrite, peBuy, peWrite, currTimeOnly) {
   518→  var extWb = _getExtWb();
   519→  var TOTAL_COLS = 5;
   520→
   521→  // Sum across all strikes
   522→  var sumCB = 0, sumCW = 0, sumPW = 0, sumPB = 0;
   523→  var allKeys = Object.keys(ceBuy).concat(Object.keys(ceWrite))
   524→    .concat(Object.keys(peWrite)).concat(Object.keys(peBuy));
   525→  var seen = {};
   526→  for (var k = 0; k < allKeys.length; k++) {
   527→    if (!seen[allKeys[k]]) {
   528→      seen[allKeys[k]] = true;
   529→      var s = Number(allKeys[k]);
   530→      sumCB += Number(ceBuy[s]   || 0);
   531→      sumCW += Number(ceWrite[s] || 0);
   532→      sumPW += Number(peWrite[s] || 0);
   533→      sumPB += Number(peBuy[s]   || 0);
   534→    }
   535→  }
   536→
   537→  var ft = extWb.getSheetByName("SF_TOTAL");
   538→  if (!ft) {
   539→    ft = extWb.insertSheet("SF_TOTAL");
   540→    _buildTotalHeaders(ft);
   541→  }
   542→
   543→  // Build 5-col row
   544→  var row = [
   545→    currTimeOnly,
   546→    Number(sumCB.toFixed(2)),              // CE Buy   (positive)
   547→    -Number(sumCW.toFixed(2)),              // CE Write (negative)
   548→    Number(sumPW.toFixed(2)),              // PE Write (positive)
   549→    -Number(sumPB.toFixed(2))               // PE Buy   (negative)
   550→  ];
   551→
   552→  // De-dup same minute
   553→  var lastR = ft.getLastRow();
   554→  var target = (lastR < FT_DATA_START) ? FT_DATA_START : lastR + 1;
   555→  if (lastR >= FT_DATA_START) {
   556→    var times = ft.getRange(FT_DATA_START, 1, lastR - FT_DATA_START + 1, 1).getValues();
   557→    for (var si = 0; si < times.length; si++) {
   558→      if (times[si][0]) {
   559→        var tstr = (times[si][0] instanceof Date) ?
   560→          Utilities.formatDate(times[si][0], "Asia/Kolkata", "HH:mm:ss") : times[si][0].toString().trim();
   561→        if (tstr === currTimeOnly) { target = si + FT_DATA_START; break; }
   562→      }
   563→    }
   564→  }
   565→
   566→  ft.getRange(target, 1, 1, TOTAL_COLS).setValues([row]);
   567→
   568→  // Color coding
   569→  var DARK = "#1a1a2e", ZERO = "#555555";
   570→  ft.getRange(target, 1).setBackground(DARK).setFontColor("#a8d8ea").setFontSize(9);
   571→
   572→  // CE Buy (col B)
   573→  if (sumCB > 0)
   574→    ft.getRange(target, 2).setBackground("#00B050").setFontColor("#fff").setFontWeight("bold");
   575→  else
   576→    ft.getRange(target, 2).setBackground(DARK).setFontColor(ZERO);
   577→  ft.getRange(target, 2).setHorizontalAlignment("center").setFontSize(9);
   578→
   579→  // CE Write (col C) — stored negative
   580→  if (sumCW > 0)
   581→    ft.getRange(target, 3).setBackground("#FFCCCC").setFontColor("#9c0006").setFontWeight("bold");
   582→  else
   583→    ft.getRange(target, 3).setBackground(DARK).setFontColor(ZERO);
   584→  ft.getRange(target, 3).setHorizontalAlignment("center").setFontSize(9);
   585→
   586→  // PE Write (col D)
   587→  if (sumPW > 0)
   588→    ft.getRange(target, 4).setBackground("#C6EFCE").setFontColor("#006100").setFontWeight("bold");
   589→  else
   590→    ft.getRange(target, 4).setBackground(DARK).setFontColor(ZERO);
   591→  ft.getRange(target, 4).setHorizontalAlignment("center").setFontSize(9);
   592→
   593→  // PE Buy (col E) — stored negative
   594→  if (sumPB > 0)
   595→    ft.getRange(target, 5).setBackground("#BD2130").setFontColor("#fff").setFontWeight("bold");
   596→  else
   597→    ft.getRange(target, 5).setBackground(DARK).setFontColor(ZERO);
   598→  ft.getRange(target, 5).setHorizontalAlignment("center").setFontSize(9);
   599→
   600→  Logger.log("[SF] SF_TOTAL row " + target + " | CB:" + sumCB.toFixed(2) + " CW:" + (-sumCW).toFixed(2) + " PW:" + sumPW.toFixed(2) + " PB:" + (-sumPB).toFixed(2));
   601→}

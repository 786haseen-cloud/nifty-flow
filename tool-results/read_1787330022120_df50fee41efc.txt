    33→// ═══════════════════════════════════════════
    34→// 4-COLOR FLOW ENGINE (from your Google Script)
    35→// OI change + Price change → 4 cases
    36→// ═══════════════════════════════════════════
    37→
    38→function computeStrikeFlow(
    39→  prev: StrikeFlowData,
    40→  curr: StrikeFlowData,
    41→  lotSize: number,
    42→): Omit<StrikeFlowCell, 'strike' | 'isATM'> {
    43→  const DIVISOR = 10000000; // Convert to Crores
    44→
    45→  // CE side
    46→  const dceOI = curr.ceOI - prev.ceOI;
    47→  const dceP = curr.ceLTP - prev.ceLTP;
    48→  let ceBuy = 0, ceWrite = 0;
    49→
    50→  if (dceOI > 0 && dceP >= 0) {
    51→    // Case 1: OI up + Price up → CE BUY (writers losing, buyers aggressive)
    52→    ceBuy = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    53→  } else if (dceOI > 0 && dceP < 0) {
    54→    // Case 2: OI up + Price down → CE WRITE (writing at higher premium, price falling)
    55→    ceWrite = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    56→  } else if (dceOI < 0 && dceP < 0) {
    57→    // Case 3: OI down + Price down → CE WRITE (short covering at lower price = write)
    58→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    59→  } else if (dceOI < 0 && dceP >= 0) {
    60→    // Case 4: OI down + Price up → CE WRITE (long unwinding at higher price = write)
    61→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    62→  }
    63→
    64→  // Volume fallback when OI unchanged (from your script v5)
    65→  if (ceBuy === 0 && ceWrite === 0 && curr.ceVol > 0) {
    66→    const vf = 0.4;
    67→    if (dceP >= 0) ceBuy = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    68→    else ceWrite = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    69→  }
    70→
    71→  // PE side (same 4 cases, mirrored)
    72→  const dpeOI = curr.peOI - prev.peOI;
    73→  const dpeP = curr.peLTP - prev.peLTP;
    74→  let peBuy = 0, peWrite = 0;
    75→
    76→  if (dpeOI > 0 && dpeP >= 0) {
    77→    peBuy = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    78→  } else if (dpeOI > 0 && dpeP < 0) {
    79→    peWrite = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    80→  } else if (dpeOI < 0 && dpeP < 0) {
    81→    peWrite = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    82→  } else if (dpeOI < 0 && dpeP >= 0) {
    83→    peBuy = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    84→  }
    85→
    86→  // Volume fallback for PE
    87→  if (peBuy === 0 && peWrite === 0 && curr.peVol > 0) {
    88→    const vf = 0.4;
    89→    if (dpeP >= 0) peBuy = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    90→    else peWrite = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    91→  }
    92→
    93→  // Determine display values (show dominant flow per side)
    94→  const ceDisplay = ceBuy >= ceWrite ? ceBuy : ceWrite;
    95→  const peDisplay = peWrite >= peBuy ? peWrite : peBuy;
    96→  const ceType = ceBuy > 0 && ceBuy >= ceWrite ? 'buy' : ceWrite > 0 ? 'write' : 'none';
    97→  const peType = peWrite > 0 && peWrite >= peBuy ? 'write' : peBuy > 0 ? 'buy' : 'none';
    98→
    99→  return { ceBuy, ceWrite, peBuy, peWrite, ceDisplay, peDisplay, ceType, peType };
   100→}

   134→
   135→export default function StrikeFlowMap() {
   136→  const [symbol, setSymbol] = useState('NIFTY');
   137→  const [prevSnapshot, setPrevSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   138→  const [currSnapshot, setCurrSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   139→  const [loading, setLoading] = useState(false);
   140→  const [error, setError] = useState<string | null>(null);
   141→  const [isLive, setIsLive] = useState(false);
   142→  const [lastUpdate, setLastUpdate] = useState<string>('');
   143→  const [flowHistory, setFlowHistory] = useState<FlowTotals[]>([]);
   144→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   145→
   146→  const fetchData = useCallback(async () => {
   147→    setLoading(true);
   148→    setError(null);
   149→    try {
   150→      const res = await fetch(`/api/kite/strike-flow?symbol=${symbol}`);
   151→      const data = await res.json();
   152→
   153→      if (data.mode === 'demo') {
   154→        setIsLive(false);
   155→        setError('Kite API not configured. Set KITE_API_KEY + KITE_ACCESS_TOKEN in Settings.');
   156→        setLoading(false);
   157→        return;
   158→      }
   159→
   160→      if (data.mode === 'error') {
   161→        setError(data.error || 'Unknown error');
   162→        setLoading(false);
   163→        return;
   164→      }
   165→
   166→      setIsLive(true);
   167→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   168→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   169→      }));
   170→
   171→      // Shift current to prev, set new current
   172→      setPrevSnapshot(prev => {
   173→        if (prev && prev.symbol === data.symbol && data.strikes?.length >= 11) {
   174→          // We have both snapshots — compute flow
   175→          const cells = computeFlowCells(prev, data);
   176→          const totals = computeTotals(cells);
   177→          setFlowHistory(h => {
   178→            const updated = [...h, { ...totals, netFlow: totals.netFlow }];
   179→            return updated.slice(-60); // keep last 60 intervals
   180→          });
   181→        }
   182→        return currSnapshot;
   183→      });
   184→      setCurrSnapshot(data);
   185→    } catch (err) {
   186→      setError(err instanceof Error ? err.message : 'Fetch failed');
   187→    } finally {
   188→      setLoading(false);
   189→    }
   190→  }, [symbol, currSnapshot]);
   191→
   192→  // Initial fetch + interval
   193→  useEffect(() => {
   194→    fetchData();
   195→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   196→    return () => {
   197→      if (intervalRef.current) clearInterval(intervalRef.current);
   198→    };
   199→  }, [fetchData]);
   200→
   201→  // Compute flow cells from two snapshots
   202→  function computeFlowCells(prev: StrikeFlowSnapshot, curr: StrikeFlowSnapshot): StrikeFlowCell[] {
   203→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   204→    return curr.strikes
   205→      .filter(s => prevMap.has(s.strike))
   206→      .map(s => ({
   207→        strike: s.strike,
   208→        isATM: s.isATM,
   209→        ...computeStrikeFlow(prevMap.get(s.strike)!, s, curr.lotSize),
   210→      }))
   211→      .sort((a, b) => a.strike - b.strike);
   212→  }
   213→
   214→  // Compute totals
   215→  function computeTotals(cells: StrikeFlowCell[]): FlowTotals {
   216→    let ceBuy = 0, ceWrite = 0, peBuy = 0, peWrite = 0;
   217→    for (const c of cells) {
   218→      ceBuy += c.ceBuy;
   219→      ceWrite += c.ceWrite;
   220→      peBuy += c.peBuy;
   221→      peWrite += c.peWrite;
   222→    }
   223→    return {
   224→      ceBuy: Math.round(ceBuy * 100) / 100,
   225→      ceWrite: Math.round(ceWrite * 100) / 100,
   226→      peBuy: Math.round(peBuy * 100) / 100,
   227→      peWrite: Math.round(peWrite * 100) / 100,
   228→      netFlow: Math.round((ceBuy + peWrite - ceWrite - peBuy) * 100) / 100,
   229→    };
   230→  }
   231→
   232→  // Compute current flow from stored prev/curr
   233→  const flowCells: StrikeFlowCell[] = (() => {
   234→    if (!prevSnapshot || !currSnapshot) return [];
   235→    if (prevSnapshot.symbol !== currSnapshot.symbol) return [];
   236→    if (currSnapshot.strikes.length < 11) return [];
   237→    return computeFlowCells(prevSnapshot, currSnapshot);
   238→  })();
   239→
   240→  const totals: FlowTotals = flowCells.length > 0 ? computeTotals(flowCells) : { ceBuy: 0, ceWrite: 0, peBuy: 0, peWrite: 0, netFlow: 0 };
   241→
   242→  // Find max single bet for thick border
   243→  let maxBetVal = 0, maxBetSide: 'ce' | 'pe' | null = null, maxBetIdx = -1;
   244→  for (let i = 0; i < flowCells.length; i++) {
   245→    const c = flowCells[i];
   246→    if (c.ceDisplay > maxBetVal) { maxBetVal = c.ceDisplay; maxBetSide = 'ce'; maxBetIdx = i; }
   247→    if (c.peDisplay > maxBetVal) { maxBetVal = c.peDisplay; maxBetSide = 'pe'; maxBetIdx = i; }
   248→  }
   249→
   250→  // CE offsets: ATM+250, +200, +150, +100, +50, 0, -50, -100, -150, -200, -250
   251→  const CE_OFFSETS = [250, 200, 150, 100, 50, 0, -50, -100, -150, -200, -250];
   252→  const PE_OFFSETS = [-250, -200, -150, -100, -50, 0, 50, 100, 150, 200, 250];
   253→  const CE_CATS = ['OTM','OTM','OTM','OTM','OTM','ATM','ITM','ITM','ITM','ITM','ITM'];
   254→  const PE_CATS = ['ITM','ITM','ITM','ITM','ITM','ATM','OTM','OTM','OTM','OTM','OTM'];
   255→
   256→  function getCellByOffset(cells: StrikeFlowCell[], offset: number): StrikeFlowCell | undefined {
   257→    if (!currSnapshot) return undefined;
   258→    const strike = currSnapshot.atmStrike + offset;
   259→    return cells.find(c => c.strike === strike);
   260→  }
   261→
   262→  function cellBg(type: 'buy' | 'write' | 'none', side: 'ce' | 'pe'): string {
   263→    if (type === 'none') return COLORS.dark;
   264→    if (side === 'ce') return type === 'buy' ? COLORS.ceBuy : COLORS.ceWrite;
   265→    return type === 'write' ? COLORS.peWrite : COLORS.peBuy;
   266→  }
   267→
   268→  function cellText(type: 'buy' | 'write' | 'none', side: 'ce' | 'pe'): string {
   269→    if (type === 'none') return COLORS.zero;
   270→    if (side === 'ce') return type === 'buy' ? COLORS.ceBuyText : COLORS.ceWriteText;
   271→    return type === 'write' ? COLORS.peWriteText : COLORS.peBuyText;
   272→  }
   273→
   274→  function catStyle(cat: string) {
   275→    const c = cat === 'OTM' ? COLORS.otm : cat === 'ATM' ? COLORS.atm : COLORS.itm;
   276→    return { background: c.bg, color: c.fg };
   277→  }
   278→
   279→  function fmtVal(v: number): string {
   280→    if (v === 0) return '—';
   281→    return v >= 0 ? `+${v.toFixed(2)}` : v.toFixed(2);
   282→  }

   284→  // ═══════════════════════════════════════════
   285→  // NET FLOW MINI CHART (last 60 intervals)
   286→  // ═══════════════════════════════════════════
   287→  function NetFlowMiniChart() {
   288→    if (flowHistory.length < 2) return null;
   289→    const vals = flowHistory.map(h => h.netFlow);
   290→    const maxAbs = Math.max(...vals.map(Math.abs), 0.01);
   291→    const w = 300, h = 40;
   292→    const step = w / (vals.length - 1);
   293→
   294→    const points = vals.map((v, i) => `${i * step},${h / 2 - (v / maxAbs) * (h / 2 - 2)}`).join(' ');
   295→    const zeroLine = `M0,${h / 2} L${w},${h / 2}`;
   296→
   297→    // Fill area
   298→    const areaPoints = `0,${h / 2} ${points} ${w},${h / 2}`;
   299→    const lastVal = vals[vals.length - 1];
   300→    const fill = lastVal >= 0 ? 'rgba(0,176,80,0.15)' : 'rgba(189,33,48,0.15)';
   301→    const stroke = lastVal >= 0 ? '#00B050' : '#BD2130';
   302→
   303→    return (
   304→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   305→        <line x1="0" y1={h/2} x2={w} y2={h/2} stroke="#333" strokeWidth="0.5" />
   306→        <polygon points={areaPoints} fill={fill} />
   307→        <polyline points={points} fill="none" stroke={stroke} strokeWidth="1.5" />
   308→        {/* Current value dot */}
   309→        <circle cx={(vals.length - 1) * step} cy={h / 2 - (lastVal / maxAbs) * (h / 2 - 2)} r="2.5" fill={stroke} />
   310→      </svg>
   311→    );
   312→  }

   315→  // TOTALS STACKED BAR (matching your SF_TOTAL sheet)
   316→  // ═══════════════════════════════════════════
   317→  function TotalFlowBar() {
   318→    const maxTotal = Math.max(Math.abs(totals.ceBuy), Math.abs(totals.ceWrite),
   319→      Math.abs(totals.peWrite), Math.abs(totals.peBuy), 0.01);
   320→    const barH = 20;
   321→
   322→    return (
   323→      <div className="space-y-1">

   331→            }} />
   332→          </div>
   333→          <span className="w-16 text-right font-mono" style={{ color: COLORS.ceBuy }}>{totals.ceBuy.toFixed(2)} Cr</span>
   334→        </div>
   335→        <div className="flex items-center gap-2 text-[10px]">
   336→          <span className="text-muted-foreground w-16">CE Write</span>
   337→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   338→            <div className="h-full rounded-sm transition-all duration-500" style={{
   339→              width: `${Math.min(100, (totals.ceWrite / maxTotal) * 100)}%`,
   340→              background: COLORS.ceWrite,
   341→              minWidth: totals.ceWrite > 0 ? '2px' : '0',
   342→            }} />
   343→          </div>
   344→          <span className="w-16 text-right font-mono" style={{ color: COLORS.ceWriteText }}>{totals.ceWrite.toFixed(2)} Cr</span>
   345→        </div>
   346→        <div className="flex items-center gap-2 text-[10px]">
   347→          <span className="text-muted-foreground w-16">PE Write</span>
   348→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   349→            <div className="h-full rounded-sm transition-all duration-500" style={{
   350→              width: `${Math.min(100, (totals.peWrite / maxTotal) * 100)}%`,
   351→              background: COLORS.peWrite,
   352→              minWidth: totals.peWrite > 0 ? '2px' : '0',
   353→            }} />
   354→          </div>
   355→          <span className="w-16 text-right font-mono" style={{ color: COLORS.peWriteText }}>{totals.peWrite.toFixed(2)} Cr</span>
   356→        </div>
   357→        <div className="flex items-center gap-2 text-[10px]">
   358→          <span className="text-muted-foreground w-16">PE Buy</span>
   359→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   360→            <div className="h-full rounded-sm transition-all duration-500" style={{
   361→              width: `${Math.min(100, (totals.peBuy / maxTotal) * 100)}%`,
   362→              background: COLORS.peBuy,
   363→              minWidth: totals.peBuy > 0 ? '2px' : '0',
   364→            }} />
   365→          </div>
   366→          <span className="w-16 text-right font-mono" style={{ color: COLORS.peBuyText }}>{totals.peBuy.toFixed(2)} Cr</span>
   367→        </div>
   368→      </div>
   369→    );
   370→  }
   371→
   372→  // ═══════════════════════════════════════════
   373→  // RENDER
   374→  // ═══════════════════════════════════════════
   375→
   376→  return (
   377→    <div className="space-y-3">
   378→      {/* Header Bar */}
   379→      <div className="flex items-center justify-between flex-wrap gap-2">
   380→        <div className="flex items-center gap-2">
   381→          <h2 className="text-sm font-bold" style={{ color: COLORS.accent }}>
   382→            STRIKE FLOW MAP
   383→          </h2>
   384→          {isLive ? (
   385→            <Badge variant="outline" className="text-[10px] border-emerald-500/40 text-emerald-300">
   386→              <Wifi className="mr-1 h-3 w-3" />Kite LIVE
   387→            </Badge>
   388→          ) : (
   389→            <Badge variant="outline" className="text-[10px] border-red-500/40 text-red-300">
   390→              <WifiOff className="mr-1 h-3 w-3" />Offline
   391→            </Badge>
   392→          )}
   393→          {loading && <RefreshCw className="h-3 w-3 animate-spin text-muted-foreground" />}
   394→          {lastUpdate && (
   395→            <span className="text-[10px] font-mono text-muted-foreground">Updated: {lastUpdate}</span>
   396→          )}
   397→        </div>
   398→
   399→        {/* Symbol Selector */}
   400→        <div className="flex gap-1">
   401→          {SYMBOLS.map(s => (
   402→            <button
   403→              key={s}
   404→              onClick={() => { setSymbol(s); setPrevSnapshot(null); setCurrSnapshot(null); setFlowHistory([]); }}
   405→              className={`px-2 py-1 text-[10px] font-bold rounded transition-colors ${

   407→                  ? 'bg-purple-500/20 text-purple-300 border border-purple-500/40'
   408→                  : 'bg-muted/50 text-muted-foreground border border-transparent hover:border-border'

   419→          {error}

   426→          <Crosshair className="h-5 w-5 mx-auto mb-2 animate-pulse" style={{ color: COLORS.info }} />
   427→          <p className="font-semibold" style={{ color: COLORS.info }}>Collecting snapshots...</p>
   428→          <p className="mt-1">Need 2 consecutive data points to compute flow. Next poll in {REFRESH_INTERVAL / 1000}s.</p>

   430→            <p className="mt-1 font-mono">Snapshot 1 received: {currSnapshot.strikes.length} strikes, Spot: {currSnapshot.spotPrice.toFixed(0)}</p>
   431→          )}
   432→        </div>
   433→      )}
   434→
   435→      {/* Main Grid */}
   436→      {flowCells.length > 0 && currSnapshot && (
   437→        <div className="overflow-x-auto">
   438→          {/* Spot + ATM Info */}
   439→          <div className="flex items-center gap-3 mb-2 px-1">
   440→            <span className="text-[10px] font-mono" style={{ color: COLORS.info }}>
   441→              Spot: <b className="text-white">{currSnapshot.spotPrice.toFixed(0)}</b>
   442→            </span>

   445→            </span>

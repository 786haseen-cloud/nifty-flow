    14→/**
    15→ * 4-Color OI classification (matches Strike Flow Map convention):
    16→ *
    17→ *  CE  Buy   = ΔOI>0 & ΔLTP≥0  → bright green  #00B050
    18→ *  CE  Write = ΔOI>0 & ΔLTP<0  → light red     #FFCCCC (dark red border)
    19→ *  CE  Close = ΔOI<0           → grey          #6b7280
    20→ *  PE  Write = ΔOI>0 & ΔLTP≥0  → light green   #C6EFCE (dark green border)
    21→ *  PE  Buy   = ΔOI>0 & ΔLTP<0  → bright red    #BD2130
    22→ *  PE  Close = ΔOI<0           → grey          #6b7280
    23→ */
    24→type OIColor = {
    25→  bg: string;        // inline background color
    26→  border?: string;   // inline border color (for Write classes)
    27→  text: string;      // tailwind text class for the OI label
    28→  label?: string;    // short label: "Buy", "Write", "Close"
    29→};
    30→
    31→function getOIColor(oiChange: number, ltpChange: number, isPE: boolean): OIColor {
    32→  // No previous data — neutral
    33→  if (oiChange === 0 && ltpChange === 0) {
    34→    return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    35→  }
    36→  // Unwinding / short covering (ΔOI < 0)
    37→  if (oiChange < 0) {
    38→    return { bg: '#6b7280', text: 'text-zinc-400', label: 'Close' };
    39→  }
    40→  // oiChange > 0 — new positions opened
    41→  if (isPE) {
    42→    // PE: Write = bullish (price up/stable + OI up), Buy = bearish (price down + OI up)
    43→    if (ltpChange >= 0) return { bg: '#C6EFCE', border: '#16a34a', text: 'text-emerald-700', label: 'Write' };
    44→    if (ltpChange < 0)  return { bg: '#BD2130', text: 'text-red-100', label: 'Buy' };
    45→  } else {
    46→    // CE: Buy = bullish (price up/stable + OI up), Write = bearish (price down + OI up)
    47→    if (ltpChange >= 0) return { bg: '#00B050', text: 'text-green-100', label: 'Buy' };
    48→    if (ltpChange < 0)  return { bg: '#FFCCCC', border: '#dc2626', text: 'text-red-700', label: 'Write' };
    49→  }
    50→  // Fallback (shouldn't reach here)
    51→  return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    52→}

   138→function formatLakhs(n: number): string {
   139→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
   140→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
   141→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
   142→  return n.toLocaleString('en-IN');
   143→}

   149→// Compute Max Pain: strike where option buyers lose the most
   150→function computeMaxPain(strikes: StrikeFlowData[]): number {
   151→  if (strikes.length === 0) return 0;
   152→  let minLoss = Infinity;
   153→  let maxPainStrike = strikes[0].strike;
   154→
   155→  for (const k of strikes) {
   156→    let totalLoss = 0;
   157→    for (const s of strikes) {
   158→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
   159→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
   160→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
   161→      // But we only have OI, not premium. Standard Max Pain uses:
   162→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
   163→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
   164→      // Simplified: use intrinsic value approach
   165→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
   166→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
   167→    }
   168→    if (totalLoss < minLoss) {
   169→      minLoss = totalLoss;
   170→      maxPainStrike = k.strike;
   171→    }
   172→  }
   173→  return maxPainStrike;
   174→}

   176→// Generate demo data for 11 strikes
   177→function generateDemoData(symbol: string): StrikeFlowSnapshot {
   178→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   179→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   180→  const atmStrike = Math.round(spot / step) * step;
   181→  const strikes: StrikeFlowData[] = [];
   182→
   183→  for (let i = -5; i <= 5; i++) {
   184→    const strike = atmStrike + i * step;
   185→    const dist = Math.abs(i);
   186→    // Heavier OI near ATM, with some randomness
   187→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   188→    const noise = 0.6 + Math.random() * 0.8;
   189→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   190→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   191→    const moneyness = (spot - strike) / spot;
   192→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   193→    const ceITM = spot > strike;
   194→    const peITM = spot < strike;
   195→
   196→    strikes.push({
   197→      strike,
   198→      isATM: i === 0,
   199→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   200→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   201→      ceOI,
   202→      peOI,
   203→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   204→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   205→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   206→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   207→      ceToken: 100 + i,
   208→      peToken: 200 + i,
   209→    });
   210→  }
   211→
   212→  return {
   213→    timestamp: new Date().toISOString(),
   214→    symbol,
   215→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   216→    atmStrike,
   217→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   218→    strikeStep: step,
   219→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   220→    strikes,
   221→  };
   222→}

   224→// ═══════════════════════════════════════════
   225→// PCR HISTORY (localStorage)
   226→// ═══════════════════════════════════════════
   227→
   228→function getPCRHistoryKey(symbol: string): string {
   229→  const today = new Date().toISOString().split('T')[0];
   230→  return `oi-walls-pcr-history-${today}-${symbol}`;
   231→}
   232→
   233→function loadPCRHistory(symbol: string): number[] {
   234→  if (typeof window === 'undefined') return [];
   235→  try {
   236→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   237→    return raw ? JSON.parse(raw) : [];
   238→  } catch {
   239→    return [];
   240→  }
   241→}
   242→
   243→function savePCRHistory(symbol: string, history: number[]) {
   244→  if (typeof window === 'undefined') return;
   245→  try {
   246→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   247→  } catch {
   248→    // localStorage full or unavailable
   249→  }
   250→}

   256→export default function OIWallsTab() {
   257→  const [symbol, setSymbol] = useState('NIFTY');
   258→  const [snapshot, setSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   259→  const [loading, setLoading] = useState(false);
   260→  const [error, setError] = useState<string | null>(null);
   261→  const [isLive, setIsLive] = useState(false);
   262→  const [lastUpdate, setLastUpdate] = useState<string>('');
   263→  const [pcrHistory, setPcrHistory] = useState<number[]>([]);
   264→  const [demoData, setDemoData] = useState<StrikeFlowSnapshot | null>(null);
   265→  const [scanData, setScanData] = useState<MaxPainScanItem[]>([]);
   266→  const [scanLoading, setScanLoading] = useState(false);
   267→  const [scanError, setScanError] = useState<string | null>(null);
   268→  const [gravitySignal, setGravitySignal] = useState<GravitySignal | null>(null);
   269→
   270→  const prevSnapshotRef = useRef<StrikeFlowSnapshot | null>(null);
   271→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   272→
   273→  // Load PCR history on mount / symbol change
   274→  useEffect(() => {
   275→    setPcrHistory(loadPCRHistory(symbol));
   276→  }, [symbol]);
   277→
   278→  const fetchData = useCallback(async () => {
   279→    setLoading(true);
   280→    setError(null);
   281→    try {
   282→      const res = await fetch(withCreds(`/api/kite/strike-flow?symbol=${symbol}`));
   283→      const data = await res.json();
   284→
   285→      if (data.mode !== 'live') {
   286→        // Generate demo data
   287→        setIsLive(false);
   288→        const demo = generateDemoData(symbol);
   289→        setDemoData(demo);
   290→        setLastUpdate(new Date().toLocaleTimeString('en-IN', {
   291→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   292→        }));
   293→        setLoading(false);
   294→        return;
   295→      }
   296→
   297→      setIsLive(true);
   298→      setDemoData(null);
   299→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   300→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   301→      }));
   302→
   303→      // Compute OI change before updating ref
   304→      const prev = prevSnapshotRef.current;
   305→      if (prev && prev.symbol === data.symbol) {
   306→        // OI change is computed in metrics; we just need to store prev for diff
   307→      }
   308→      prevSnapshotRef.current = data;
   309→      setSnapshot(data);
   310→    } catch (err) {
   311→      setError(err instanceof Error ? err.message : 'Fetch failed');
   312→    } finally {
   313→      setLoading(false);
   314→    }
   315→  }, [symbol]);
   316→
   317→  // Initial fetch + polling for per-symbol OI walls
   318→  useEffect(() => {
   319→    fetchData();
   320→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   321→    return () => {
   322→      if (intervalRef.current) clearInterval(intervalRef.current);
   323→    };
   324→  }, [fetchData]);
   325→
   326→  // Fetch max pain scan for all symbols (longer interval — 120s)
   327→  const scanIntervalRef = useRef<NodeJS.Timeout | null>(null);
   328→  const fetchScan = useCallback(async () => {
   329→    setScanLoading(true);
   330→    setScanError(null);
   331→    try {
   332→      const res = await fetch(withCreds('/api/kite/max-pain-scan'));
   333→      const json = await res.json();
   334→
   335→      if (json.mode === 'live' && json.symbols?.length > 0) {
   336→        const items: MaxPainScanItem[] = json.symbols;
   337→        setScanData(items);
   338→
   339→        // Compute gravity signal — works BOTH directions
   340→        const indices = items.filter(s => s.type === 'index');
   341→        const stocks = items.filter(s => s.type === 'stock');
   342→        const indicesAbove = indices.filter(s => s.dist > 0).length;
   343→        const indicesBelow = indices.filter(s => s.dist < 0).length;
   344→        const stocksAbove = stocks.filter(s => s.dist > 0).length;
   345→        const stocksBelow = stocks.filter(s => s.dist < 0).length;
   346→        const avgIdxDist = indices.length > 0 ? indices.reduce((s, i) => s + i.distPct, 0) / indices.length : 0;
   347→        const avgStkDist = stocks.length > 0 ? stocks.reduce((s, i) => s + i.distPct, 0) / stocks.length : 0;
   348→
   349→        // Determine direction: which side has more index alignment?
   350→        let direction: GravitySignal['direction'] = 'mixed';
   351→        if (indicesAbove > indicesBelow) direction = 'down';   // spot above MP → gravity pulls DOWN
   352→        else if (indicesBelow > indicesAbove) direction = 'up'; // spot below MP → gravity pulls UP
   353→
   354→        // Determine aligned count (whichever direction is dominant)
   355→        const alignedIndices = Math.max(indicesAbove, indicesBelow);
   356→        const alignedStocks = direction === 'down' ? stocksAbove : stocksBelow;
   357→        const stockAvgConfirms = direction === 'down' ? avgStkDist > 0.1 : avgStkDist < -0.1;
   358→
   359→        let strength: GravitySignal['strength'] = 'divergent';
   360→        if (alignedIndices === 4 && stockAvgConfirms) {
   361→          strength = 'strong';
   362→        } else if (alignedIndices >= 3 && stockAvgConfirms) {
   363→          strength = 'moderate';
   364→        } else if (alignedIndices >= 2 && direction !== 'mixed') {
   365→          strength = 'weak';
   366→        }
   367→
   368→        setGravitySignal({
   369→          indicesAbove,
   370→          indicesBelow,
   371→          indicesTotal: indices.length,
   372→          stocksAbove,
   373→          stocksBelow,
   374→          stocksTotal: stocks.length,
   375→          avgIndexDistPct: avgIdxDist,
   376→          avgStockDistPct: avgStkDist,
   377→          direction,
   378→          strength,
   379→        });
   380→      } else if (json.mode === 'demo') {
   381→        setScanData([]);
   382→        setGravitySignal(null);
   383→      } else if (json.error) {
   384→        setScanError(json.error);
   385→      }
   386→    } catch (err) {
   387→      setScanError(err instanceof Error ? err.message : 'Scan failed');
   388→    } finally {
   389→      setScanLoading(false);
   390→    }
   391→  }, []);
   392→
   393→  useEffect(() => {
   394→    fetchScan();
   395→    scanIntervalRef.current = setInterval(fetchScan, 120000); // every 2 min
   396→    return () => {
   397→      if (scanIntervalRef.current) clearInterval(scanIntervalRef.current);
   398→    };
   399→  }, [fetchScan]);
   400→
   401→  // Compute OI change using ref to prev snapshot
   402→  const oiChange = (() => {
   403→    const prev = prevSnapshotRef.current;
   404→    const curr = snapshot;
   405→    if (!prev || !curr || prev.symbol !== curr.symbol) {
   406→      // Check demo data diff
   407→      if (demoData) return { ce: 0, pe: 0 };
   408→      return { ce: 0, pe: 0 };
   409→    }
   410→    let ceChange = 0, peChange = 0;
   411→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   412→    for (const s of curr.strikes) {
   413→      const p = prevMap.get(s.strike);
   414→      if (p) {
   415→        ceChange += s.ceOI - p.ceOI;
   416→        peChange += s.peOI - p.peOI;
   417→      }
   418→    }
   419→    return { ce: ceChange, pe: peChange };
   420→  })();
   421→
   422→  // Use demo or live data
   423→  const data = isLive ? snapshot : demoData;
   424→
   425→  // Compute walls and metrics
   426→  const { walls, metrics } = (() => {
   427→    if (!data || !data.strikes?.length) {
   428→      return { walls: [] as OIWallsStrike[], metrics: null as ComputedMetrics | null };
   429→    }
   430→
   431→    const maxOI = Math.max(
   432→      ...data.strikes.map(s => Math.max(s.ceOI, s.peOI)),
   433→      1
   434→    );
   435→
   436→    // Find ATM strike for ITM determination
   437→    const atmStrike = data.atmStrike || data.strikes.find(s => s.isATM)?.strike || 0;
   438→
   439→    // Build prev strike map for per-strike diff
   440→    const prev = prevSnapshotRef.current;
   441→    const prevMap = new Map<number, StrikeFlowData>();
   442→    if (prev && prev.symbol === data.symbol) {
   443→      for (const s of prev.strikes) prevMap.set(s.strike, s);
   444→    }
   445→
   446→    const walls: OIWallsStrike[] = data.strikes.map(s => {
   447→      const p = prevMap.get(s.strike);
   448→      const ceOiChg = p ? s.ceOI - p.ceOI : 0;
   449→      const peOiChg = p ? s.peOI - p.peOI : 0;
   450→      const ceLtpChg = p ? s.ceLTP - p.ceLTP : 0;
   451→      const peLtpChg = p ? s.peLTP - p.peLTP : 0;
   452→
   453→      return {
   454→        strike: s.strike,
   455→        isATM: s.isATM,
   456→        isCeITM: !s.isATM && s.strike < atmStrike,
   457→        isPeITM: !s.isATM && s.strike > atmStrike,
   458→        ceOI: s.ceOI,
   459→        peOI: s.peOI,
   460→        ceLTP: s.ceLTP,
   461→        peLTP: s.peLTP,
   462→        cePct: (s.ceOI / maxOI) * 100,
   463→        pePct: (s.peOI / maxOI) * 100,
   464→        ceOiChange: ceOiChg,
   465→        peOiChange: peOiChg,
   466→        ceColor: getOIColor(ceOiChg, ceLtpChg, false),
   467→        peColor: getOIColor(peOiChg, peLtpChg, true),
   468→      };
   469→    });
   470→
   471→    const totalCEOI = data.strikes.reduce((sum, s) => sum + s.ceOI, 0);
   472→    const totalPEOI = data.strikes.reduce((sum, s) => sum + s.peOI, 0);
   473→    const pcr = totalCEOI > 0 ? totalPEOI / totalCEOI : 0;
   474→    const maxPain = computeMaxPain(data.strikes);
   475→    const maxPainDist = data.spotPrice - maxPain;
   476→
   477→    const m: ComputedMetrics = {
   478→      pcr,
   479→      maxPain,
   480→      maxPainDist,
   481→      totalCEOI,
   482→      totalPEOI,
   483→      totalOI: totalCEOI + totalPEOI,
   484→      ceOIChange: oiChange.ce,
   485→      peOIChange: oiChange.pe,
   486→    };
   487→
   488→    return { walls, metrics: m };
   489→  })();
   490→
   491→  // Append PCR to history when metrics change
   492→  useEffect(() => {
   493→    if (!metrics) return;
   494→    setPcrHistory(prev => {
   495→      const updated = [...prev, metrics.pcr];
   496→      const trimmed = updated.slice(-60);
   497→      savePCRHistory(symbol, trimmed);
   498→      return trimmed;
   499→    });
   500→  }, [isLive ? snapshot?.timestamp : demoData?.timestamp, metrics?.pcr, symbol]);

   502→  // PCR label

   514→  function PCRSparkline() {
   515→    if (pcrHistory.length < 2) {
   516→      return (
   517→        <div className="flex items-center justify-center h-10 text-zinc-600 text-[10px]">
   518→          Waiting for data points...
   519→        </div>
   520→      );
   521→    }
   522→
   523→    const w = 280, h = 36;
   524→    const vals = pcrHistory;
   525→    const minV = Math.min(...vals) * 0.95;
   526→    const maxV = Math.max(...vals) * 1.05;
   527→    const rangeV = maxV - minV || 1;
   528→    const step = w / (vals.length - 1);
   529→
   530→    const points = vals.map((v, i) => {
   531→      const x = i * step;
   532→      const y = h - 2 - ((v - minV) / rangeV) * (h - 4);
   533→      return `${x},${y}`;
   534→    }).join(' ');
   535→
   536→    // PCR = 1 reference line
   537→    const pcr1Y = h - 2 - ((1 - minV) / rangeV) * (h - 4);
   538→    const lastVal = vals[vals.length - 1];
   539→    const lastX = (vals.length - 1) * step;
   540→    const lastY = h - 2 - ((lastVal - minV) / rangeV) * (h - 4);
   541→    const lineColor = lastVal >= 1 ? '#34d399' : '#f87171';
   542→
   543→    return (
   544→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   545→        {/* PCR = 1 line */}
   546→        {pcr1Y > 0 && pcr1Y < h && (
   547→          <line x1="0" y1={pcr1Y} x2={w} y2={pcr1Y}
   548→            stroke="#525252" strokeWidth="0.5" strokeDasharray="3,3" />
   549→        )}
   550→        {/* Area fill */}
   551→        <polygon
   552→          points={`0,${h} ${points} ${lastX},${h}`}
   553→          fill={lastVal >= 1 ? 'rgba(52,211,153,0.1)' : 'rgba(248,113,113,0.1)'}
   554→        />
   555→        {/* Line */}
   556→        <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" />
   557→        {/* End dot */}
   558→        <circle cx={lastX} cy={lastY} r="2.5" fill={lineColor} />
   559→      </svg>
   560→    );
   561→  }

   120→function toIST(isoStr: string): string {
   121→  const d = new Date(isoStr);
   122→  const utc = d.getTime() + d.getTimezoneOffset() * 60000;
   123→  const ist = new Date(utc + 5.5 * 60 * 60 * 1000);
   124→  return ist.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false, timeZone: 'UTC' });
   125→}

   146→function computeIntervalFlows(
   147→  prev: BatchSnapshot,
   148→  curr: BatchSnapshot,
   149→): IntervalFlow[] {
   150→  const results: IntervalFlow[] = [];
   151→
   152→  for (const cSym of curr.symbols) {
   153→    const pSym = prev.symbols.find(s => s.symbol === cSym.symbol);
   154→    if (!pSym || pSym.strikes.length === 0) continue;
   155→
   156→    // Cash turnover delta: (curr cumulative vol - prev cumulative vol) × curr price
   157→    const cashDeltaVol = Math.max(0, cSym.spotVolume - pSym.spotVolume);
   158→    const cashTurnover = (cashDeltaVol * cSym.spotPrice) / DIVISOR;
   159→
   160→    // Future OI change × lotSize × price
   161→    const futOIDelta = cSym.futOI - pSym.futOI;
   162→    const futFlow = (Math.abs(futOIDelta) * cSym.futLotSize * cSym.futPrice) / DIVISOR;
   163→    const futFlowSigned = futOIDelta > 0 ? futFlow : -futFlow;
   164→
   165→    // Options: 4-color flow per strike, find max per type
   166→    let maxCeBuy = 0, maxCeWrite = 0, maxPeBuy = 0, maxPeWrite = 0;
   167→    let ceBuyStrike = 0, ceWriteStrike = 0, peBuyStrike = 0, peWriteStrike = 0;
   168→    let totalCeBuy = 0, totalCeWrite = 0, totalPeBuy = 0, totalPeWrite = 0;
   169→    let maxSingleStrike = 0, maxSingleType = '', maxSingleStrikeVal = 0;
   170→
   171→    for (const cStrike of cSym.strikes) {
   172→      const pStrike = pSym.strikes.find(s => s.strike === cStrike.strike);
   173→      if (!pStrike) continue;
   174→
   175→      const flow = computeStrikeFlow(pStrike, cStrike, cSym.lotSize);
   176→      totalCeBuy += flow.ceBuy;
   177→      totalCeWrite += flow.ceWrite;
   178→      totalPeBuy += flow.peBuy;
   179→      totalPeWrite += flow.peWrite;
   180→
   181→      if (flow.ceBuy > maxCeBuy) { maxCeBuy = flow.ceBuy; ceBuyStrike = cStrike.strike; }
   182→      if (flow.ceWrite > maxCeWrite) { maxCeWrite = flow.ceWrite; ceWriteStrike = cStrike.strike; }
   183→      if (flow.peBuy > maxPeBuy) { maxPeBuy = flow.peBuy; peBuyStrike = cStrike.strike; }
   184→      if (flow.peWrite > maxPeWrite) { maxPeWrite = flow.peWrite; peWriteStrike = cStrike.strike; }
   185→
   186→      // Track absolute max single-strike flow
   187→      const absMax = Math.max(flow.ceBuy, flow.ceWrite, flow.peBuy, flow.peWrite);
   188→      if (absMax > maxSingleStrike) {
   189→        maxSingleStrike = absMax;
   190→        maxSingleStrikeVal = absMax;
   191→        if (absMax === flow.ceBuy) { maxSingleType = 'CE Buy'; maxSingleStrikeVal = flow.ceBuy; }
   192→        else if (absMax === flow.ceWrite) { maxSingleType = 'CE Write'; maxSingleStrikeVal = flow.ceWrite; }
   193→        else if (absMax === flow.peBuy) { maxSingleType = 'PE Buy'; maxSingleStrikeVal = flow.peBuy; }
   194→        else { maxSingleType = 'PE Write'; maxSingleStrikeVal = flow.peWrite; }
   195→      }
   196→    }
   197→
   198→    const netFlow = (totalCeBuy + totalPeWrite) - (totalPeBuy + totalCeWrite);
   199→
   200→    // Find the absolute peak across all 6 categories
   201→    const allValues = [
   202→      { type: 'Cash', value: cashTurnover },
   203→      { type: 'Future', value: Math.abs(futFlowSigned) },
   204→      { type: 'CE Buy', value: maxCeBuy },
   205→      { type: 'CE Write', value: maxCeWrite },
   206→      { type: 'PE Buy', value: maxPeBuy },
   207→      { type: 'PE Write', value: maxPeWrite },
   208→    ];
   209→    const peak = allValues.reduce((a, b) => a.value > b.value ? a : b, allValues[0]);
   210→
   211→    results.push({
   212→      symbol: cSym.symbol,
   213→      cash: cashTurnover,
   214→      future: futFlowSigned,
   215→      ceBuy: maxCeBuy,
   216→      ceWrite: maxCeWrite,
   217→      peBuy: maxPeBuy,
   218→      peWrite: maxPeWrite,
   219→      netFlow,
   220→      maxStrike: maxSingleStrike > 0.01 ? { type: maxSingleType, strike: maxSingleStrikeVal > 0 ? (ceBuyStrike || peBuyStrike || ceWriteStrike || peWriteStrike || 0) : 0, value: maxSingleStrike } : null,
   221→    });
   222→  }
   223→
   224→  return results;
   225→}

   233→function getTodayKey(): string {
   234→  const d = new Date();
   235→  const utc = d.getTime() + d.getTimezoneOffset() * 60000;
   236→  const ist = new Date(utc + 5.5 * 60 * 60 * 1000);

   240→function loadFromLS(): Record<string, SymbolDayHighest> | null {
   241→  try {
   242→    const raw = localStorage.getItem(LS_KEY);
   243→    if (!raw) return null;
   244→    const parsed = JSON.parse(raw);
   245→    if (parsed.date === getTodayKey()) return parsed.data;
   246→    return null;
   247→  } catch { return null; }
   248→}

   250→function saveToLS(data: Record<string, SymbolDayHighest>) {
   251→  try {
   252→    localStorage.setItem(LS_KEY, JSON.stringify({ date: getTodayKey(), data }));
   253→  } catch { /* ignore */ }
   254→}

   271→export default function HighestBetTracker() {
   272→  const [mode, setMode] = useState<string>('demo');
   273→  const [loading, setLoading] = useState(true);
   274→  const [error, setError] = useState('');
   275→  const [lastUpdate, setLastUpdate] = useState('');
   276→  const [intervalCount, setIntervalCount] = useState(0);
   277→
   278→  // Snapshots for diffing
   279→  const prevSnapRef = useRef<BatchSnapshot | null>(null);
   280→  const currSnapRef = useRef<BatchSnapshot | null>(null);
   281→
   282→  // Day's highest per symbol
   283→  const [dayHighest, setDayHighest] = useState<Record<string, SymbolDayHighest>>(() => {
   284→    return loadFromLS() || {};
   285→  });
   286→
   287→  // Current interval flows (for display)
   288→  const [currentFlows, setCurrentFlows] = useState<IntervalFlow[]>([]);
   289→
   290→  // Current spot prices (for display in table)
   291→  const [spotPrices, setSpotPrices] = useState<Record<string, { price: number; change: number }>>({});
   292→
   293→  // Sort state
   294→  const [sortKey, setSortKey] = useState<SortKey>('netFlow');
   295→  const [sortAsc, setSortAsc] = useState(false);
   296→
   297→  // Auto-refresh
   298→  const timerRef = useRef<NodeJS.Timeout | null>(null);
   299→  const mountedRef = useRef(true);
   300→
   301→  const fetchData = useCallback(async () => {
   302→    try {
   303→      const res = await fetch(withCreds('/api/kite/highest-bet'));
   304→      const data = await res.json();
   305→
   306→      if (!mountedRef.current) return;
   307→
   308→      if (data.mode === 'error') {
   309→        setError(data.error || 'Unknown error');
   310→        return;
   311→      }
   312→
   313→      setMode(data.mode);
   314→      setError('');
   315→      setLastUpdate(toIST(data.timestamp));
   316→
   317→      const snapshot: BatchSnapshot = {
   318→        timestamp: data.timestamp,
   319→        mode: data.mode,
   320→        symbols: data.symbols,
   321→      };
   322→
   323→      // Update spot prices
   324→      const sp: Record<string, { price: number; change: number }> = {};
   325→      for (const s of data.symbols) {
   326→        sp[s.symbol] = { price: s.spotPrice, change: s.spotChange };
   327→      }
   328→      setSpotPrices(sp);
   329→
   330→      // Seed dayHighest with all symbols on first load (so table always renders)
   331→      if (Object.keys(dayHighest).length === 0 && data.symbols.length > 0) {
   332→        const seed: Record<string, SymbolDayHighest> = {};
   333→        for (const s of data.symbols) {
   334→          seed[s.symbol] = emptyDayHighest();
   335→        }
   336→        setDayHighest(seed);
   337→      }
   338→
   339→      // Compute flows if we have a previous snapshot
   340→      if (prevSnapRef.current && prevSnapRef.current.symbols.length > 0) {
   341→        const flows = computeIntervalFlows(prevSnapRef.current, snapshot);
   342→        setCurrentFlows(flows);
   343→        setIntervalCount(c => c + 1);
   344→
   345→        // Update day's highest
   346→        const istTime = toIST(data.timestamp);
   347→        setDayHighest(prev => {
   348→          const next = { ...prev };
   349→          let changed = false;
   350→
   351→          for (const flow of flows) {
   352→            const existing = next[flow.symbol] || emptyDayHighest();
   353→            const updated = { ...existing };
   354→
   355→            // Check each category
   356→            if (flow.cash > existing.cash.value) {
   357→              updated.cash = { value: flow.cash, time: istTime };
   358→              changed = true;
   359→            }
   360→            if (Math.abs(flow.future) > Math.abs(existing.future.value)) {
   361→              updated.future = { value: flow.future, time: istTime };
   362→              changed = true;
   363→            }
   364→            if (flow.ceBuy > existing.ceBuy.value) {
   365→              updated.ceBuy = { value: flow.ceBuy, time: istTime, strike: 0 };
   366→              changed = true;
   367→            }
   368→            if (flow.ceWrite > existing.ceWrite.value) {
   369→              updated.ceWrite = { value: flow.ceWrite, time: istTime, strike: 0 };
   370→              changed = true;
   371→            }
   372→            if (flow.peBuy > existing.peBuy.value) {
   373→              updated.peBuy = { value: flow.peBuy, time: istTime, strike: 0 };
   374→              changed = true;
   375→            }
   376→            if (flow.peWrite > existing.peWrite.value) {
   377→              updated.peWrite = { value: flow.peWrite, time: istTime, strike: 0 };
   378→              changed = true;
   379→            }
   380→
   381→            // Net flow = cumulative max absolute
   382→            if (Math.abs(flow.netFlow) > Math.abs(existing.netFlow)) {
   383→              updated.netFlow = flow.netFlow;
   384→            }
   385→
   386→            // Peak info: which type had the absolute highest
   387→            const cats = [
   388→              { type: 'CE Buy', val: updated.ceBuy.value, strike: updated.ceBuy.strike },
   389→              { type: 'CE Write', val: updated.ceWrite.value, strike: updated.ceWrite.strike },
   390→              { type: 'PE Buy', val: updated.peBuy.value, strike: updated.peBuy.strike },
   391→              { type: 'PE Write', val: updated.peWrite.value, strike: updated.peWrite.strike },
   392→              { type: 'Cash', val: updated.cash.value },
   393→              { type: 'Future', val: Math.abs(updated.future.value) },
   394→            ];
   395→            const peakCat = cats.reduce((a, b) => a.val > b.val ? a : b, cats[0]);
   396→            updated.peakType = peakCat.type;
   397→            updated.peakTime = peakCat.strike !== undefined ? (cats.find(c => c.type === peakCat.type)?.time || istTime) : istTime;
   398→
   399→            next[flow.symbol] = updated;
   400→          }
   401→
   402→          if (changed) saveToLS(next);
   403→          return next;
   404→        });
   405→      }
   406→
   407→      // Shift snapshots
   408→      prevSnapRef.current = currSnapRef.current;
   409→      currSnapRef.current = snapshot;
   410→
   411→      setLoading(false);
   412→    } catch (err) {
   413→      if (!mountedRef.current) return;
   414→      setError(err instanceof Error ? err.message : 'Fetch failed');
   415→      setLoading(false);
   416→    }
   417→  }, []);
   418→
   419→  useEffect(() => {
   420→    mountedRef.current = true;
   421→    fetchData();
   422→    timerRef.current = setInterval(fetchData, 30000);
   423→    return () => {
   424→      mountedRef.current = false;
   425→      if (timerRef.current) clearInterval(timerRef.current);
   426→    };
   427→  }, [fetchData]);
   428→
   429→  // Manual refresh
   430→  const handleRefresh = () => {
   431→    setLoading(true);
   432→    prevSnapRef.current = null; // Reset to get fresh diff
   433→    fetchData();
   434→  };
   435→
   436→  // Sort logic
   437→  const toggleSort = (key: SortKey) => {
   438→    if (sortKey === key) setSortAsc(!sortAsc);
   439→    else { setSortKey(key); setSortAsc(false); }
   440→  };
   441→
   442→  const sortedSymbols = Object.entries(dayHighest)
   443→    .map(([symbol, h]) => {
   444→      const sp = spotPrices[symbol];
   445→      const currentFlow = currentFlows.find(f => f.symbol === symbol);
   446→      return { symbol, h, spotPrice: sp?.price || 0, spotChange: sp?.change || 0, currentFlow };
   447→    })
   448→    .sort((a, b) => {
   449→      let va = 0, vb = 0;
   450→      switch (sortKey) {
   451→        case 'symbol': return sortAsc ? a.symbol.localeCompare(b.symbol) : b.symbol.localeCompare(a.symbol);
   452→        case 'cash': va = a.h.cash.value; vb = b.h.cash.value; break;
   453→        case 'future': va = Math.abs(a.h.future.value); vb = Math.abs(b.h.future.value); break;
   454→        case 'ceBuy': va = a.h.ceBuy.value; vb = b.h.ceBuy.value; break;
   455→        case 'ceWrite': va = a.h.ceWrite.value; vb = b.h.ceWrite.value; break;
   456→        case 'peBuy': va = a.h.peBuy.value; vb = b.h.peBuy.value; break;
   457→        case 'peWrite': va = a.h.peWrite.value; vb = b.h.peWrite.value; break;
   458→        case 'netFlow': va = Math.abs(a.h.netFlow); vb = Math.abs(b.h.netFlow); break;
   459→      }
   460→      return sortAsc ? va - vb : vb - va;
   461→    });
   462→
   463→  // Find overall biggest bet of the day
   464→  const biggestBet = sortedSymbols.reduce((best, row) => {
   465→    const cats = [
   466→      { val: row.h.ceBuy.value, type: 'CE Buy', time: row.h.ceBuy.time, sym: row.symbol },
   467→      { val: row.h.ceWrite.value, type: 'CE Write', time: row.h.ceWrite.time, sym: row.symbol },
   468→      { val: row.h.peBuy.value, type: 'PE Buy', time: row.h.peBuy.time, sym: row.symbol },
   469→      { val: row.h.peWrite.value, type: 'PE Write', time: row.h.peWrite.time, sym: row.symbol },
   470→      { val: row.h.cash.value, type: 'Cash', time: row.h.cash.time, sym: row.symbol },
   471→      { val: Math.abs(row.h.future.value), type: 'Future', time: row.h.future.time, sym: row.symbol },
   472→    ];
   473→    const peak = cats.reduce((a, b) => a.val > b.val ? a : b, cats[0]);
   474→    if (peak.val > (best?.val || 0)) return { val: peak.val, type: peak.type, time: peak.time, sym: peak.sym };
   475→    return best;
   476→  }, null as { val: number; type: string; time: string; sym: string } | null);
   477→
   478→  // ═══ SECTOR FLOW ROTATION ═══
   479→  const SECTORS: Record<string, string[]> = {
   480→    'Banking': ['HDFCBANK', 'ICICIBANK', 'SBIN', 'KOTAKBANK', 'AXISBANK'],
   481→    'IT': ['TCS', 'INFY'],
   482→    'FMCG': ['HINDUNILVR', 'ITC'],
   483→    'Energy': ['RELIANCE'],
   484→    'Auto': ['MARUTI', 'TATAMOTORS'],
   485→    'Finance': ['BAJFINANCE'],
   486→    'Infra': ['LT'],
   487→    'Telecom': ['BHARTIARTL'],

   509→    if (abs >= 0.01) return `${(abs * 100).toFixed(0)}L`;
   510→    return `${(abs * 10000).toFixed(0)}K`;
   511→  };
   512→
   513→  const cellColor = (value: number, isBullish?: boolean) => {
   514→    if (value <= 0) return 'text-muted-foreground/50';
   515→    if (isBullish === true) return 'text-emerald-400';
   516→    if (isBullish === false) return 'text-red-400';
   517→    return 'text-amber-400';
   518→  };
   519→
   520→  return (
   521→    <div className="space-y-3">
   522→      {/* Header Bar */}
   523→      <div className="flex items-center justify-between flex-wrap gap-2">
   524→        <div className="flex items-center gap-2">
   525→          <Badge variant="outline" className={
   526→            mode === 'live' ? 'border-emerald-500/40 text-emerald-300' : 'border-amber-500/40 text-amber-300'
   527→          }>
   528→            {mode === 'live' ? <Wifi className="mr-1 h-3 w-3" /> : <WifiOff className="mr-1 h-3 w-3" />}
   529→            {mode === 'live' ? 'LIVE' : 'DEMO'}
   530→          </Badge>
   531→          <span className="text-[10px] text-muted-foreground">
   532→            {intervalCount > 0 ? `${intervalCount} intervals tracked` : 'Waiting for 2nd snapshot...'}
   533→          </span>

    66→function formatLakhs(n: number): string {
    67→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
    68→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
    69→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
    70→  return n.toLocaleString('en-IN');
    71→}

    73→function formatNumber(n: number): string {
    74→  return n.toLocaleString('en-IN');

    77→// Compute Max Pain: strike where option buyers lose the most
    78→function computeMaxPain(strikes: StrikeFlowData[]): number {
    79→  if (strikes.length === 0) return 0;
    80→  let minLoss = Infinity;
    81→  let maxPainStrike = strikes[0].strike;
    82→
    83→  for (const k of strikes) {
    84→    let totalLoss = 0;
    85→    for (const s of strikes) {
    86→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
    87→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
    88→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
    89→      // But we only have OI, not premium. Standard Max Pain uses:
    90→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
    91→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
    92→      // Simplified: use intrinsic value approach
    93→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
    94→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
    95→    }
    96→    if (totalLoss < minLoss) {
    97→      minLoss = totalLoss;
    98→      maxPainStrike = k.strike;
    99→    }
   100→  }
   101→  return maxPainStrike;
   102→}

   104→// Generate demo data for 11 strikes
   105→function generateDemoData(symbol: string): StrikeFlowSnapshot {
   106→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   107→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   108→  const atmStrike = Math.round(spot / step) * step;
   109→  const strikes: StrikeFlowData[] = [];
   110→
   111→  for (let i = -5; i <= 5; i++) {
   112→    const strike = atmStrike + i * step;
   113→    const dist = Math.abs(i);
   114→    // Heavier OI near ATM, with some randomness
   115→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   116→    const noise = 0.6 + Math.random() * 0.8;
   117→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   118→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   119→    const moneyness = (spot - strike) / spot;
   120→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   121→    const ceITM = spot > strike;
   122→    const peITM = spot < strike;
   123→
   124→    strikes.push({
   125→      strike,
   126→      isATM: i === 0,
   127→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   128→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   129→      ceOI,
   130→      peOI,
   131→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   132→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   133→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   134→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   135→      ceToken: 100 + i,
   136→      peToken: 200 + i,
   137→    });
   138→  }
   139→
   140→  return {
   141→    timestamp: new Date().toISOString(),
   142→    symbol,
   143→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   144→    atmStrike,
   145→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   146→    strikeStep: step,
   147→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   148→    strikes,
   149→  };
   150→}

   152→// ═══════════════════════════════════════════
   153→// PCR HISTORY (localStorage)
   154→// ═══════════════════════════════════════════
   155→
   156→function getPCRHistoryKey(symbol: string): string {
   157→  const today = new Date().toISOString().split('T')[0];
   158→  return `oi-walls-pcr-history-${today}-${symbol}`;
   159→}
   160→
   161→function loadPCRHistory(symbol: string): number[] {
   162→  if (typeof window === 'undefined') return [];
   163→  try {
   164→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   165→    return raw ? JSON.parse(raw) : [];
   166→  } catch {
   167→    return [];
   168→  }
   169→}
   170→
   171→function savePCRHistory(symbol: string, history: number[]) {
   172→  if (typeof window === 'undefined') return;
   173→  try {
   174→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   175→  } catch {
   176→    // localStorage full or unavailable
   177→  }
   178→}

   184→export default function OIWallsTab() {
   185→  const [symbol, setSymbol] = useState('NIFTY');
   186→  const [snapshot, setSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   187→  const [loading, setLoading] = useState(false);
   188→  const [error, setError] = useState<string | null>(null);
   189→  const [isLive, setIsLive] = useState(false);
   190→  const [lastUpdate, setLastUpdate] = useState<string>('');
   191→  const [pcrHistory, setPcrHistory] = useState<number[]>([]);
   192→  const [demoData, setDemoData] = useState<StrikeFlowSnapshot | null>(null);
   193→
   194→  const prevSnapshotRef = useRef<StrikeFlowSnapshot | null>(null);
   195→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   196→
   197→  // Load PCR history on mount / symbol change
   198→  useEffect(() => {
   199→    setPcrHistory(loadPCRHistory(symbol));
   200→  }, [symbol]);
   201→
   202→  const fetchData = useCallback(async () => {
   203→    setLoading(true);
   204→    setError(null);
   205→    try {
   206→      const res = await fetch(`/api/kite/strike-flow?symbol=${symbol}`);
   207→      const data = await res.json();
   208→
   209→      if (data.mode !== 'live') {
   210→        // Generate demo data
   211→        setIsLive(false);
   212→        const demo = generateDemoData(symbol);
   213→        setDemoData(demo);
   214→        setLastUpdate(new Date().toLocaleTimeString('en-IN', {
   215→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   216→        }));
   217→        setLoading(false);
   218→        return;
   219→      }
   220→
   221→      setIsLive(true);
   222→      setDemoData(null);
   223→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   224→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   225→      }));
   226→
   227→      // Compute OI change before updating ref
   228→      const prev = prevSnapshotRef.current;
   229→      if (prev && prev.symbol === data.symbol) {
   230→        // OI change is computed in metrics; we just need to store prev for diff
   231→      }
   232→      prevSnapshotRef.current = data;
   233→      setSnapshot(data);
   234→    } catch (err) {
   235→      setError(err instanceof Error ? err.message : 'Fetch failed');
   236→    } finally {
   237→      setLoading(false);
   238→    }
   239→  }, [symbol]);
   240→
   241→  // Initial fetch + polling
   242→  useEffect(() => {
   243→    fetchData();
   244→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   245→    return () => {
   246→      if (intervalRef.current) clearInterval(intervalRef.current);
   247→    };
   248→  }, [fetchData]);
   249→
   250→  // Compute OI change using ref to prev snapshot
   251→  const oiChange = (() => {
   252→    const prev = prevSnapshotRef.current;
   253→    const curr = snapshot;
   254→    if (!prev || !curr || prev.symbol !== curr.symbol) {
   255→      // Check demo data diff
   256→      if (demoData) return { ce: 0, pe: 0 };
   257→      return { ce: 0, pe: 0 };
   258→    }
   259→    let ceChange = 0, peChange = 0;
   260→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   261→    for (const s of curr.strikes) {
   262→      const p = prevMap.get(s.strike);
   263→      if (p) {
   264→        ceChange += s.ceOI - p.ceOI;
   265→        peChange += s.peOI - p.peOI;
   266→      }
   267→    }
   268→    return { ce: ceChange, pe: peChange };
   269→  })();
   270→
   271→  // Use demo or live data
   272→  const data = isLive ? snapshot : demoData;
   273→
   274→  // Compute walls and metrics
   275→  const { walls, metrics } = (() => {
   276→    if (!data || !data.strikes?.length) {
   277→      return { walls: [] as OIWallsStrike[], metrics: null as ComputedMetrics | null };
   278→    }
   279→
   280→    const maxOI = Math.max(
   281→      ...data.strikes.map(s => Math.max(s.ceOI, s.peOI)),
   282→      1
   283→    );
   284→
   285→    const walls: OIWallsStrike[] = data.strikes.map(s => ({
   286→      strike: s.strike,
   287→      isATM: s.isATM,
   288→      ceOI: s.ceOI,
   289→      peOI: s.peOI,
   290→      ceLTP: s.ceLTP,
   291→      peLTP: s.peLTP,
   292→      pePct: (s.peOI / maxOI) * 100,
   293→      cePct: (s.ceOI / maxOI) * 100,
   294→    }));
   295→
   296→    const totalCEOI = data.strikes.reduce((sum, s) => sum + s.ceOI, 0);
   297→    const totalPEOI = data.strikes.reduce((sum, s) => sum + s.peOI, 0);
   298→    const pcr = totalCEOI > 0 ? totalPEOI / totalCEOI : 0;
   299→    const maxPain = computeMaxPain(data.strikes);
   300→    const maxPainDist = data.spotPrice - maxPain;
   301→
   302→    const m: ComputedMetrics = {
   303→      pcr,
   304→      maxPain,
   305→      maxPainDist,
   306→      totalCEOI,
   307→      totalPEOI,
   308→      totalOI: totalCEOI + totalPEOI,
   309→      ceOIChange: oiChange.ce,
   310→      peOIChange: oiChange.pe,
   311→    };
   312→
   313→    return { walls, metrics: m };
   314→  })();
   315→
   316→  // Append PCR to history when metrics change
   317→  useEffect(() => {
   318→    if (!metrics) return;
   319→    setPcrHistory(prev => {
   320→      const updated = [...prev, metrics.pcr];
   321→      const trimmed = updated.slice(-60);
   322→      savePCRHistory(symbol, trimmed);
   323→      return trimmed;
   324→    });
   325→  }, [isLive ? snapshot?.timestamp : demoData?.timestamp, metrics?.pcr, symbol]);
   326→
   327→  // PCR label
   328→  function getPCRLabel(pcr: number): { text: string; color: string } {
   329→    if (pcr >= 1.5) return { text: 'Extreme Bearish', color: 'text-emerald-400' };
   330→    if (pcr >= 1.1) return { text: 'Bearish', color: 'text-emerald-300' };
   331→    if (pcr >= 0.9) return { text: 'Neutral', color: 'text-zinc-400' };
   332→    if (pcr >= 0.6) return { text: 'Bullish', color: 'text-red-400' };
   333→    return { text: 'Extreme Bullish', color: 'text-red-500' };
   334→  }
   335→
   336→  // Spot price position (0-100%) within strike range for the dashed line
   337→  const spotPosition = (() => {
   338→    if (!data || walls.length < 2) return 50;
   339→    const minStrike = walls[0].strike;
   340→    const maxStrike = walls[walls.length - 1].strike;
   341→    const range = maxStrike - minStrike;
   342→    if (range === 0) return 50;
   343→    return Math.max(0, Math.min(100, ((data.spotPrice - minStrike) / range) * 100));
   344→  })();
   345→
   346→  // ═══════════════════════════════════════════
   347→  // PCR SPARKLINE (SVG)
   348→  // ═══════════════════════════════════════════
   349→  function PCRSparkline() {
   350→    if (pcrHistory.length < 2) {
   351→      return (
   352→        <div className="flex items-center justify-center h-10 text-zinc-600 text-[10px]">
   353→          Waiting for data points...
   354→        </div>
   355→      );
   356→    }
   357→
   358→    const w = 280, h = 36;
   359→    const vals = pcrHistory;
   360→    const minV = Math.min(...vals) * 0.95;
   361→    const maxV = Math.max(...vals) * 1.05;
   362→    const rangeV = maxV - minV || 1;
   363→    const step = w / (vals.length - 1);
   364→
   365→    const points = vals.map((v, i) => {
   366→      const x = i * step;
   367→      const y = h - 2 - ((v - minV) / rangeV) * (h - 4);
   368→      return `${x},${y}`;
   369→    }).join(' ');
   370→
   371→    // PCR = 1 reference line
   372→    const pcr1Y = h - 2 - ((1 - minV) / rangeV) * (h - 4);
   373→    const lastVal = vals[vals.length - 1];
   374→    const lastX = (vals.length - 1) * step;
   375→    const lastY = h - 2 - ((lastVal - minV) / rangeV) * (h - 4);
   376→    const lineColor = lastVal >= 1 ? '#34d399' : '#f87171';
   377→
   378→    return (
   379→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   380→        {/* PCR = 1 line */}
   381→        {pcr1Y > 0 && pcr1Y < h && (
   382→          <line x1="0" y1={pcr1Y} x2={w} y2={pcr1Y}
   383→            stroke="#525252" strokeWidth="0.5" strokeDasharray="3,3" />
   384→        )}
   385→        {/* Area fill */}
   386→        <polygon
   387→          points={`0,${h} ${points} ${lastX},${h}`}
   388→          fill={lastVal >= 1 ? 'rgba(52,211,153,0.1)' : 'rgba(248,113,113,0.1)'}
   389→        />
   390→        {/* Line */}
   391→        <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" />
   392→        {/* End dot */}
   393→        <circle cx={lastX} cy={lastY} r="2.5" fill={lineColor} />
   394→      </svg>
   395→    );
   396→  }

   406→        {ALL_SYMBOLS.map(s => (
   407→          <button
   408→            key={s.symbol}
   409→            onClick={() => {
   410→              setSymbol(s.symbol);
   411→              setSnapshot(null);
   412→              setDemoData(null);
   413→              prevSnapshotRef.current = null;
   414→            }}
   415→            className={
   416→              `text-[10px] px-2 py-1 rounded border transition-all duration-150 cursor-pointer whitespace-nowrap ` +
   417→              (symbol === s.symbol
   418→                ? 'bg-primary/20 border-primary/50 text-primary-foreground font-semibold'
   419→                : 'bg-card border-border/30 text-zinc-400 hover:border-zinc-500 hover:text-zinc-300')
   420→            }
   421→          >
   422→            {s.symbol}
   423→          </button>
   424→        ))}
   425→      </div>
   426→
   427→      {/* ─── Status Bar ─── */}
   428→      <div className="flex items-center justify-between">
   429→        <div className="flex items-center gap-2">
   430→          {isLive ? (
   431→            <Badge variant="outline" className="gap-1 text-emerald-400 border-emerald-500/30 bg-emerald-500/5">

   448→              Spot: <span className="text-foreground font-semibold">{formatNumber(data.spotPrice)}</span>
   449→            </span>

   454→          <button

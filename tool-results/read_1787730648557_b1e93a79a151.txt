    35→// ═══════════════════════════════════════════
    36→// 4-COLOR FLOW ENGINE (from your Google Script)
    37→// OI change + Price change → 4 cases
    38→// ═══════════════════════════════════════════
    39→
    40→function computeStrikeFlow(
    41→  prev: StrikeFlowData,
    42→  curr: StrikeFlowData,
    43→  lotSize: number,
    44→): Omit<StrikeFlowCell, 'strike' | 'isATM'> {
    45→  const DIVISOR = 10000000; // Convert to Crores
    46→
    47→  // CE side
    48→  const dceOI = curr.ceOI - prev.ceOI;
    49→  const dceP = curr.ceLTP - prev.ceLTP;
    50→  let ceBuy = 0, ceWrite = 0;
    51→
    52→  if (dceOI > 0 && dceP >= 0) {
    53→    // Case 1: OI up + Price up → CE BUY (writers losing, buyers aggressive)
    54→    ceBuy = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    55→  } else if (dceOI > 0 && dceP < 0) {
    56→    // Case 2: OI up + Price down → CE WRITE (writing at higher premium, price falling)
    57→    ceWrite = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    58→  } else if (dceOI < 0 && dceP < 0) {
    59→    // Case 3: OI down + Price down → CE WRITE (short covering at lower price = write)
    60→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    61→  } else if (dceOI < 0 && dceP >= 0) {
    62→    // Case 4: OI down + Price up → CE WRITE (long unwinding at higher price = write)
    63→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    64→  }
    65→
    66→  // Volume fallback when OI unchanged (from your script v5)
    67→  if (ceBuy === 0 && ceWrite === 0 && curr.ceVol > 0) {
    68→    const vf = 0.4;
    69→    if (dceP >= 0) ceBuy = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    70→    else ceWrite = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    71→  }
    72→
    73→  // PE side (same 4 cases, mirrored)
    74→  const dpeOI = curr.peOI - prev.peOI;
    75→  const dpeP = curr.peLTP - prev.peLTP;
    76→  let peBuy = 0, peWrite = 0;
    77→
    78→  if (dpeOI > 0 && dpeP >= 0) {
    79→    peBuy = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    80→  } else if (dpeOI > 0 && dpeP < 0) {
    81→    peWrite = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    82→  } else if (dpeOI < 0 && dpeP < 0) {
    83→    peWrite = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    84→  } else if (dpeOI < 0 && dpeP >= 0) {
    85→    peBuy = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    86→  }
    87→
    88→  // Volume fallback for PE
    89→  if (peBuy === 0 && peWrite === 0 && curr.peVol > 0) {
    90→    const vf = 0.4;
    91→    if (dpeP >= 0) peBuy = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    92→    else peWrite = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    93→  }
    94→
    95→  // Determine display values (show dominant flow per side)
    96→  const ceDisplay = ceBuy >= ceWrite ? ceBuy : ceWrite;
    97→  const peDisplay = peWrite >= peBuy ? peWrite : peBuy;
    98→  const ceType = ceBuy > 0 && ceBuy >= ceWrite ? 'buy' : ceWrite > 0 ? 'write' : 'none';
    99→  const peType = peWrite > 0 && peWrite >= peBuy ? 'write' : peBuy > 0 ? 'buy' : 'none';
   100→
   101→  return { ceBuy, ceWrite, peBuy, peWrite, ceDisplay, peDisplay, ceType, peType };
   102→}

   139→
   140→export default function StrikeFlowMap() {
   141→  const [symbol, setSymbol] = useState('NIFTY');
   142→  const [prevSnapshot, setPrevSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   143→  const [currSnapshot, setCurrSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   144→  const [loading, setLoading] = useState(false);
   145→  const [error, setError] = useState<string | null>(null);
   146→  const [isLive, setIsLive] = useState(false);
   147→  const [lastUpdate, setLastUpdate] = useState<string>('');
   148→  const [flowHistory, setFlowHistory] = useState<FlowTotals[]>([]);
   149→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   150→
   151→  const fetchData = useCallback(async () => {
   152→    setLoading(true);
   153→    setError(null);
   154→    try {
   155→      const res = await fetch(withCreds(`/api/kite/strike-flow?symbol=${symbol}`));
   156→      const data = await res.json();
   157→
   158→      if (data.mode === 'demo') {
   159→        setIsLive(false);
   160→        setError('Kite API not configured. Set KITE_API_KEY + KITE_ACCESS_TOKEN in Settings.');
   161→        setLoading(false);
   162→        return;
   163→      }
   164→
   165→      if (data.mode === 'error') {
   166→        setError(data.error || 'Unknown error');
   167→        setLoading(false);
   168→        return;
   169→      }
   170→
   171→      setIsLive(true);
   172→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   173→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   174→      }));
   175→
   176→      // Shift current to prev, set new current
   177→      setPrevSnapshot(prev => {
   178→        if (prev && prev.symbol === data.symbol && data.strikes?.length >= 11) {
   179→          // We have both snapshots — compute flow
   180→          const cells = computeFlowCells(prev, data);
   181→          const totals = computeTotals(cells);
   182→          setFlowHistory(h => {
   183→            const updated = [...h, { ...totals, netFlow: totals.netFlow }];
   184→            return updated.slice(-60); // keep last 60 intervals
   185→          });
   186→        }
   187→        return currSnapshot;
   188→      });
   189→      setCurrSnapshot(data);
   190→    } catch (err) {
   191→      setError(err instanceof Error ? err.message : 'Fetch failed');
   192→    } finally {
   193→      setLoading(false);
   194→    }
   195→  }, [symbol, currSnapshot]);
   196→
   197→  // Initial fetch + interval
   198→  useEffect(() => {
   199→    fetchData();
   200→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   201→    return () => {
   202→      if (intervalRef.current) clearInterval(intervalRef.current);
   203→    };
   204→  }, [fetchData]);
   205→
   206→  // Compute flow cells from two snapshots
   207→  function computeFlowCells(prev: StrikeFlowSnapshot, curr: StrikeFlowSnapshot): StrikeFlowCell[] {
   208→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   209→    return curr.strikes
   210→      .filter(s => prevMap.has(s.strike))
   211→      .map(s => ({
   212→        strike: s.strike,
   213→        isATM: s.isATM,
   214→        ...computeStrikeFlow(prevMap.get(s.strike)!, s, curr.lotSize),
   215→      }))
   216→      .sort((a, b) => a.strike - b.strike);
   217→  }
   218→
   219→  // Compute totals
   220→  function computeTotals(cells: StrikeFlowCell[]): FlowTotals {
   221→    let ceBuy = 0, ceWrite = 0, peBuy = 0, peWrite = 0;
   222→    for (const c of cells) {
   223→      ceBuy += c.ceBuy;
   224→      ceWrite += c.ceWrite;
   225→      peBuy += c.peBuy;
   226→      peWrite += c.peWrite;
   227→    }
   228→    return {
   229→      ceBuy: Math.round(ceBuy * 100) / 100,
   230→      ceWrite: Math.round(ceWrite * 100) / 100,
   231→      peBuy: Math.round(peBuy * 100) / 100,
   232→      peWrite: Math.round(peWrite * 100) / 100,
   233→      netFlow: Math.round((ceBuy + peWrite - ceWrite - peBuy) * 100) / 100,
   234→    };
   235→  }
   236→
   237→  // Compute current flow from stored prev/curr
   238→  const flowCells: StrikeFlowCell[] = (() => {
   239→    if (!prevSnapshot || !currSnapshot) return [];
   240→    if (prevSnapshot.symbol !== currSnapshot.symbol) return [];
   241→    if (currSnapshot.strikes.length < 11) return [];
   242→    return computeFlowCells(prevSnapshot, currSnapshot);
   243→  })();
   244→
   245→  const totals: FlowTotals = flowCells.length > 0 ? computeTotals(flowCells) : { ceBuy: 0, ceWrite: 0, peBuy: 0, peWrite: 0, netFlow: 0 };
   246→
   247→  // Find max single bet for thick border
   248→  let maxBetVal = 0, maxBetSide: 'ce' | 'pe' | null = null, maxBetIdx = -1;
   249→  for (let i = 0; i < flowCells.length; i++) {
   250→    const c = flowCells[i];
   251→    if (c.ceDisplay > maxBetVal) { maxBetVal = c.ceDisplay; maxBetSide = 'ce'; maxBetIdx = i; }
   252→    if (c.peDisplay > maxBetVal) { maxBetVal = c.peDisplay; maxBetSide = 'pe'; maxBetIdx = i; }
   253→  }
   254→
   255→  // Dynamic offsets based on strikeStep (works for Nifty=50, HDFCBANK=10, etc.)
   256→  const step = currSnapshot ? currSnapshot.strikeStep : 50;
   257→  const n = 5;
   258→  const CE_OFFSETS = Array.from({ length: n * 2 + 1 }, function(_, i) { return (n - i) * step; });
   259→  const PE_OFFSETS = Array.from({ length: n * 2 + 1 }, function(_, i) { return (i - n) * step; });
   260→  const CE_CATS = Array.from({ length: n * 2 + 1 }, function(_, i) { return i < n ? 'OTM' : i === n ? 'ATM' : 'ITM'; });
   261→  const PE_CATS = Array.from({ length: n * 2 + 1 }, function(_, i) { return i < n ? 'ITM' : i === n ? 'ATM' : 'OTM'; });
   262→
   263→  function getCellByOffset(cells: StrikeFlowCell[], offset: number): StrikeFlowCell | undefined {
   264→    if (!currSnapshot) return undefined;
   265→    const strike = currSnapshot.atmStrike + offset;
   266→    return cells.find(c => c.strike === strike);
   267→  }
   268→
   269→  function cellBg(type: 'buy' | 'write' | 'none', side: 'ce' | 'pe'): string {
   270→    if (type === 'none') return COLORS.dark;
   271→    if (side === 'ce') return type === 'buy' ? COLORS.ceBuy : COLORS.ceWrite;
   272→    return type === 'write' ? COLORS.peWrite : COLORS.peBuy;
   273→  }
   274→
   275→  function cellText(type: 'buy' | 'write' | 'none', side: 'ce' | 'pe'): string {
   276→    if (type === 'none') return COLORS.zero;
   277→    if (side === 'ce') return type === 'buy' ? COLORS.ceBuyText : COLORS.ceWriteText;
   278→    return type === 'write' ? COLORS.peWriteText : COLORS.peBuyText;
   279→  }
   280→
   281→  function catStyle(cat: string) {
   282→    const c = cat === 'OTM' ? COLORS.otm : cat === 'ATM' ? COLORS.atm : COLORS.itm;
   283→    return { background: c.bg, color: c.fg };
   284→  }
   285→
   286→  function fmtVal(v: number): string {
   287→    if (v === 0) return '—';
   288→    return v >= 0 ? `+${v.toFixed(2)}` : v.toFixed(2);
   289→  }
   290→
   291→  // ═══════════════════════════════════════════
   292→  // NET FLOW MINI CHART (last 60 intervals)
   293→  // ═══════════════════════════════════════════
   294→  function NetFlowMiniChart() {
   295→    if (flowHistory.length < 2) return null;
   296→    const vals = flowHistory.map(h => h.netFlow);
   297→    const maxAbs = Math.max(...vals.map(Math.abs), 0.01);
   298→    const w = 300, h = 40;
   299→    const step = w / (vals.length - 1);
   300→
   301→    const points = vals.map((v, i) => `${i * step},${h / 2 - (v / maxAbs) * (h / 2 - 2)}`).join(' ');
   302→    const zeroLine = `M0,${h / 2} L${w},${h / 2}`;
   303→
   304→    // Fill area
   305→    const areaPoints = `0,${h / 2} ${points} ${w},${h / 2}`;
   306→    const lastVal = vals[vals.length - 1];
   307→    const fill = lastVal >= 0 ? 'rgba(0,176,80,0.15)' : 'rgba(189,33,48,0.15)';
   308→    const stroke = lastVal >= 0 ? '#00B050' : '#BD2130';
   309→
   310→    return (
   311→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   312→        <line x1="0" y1={h/2} x2={w} y2={h/2} stroke="#333" strokeWidth="0.5" />
   313→        <polygon points={areaPoints} fill={fill} />
   314→        <polyline points={points} fill="none" stroke={stroke} strokeWidth="1.5" />
   315→        {/* Current value dot */}
   316→        <circle cx={(vals.length - 1) * step} cy={h / 2 - (lastVal / maxAbs) * (h / 2 - 2)} r="2.5" fill={stroke} />
   317→      </svg>
   318→    );
   319→  }

   321→  // ═══════════════════════════════════════════
   322→  // TOTALS STACKED BAR (matching your SF_TOTAL sheet)
   323→  // ═══════════════════════════════════════════
   324→  function TotalFlowBar() {
   325→    const maxTotal = Math.max(Math.abs(totals.ceBuy), Math.abs(totals.ceWrite),
   326→      Math.abs(totals.peWrite), Math.abs(totals.peBuy), 0.01);
   327→    const barH = 20;
   328→
   329→    return (
   330→      <div className="space-y-1">
   331→        <div className="flex items-center gap-2 text-[10px]">
   332→          <span className="text-muted-foreground w-16">CE Buy</span>
   333→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   334→            <div className="h-full rounded-sm transition-all duration-500" style={{
   335→              width: `${Math.min(100, (totals.ceBuy / maxTotal) * 100)}%`,
   336→              background: COLORS.ceBuy,
   337→              minWidth: totals.ceBuy > 0 ? '2px' : '0',
   338→            }} />
   339→          </div>
   340→          <span className="w-16 text-right font-mono" style={{ color: COLORS.ceBuy }}>{totals.ceBuy.toFixed(2)} Cr</span>
   341→        </div>
   342→        <div className="flex items-center gap-2 text-[10px]">
   343→          <span className="text-muted-foreground w-16">CE Write</span>
   344→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   345→            <div className="h-full rounded-sm transition-all duration-500" style={{
   346→              width: `${Math.min(100, (totals.ceWrite / maxTotal) * 100)}%`,
   347→              background: COLORS.ceWrite,
   348→              minWidth: totals.ceWrite > 0 ? '2px' : '0',
   349→            }} />
   350→          </div>
   351→          <span className="w-16 text-right font-mono" style={{ color: COLORS.ceWriteText }}>{totals.ceWrite.toFixed(2)} Cr</span>
   352→        </div>
   353→        <div className="flex items-center gap-2 text-[10px]">
   354→          <span className="text-muted-foreground w-16">PE Write</span>
   355→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   356→            <div className="h-full rounded-sm transition-all duration-500" style={{
   357→              width: `${Math.min(100, (totals.peWrite / maxTotal) * 100)}%`,
   358→              background: COLORS.peWrite,
   359→              minWidth: totals.peWrite > 0 ? '2px' : '0',
   360→            }} />
   361→          </div>
   362→          <span className="w-16 text-right font-mono" style={{ color: COLORS.peWriteText }}>{totals.peWrite.toFixed(2)} Cr</span>
   363→        </div>
   364→        <div className="flex items-center gap-2 text-[10px]">
   365→          <span className="text-muted-foreground w-16">PE Buy</span>
   366→          <div className="flex-1 h-4 rounded-sm overflow-hidden" style={{ background: COLORS.dark }}>
   367→            <div className="h-full rounded-sm transition-all duration-500" style={{
   368→              width: `${Math.min(100, (totals.peBuy / maxTotal) * 100)}%`,
   369→              background: COLORS.peBuy,
   370→              minWidth: totals.peBuy > 0 ? '2px' : '0',
   371→            }} />
   372→          </div>
   373→          <span className="w-16 text-right font-mono" style={{ color: COLORS.peBuyText }}>{totals.peBuy.toFixed(2)} Cr</span>
   374→        </div>
   375→      </div>
   376→    );
   377→  }
   378→
   379→  // ═══════════════════════════════════════════
   380→  // RENDER
   381→  // ═══════════════════════════════════════════
   382→
   383→  return (
   384→    <div className="space-y-3">
   385→      {/* Header Bar */}
   386→      <div className="flex items-center justify-between flex-wrap gap-2">
   387→        <div className="flex items-center gap-2">
   388→          <h2 className="text-sm font-bold" style={{ color: COLORS.accent }}>
   389→            STRIKE FLOW MAP
   390→          </h2>
   391→          {isLive ? (
   392→            <Badge variant="outline" className="text-[10px] border-emerald-500/40 text-emerald-300">
   393→              <Wifi className="mr-1 h-3 w-3" />Kite LIVE
   394→            </Badge>
   395→          ) : (
   396→            <Badge variant="outline" className="text-[10px] border-red-500/40 text-red-300">
   397→              <WifiOff className="mr-1 h-3 w-3" />Offline
   398→            </Badge>
   399→          )}
   400→          {loading && <RefreshCw className="h-3 w-3 animate-spin text-muted-foreground" />}
   401→          {lastUpdate && (
   402→            <span className="text-[10px] font-mono text-muted-foreground">Updated: {lastUpdate}</span>
   403→          )}
   404→        </div>
   405→
   406→        {/* Symbol Selector — dropdown for 19 symbols */}
   407→        <div className="flex items-center gap-2">
   408→          <select
   409→            value={symbol}
   410→            onChange={function(e) { setSymbol(e.target.value); setPrevSnapshot(null); setCurrSnapshot(null); setFlowHistory([]); }}
   411→            className="bg-muted/50 border border-border/50 rounded-md px-2 py-1 text-xs font-mono focus:outline-none focus:ring-1 focus:ring-purple-500/50"
   412→          >
   413→            <optgroup label="Indices">
   414→              {ALL_SYMBOLS.filter(function(s) { return s.type === 'index'; }).map(function(s) {
   415→                return <option key={s.symbol} value={s.symbol}>{s.name}</option>;
   416→              })}
   417→            </optgroup>
   418→            <optgroup label="Stocks">
   419→              {ALL_SYMBOLS.filter(function(s) { return s.type === 'stock'; }).map(function(s) {
   420→                return <option key={s.symbol} value={s.symbol}>{s.name}</option>;
   421→              })}
   422→            </optgroup>
   423→          </select>
   424→          <Badge variant="outline" className="text-[9px] border-purple-500/30 text-purple-300">
   425→            {ALL_SYMBOLS.find(function(s) { return s.symbol === symbol; }) ? 'INDEX/STOCK' : ''}
   426→          </Badge>
   427→        </div>
   428→      </div>
   429→
   430→      {error && (
   431→        <div className="rounded-md p-3 text-xs" style={{ background: '#2e0a0a', color: '#ff6666', border: '1px solid #5a1a1a' }}>
   432→          {error}
   433→        </div>
   434→      )}
   435→
   436→      {/* Waiting for 2 snapshots */}
   437→      {isLive && !error && flowCells.length === 0 && (
   438→        <div className="rounded-md p-4 text-center text-xs text-muted-foreground" style={{ background: COLORS.dark }}>
   439→          <Crosshair className="h-5 w-5 mx-auto mb-2 animate-pulse" style={{ color: COLORS.info }} />
   440→          <p className="font-semibold" style={{ color: COLORS.info }}>Collecting snapshots...</p>

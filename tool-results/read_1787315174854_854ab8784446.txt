   206→   406→   406→  function calculateRealIV(marketPrice, strike, T, r, optionPrice, type) {
   207→   407→   407→    if (optionPrice <= 0 || marketPrice <= 0 || T <= 0.001) return 0.18;
   208→   408→   408→    
   209→   409→   409→    var sigma = 0.20;
   210→   410→   410→    var tolerance = 0.0001;
   211→   411→   411→    var maxIterations = 30;
   212→   412→   412→    
   213→   413→   413→    for (var iter = 0; iter < maxIterations; iter++) {
   214→   414→   414→      var d1 = (Math.log(marketPrice / strike) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   215→   415→   415→      var d2 = d1 - sigma * Math.sqrt(T);
   216→   416→   416→      
   217→   417→   417→      var priceDiff;
   218→   418→   418→      if (type === "CE") {
   219→   419→   419→        priceDiff = marketPrice * normCDF(d1) - strike * Math.exp(-r * T) * normCDF(d2) - optionPrice;
   220→   420→   420→      } else {
   221→   421→   421→        priceDiff = strike * Math.exp(-r * T) * normCDF(-d2) - marketPrice * normCDF(-d1) - optionPrice;
   222→   422→   422→      }
   223→   423→   423→      
   224→   424→   424→      if (Math.abs(priceDiff) < tolerance) break;
   225→   425→   425→      
   226→   426→   426→      var vega = marketPrice * Math.sqrt(T) * normPDF(d1);
   227→   427→   427→      if (vega < 0.0001) vega = 0.0001;
   228→   428→   428→      
   229→   429→   429→      var newSigma = sigma - priceDiff / vega;
   230→   430→   430→      if (newSigma <= 0.05) newSigma = 0.05;
   231→   431→   431→      if (newSigma > 1.5) newSigma = 1.5;
   232→   432→   432→      
   233→   433→   433→      sigma = newSigma;
   234→   434→   434→    }
   235→   435→   435→    
   236→   436→   436→    return sigma;
   237→   437→   437→  }

   239→   439→   439→  function calculateCompleteGreeks(S, K, T, r, sigma, type) {
   240→   440→   440→    if (T <= 0.001) T = 0.001;
   241→   441→   441→    if (sigma <= 0.01) sigma = 0.15;
   242→   442→   442→    
   243→   443→   443→    var d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   244→   444→   444→    var d2 = d1 - sigma * Math.sqrt(T);
   245→   445→   445→    
   246→   446→   446→    var delta, gamma, theta, vega, rho;
   247→   447→   447→    var pdfD1 = normPDF(d1);
   248→   448→   448→    
   249→   449→   449→    if (type === "CE") {
   250→   450→   450→      delta = normCDF(d1);
   251→   451→   451→      theta = (-S * pdfD1 * sigma / (2 * Math.sqrt(T))) - (r * K * Math.exp(-r * T) * normCDF(d2));
   252→   452→   452→      rho = K * T * Math.exp(-r * T) * normCDF(d2);
   253→   453→   453→    } else {
   254→   454→   454→      delta = normCDF(d1) - 1;
   255→   455→   455→      theta = (-S * pdfD1 * sigma / (2 * Math.sqrt(T))) + (r * K * Math.exp(-r * T) * normCDF(-d2));
   256→   456→   456→      rho = -K * T * Math.exp(-r * T) * normCDF(-d2);
   257→   457→   457→    }
   258→   458→   458→    
   259→   459→   459→    gamma = pdfD1 / (S * sigma * Math.sqrt(T));
   260→   460→   460→    vega = S * pdfD1 * Math.sqrt(T);
   261→   461→   461→    
   262→   462→   462→    var tradingDays = 252;
   263→   463→   463→    theta = theta / tradingDays;
   264→   464→   464→    vega = vega / 100;
   265→   465→   465→    
   266→   466→   466→    return {
   267→   467→   467→      delta: delta,
   268→   468→   468→      gamma: gamma,
   269→   469→   469→      theta: theta,
   270→   470→   470→      vega: vega,
   271→   471→   471→      rho: rho,
   272→   472→   472→      iv: sigma
   273→   473→   473→    };
   274→   474→   474→  }

   766→   966→   966→function erf(x) {
   767→   967→   967→  var sign = (x >= 0) ? 1 : -1;
   768→   968→   968→  x = Math.abs(x);
   769→   969→   969→
   770→   970→   970→  var a1 = 0.254829592;
   771→   971→   971→  var a2 = -0.284496736;
   772→   972→   972→  var a3 = 1.421413741;
   773→   973→   973→  var a4 = -1.453152027;
   774→   974→   974→  var a5 = 1.061405429;
   775→   975→   975→  var p  = 0.3275911;
   776→   976→   976→
   777→   977→   977→  var t = 1.0 / (1.0 + p * x);
   778→   978→   978→  var y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);
   779→   979→   979→
   780→   980→   980→  return sign * y;
   781→   981→   981→}

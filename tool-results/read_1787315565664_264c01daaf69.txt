     3→     3→// =============================================================================
     4→     4→// Standalone script — paste into a NEW .gs file in Apps Script editor
     5→     5→// Does NOT touch your existing NIFTY_SMART_MONEY_TERMINAL script
     6→     6→//
     7→     7→// HOW IT WORKS:
     8→     8→//   1. Reads RAW_OPTIONS sheet (populated by your main script every minute)
     9→     9→//   2. Compares last 2 snapshots to calculate per-strike OI + price changes
    10→    10→//   3. Applies the same Greek-powered 4-color flow logic
    11→    11→//   4. Writes 1 row/min to STRIKE_FLOW tab (24-col color table + NET FLOW in col M)
    12→    12→//   5. Writes 4 raw values per strike to SF_RAWFLOW tab (for Google Sheets charts)
    13→    13→//   6. Colors: CE Write=Light Red, CE Buy=Bright Green, PE Write=Light Green, PE Buy=Bright Red
    14→    14→//   7. Thick black border on highest single bet per minute
    15→    15→//
    16→    16→// SHEETS CREATED:
    17→    17→//   STRIKE_FLOW  — minute-by-minute color table with NET FLOW (rows grow all day)
    18→    18→//   SF_RAWFLOW   — raw 4 values per strike (for native Google Sheets stacked bar chart)
    19→    19→//   SF_TOTAL     — total flow across all strikes (4 cols: CE Buy, CE Write, PE Write, PE Buy)
    20→    20→//                  CE Write & PE Buy stored negative → select all data → Insert stacked bar chart
    21→    21→//
    22→    22→// SETUP (one-time):
    23→    23→//   1. In Apps Script editor, click + next to Files > New Script
    24→    24→//   2. Name it "STRIKE_FLOW_MAP"
    25→    25→//   3. Select ALL > Delete > Paste this ENTIRE file > Save
    26→    26→//   4. Set trigger: Triggers > Add Trigger > runStrikeFlowMap > Time-driven > Every minute
    27→    27→//      (offset by 15-20 sec so it runs AFTER your main script)
    28→    28→//

    32→    32→var STRIKE_COUNT = 11;

    38→    38→var EXPIRY_DAY = 2;        // 0=Sun, 1=Mon, 2=Tue, 3=Wed, 4=Thu, 5=Fri, 6=Sat
    39→    39→var ROLLOVER_HOUR = 12;    // Switch to next expiry after 12:00 IST on expiry day

    49→    49→function _sfNormTime(rawTime) {
    50→    50→  if (!rawTime) return "";
    51→    51→  if (rawTime instanceof Date)
    52→    52→    return Utilities.formatDate(rawTime, "Asia/Kolkata", "yyyy-MM-dd HH:mm:ss");
    53→    53→  return String(rawTime).trim();
    54→    54→}
    55→    55→
    56→    56→function _sfTimeOnly(rawTime) {
    57→    57→  if (!rawTime) return "";
    58→    58→  if (rawTime instanceof Date)
    59→    59→    return Utilities.formatDate(rawTime, "Asia/Kolkata", "HH:mm:ss");
    60→    60→  var s = String(rawTime).trim();
    61→    61→  var p = s.split(" ");
    62→    62→  return p.length > 1 ? p[p.length - 1] : s;
    63→    63→}
    64→    64→

    67→    67→}

    70→    70→// MAIN FUNCTION
    71→    71→// =============================================================================
    72→    72→function runStrikeFlowMap() {
    73→    73→
    74→    74→  // --- Market hours filter ---
    75→    75→  var now = new Date();
    76→    76→  var ist = new Date(now.toLocaleString("en-US", {timeZone: "Asia/Kolkata"}));
    77→    77→  var day = ist.getDay();
    78→    78→  if (day === 0 || day === 6) { Logger.log("[SF] Weekend"); return; }
    79→    79→  var mins = ist.getHours() * 60 + ist.getMinutes();
    80→    80→  if (mins < 555 || mins > 931) { Logger.log("[SF] Outside Market Hours"); return; }
    81→    81→
    82→    82→  // --- Read RAW_OPTIONS ---
    83→    83→  var ss = SpreadsheetApp.getActiveSpreadsheet();
    84→    84→  var rawSheet = ss.getSheetByName("RAW_OPTIONS");
    85→    85→  if (!rawSheet) { Logger.log("[SF] ERROR: RAW_OPTIONS not found"); return; }
    86→    86→
    87→    87→  var lastRow = rawSheet.getLastRow();
    88→    88→  if (lastRow < START_ROW + STRIKE_COUNT) {
    89→    89→    Logger.log("[SF] Not enough data. Rows: " + lastRow); return;
    90→    90→  }
    91→    91→
    92→    92→  var scanRows = Math.min(100, lastRow - START_ROW + 1);
    93→    93→  var scanStart = lastRow - scanRows + 1;
    94→    94→  var rawData = rawSheet.getRange(scanStart, 1, scanRows, 13).getValues();
    95→    95→
    96→    96→  // --- Find 2 unique timestamps ---
    97→    97→  var timestamps = [];
    98→    98→  for (var i = 0; i < rawData.length; i++) {
    99→    99→    var t = _sfNormTime(rawData[i][0]);
   100→   100→    if (t === "") continue;
   101→   101→    if (timestamps.length === 0 || timestamps[timestamps.length - 1] !== t)
   102→   102→      timestamps.push(t);
   103→   103→  }
   104→   104→  if (timestamps.length < 2) { Logger.log("[SF] Only 1 snapshot"); return; }
   105→   105→
   106→   106→  var prevTs = timestamps[timestamps.length - 2];
   107→   107→  var currTs = timestamps[timestamps.length - 1];
   108→   108→
   109→   109→  // --- Validate timestamps are within 2 minutes (skip if gap/spike) ---
   110→   110→  var prevTime = new Date(prevTs);
   111→   111→  var currTime = new Date(currTs);
   112→   112→  var tsGapMs = currTime.getTime() - prevTime.getTime();
   113→   113→  if (tsGapMs > 2 * 60 * 1000 || tsGapMs <= 0) {
   114→   114→    Logger.log("[SF] SKIP: Timestamp gap " + (tsGapMs / 1000).toFixed(0) + "s (>2 min). Prev: " + prevTs + " | Curr: " + currTs); return;
   115→   115→  }
   116→   116→
   117→   117→  var currTimeOnly = _sfTimeOnly(rawData[rawData.length - 1][0]);
   118→   118→  Logger.log("[SF] Prev: " + prevTs + " | Curr: " + currTs + " | Gap: " + (tsGapMs / 1000).toFixed(0) + "s");
   119→   119→
   120→   120→  // --- Smart expiry selection (auto-rollover on expiry day) ---
   121→   121→  var allExpiries = [];
   122→   122→  for (var i = 0; i < rawData.length; i++) {
   123→   123→    if (_sfNormTime(rawData[i][0]) === currTs) {
   124→   124→      var e = String(rawData[i][1]).trim();
   125→   125→      if (e !== "" && allExpiries.indexOf(e) === -1) allExpiries.push(e);
   126→   126→    }
   127→   127→  }
   128→   128→  if (allExpiries.length === 0) { Logger.log("[SF] ERROR: No expiry found"); return; }
   129→   129→
   130→   130→  // On expiry day after ROLLOVER_HOUR, use next expiry (index 1)
   131→   131→  var expiryIndex = 0;
   132→   132→  if (day === EXPIRY_DAY && ist.getHours() >= ROLLOVER_HOUR && allExpiries.length > 1) {
   133→   133→    expiryIndex = 1;
   134→   134→  }
   135→   135→  var expiry = allExpiries[expiryIndex];
   136→   136→  Logger.log("[SF] Expiries: " + allExpiries.join(", ") + " | Using: " + expiry +
   137→   137→    (expiryIndex === 1 ? " (ROLLOVER)" : " (normal)"));
   138→   138→
   139→   139→  // --- Parse prev & curr data ---
   140→   140→  var prevData = {}, currData = {};
   141→   141→  for (var i = 0; i < rawData.length; i++) {
   142→   142→    var t = _sfNormTime(rawData[i][0]);
   143→   143→    var exp = String(rawData[i][1]).trim();
   144→   144→    var strike = Number(rawData[i][2]);
   145→   145→    if (exp !== expiry || strike <= 0) continue;
   146→   146→
   147→   147→    var entry = {
   148→   148→      ceLTP: Number(rawData[i][3]) || 0,
   149→   149→      peLTP: Number(rawData[i][4]) || 0,
   150→   150→      ceOI:  Number(rawData[i][5]) || 0,
   151→   151→      peOI:  Number(rawData[i][6]) || 0,
   152→   152→      ceVol: Number(rawData[i][7]) || 0,
   153→   153→      peVol: Number(rawData[i][8]) || 0,
   154→   154→      ceDelta: Math.abs(Number(rawData[i][11]) || 0.5),
   155→   155→      peDelta: Math.abs(Number(rawData[i][12]) || 0.5)
   156→   156→    };
   157→   157→
   158→   158→    if (t === prevTs) prevData[strike] = entry;
   159→   159→    else if (t === currTs) currData[strike] = entry;
   160→   160→  }
   161→   161→
   162→   162→  var strikes = Object.keys(currData).map(Number).sort(function(a,b){return a-b;});
   163→   163→  if (strikes.length < STRIKE_COUNT) {
   164→   164→    Logger.log("[SF] Only " + strikes.length + " strikes"); return;
   165→   165→  }
   166→   166→  if (Object.keys(prevData).length < STRIKE_COUNT) {
   167→   167→    Logger.log("[SF] Prev snapshot incomplete"); return;
   168→   168→  }
   169→   169→
   170→   170→  // --- Get Nifty spot ---
   171→   171→  var niftySpot = 0;
   172→   172→  try {
   173→   173→    var extWb = _getExtWb();
   174→   174→    var dyn = extWb.getSheetByName("NIFTY50 DYNAMICS");
   175→   175→    if (dyn) {
   176→   176→      var dr = dyn.getLastRow();
   177→   177→      if (dr >= 5) {
   178→   178→        var spotVal = Number(dyn.getRange(dr, 2).getValue()) || 0;
   179→   179→        if (spotVal > 10000) niftySpot = spotVal;  // Nifty50 is 22000+, reject bad values
   180→   180→      }
   181→   181→    }
   182→   182→  } catch(e) { niftySpot = 0; }
   183→   183→  // Fallback — use midpoint of actual strikes from data
   184→   184→  if (niftySpot === 0 || niftySpot < 10000)
   185→   185→    niftySpot = strikes[Math.floor(strikes.length / 2)];
   186→   186→
   187→   187→  var liveATM = Math.round(niftySpot / 50) * 50;
   188→   188→  Logger.log("[SF] Spot: " + niftySpot + " | ATM: " + liveATM);
   189→   189→
   190→   190→  // --- Calculate 4-color flows ---
   191→   191→  var ceBuy = {}, ceWrite = {}, peBuy = {}, peWrite = {};
   192→   192→
   193→   193→  for (var si = 0; si < strikes.length; si++) {
   194→   194→    var stk = strikes[si];
   195→   195→    if (!prevData[stk]) continue;
   196→   196→    var c = currData[stk], p = prevData[stk];
   197→   197→    var dceOI = c.ceOI - p.ceOI;
   198→   198→    var dpeOI = c.peOI - p.peOI;
   199→   199→    var dceP = c.ceLTP - p.ceLTP;
   200→   200→    var dpeP = c.peLTP - p.peLTP;
   201→   201→
   202→   202→    // === CE Flow (v5: all 4 cases) ===
   203→   203→    if (dceOI > 0 && dceP >= 0)
   204→   204→      ceBuy[stk] = (dceOI * c.ceDelta * LOT_SIZE) / 10000000;
   205→   205→    else if (dceOI > 0 && dceP < 0)
   206→   206→      ceWrite[stk] = (dceOI * c.ceDelta * LOT_SIZE) / 10000000;
   207→   207→    else if (dceOI < 0 && dceP < 0)
   208→   208→      ceWrite[stk] = (Math.abs(dceOI) * c.ceDelta * LOT_SIZE * 0.3) / 10000000;
   209→   209→    else if (dceOI < 0 && dceP >= 0)
   210→   210→      ceWrite[stk] = (Math.abs(dceOI) * c.ceDelta * LOT_SIZE * 0.3) / 10000000;
   211→   211→
   212→   212→    // === PE Flow (v5: all 4 cases) ===
   213→   213→    if (dpeOI > 0 && dpeP >= 0)
   214→   214→      peBuy[stk] = (dpeOI * c.peDelta * LOT_SIZE) / 10000000;
   215→   215→    else if (dpeOI > 0 && dpeP < 0)
   216→   216→      peWrite[stk] = (dpeOI * c.peDelta * LOT_SIZE) / 10000000;
   217→   217→    else if (dpeOI < 0 && dpeP < 0)
   218→   218→      peWrite[stk] = (Math.abs(dpeOI) * c.peDelta * LOT_SIZE * 0.3) / 10000000;
   219→   219→    else if (dpeOI < 0 && dpeP >= 0)
   220→   220→      peBuy[stk] = (Math.abs(dpeOI) * c.peDelta * LOT_SIZE * 0.3) / 10000000;
   221→   221→
   222→   222→    // === v5: Volume fallback when OI unchanged ===
   223→   223→    if (!ceBuy[stk] && !ceWrite[stk] && (c.ceVol > 0)) {
   224→   224→      var vf = 0.4;
   225→   225→      if (dceP >= 0) ceBuy[stk] = (c.ceVol * c.ceDelta * LOT_SIZE * vf) / 10000000;
   226→   226→      else ceWrite[stk] = (c.ceVol * c.ceDelta * LOT_SIZE * vf) / 10000000;
   227→   227→    }
   228→   228→    if (!peBuy[stk] && !peWrite[stk] && (c.peVol > 0)) {
   229→   229→      var vf = 0.4;
   230→   230→      if (dpeP >= 0) peBuy[stk] = (c.peVol * c.peDelta * LOT_SIZE * vf) / 10000000;
   231→   231→      else peWrite[stk] = (c.peVol * c.peDelta * LOT_SIZE * vf) / 10000000;
   232→   232→    }
   233→   233→  }
   234→   234→
   235→   235→  // =========================
   236→   236→  // WRITE TO STRIKE_FLOW TAB
   237→   237→  // =========================
   238→   238→  var extWb = _getExtWb();
   239→   239→  var sfSheet = extWb.getSheetByName("STRIKE_FLOW");
   240→   240→  if (!sfSheet) {
   241→   241→    sfSheet = extWb.insertSheet("STRIKE_FLOW");
   242→   242→    _buildHeaders(sfSheet);
   243→   243→  }
   244→   244→
   245→   245→  var ceOff = [250,200,150,100,50,0,-50,-100,-150,-200,-250];
   246→   246→  var peOff = [-250,-200,-150,-100,-50,0,50,100,150,200,250];
   247→   247→
   248→   248→  // Update strike headers
   249→   249→  for (var ci = 0; ci < 11; ci++) {
   250→   250→    sfSheet.getRange(4, 2+ci).setValue((liveATM + ceOff[ci]) + " CE");
   251→   251→    sfSheet.getRange(4, 14+ci).setValue((liveATM + peOff[ci]) + " PE");
   252→   252→  }
   253→   253→
   254→   254→  // Build 24-col row
   255→   255→  var sfRow = new Array(24);
   256→   256→  sfRow[0] = currTimeOnly;
   257→   257→
   258→   258→  // Calculate NET FLOW for column M (index 12)
   259→   259→  // Positive = CE Buy + PE Write (bullish) | Negative = CE Write + PE Buy (bearish)
   260→   260→  var totalCEBuy = 0, totalCEWrite = 0, totalPEBuy = 0, totalPEWrite = 0;
   261→   261→  var allFlowKeys = Object.keys(ceBuy).concat(Object.keys(ceWrite))
   262→   262→    .concat(Object.keys(peBuy)).concat(Object.keys(peWrite));
   263→   263→  var seenStk = {};
   264→   264→  for (var k = 0; k < allFlowKeys.length; k++) {
   265→   265→    if (!seenStk[allFlowKeys[k]]) {
   266→   266→      seenStk[allFlowKeys[k]] = true;
   267→   267→      var s = Number(allFlowKeys[k]);
   268→   268→      totalCEBuy   += Number(ceBuy[s]   || 0);
   269→   269→      totalCEWrite += Number(ceWrite[s] || 0);
   270→   270→      totalPEBuy   += Number(peBuy[s]   || 0);
   271→   271→      totalPEWrite += Number(peWrite[s] || 0);
   272→   272→    }
   273→   273→  }
   274→   274→  var netFlow = (totalCEBuy + totalPEWrite) - (totalCEWrite + totalPEBuy);
   275→   275→  sfRow[12] = Number(netFlow.toFixed(2));
   276→   276→
   277→   277→  for (var ci = 0; ci < 11; ci++) {
   278→   278→    var stk = liveATM + ceOff[ci];
   279→   279→    var bv = Number(ceBuy[stk]||0), wv = Number(ceWrite[stk]||0);
   280→   280→    sfRow[ci+1] = (bv > 0 || wv > 0) ? (bv >= wv ? Number(bv.toFixed(2)) : Number(wv.toFixed(2))) : 0;
   281→   281→  }
   282→   282→  for (var pi = 0; pi < 11; pi++) {
   283→   283→    var stk = liveATM + peOff[pi];
   284→   284→    var bv = Number(peBuy[stk]||0), wv = Number(peWrite[stk]||0);
   285→   285→    sfRow[pi+13] = (bv > 0 || wv > 0) ? (wv >= bv ? Number(wv.toFixed(2)) : Number(bv.toFixed(2))) : 0;
   286→   286→  }
   287→   287→
   288→   288→  // De-dup same minute
   289→   289→  var sfLast = sfSheet.getLastRow();
   290→   290→  var target = (sfLast < SF_DATA_START) ? SF_DATA_START : sfLast + 1;
   291→   291→  if (sfLast >= SF_DATA_START) {
   292→   292→    var times = sfSheet.getRange(SF_DATA_START, 1, sfLast - SF_DATA_START + 1, 1).getValues();
   293→   293→    for (var si = 0; si < times.length; si++) {
   294→   294→      if (times[si][0]) {
   295→   295→        var ft = (times[si][0] instanceof Date) ?
   296→   296→          Utilities.formatDate(times[si][0], "Asia/Kolkata", "HH:mm:ss") : times[si][0].toString().trim();
   297→   297→        if (ft === currTimeOnly) { target = si + SF_DATA_START; break; }
   298→   298→      }
   299→   299→    }
   300→   300→  }
   301→   301→
   302→   302→  sfSheet.getRange(target, 1, 1, 24).setValues([sfRow]);
   303→   303→
   304→   304→  // --- Color coding ---
   305→   305→  var CE_W = "#FFCCCC", CE_B = "#00B050", PE_W = "#C6EFCE", PE_B = "#BD2130";
   306→   306→  var DARK = "#1a1a2e", ZERO = "#555555";
   307→   307→
   308→   308→  sfSheet.getRange(target, 1).setBackground(DARK).setFontColor("#a8d8ea").setFontSize(9);
   309→   309→
   310→   310→  // Column M: NET FLOW color — green if bullish, red if bearish
   311→   311→  var netCell = sfSheet.getRange(target, 13);
   312→   312→  if (netFlow > 0)
   313→   313→    netCell.setBackground("#00B050").setFontColor("#ffffff").setFontWeight("bold");
   314→   314→  else if (netFlow < 0)
   315→   315→    netCell.setBackground("#BD2130").setFontColor("#ffffff").setFontWeight("bold");
   316→   316→  else
   317→   317→    netCell.setBackground(DARK).setFontColor(ZERO).setFontWeight("normal");
   318→   318→  netCell.setHorizontalAlignment("center").setFontSize(9);
   319→   319→
   320→   320→  var maxVal = 0, maxCol = 0;
   321→   321→
   322→   322→  // CE columns B-L
   323→   323→  for (var ci = 0; ci < 11; ci++) {
   324→   324→    var stk = liveATM + ceOff[ci];
   325→   325→    var bv = Number(ceBuy[stk]||0), wv = Number(ceWrite[stk]||0);
   326→   326→    var cell = sfSheet.getRange(target, 2+ci);
   327→   327→    if (bv <= 0 && wv <= 0)
   328→   328→      cell.setBackground(DARK).setFontColor(ZERO).setFontWeight("normal");
   329→   329→    else if (bv >= wv)
   330→   330→      cell.setBackground(CE_B).setFontColor("#ffffff").setFontWeight("bold");
   331→   331→    else
   332→   332→      cell.setBackground(CE_W).setFontColor("#9c0006").setFontWeight("bold");
   333→   333→    cell.setHorizontalAlignment("center").setFontSize(9);
   334→   334→    var dom = Math.max(bv, wv);
   335→   335→    if (dom > maxVal) { maxVal = dom; maxCol = 2+ci; }
   336→   336→  }
   337→   337→
   338→   338→  // PE columns N-X
   339→   339→  for (var pi = 0; pi < 11; pi++) {
   340→   340→    var stk = liveATM + peOff[pi];
   341→   341→    var bv = Number(peBuy[stk]||0), wv = Number(peWrite[stk]||0);
   342→   342→    var cell = sfSheet.getRange(target, 14+pi);
   343→   343→    if (bv <= 0 && wv <= 0)
   344→   344→      cell.setBackground(DARK).setFontColor(ZERO).setFontWeight("normal");
   345→   345→    else if (wv >= bv)
   346→   346→      cell.setBackground(PE_W).setFontColor("#006100").setFontWeight("bold");
   347→   347→    else
   348→   348→      cell.setBackground(PE_B).setFontColor("#ffffff").setFontWeight("bold");
   349→   349→    cell.setHorizontalAlignment("center").setFontSize(9);
   350→   350→    var dom = Math.max(bv, wv);
   351→   351→    if (dom > maxVal) { maxVal = dom; maxCol = 14+pi; }
   352→   352→  }
   353→   353→
   354→   354→  // Thick black border on highest bet
   355→   355→  if (maxVal > 0) {
   356→   356→    sfSheet.getRange(target, maxCol).setBorder(
   357→   357→      true, true, true, true, null, null, "#000000", SpreadsheetApp.BorderStyle.SOLID_THICK
   358→   358→    );
   359→   359→  }
   360→   360→
   361→   361→  Logger.log("[SF] Row " + target + " | Net: " + netFlow.toFixed(2) + " Cr | Max: " + maxVal.toFixed(2) + " Cr | Spot: " + niftySpot);
   362→   362→
   363→   363→  // =========================
   364→   364→  // WRITE TO SF_RAWFLOW TAB
   365→   365→  // =========================
   366→   366→  _writeRawFlow(ceBuy, ceWrite, peBuy, peWrite, liveATM, currTimeOnly);
   367→   367→
   368→   368→  // =========================
   369→   369→  // WRITE TO SF_TOTAL TAB
   370→   370→  // =========================
   371→   371→  _writeTotalFlow(ceBuy, ceWrite, peBuy, peWrite, currTimeOnly);
   372→   372→}

   378→   378→// Layout: 45 columns total
   379→   379→//   Col A: TIME
   380→   380→//   Cols B-E: ATM-250 -> CE Buy | CE Write | PE Write | PE Buy
   381→   381→//   Cols F-I: ATM-200 -> CE Buy | CE Write | PE Write | PE Buy
   382→   382→//   ... (11 strikes x 4 sub-cols = 44 data cols)
   383→   383→//   CE Write and PE Buy stored as NEGATIVE numbers for charting.
   384→   384→//   User can select all data -> Insert -> Stacked bar chart in Google Sheets.
   385→   385→//   Green bars go UP (CE Buy + PE Write), Red bars go DOWN (CE Write + PE Buy).
   386→   386→// =============================================================================
   387→   387→var RF_DATA_START = 7;
   388→   388→
   389→   389→function _writeRawFlow(ceBuy, ceWrite, peBuy, peWrite, liveATM, currTimeOnly) {
   390→   390→  var extWb = _getExtWb();
   391→   391→  var off = [-250,-200,-150,-100,-50,0,50,100,150,200,250];
   392→   392→  var TOTAL_COLS = 1 + 11 * 4; // 45
   393→   393→
   394→   394→  var rf = extWb.getSheetByName("SF_RAWFLOW");
   395→   395→  if (!rf) {
   396→   396→    rf = extWb.insertSheet("SF_RAWFLOW");
   397→   397→    _buildRawFlowHeaders(rf, liveATM, off);
   398→   398→  }
   399→   399→
   400→   400→  // Update strike labels in row 1 (batch — no merge, avoids timeout on external sheet)
   401→   401→  var hdrRow = ["TIME"];
   402→   402→  for (var i = 0; i < 11; i++) {
   403→   403→    var stk = liveATM + off[i];
   404→   404→    var label = off[i] === 0 ? "ATM " + stk : String(stk);
   405→   405→    hdrRow.push(label, label, label, label);
   406→   406→  }
   407→   407→  rf.getRange(1, 1, 1, 45).setValues([hdrRow]);
   408→   408→
   409→   409→  // Build 45-col row
   410→   410→  var row = new Array(TOTAL_COLS);
   411→   411→  row[0] = currTimeOnly;
   412→   412→
   413→   413→  for (var i = 0; i < 11; i++) {
   414→   414→    var stk = liveATM + off[i];
   415→   415→    var b = 1 + i * 4; // 0-indexed position in row array
   416→   416→    row[b]     =  Number((ceBuy[stk]   || 0).toFixed(2)); // CE Buy   (positive)
   417→   417→    row[b + 1] = -Number((ceWrite[stk] || 0).toFixed(2)); // CE Write (negative)
   418→   418→    row[b + 2] =  Number((peWrite[stk] || 0).toFixed(2)); // PE Write (positive)
   419→   419→    row[b + 3] = -Number((peBuy[stk]   || 0).toFixed(2)); // PE Buy   (negative)
   420→   420→  }
   421→   421→
   422→   422→  // De-dup same minute
   423→   423→  var lastR = rf.getLastRow();
   424→   424→  var target = (lastR < RF_DATA_START) ? RF_DATA_START : lastR + 1;
   425→   425→  if (lastR >= RF_DATA_START) {
   426→   426→    var times = rf.getRange(RF_DATA_START, 1, lastR - RF_DATA_START + 1, 1).getValues();
   427→   427→    for (var si = 0; si < times.length; si++) {
   428→   428→      if (times[si][0]) {
   429→   429→        var ft = (times[si][0] instanceof Date) ?
   430→   430→          Utilities.formatDate(times[si][0], "Asia/Kolkata", "HH:mm:ss") : times[si][0].toString().trim();
   431→   431→        if (ft === currTimeOnly) { target = si + RF_DATA_START; break; }
   432→   432→      }
   433→   433→    }
   434→   434→  }
   435→   435→
   436→   436→  rf.getRange(target, 1, 1, TOTAL_COLS).setValues([row]);
   437→   437→
   438→   438→  // Color coding — use 4 batch getRangedList calls (non-contiguous but single API call each)
   439→   439→  // Google Sheets supports getRangeList for styling multiple non-adjacent ranges at once
   440→   440→  var DARK = "#1a1a2e", ZERO = "#555555";
   441→   441→  rf.getRange(target, 1).setBackground(DARK).setFontColor("#a8d8ea").setFontSize(8);
   442→   442→
   443→   443→  // Collect ranges per flow type, then batch-style
   444→   444→  var cbCols = [], cwCols = [], pwCols = [], pbCols = [];
   445→   445→  for (var i = 0; i < 11; i++) {
   446→   446→    var stk = liveATM + off[i];
   447→   447→    var base = 2 + i * 4;
   448→   448→    var cb = Number(ceBuy[stk]||0), cw = Number(ceWrite[stk]||0);
   449→   449→    var pw = Number(peWrite[stk]||0), pb = Number(peBuy[stk]||0);
   450→   450→    if (cb > 0) cbCols.push("R" + target + "C" + base);
   451→   451→    else rf.getRange(target, base).setBackground(DARK).setFontColor(ZERO);
   452→   452→    if (cw > 0) cwCols.push("R" + target + "C" + (base+1));
   453→   453→    else rf.getRange(target, base+1).setBackground(DARK).setFontColor(ZERO);
   454→   454→    if (pw > 0) pwCols.push("R" + target + "C" + (base+2));
   455→   455→    else rf.getRange(target, base+2).setBackground(DARK).setFontColor(ZERO);
   456→   456→    if (pb > 0) pbCols.push("R" + target + "C" + (base+3));
   457→   457→    else rf.getRange(target, base+3).setBackground(DARK).setFontColor(ZERO);
   458→   458→  }
   459→   459→
   460→   460→  // Batch style non-zero cells by type (each is 1 API call regardless of how many cells)
   461→   461→  if (cbCols.length > 0) rf.getRangeList(cbCols).setBackground("#00B050").setFontColor("#fff").setFontSize(8).setHorizontalAlignment("center");
   462→   462→  if (cwCols.length > 0) rf.getRangeList(cwCols).setBackground("#FFCCCC").setFontColor("#9c0006").setFontSize(8).setHorizontalAlignment("center");
   463→   463→  if (pwCols.length > 0) rf.getRangeList(pwCols).setBackground("#C6EFCE").setFontColor("#006100").setFontSize(8).setHorizontalAlignment("center");
   464→   464→  if (pbCols.length > 0) rf.getRangeList(pbCols).setBackground("#BD2130").setFontColor("#fff").setFontSize(8).setHorizontalAlignment("center");
   465→   465→
   466→   466→  Logger.log("[SF] SF_RAWFLOW row " + target + " written");
   467→   467→}

   509→   509→// SF_TOTAL — Total flow across all 11 strikes, 4 values per minute

   512→   512→// CE Write & PE Buy stored as NEGATIVE so stacked bar chart shows red going DOWN.

   517→   517→function _writeTotalFlow(ceBuy, ceWrite, peBuy, peWrite, currTimeOnly) {
   518→   518→  var extWb = _getExtWb();
   519→   519→  var TOTAL_COLS = 5;
   520→   520→
   521→   521→  // Sum across all strikes
   522→   522→  var sumCB = 0, sumCW = 0, sumPW = 0, sumPB = 0;
   523→   523→  var allKeys = Object.keys(ceBuy).concat(Object.keys(ceWrite))
   524→   524→    .concat(Object.keys(peWrite)).concat(Object.keys(peBuy));
   525→   525→  var seen = {};
   526→   526→  for (var k = 0; k < allKeys.length; k++) {
   527→   527→    if (!seen[allKeys[k]]) {
   528→   528→      seen[allKeys[k]] = true;
   529→   529→      var s = Number(allKeys[k]);
   530→   530→      sumCB += Number(ceBuy[s]   || 0);
   531→   531→      sumCW += Number(ceWrite[s] || 0);
   532→   532→      sumPW += Number(peWrite[s] || 0);
   533→   533→      sumPB += Number(peBuy[s]   || 0);
   534→   534→    }
   535→   535→  }
   536→   536→
   537→   537→  var ft = extWb.getSheetByName("SF_TOTAL");
   538→   538→  if (!ft) {
   539→   539→    ft = extWb.insertSheet("SF_TOTAL");
   540→   540→    _buildTotalHeaders(ft);
   541→   541→  }
   542→   542→
   543→   543→  // Build 5-col row
   544→   544→  var row = [
   545→   545→    currTimeOnly,
   546→   546→    Number(sumCB.toFixed(2)),              // CE Buy   (positive)
   547→   547→    -Number(sumCW.toFixed(2)),              // CE Write (negative)
   548→   548→    Number(sumPW.toFixed(2)),              // PE Write (positive)
   549→   549→    -Number(sumPB.toFixed(2))               // PE Buy   (negative)
   550→   550→  ];
   551→   551→
   552→   552→  // De-dup same minute
   553→   553→  var lastR = ft.getLastRow();
   554→   554→  var target = (lastR < FT_DATA_START) ? FT_DATA_START : lastR + 1;
   555→   555→  if (lastR >= FT_DATA_START) {
   556→   556→    var times = ft.getRange(FT_DATA_START, 1, lastR - FT_DATA_START + 1, 1).getValues();

   558→   558→      if (times[si][0]) {
   559→   559→        var tstr = (times[si][0] instanceof Date) ?
   560→   560→          Utilities.formatDate(times[si][0], "Asia/Kolkata", "HH:mm:ss") : times[si][0].toString().trim();
   561→   561→        if (tstr === currTimeOnly) { target = si + FT_DATA_START; break; }
   562→   562→      }
   563→   563→    }
   564→   564→  }
   565→   565→
   566→   566→  ft.getRange(target, 1, 1, TOTAL_COLS).setValues([row]);
   567→   567→

   575→   575→  else
   576→   576→    ft.getRange(target, 2).setBackground(DARK).setFontColor(ZERO);

   582→   582→  else
   583→   583→    ft.getRange(target, 3).setBackground(DARK).setFontColor(ZERO);

   589→   589→  else

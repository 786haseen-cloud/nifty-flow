     1→/**
     2→ * Zerodha Kite API Integration
     3→ *
     4→ * Connects to Kite Connect REST API for real NSE/BSE market data.
     5→ * When KITE_API_KEY + KITE_ACCESS_TOKEN are set in .env, real data flows.
     6→ * Otherwise, falls back to demo data.
     7→ *
     8→ * Setup:
     9→ * 1. Go to https://developers.kite.trade
    10→ * 2. Create an app → get API Key + Secret
    11→ * 3. Login: https://kite.zerodha.com/connect/login?api_key=YOUR_KEY
    12→ * 4. After login, capture request_token from redirect URL
    13→ * 5. Call /api/kite/auth?request_token=xxx to generate access_token
    14→ * 6. Put access_token in .env → KITE_ACCESS_TOKEN=xxx
    15→ * 7. Redeploy
    16→ *
    17→ * Access tokens expire daily at midnight. Re-login each morning.
    18→ */
    19→

    31→export function setKiteOverride(apiKey?: string, accessToken?: string): void {

    36→export function isKiteConfigured(): boolean {
    37→  return !!(
    38→    (process.env.KITE_API_KEY && process.env.KITE_ACCESS_TOKEN) ||
    39→    (_overrideApiKey && _overrideAccessToken)
    40→  );
    41→}

    43→function kiteHeaders() {

    46→  return {
    47→    'Authorization': `token ${apiKey}:${accessToken}`,
    48→    'X-Kite-Version': '3',
    49→    'Content-Type': 'application/x-www-form-urlencoded',
    50→  };
    51→}

   100→/**
   101→ * Generate access_token from request_token (one-time, after login redirect)
   102→ */
   103→export async function generateSession(requestToken: string): Promise<{
   104→  accessToken: string;
   105→  refreshToken: string;
   106→} | null> {
   107→  const apiKey = process.env.KITE_API_KEY;
   108→  const apiSecret = process.env.KITE_API_SECRET;
   109→  if (!apiKey || !apiSecret) return null;
   110→
   111→  try {
   112→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   113→    const { createHash } = await import('crypto');
   114→    const checksum = createHash('sha256')
   115→      .update(apiKey + requestToken + apiSecret)
   116→      .digest('hex');
   117→
   118→    const res = await fetch(`${KITE_BASE}/session/token`, {
   119→      method: 'POST',
   120→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   121→      body: new URLSearchParams({
   122→        api_key: apiKey,
   123→        request_token: requestToken,
   124→        checksum,
   125→      }).toString(),
   126→    });
   127→
   128→    const data = await res.json();
   129→    if (data.status === 'success' && data.data) {
   130→      return {
   131→        accessToken: data.data.access_token,
   132→        refreshToken: data.data.refresh_token || '',
   133→      };
   134→    }
   135→    console.error('[Kite] Session error:', data.message);
   136→    return null;
   137→  } catch (err) {
   138→    console.error('[Kite] generateSession failed:', err);
   139→    return null;
   140→  }
   141→}

   145→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   146→let instrumentsCache: KiteInstrument[] | null = null;
   147→let instrumentsCacheTime = 0;
   148→
   149→/**
   150→ * Invalidate the in-memory instruments cache.
   151→ * Use after credential changes (e.g. user switched from expired → valid token)
   152→ * to force a fresh CSV download on the next getInstruments() call.
   153→ */
   154→export function invalidateInstrumentsCache(): void {
   155→  instrumentsCache = null;
   156→  instrumentsCacheTime = 0;
   157→}
   158→
   159→/**
   160→ * Fetch all instruments from Kite (cached for 1 hour)
   161→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   162→ * We filter by exchange after caching
   163→ *
   164→ * @param exchange Optional exchange filter (e.g. 'NFO' returns only NFO F&O instruments)
   165→ * @param forceRefresh Bypass the 1-hour cache and re-download. Useful when creds change.
   166→ */
   167→export async function getInstruments(exchange?: string, forceRefresh?: boolean): Promise<KiteInstrument[]> {
   168→  // Return from cache if fresh (and not forced to refresh)
   169→  if (!forceRefresh && instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   170→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   171→  }
   172→
   173→  try {
   174→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   175→    if (!res.ok) {
   176→      const errText = await res.text();
   177→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   178→      return [];
   179→    }
   180→    const text = await res.text();
   181→
   182→    // Kite returns CSV format
   183→    const lines = text.trim().split('\n');
   184→    const instruments: KiteInstrument[] = [];
   185→
   186→    // Kite CSV column order (2025+ format):
   187→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   188→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   189→    //
   190→    // Kite changed the CSV format in 2025:
   191→    //   OLD: instrument_type was 'OPTIDX'/'OPTSTK'/'FUTIDX'/'FUTSTK'/'EQ'/'INDEX'
   192→    //        segment was 'NFO'/'BFO'/'NSE'/'BSE'
   193→    //   NEW: instrument_type is 'CE'/'PE'/'FUT'/'EQ'
   194→    //        segment is 'NFO-OPT'/'NFO-FUT'/'NFO'/'BFO-OPT'/'BFO-FUT'/'NSE'/'BSE'
   195→    // To avoid breaking the rest of the code (which filters by OPTIDX/OPTSTK/FUTIDX/FUTSTK),
   196→    // we normalize the new format back to the legacy types.
   197→    //
   198→    // Disambiguation: in the new CSV, ALL options (index + stock) are CE/PE — there is no
   199→    // index-vs-stock hint in the row itself. We use the `name` column to decide:
   200→    //   - If the underlying name matches a known F&O index (NIFTY, BANKNIFTY, FINNIFTY,
   201→    //     SENSEX, BANKEX, MIDCPNIFTY, etc.) → OPTIDX / FUTIDX
   202→    //   - Otherwise (RELIANCE, TCS, HDFCBANK, ...) → OPTSTK / FUTSTK
   203→    // The set below is checked at parse time (cheap Set.has lookup).
   204→    const KNOWN_FNO_INDICES = new Set([
   205→      'NIFTY', 'BANKNIFTY', 'FINNIFTY', 'SENSEX', 'BANKEX',
   206→      'MIDCPNIFTY', 'NIFTY MID SELECT', 'NIFTY PSE',
   207→    ]);
   208→    const normalizeInstrumentType = (rawType: string, segment: string, name: string): string => {
   209→      // Keep legacy values as-is (in case Kite reverts or some entries still use old format)
   210→      if (rawType === 'OPTIDX' || rawType === 'OPTSTK' || rawType === 'FUTIDX' || rawType === 'FUTSTK' ||
   211→          rawType === 'EQ' || rawType === 'INDEX') {
   212→        return rawType;
   213→      }
   214→      const nameUpper = name.toUpperCase();
   215→      const isIndex = KNOWN_FNO_INDICES.has(nameUpper);
   216→      if (rawType === 'CE' || rawType === 'PE') {
   217→        // BFO always has index options only (SENSEX/BANKEX). NFO has both index + stock.
   218→        return isIndex ? 'OPTIDX' : 'OPTSTK';
   219→      }
   220→      if (rawType === 'FUT') {
   221→        return isIndex ? 'FUTIDX' : 'FUTSTK';
   222→      }
   223→      return rawType;
   224→    };
   225→
   226→    for (let i = 1; i < lines.length; i++) { // skip header
   227→      const cols = lines[i].split(',');
   228→      if (cols.length < 12) continue;
   229→      const rawType = cols[9] || '';
   230→      const segment = cols[10] || '';
   231→      // Kite wraps the `name` field in double quotes (e.g. "NIFTY").
   232→      // Strip them so name-based filters (e.g. .includes('NIFTY')) work correctly.
   233→      const rawName = cols[3] || '';
   234→      const name = rawName.startsWith('"') && rawName.endsWith('"')
   235→        ? rawName.slice(1, -1)
   236→        : rawName;
   237→      instruments.push({
   238→        instrumentToken: parseInt(cols[0]) || 0,
   239→        exchangeToken: parseInt(cols[1]) || 0,
   240→        tradingSymbol: cols[2],
   241→        name,
   242→        exchange: cols[11] || '',
   243→        segment,
   244→        instrumentType: normalizeInstrumentType(rawType, segment, name),
   245→        strike: parseFloat(cols[6]) || 0,
   246→        lotSize: parseInt(cols[8]) || 1,
   247→        expiry: cols[5] || '',
   248→      });
   249→    }
   250→
   251→    instrumentsCache = instruments;
   252→    instrumentsCacheTime = Date.now();
   253→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   254→  } catch (err) {
   255→    const errMsg = err instanceof Error ? err.message : String(err);
   256→    console.error('[Kite] instruments fetch failed:', errMsg);
   257→    // Return empty but set error flag for callers to detect
   258→    return [];
   259→  }
   260→}

   264→/**
   265→ * Get real-time quotes for instruments
   266→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   267→ */
   268→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   269→  if (!isKiteConfigured()) return {};
   270→
   271→  try {
   272→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   273→    // Tokens have no encoding issues on any platform.
   274→    // Build a map from token → original instrument key for response mapping.
   275→    const tokenToKey: Record<string, string> = {};
   276→    const tokenList: string[] = [];
   277→
   278→    for (const inst of instruments) {
   279→      // If it's a numeric token, use directly
   280→      if (/^\d+$/.test(inst)) {
   281→        tokenToKey[inst] = inst;
   282→        tokenList.push(inst);
   283→      } else {
   284→        // Look up token from KITE_INDEX_INSTRUMENTS
   285→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   286→          ([, v]) => v.symbol === inst
   287→        );
   288→        if (found) {
   289→          const tok = String(found[1].token);
   290→          tokenToKey[tok] = inst; // map token back to symbol name
   291→          tokenList.push(tok);
   292→        } else {
   293→          // Fallback: use the instrument string as-is with encodeURI
   294→          tokenToKey[inst] = inst;
   295→          tokenList.push(inst);
   296→        }
   297→      }
   298→    }
   299→
   300→    const iList = tokenList.join(',');
   301→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   302→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   303→    const params = tokenList.map(t => `i=${t}`).join('&');
   304→    const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   305→      headers: kiteHeaders(),
   306→    });
   307→
   308→    // Log non-success for debugging
   309→    if (!res.ok) {
   310→      const errText = await res.text();
   311→      console.error(`[Kite] quote API ${res.status}: ${errText}`);
   312→      return { _error: `HTTP ${res.status}: ${errText}` } as any;
   313→    }
   314→
   315→    const data = await res.json();
   316→    if (data.status !== 'success' || !data.data) {
   317→      console.error('[Kite] quote API error:', JSON.stringify(data));
   318→      return { _error: data.message || 'Unknown error' } as any;
   319→    }
   320→
   321→    const quotes: Record<string, KiteQuote> = {};
   322→    for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   323→      // Kite returns the key as the token string (e.g. "256265")
   324→      // Map it back to the original instrument name (e.g. "NSE:NIFTY 50")
   325→      const displayKey = tokenToKey[key] || key;
   326→      quotes[displayKey] = {
   327→        instrumentToken: q.instrument_token || 0,
   328→        lastPrice: q.last_price || 0,
   329→        open: q.ohlc?.open || 0,
   330→        high: q.ohlc?.high || 0,
   331→        low: q.ohlc?.low || 0,
   332→        close: q.ohlc?.close || 0,
   333→        volume: q.volume || 0,
   334→        netChange: q.net_change || 0,
   335→        averagePrice: q.average_price || 0,
   336→        oi: q.oi || 0,
   337→        oiDayHigh: q.oi_day_high || 0,
   338→        oiDayLow: q.oi_day_low || 0,
   339→        dayHigh: q.ohlc?.high || 0,
   340→        dayLow: q.ohlc?.low || 0,
   341→      };
   342→    }
   343→    return quotes;
   344→  } catch (err) {
   345→    const errMsg = err instanceof Error ? err.message : String(err);
   346→    console.error('[Kite] quote error:', errMsg);
   347→    return { _error: `Exception: ${errMsg}` } as any;
   348→  }
   349→}

   353→/**
   354→ * Get historical candle data for price line chart
   355→ * instrumentToken: numeric token from instruments list
   356→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   357→ */
   358→export async function getCandles(
   359→  instrumentToken: number,
   360→  interval: string = '15minute',
   361→  days: number = 1,
   362→): Promise<KiteHistoricalCandle[]> {
   363→  if (!isKiteConfigured()) return [];
   364→
   365→  // Kite historical API expects IST dates.
   366→  // On Vercel (UTC), we must convert to IST before formatting.
   367→  const IST_OFFSET = 5.5 * 60 * 60 * 1000;
   368→  const toIST = (d: Date) => new Date(d.getTime() + IST_OFFSET);
   369→
   370→  const toDate = toIST(new Date());
   371→  const fromDate = toIST(new Date());
   372→  fromDate.setDate(fromDate.getDate() - days);
   373→
   374→  // Kite historical API requires format "YYYY-MM-DD HH:MM:SS" (WITH seconds).
   375→  // Sending "YYYY-MM-DD HH:MM" (no seconds) returns HTTP 400 "invalid from date".
   376→  const fmt = (d: Date) =>
   377→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}:00`;
   378→
   379→  try {
   380→    const fromStr = fmt(fromDate);
   381→    const toStr = fmt(toDate);
   382→    // URL-encode the date strings (they contain a space: "2026-08-25 09:15")
   383→    const url = `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${encodeURIComponent(fromStr)}&to=${encodeURIComponent(toStr)}&continuous=0`;
   384→
   385→    const res = await fetch(url, { headers: kiteHeaders() });
   386→
   387→    // Surface HTTP-level errors (403 expired token, 429 rate limit, etc.)
   388→    if (!res.ok) {
   389→      const errText = await res.text();
   390→      console.error(`[Kite] candles HTTP ${res.status} for token ${instrumentToken}: ${errText.substring(0, 300)}`);
   391→      return [];
   392→    }
   393→
   394→    const data = await res.json();
   395→    if (data.status !== 'success' || !data.data?.candles) {
   396→      console.error('[Kite] candles non-success: status=%s, message=%s, from=%s, to=%s',
   397→        data.status, data.message || 'no message', fromStr, toStr);
   398→      return [];
   399→    }
   400→
   401→    return (data.data.candles as any[][]).map(c => ({
   402→      timestamp: c[0],
   403→      open: c[1],
   404→      high: c[2],
   405→      low: c[3],
   406→      close: c[4],
   407→      volume: c[5],
   408→      oi: c[6] || 0,  // c[6] = OI (F&O only), undefined for cash
   409→    }));
   410→  } catch (err) {
   411→    const errMsg = err instanceof Error ? err.message : String(err);
   412→    console.error('[Kite] candles error:', errMsg);
   413→    return [];
   414→  }
   415→}

   419→/**
   420→ * Get option chain for an index or stock
   421→ * Uses InstrumentSpec for correct exchange, segment
   422→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   423→ * Returns all CE + PE instruments for the current expiry near ATM
   424→ */
   425→export async function getOptionInstruments(
   426→  symbol: string,
   427→  spotPrice: number,
   428→  strikesAroundOverride?: number,
   429→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   430→  const spec = getInstrumentSpec(symbol);
   431→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   432→
   433→  // IMPORTANT: F&O options/futures are NOT on the cash exchange.
   434→  //   - NIFTY index spot is on exchange='NSE', but NIFTY OPTIONS are on exchange='NFO'
   435→  //   - SENSEX spot is on exchange='BSE', but SENSEX OPTIONS are on exchange='BFO'
   436→  //   - Stock (RELIANCE/TCS/...) spot is on exchange='NSE', but their OPTIONS are on exchange='NFO'
   437→  // So we use spec.segment ('NFO' or 'BFO') — which equals the F&O exchange column in
   438→  // Kite's CSV — NOT spec.exchange ('NSE'/'BSE', which is the cash exchange).
   439→  // If you call getInstruments(spec.exchange) you'll only get cash/index instruments,
   440→  // and the option filter below will return ZERO matches.
   441→  const instruments = await getInstruments(spec.segment);
   442→
   443→  // Filter options using spec.
   444→  //
   445→  // Kite changed their CSV format in 2025:
   446→  //   OLD: segment='NFO'/'BFO', instrument_type='OPTIDX'/'OPTSTK'
   447→  //   NEW: segment='NFO-OPT'/'BFO-OPT', instrument_type='CE'/'PE' (we normalize to OPTIDX)
   448→  //
   449→  // So we use a "starts with" match on segment (so 'NFO' matches both 'NFO' and 'NFO-OPT'),
   450→  // and an equality match on instrumentType (we've already normalized new → legacy).
   451→  const symUpper = symbol.toUpperCase();
   452→  const indexOptions = instruments.filter(i =>
   453→    i.segment.startsWith(spec.segment) &&
   454→    i.instrumentType === spec.instrumentType &&
   455→    (i.name.toUpperCase().includes(symUpper) ||
   456→     i.tradingSymbol.toUpperCase().includes(symUpper))
   457→  );
   458→
   459→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   460→
   461→  // Get unique expiries, sort by nearest
   462→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   463→
   464→  // Use nearest expiry that's not expired
   465→  const today = new Date();
   466→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   467→
   468→  // Filter to nearest expiry
   469→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   470→
   471→  // Get lot size from first option (all same for same underlying)
   472→  const lotSize = expiryOptions[0]?.lotSize || 1;
   473→
   474→  // Derive strike step DYNAMICALLY from actual strike gaps
   475→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   476→  let strikeStep = 50;
   477→  if (strikes.length >= 2) {
   478→    const gaps: Record<number, number> = {};
   479→    for (let i = 1; i < strikes.length; i++) {
   480→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   481→      gaps[gap] = (gaps[gap] || 0) + 1;
   482→    }
   483→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   484→  }
   485→
   486→  // Round ATM strike using the DYNAMIC strike step
   487→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   488→
   489→  // Get strikes around ATM
   490→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   491→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   492→    atmStrike - strikesAround * strikeStep + i * strikeStep
   493→  );
   494→
   495→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   496→
   497→  return {
   498→    instruments: filtered,
   499→    meta: { lotSize, strikeStep },
   500→  };
   501→}

   503→/**
   504→ * Build options flow data from quotes + OI changes
   505→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   506→ */
   507→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   508→  const spec = getInstrumentSpec(symbol);
   509→  if (!spec) return null;
   510→
   511→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   512→  if (optInstruments.length === 0) return null;
   513→
   514→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   515→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   516→
   517→  // Batch fetch quotes
   518→  const quotes = await getQuotes(iKeys);
   519→  if (Object.keys(quotes).length === 0) return null;
   520→
   521→  // Group by strike → build flow data
   522→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   523→
   524→  // Use DYNAMIC strike step and lot size from CSV
   525→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   526→
   527→  for (const inst of optInstruments) {
   528→    // Quotes are keyed by instrument token (string)
   529→    const key = String(inst.instrumentToken);
   530→    const q = quotes[key];
   531→    if (!q) continue;
   532→
   533→    if (!strikes.has(inst.strike)) {
   534→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   535→    }
   536→    const s = strikes.get(inst.strike)!;
   537→
   538→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   539→    const flow = q.volume * meta.lotSize * q.averagePrice;
   540→
   541→    if (inst.tradingSymbol.endsWith('CE')) {
   542→      s.callBuy = flow * 0.6;
   543→      s.callWrite = flow * 0.4;
   544→      s.callOI = q.oi;
   545→      s.callVol = q.volume;
   546→    } else if (inst.tradingSymbol.endsWith('PE')) {
   547→      s.putBuy = flow * 0.6;
   548→      s.putWrite = flow * 0.4;
   549→      s.putOI = q.oi;
   550→      s.putVol = q.volume;
   551→    }
   552→  }
   553→
   554→  // Build result
   555→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   556→    strike,
   557→    isATM: strike === atmStrike,
   558→    ...flow,
   559→    bullishFlow: flow.callBuy + flow.putWrite,
   560→    bearishFlow: flow.putBuy + flow.callWrite,
   561→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   562→  })).sort((a, b) => a.strike - b.strike);
   563→}

   644→/**
   645→ * Derive lot size and strike step from Kite instruments CSV
   646→ * - lotSize: from the option instrument's lot_size field
   647→ * - strikeStep: derived from the gap between adjacent strikes
   648→ */
   649→export async function getInstrumentMeta(
   650→  symbol: string,
   651→  spotPrice: number,
   652→): Promise<InstrumentMeta> {
   653→  const spec = getInstrumentSpec(symbol);
   654→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   655→
   656→  // Fetch instruments from the F&O exchange (NFO/BFO), NOT the cash exchange (NSE/BSE).
   657→  // See getOptionInstruments() for the full explanation.
   658→  const allInstruments = await getInstruments(spec.segment);
   659→
   660→  // Filter to this symbol's options for current/near expiry.
   661→  // Use startsWith on segment to match both legacy 'NFO' and new 'NFO-OPT' formats.
   662→  const symbolOpts = allInstruments.filter(i =>
   663→    i.segment.startsWith(spec.segment) &&
   664→    i.instrumentType === spec.instrumentType &&
   665→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   666→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   667→  );
   668→
   669→  if (symbolOpts.length === 0) {
   670→    return { lotSize: 1, strikeStep: 50 }; // fallback
   671→  }
   672→
   673→  // Get lot size from the first matching option (all have same lot size for same underlying)
   674→  const lotSize = symbolOpts[0].lotSize || 1;
   675→
   676→  // Derive strike step from unique sorted strikes near ATM
   677→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   678→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   679→
   680→  const expiryOpts = nearestExpiry
   681→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   682→    : symbolOpts;
   683→
   684→  // Get unique strikes, sorted
   685→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   686→
   687→  // Find strikes near ATM and compute step from adjacent gaps
   688→  let strikeStep = 50; // default fallback
   689→  if (strikes.length >= 2) {
   690→    // Use the most common gap between adjacent strikes
   691→    const gaps: Record<number, number> = {};
   692→    for (let i = 1; i < strikes.length; i++) {
   693→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   694→      gaps[gap] = (gaps[gap] || 0) + 1;
   695→    }
   696→    // Most frequent gap = the strike step
   697→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   698→  }
   699→
   700→  return { lotSize, strikeStep };
   701→}

   712→// ─── Black-Scholes Delta Calculator ───
   713→// Approximate delta for options when Kite doesn't provide Greeks.
   714→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   715→
   716→function normCDF(x: number): number {
   717→  // Rational approximation (max error ~7.5e-8)
   718→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   719→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   720→  const sign = x < 0 ? -1 : 1;
   721→  const absX = Math.abs(x);
   722→  const t = 1.0 / (1.0 + p * absX);
   723→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   724→  return 0.5 * (1.0 + sign * y);
   725→}
   726→
   727→/**
   728→ * Calculate Black-Scholes delta for an option
   729→ * @param isCall - true for CE, false for PE
   730→ * @param S - Spot price
   731→ * @param K - Strike price
   732→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   733→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   734→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   735→ */
   736→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   737→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   738→  // If no IV provided, estimate from moneyness
   739→  if (sigma <= 0) {
   740→    const moneyness = (S - K) / S;
   741→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   742→  }
   743→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   744→  const delta = normCDF(d1);
   745→  return isCall ? delta : delta - 1;
   746→}

   748→// ─── Strike Flow Snapshot ───

   753→export interface StrikeFlowSnapshot {
   754→  timestamp: string;
   755→  symbol: string;

   760→  expiry: string;
   761→  strikes: StrikeFlowData[];

   765→  strike: number;
   766→  isATM: boolean;
   767→  ceLTP: number;
   768→  peLTP: number;
   769→  ceOI: number;
   770→  peOI: number;
   771→  ceVol: number;

   774→  peDelta: number;
   775→  ceToken: number;
   776→  peToken: number;

   779→/**
   780→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   781→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   782→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   783→ */
   784→export async function getStrikeFlowSnapshot(
   785→  symbol: string,
   786→  spotPrice: number,
   787→): Promise<StrikeFlowSnapshot | null> {
   788→  const spec = getInstrumentSpec(symbol);
   789→  if (!spec) return null;
   790→
   791→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   792→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   793→  if (optInstruments.length === 0) return null;
   794→
   795→  // Batch fetch quotes using instrument tokens
   796→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   797→  const quotes = await getQuotes(iKeys);
   798→  if ('_error' in quotes) return null;
   799→
   800→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   801→
   802→  // Group by strike
   803→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   804→  for (const inst of optInstruments) {
   805→    if (!strikeMap.has(inst.strike)) {
   806→      strikeMap.set(inst.strike, { ce: null, pe: null });
   807→    }
   808→    const entry = strikeMap.get(inst.strike)!;
   809→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   810→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   811→  }
   812→
   813→  // Calculate time to expiry
   814→  const expiry = optInstruments[0]?.expiry || '';
   815→  const now = new Date();
   816→  const expiryDate = new Date(expiry);
   817→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   818→  const T = daysToExpiry / 365;
   819→
   820→  // Build strike data
   821→  const strikes: StrikeFlowData[] = [];
   822→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   823→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   824→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   825→
   826→    const ceLTP = ceQuote?.lastPrice || 0;
   827→    const peLTP = peQuote?.lastPrice || 0;
   828→    const ceOI = ceQuote?.oi || 0;
   829→    const peOI = peQuote?.oi || 0;
   830→    const ceVol = ceQuote?.volume || 0;
   831→    const peVol = peQuote?.volume || 0;
   832→
   833→    // Compute delta via Black-Scholes
   834→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   835→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   836→
   837→    strikes.push({
   838→      strike,
   839→      isATM: strike === atmStrike,
   840→      ceLTP,
   841→      peLTP,
   842→      ceOI,
   843→      peOI,
   844→      ceVol,
   845→      peVol,
   846→      ceDelta,
   847→      peDelta,
   848→      ceToken: ce?.instrumentToken || 0,
   849→      peToken: pe?.instrumentToken || 0,
   850→    });
   851→  }
   852→
   853→  strikes.sort((a, b) => a.strike - b.strike);
   854→
   855→  return {
   856→    timestamp: new Date().toISOString(),
   857→    symbol,
   858→    spotPrice,
   859→    atmStrike,
   860→    lotSize: meta.lotSize,
   861→    strikeStep: meta.strikeStep,
   862→    expiry,
   863→    strikes,
   864→  };
   865→}

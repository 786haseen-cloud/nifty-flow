    78→  if (mins < 555 || mins > 940) {
    79→    Logger.log("Market CLOSED (outside hours). IST: " + istStr + " Minutes: " + mins + " (need 555-940)");
    80→    return false;
    81→  }
    82→
    83→  Logger.log("Market OPEN. IST: " + istStr + " Minutes: " + mins);

    98→    return d;

   102→  var mo = now.getMonth();
   103→  var exp = lastTuesday(yr, mo);

   106→  var today = new Date(yr, mo, now.getDate());

   108→  if (today > expDate || (today.getTime() === expDate.getTime() && istMinutesOfDay(now) > 940)) {
   109→    mo = (mo + 1) % 12;

   111→    exp = lastThursday(yr, mo);

   113→

   121→function kiteFetch(url, retries) {
   122→  retries = retries || 3;
   123→  var lastError = "";
   124→  for (var i = 0; i < retries; i++) {
   125→    try {
   126→      var resp = UrlFetchApp.fetch(url, { method: "get", headers: API_HEADERS, muteHttpExceptions: true });
   127→      var code = resp.getResponseCode();
   128→      var body = resp.getContentText();
   129→      var data = JSON.parse(body);
   130→
   131→      // Check for Kite API-level errors
   132→      if (data.status === "error") {
   133→        lastError = data.message || "Unknown Kite API error";
   134→        Logger.log("Kite API error (attempt " + (i+1) + "/" + retries + "): " + lastError);
   135→
   136→        // Token expired or invalid — no point retrying
   137→        if (lastError.indexOf("token") > -1 || lastError.indexOf("Token") > -1 ||
   138→            lastError.indexOf("expired") > -1 || lastError.indexOf("invalid") > -1) {
   139→          logRunStatus("TOKEN_ERROR", lastError);
   140→          return null;  // return null instead of throwing
   141→        }
   142→        // Rate limit — wait longer
   143→        if (code === 429 || lastError.indexOf("rate") > -1) {
   144→          Utilities.sleep(2000 * (i + 1));
   145→          continue;
   146→        }
   147→        // Other API error — retry
   148→        Utilities.sleep(1000 * (i + 1));
   149→        continue;
   150→      }
   151→
   152→      // Success
   153→      return data;
   154→    } catch (e) {
   155→      lastError = e.toString();
   156→      Logger.log("Fetch error (attempt " + (i+1) + "/" + retries + "): " + lastError);
   157→      Utilities.sleep(1000 * (i + 1));
   158→    }
   159→  }
   160→  // All retries failed — return null (caller handles gracefully)
   161→  logRunStatus("FETCH_FAILED", lastError);
   162→  return null;
   163→}

   192→function normCDF(x) {
   193→  return (1.0 + erf(x / Math.sqrt(2.0))) / 2.0;
   194→}
   195→
   196→function erf(x) {
   197→  var sign = (x >= 0) ? 1 : -1;
   198→  x = Math.abs(x);
   199→  var a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   200→  var a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   201→  var t = 1.0 / (1.0 + p * x);
   202→  var y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);
   203→  return sign * y;
   204→}

   206→function normPDF(x) {
   207→  return Math.exp(-x * x / 2) / Math.sqrt(2 * Math.PI);
   208→}

   210→function calcStdDev(arr) {
   211→  if (!arr || arr.length < 2) return 0;
   212→  var mean = arr.reduce(function(a,b){return a+b;}) / arr.length;
   213→  return Math.sqrt(arr.reduce(function(sq,n){return sq + Math.pow(n-mean,2);}, 0) / arr.length);
   214→}

   223→// IV & GREEKS (Robust with bounds)
   224→// ============================================================================
   225→function calcIV(S, K, T, r, price, type) {
   226→  if (price <= 0.1 || S <= 0 || T <= 0.0001) return 0.25;
   227→  var sigma = 0.25, tol = 0.0001, maxIter = 30;
   228→  for (var i = 0; i < maxIter; i++) {
   229→    var d1 = (Math.log(S/K) + (r + sigma*sigma/2)*T) / (sigma*Math.sqrt(T));
   230→    var d2 = d1 - sigma*Math.sqrt(T);
   231→    var diff;
   232→    if (type === "CE") diff = S*normCDF(d1) - K*Math.exp(-r*T)*normCDF(d2) - price;
   233→    else diff = K*Math.exp(-r*T)*normCDF(-d2) - S*normCDF(-d1) - price;
   234→    if (Math.abs(diff) < tol) break;
   235→    var vega = S * Math.sqrt(T) * normPDF(d1);
   236→    if (vega < 0.0001) vega = 0.0001;
   237→    sigma -= diff / vega;
   238→    if (sigma < 0.05) sigma = 0.05;
   239→    if (sigma > 2.0) sigma = 2.0;
   240→  }
   241→  return sigma;
   242→}

   244→function calcGreeks(S, K, T, r, sigma, type) {
   245→  if (T < 0.0001) T = 0.0001;
   246→  if (sigma < 0.01) sigma = 0.15;
   247→  var d1 = (Math.log(S/K) + (r + sigma*sigma/2)*T) / (sigma*Math.sqrt(T));
   248→  var d2 = d1 - sigma*Math.sqrt(T);
   249→  var pdf = normPDF(d1);
   250→  var gamma = pdf / (S * sigma * Math.sqrt(T));
   251→  var vega = S * pdf * Math.sqrt(T) / 100;
   252→  var delta, theta, rho;
   253→  if (type === "CE") {
   254→    delta = normCDF(d1);
   255→    theta = (-S*pdf*sigma/(2*Math.sqrt(T))) - (r*K*Math.exp(-r*T)*normCDF(d2));
   256→    rho = K * T * Math.exp(-r*T) * normCDF(d2);
   257→  } else {
   258→    delta = normCDF(d1) - 1;
   259→    theta = (-S*pdf*sigma/(2*Math.sqrt(T))) + (r*K*Math.exp(-r*T)*normCDF(-d2));
   260→    rho = -K * T * Math.exp(-r*T) * normCDF(-d2);
   261→  }
   262→  theta = theta / 252;
   263→  return { delta: delta, gamma: gamma, theta: theta, vega: vega, rho: rho, iv: sigma };
   264→}

   273→function calcMaxPain(strikesMap) {
   274→  var strikes = Object.keys(strikesMap).map(Number).sort(function(a,b){return a-b;});
   275→  var minPain = Infinity, maxPainStrike = strikes[0];
   276→  for (var i = 0; i < strikes.length; i++) {
   277→    var k = strikes[i];
   278→    var pain = 0;
   279→    for (var j = 0; j < strikes.length; j++) {
   280→      var sk = strikes[j];
   281→      if (sk < k) pain += (strikesMap[sk].ceOI || 0) * (k - sk);
   282→      if (sk > k) pain += (strikesMap[sk].peOI || 0) * (sk - k);
   283→    }
   284→    if (pain < minPain) { minPain = pain; maxPainStrike = k; }
   285→  }
   286→  return { strike: maxPainStrike, pain: minPain };

   292→function calcNetDeltaGEX(strikesMap, spot) {
   293→  var strikes = Object.keys(strikesMap).map(Number);
   294→  var netDelta = 0, gex = 0, notional = 0;
   295→  for (var i = 0; i < strikes.length; i++) {
   296→    var s = strikes[i];
   297→    var d = strikesMap[s];
   298→    var ceD = safeNum(d.ceDelta, 0.5);
   299→    var peD = safeNum(d.peDelta, -0.5);
   300→    var ceG = safeNum(d.ceGamma, 0.001);
   301→    var peG = safeNum(d.peGamma, 0.001);
   302→    var ceOI = safeNum(d.ceOI, 0);
   303→    var peOI = safeNum(d.peOI, 0);
   304→    
   305→    netDelta += (ceOI * ceD) + (peOI * peD);
   306→    gex += (ceOI * ceG) + (peOI * peG);
   307→    notional += (ceOI + peOI) * s;
   308→  }
   309→  return { netDelta: netDelta, gex: gex, notional: notional };
   310→}

   347→ * Analyze IV Skew (Risk Reversal)

   349→function analyzeIVSkew(strikesMap, atmStrike) {
   350→  var strikes = Object.keys(strikesMap).map(Number).sort(function(a,b){return a-b;});
   351→  var otmCeIVs = [], otmPeIVs = [], atmIV = 0;
   352→  for (var i = 0; i < strikes.length; i++) {
   353→    var s = strikes[i];
   354→    var d = strikesMap[s];
   355→    if (Math.abs(s - atmStrike) < STRIKE_STEP/2) atmIV = (d.ceIV + d.peIV) / 2;
   356→    if (s > atmStrike + STRIKE_STEP/2) otmCeIVs.push(d.ceIV);
   357→    if (s < atmStrike - STRIKE_STEP/2) otmPeIVs.push(d.peIV);
   358→  }
   359→  var avgOtmCe = otmCeIVs.length > 0 ? otmCeIVs.reduce(function(a,b){return a+b;}) / otmCeIVs.length : atmIV;
   360→  var avgOtmPe = otmPeIVs.length > 0 ? otmPeIVs.reduce(function(a,b){return a+b;}) / otmPeIVs.length : atmIV;
   361→  return {
   362→    atmIV: atmIV,
   363→    otmCallSkew: atmIV > 0 ? (avgOtmCe / atmIV - 1) : 0,
   364→    otmPutSkew: atmIV > 0 ? (avgOtmPe / atmIV - 1) : 0,
   365→    riskReversal: avgOtmCe - avgOtmPe
   366→  };
   367→}

   372→function calcDANF(oiChange, delta, price, volume, type) {

   455→  var skew = metrics.ivSkew || {};
   456→  if (skew.otmPutSkew > 0.15 && skew.otmCallSkew < 0.05) { score += 3; factors.push("Put panic skew"); }
   457→  else if (skew.otmCallSkew > 0.15 && skew.otmPutSkew < 0.05) { score += 9; factors.push("Call frenzy skew"); }
   458→  else if (skew.riskReversal < -0.05) { score += 4; factors.push("Negative risk reversal"); }
   459→  else if (skew.riskReversal > 0.05) { score += 8; factors.push("Positive risk reversal"); }
   460→  else { score += 6; factors.push("Balanced skew"); }
   461→  
   462→  // Confidence based on data quality and momentum alignment
   463→  var confidence = 50;
   464→  if (metrics.dataQuality >= 10) confidence += 15;
   465→  if (history.scoreHistory && history.scoreHistory.length >= 3) {
   466→    var recent = history.scoreHistory.slice(-3);
   467→    var aligned = recent.every(function(s){ return (score > 60 && s > 60) || (score < 40 && s < 40) || (score >= 40 && score <= 60); });
   468→    if (aligned) confidence += 20;
   469→    else confidence -= 10;
   470→  }
   471→  if (Math.abs(flowScore) > 10) confidence += 10;
   472→  confidence = Math.max(10, Math.min(95, confidence));
   473→  
   474→  // Signal classification
   475→  var signal, signalClass;
   476→  if (score >= 80) { signal = "🚀 STRONG BUY CALLS"; signalClass = "STRONG_BULL"; }
   477→  else if (score >= 65) { signal = "📈 BUY CALLS / SELL PUTS"; signalClass = "BULL"; }
   478→  else if (score > 55) { signal = "🟢 Mild Bullish — Buy ATM CE"; signalClass = "MILD_BULL"; }
   479→  else if (score >= 45) { signal = "⚖️ NEUTRAL — Iron Condor / Avoid"; signalClass = "NEUTRAL"; }
   480→  else if (score > 35) { signal = "🔴 Mild Bearish — Buy ATM PE"; signalClass = "MILD_BEAR"; }
   481→  else if (score >= 20) { signal = "📉 BUY PUTS / SELL CALLS"; signalClass = "BEAR"; }
   482→  else { signal = "🔻 STRONG BUY PUTS"; signalClass = "STRONG_BEAR"; }
   483→  
   484→  // Target / Stop based on OI walls and Max Pain
   485→  var target = metrics.resistance;
   486→  var stop = metrics.support;
   487→  if (score > 60) {
   488→    // Bullish: target next resistance, stop at support
   489→    target = metrics.resistance;
   490→    stop = metrics.support;
   491→  } else if (score < 40) {
   492→    // Bearish: target support, stop at resistance
   493→    target = metrics.support;
   494→    stop = metrics.resistance;
   495→  } else {
   496→    target = metrics.maxPain;
   497→    stop = (spot > metrics.maxPain) ? metrics.support : metrics.resistance;
   498→  }
   499→  
   500→  return {
   501→    score: Math.round(score),
   502→    confidence: Math.round(confidence),
   503→    signal: signal,
   504→    signalClass: signalClass,
   505→    target: target,
   506→    stop: stop,
   507→    maxPain: metrics.maxPain,
   508→    factors: factors.join(" | ")
   509→  };
   510→}

   903→function runHDFCBANKHeatMap(hmSheet, optSheet, spot, expiryCode, timeOnly) {
   904→  if (!hmSheet || !optSheet) return;
   905→  var lastRow = optSheet.getLastRow();
   906→  if (lastRow < HM_START_ROW + 5) return;
   907→
   908→  var scanRows = Math.min(50, lastRow - HM_START_ROW + 1);
   909→  var rawData = optSheet.getRange(lastRow - scanRows + 1, 1, scanRows, 7).getValues();
   910→
   911→  // Find last 2 distinct timestamps
   912→  var timestamps = [];
   913→  for (var i = 0; i < rawData.length; i++) {
   914→    var t = String(rawData[i][0]).trim();
   915→    if (t !== "" && (timestamps.length === 0 || timestamps[timestamps.length - 1] !== t)) {
   916→      timestamps.push(t);
   917→    }
   918→  }
   919→  if (timestamps.length < 2) return;
   920→
   921→  var prevTs = timestamps[timestamps.length - 2];
   922→  var currTs = timestamps[timestamps.length - 1];
   923→
   924→  var prevData = {}, currData = {};
   925→  for (var i = 0; i < rawData.length; i++) {
   926→    var t = String(rawData[i][0]).trim();
   927→    var stk = Number(rawData[i][2]);
   928→    var entry = {
   929→      ceLTP: Number(rawData[i][3]) || 0, peLTP: Number(rawData[i][4]) || 0,
   930→      ceOI: Number(rawData[i][5]) || 0, peOI: Number(rawData[i][6]) || 0
   931→    };
   932→    if (t === prevTs) prevData[stk] = entry;
   933→    else if (t === currTs) currData[stk] = entry;
   934→  }
   935→
   936→  var strikes = Object.keys(currData).map(Number).sort(function(a,b){return a-b;});
   937→  if (strikes.length < 5) return;
   938→
   939→  var cePremChg = {}, pePremChg = {}, ceOIChg = {}, peOIChg = {};
   940→  for (var i = 0; i < strikes.length; i++) {
   941→    var s = strikes[i];
   942→    if (!prevData[s]) continue;
   943→    cePremChg[s] = currData[s].ceLTP - prevData[s].ceLTP;
   944→    pePremChg[s] = currData[s].peLTP - prevData[s].peLTP;
   945→    ceOIChg[s] = currData[s].ceOI - prevData[s].ceOI;
   946→    peOIChg[s] = currData[s].peOI - prevData[s].peOI;
   947→  }
   948→
   949→  // Heat score = premium change minus neighbor average
   950→  var ceHeat = {}, peHeat = {};
   951→  for (var i = 0; i < strikes.length; i++) {
   952→    var s = strikes[i];
   953→    if (cePremChg[s] === undefined) continue;
   954→    var ceSum = 0, peSum = 0, count = 0;
   955→    for (var n = -HM_NEIGHBOR_RANGE; n <= HM_NEIGHBOR_RANGE; n++) {
   956→      if (n === 0) continue;
   957→      var ns = s + (n * STRIKE_STEP);
   958→      if (cePremChg[ns] !== undefined) { ceSum += cePremChg[ns]; peSum += pePremChg[ns]; count++; }
   959→    }
   960→    if (count > 0) {
   961→      ceHeat[s] = cePremChg[s] - (ceSum / count);
   962→      peHeat[s] = pePremChg[s] - (peSum / count);
   963→    }
   964→  }
   965→
   966→  var allCeHeats = Object.values(ceHeat);
   967→  var allPeHeats = Object.values(peHeat);
   968→  var ceStd = calcStdDev(allCeHeats);
   969→  var peStd = calcStdDev(allPeHeats);
   970→  var ceThr = Math.max(HM_MIN_THRESHOLD, HM_HEAT_MULT * ceStd);
   971→  var peThr = Math.max(HM_MIN_THRESHOLD, HM_HEAT_MULT * peStd);
   972→
   973→  // Find existing strike rows for in-place update
   974→  var hmLastRow = hmSheet.getLastRow();
   975→  var strikeRowMap = {};
   976→  if (hmLastRow >= HM_START_ROW) {
   977→    var existing = hmSheet.getRange(HM_START_ROW, 11, hmLastRow - HM_START_ROW + 1, 1).getValues();
   978→    for (var i = 0; i < existing.length; i++) {
   979→      if (existing[i][0]) strikeRowMap[existing[i][0]] = HM_START_ROW + i;
   980→    }
   981→  }
   982→
   983→  for (var i = 0; i < strikes.length; i++) {
   984→    var s = strikes[i];
   985→    var targetRow = strikeRowMap[s] ? strikeRowMap[s] : hmSheet.getLastRow() + 1;
   986→    var ceSig = getHMHeatSignal(ceHeat[s] || 0, ceOIChg[s] || 0, "CE", ceThr);
   987→    var peSig = getHMHeatSignal(peHeat[s] || 0, peOIChg[s] || 0, "PE", peThr);
   988→
   989→    var rowData = [
   990→      timeOnly, (i === 0 ? spot : ""), expiryCode,
   991→      (cePremChg[s] || 0).toFixed(2), prevData[s] ? prevData[s].ceLTP : 0, currData[s].ceLTP,
   992→      ceOIChg[s] || 0, currData[s].ceOI, (ceHeat[s] || 0).toFixed(2), ceSig,
   993→      s, peSig, (peHeat[s] || 0).toFixed(2), currData[s].peOI, peOIChg[s] || 0,
   994→      currData[s].peLTP, prevData[s] ? prevData[s].peLTP : 0, (pePremChg[s] || 0).toFixed(2)
   995→    ];
   996→
   997→    hmSheet.getRange(targetRow, 1, 1, 18).setValues([rowData]);
   998→    // Color coding
   999→    var ceSigCell = hmSheet.getRange(targetRow, 10);
  1000→    var peSigCell = hmSheet.getRange(targetRow, 12);
  1001→    if (ceSig.indexOf("BIG") > -1) ceSigCell.setBackground(ceSig.indexOf("SHORT") > -1 ? "#ffcccc" : "#ccffcc");
  1002→    else ceSigCell.setBackground("#f0f0f0");
  1003→    if (peSig.indexOf("BIG") > -1) peSigCell.setBackground(peSig.indexOf("SHORT") > -1 ? "#ccffcc" : "#ffcccc");
  1004→    else peSigCell.setBackground("#f0f0f0");
  1005→  }
  1006→
  1007→  // Update title row
  1008→  hmSheet.getRange(1, 1).setValue("STRIKE HEAT MAP - HDFC BANK | Expiry: " + expiryCode + " | Last: " + timeOnly);
  1009→  Logger.log("HEATMAP updated for " + strikes.length + " strikes");
  1010→}

  1012→function getHMHeatSignal(heat, oiChange, type, thr) {
  1013→  var rising = oiChange > 0;
  1014→  if (type === "CE") {
  1015→    if (heat < -thr && rising) return "BIG SHORT";
  1016→    if (heat > thr && rising) return "BIG LONG";
  1017→    return "Normal";
  1018→  } else {
  1019→    if (heat < -thr && rising) return "BIG SHORT";
  1020→    if (heat > thr && rising) return "BIG LONG";
  1021→    return "Normal";
  1022→  }
  1023→}

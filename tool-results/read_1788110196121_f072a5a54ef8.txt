    14→    14→/**
    15→    15→ * 4-Color OI classification (matches Strike Flow Map convention):
    16→    16→ *
    17→    17→ *  CE  Buy   = ΔOI>0 & ΔLTP≥0  → bright green  #00B050
    18→    18→ *  CE  Write = ΔOI>0 & ΔLTP<0  → light red     #FFCCCC (dark red border)
    19→    19→ *  CE  Close = ΔOI<0           → grey          #6b7280
    20→    20→ *  PE  Write = ΔOI>0 & ΔLTP≥0  → light green   #C6EFCE (dark green border)
    21→    21→ *  PE  Buy   = ΔOI>0 & ΔLTP<0  → bright red    #BD2130
    22→    22→ *  PE  Close = ΔOI<0           → grey          #6b7280
    23→    23→ */
    24→    24→type OIColor = {
    25→    25→  bg: string;        // inline background color
    26→    26→  border?: string;   // inline border color (for Write classes)
    27→    27→  text: string;      // tailwind text class for the OI label
    28→    28→  label?: string;    // short label: "Buy", "Write", "Close"
    29→    29→};
    30→    30→
    31→    31→function getOIColor(oiChange: number, ltpChange: number, isPE: boolean): OIColor {
    32→    32→  // No previous data — neutral
    33→    33→  if (oiChange === 0 && ltpChange === 0) {
    34→    34→    return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    35→    35→  }
    36→    36→  // Unwinding / short covering (ΔOI < 0)
    37→    37→  if (oiChange < 0) {
    38→    38→    return { bg: '#6b7280', text: 'text-zinc-400', label: 'Close' };
    39→    39→  }
    40→    40→  // oiChange > 0 — new positions opened
    41→    41→  if (isPE) {
    42→    42→    // PE: Write = bullish (price up/stable + OI up), Buy = bearish (price down + OI up)
    43→    43→    if (ltpChange >= 0) return { bg: '#C6EFCE', border: '#16a34a', text: 'text-emerald-700', label: 'Write' };
    44→    44→    if (ltpChange < 0)  return { bg: '#BD2130', text: 'text-red-100', label: 'Buy' };
    45→    45→  } else {
    46→    46→    // CE: Buy = bullish (price up/stable + OI up), Write = bearish (price down + OI up)
    47→    47→    if (ltpChange >= 0) return { bg: '#00B050', text: 'text-green-100', label: 'Buy' };
    48→    48→    if (ltpChange < 0)  return { bg: '#FFCCCC', border: '#dc2626', text: 'text-red-700', label: 'Write' };
    49→    49→  }
    50→    50→  // Fallback (shouldn't reach here)
    51→    51→  return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    52→    52→}

   113→   113→function formatLakhs(n: number): string {
   114→   114→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
   115→   115→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
   116→   116→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
   117→   117→  return n.toLocaleString('en-IN');
   118→   118→}

   124→   124→// Compute Max Pain: strike where option buyers lose the most
   125→   125→function computeMaxPain(strikes: StrikeFlowData[]): number {
   126→   126→  if (strikes.length === 0) return 0;
   127→   127→  let minLoss = Infinity;
   128→   128→  let maxPainStrike = strikes[0].strike;
   129→   129→
   130→   130→  for (const k of strikes) {
   131→   131→    let totalLoss = 0;
   132→   132→    for (const s of strikes) {
   133→   133→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
   134→   134→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
   135→   135→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
   136→   136→      // But we only have OI, not premium. Standard Max Pain uses:
   137→   137→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
   138→   138→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
   139→   139→      // Simplified: use intrinsic value approach
   140→   140→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
   141→   141→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
   142→   142→    }
   143→   143→    if (totalLoss < minLoss) {
   144→   144→      minLoss = totalLoss;
   145→   145→      maxPainStrike = k.strike;
   146→   146→    }
   147→   147→  }
   148→   148→  return maxPainStrike;
   149→   149→}

   151→   151→// Generate demo data for 11 strikes
   152→   152→function generateDemoData(symbol: string): StrikeFlowSnapshot {
   153→   153→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   154→   154→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   155→   155→  const atmStrike = Math.round(spot / step) * step;
   156→   156→  const strikes: StrikeFlowData[] = [];
   157→   157→
   158→   158→  for (let i = -5; i <= 5; i++) {
   159→   159→    const strike = atmStrike + i * step;
   160→   160→    const dist = Math.abs(i);
   161→   161→    // Heavier OI near ATM, with some randomness
   162→   162→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   163→   163→    const noise = 0.6 + Math.random() * 0.8;
   164→   164→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   165→   165→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   166→   166→    const moneyness = (spot - strike) / spot;
   167→   167→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   168→   168→    const ceITM = spot > strike;
   169→   169→    const peITM = spot < strike;
   170→   170→
   171→   171→    strikes.push({
   172→   172→      strike,
   173→   173→      isATM: i === 0,
   174→   174→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   175→   175→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   176→   176→      ceOI,
   177→   177→      peOI,
   178→   178→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   179→   179→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   180→   180→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   181→   181→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   182→   182→      ceToken: 100 + i,
   183→   183→      peToken: 200 + i,
   184→   184→    });
   185→   185→  }
   186→   186→
   187→   187→  return {
   188→   188→    timestamp: new Date().toISOString(),
   189→   189→    symbol,
   190→   190→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   191→   191→    atmStrike,
   192→   192→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   193→   193→    strikeStep: step,
   194→   194→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   195→   195→    strikes,
   196→   196→  };
   197→   197→}

   203→   203→function getPCRHistoryKey(symbol: string): string {
   204→   204→  const today = new Date().toISOString().split('T')[0];
   205→   205→  return `oi-walls-pcr-history-${today}-${symbol}`;

   207→   207→
   208→   208→function loadPCRHistory(symbol: string): number[] {
   209→   209→  if (typeof window === 'undefined') return [];
   210→   210→  try {
   211→   211→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   212→   212→    return raw ? JSON.parse(raw) : [];
   213→   213→  } catch {
   214→   214→    return [];
   215→   215→  }
   216→   216→}
   217→   217→
   218→   218→function savePCRHistory(symbol: string, history: number[]) {
   219→   219→  if (typeof window === 'undefined') return;
   220→   220→  try {
   221→   221→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   222→   222→  } catch {
   223→   223→    // localStorage full or unavailable
   224→   224→  }
   225→   225→}

   231→   231→export default function OIWallsTab() {
   232→   232→  const [symbol, setSymbol] = useState('NIFTY');
   233→   233→  const [snapshot, setSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   234→   234→  const [loading, setLoading] = useState(false);
   235→   235→  const [error, setError] = useState<string | null>(null);
   236→   236→  const [isLive, setIsLive] = useState(false);
   237→   237→  const [lastUpdate, setLastUpdate] = useState<string>('');
   238→   238→  const [pcrHistory, setPcrHistory] = useState<number[]>([]);
   239→   239→  const [demoData, setDemoData] = useState<StrikeFlowSnapshot | null>(null);
   240→   240→
   241→   241→  const prevSnapshotRef = useRef<StrikeFlowSnapshot | null>(null);
   242→   242→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   243→   243→
   244→   244→  // Load PCR history on mount / symbol change
   245→   245→  useEffect(() => {
   246→   246→    setPcrHistory(loadPCRHistory(symbol));
   247→   247→  }, [symbol]);
   248→   248→
   249→   249→  const fetchData = useCallback(async () => {
   250→   250→    setLoading(true);
   251→   251→    setError(null);
   252→   252→    try {
   253→   253→      const res = await fetch(withCreds(`/api/kite/strike-flow?symbol=${symbol}`));
   254→   254→      const data = await res.json();
   255→   255→
   256→   256→      if (data.mode !== 'live') {
   257→   257→        // Generate demo data
   258→   258→        setIsLive(false);
   259→   259→        const demo = generateDemoData(symbol);
   260→   260→        setDemoData(demo);
   261→   261→        setLastUpdate(new Date().toLocaleTimeString('en-IN', {
   262→   262→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   263→   263→        }));
   264→   264→        setLoading(false);
   265→   265→        return;
   266→   266→      }
   267→   267→
   268→   268→      setIsLive(true);
   269→   269→      setDemoData(null);
   270→   270→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   271→   271→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   272→   272→      }));
   273→   273→
   274→   274→      // Compute OI change before updating ref
   275→   275→      const prev = prevSnapshotRef.current;
   276→   276→      if (prev && prev.symbol === data.symbol) {
   277→   277→        // OI change is computed in metrics; we just need to store prev for diff
   278→   278→      }
   279→   279→      prevSnapshotRef.current = data;
   280→   280→      setSnapshot(data);
   281→   281→    } catch (err) {
   282→   282→      setError(err instanceof Error ? err.message : 'Fetch failed');
   283→   283→    } finally {
   284→   284→      setLoading(false);
   285→   285→    }
   286→   286→  }, [symbol]);
   287→   287→
   288→   288→  // Initial fetch + polling
   289→   289→  useEffect(() => {
   290→   290→    fetchData();
   291→   291→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   292→   292→    return () => {
   293→   293→      if (intervalRef.current) clearInterval(intervalRef.current);
   294→   294→    };
   295→   295→  }, [fetchData]);
   296→   296→
   297→   297→  // Compute OI change using ref to prev snapshot
   298→   298→  const oiChange = (() => {
   299→   299→    const prev = prevSnapshotRef.current;
   300→   300→    const curr = snapshot;
   301→   301→    if (!prev || !curr || prev.symbol !== curr.symbol) {
   302→   302→      // Check demo data diff
   303→   303→      if (demoData) return { ce: 0, pe: 0 };
   304→   304→      return { ce: 0, pe: 0 };
   305→   305→    }
   306→   306→    let ceChange = 0, peChange = 0;
   307→   307→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   308→   308→    for (const s of curr.strikes) {
   309→   309→      const p = prevMap.get(s.strike);
   310→   310→      if (p) {
   311→   311→        ceChange += s.ceOI - p.ceOI;
   312→   312→        peChange += s.peOI - p.peOI;
   313→   313→      }
   314→   314→    }
   315→   315→    return { ce: ceChange, pe: peChange };
   316→   316→  })();
   317→   317→
   318→   318→  // Use demo or live data
   319→   319→  const data = isLive ? snapshot : demoData;
   320→   320→
   321→   321→  // Compute walls and metrics
   322→   322→  const { walls, metrics } = (() => {
   323→   323→    if (!data || !data.strikes?.length) {
   324→   324→      return { walls: [] as OIWallsStrike[], metrics: null as ComputedMetrics | null };
   325→   325→    }
   326→   326→
   327→   327→    const maxOI = Math.max(
   328→   328→      ...data.strikes.map(s => Math.max(s.ceOI, s.peOI)),
   329→   329→      1
   330→   330→    );
   331→   331→
   332→   332→    // Find ATM strike for ITM determination
   333→   333→    const atmStrike = data.atmStrike || data.strikes.find(s => s.isATM)?.strike || 0;
   334→   334→
   335→   335→    // Build prev strike map for per-strike diff
   336→   336→    const prev = prevSnapshotRef.current;
   337→   337→    const prevMap = new Map<number, StrikeFlowData>();
   338→   338→    if (prev && prev.symbol === data.symbol) {
   339→   339→      for (const s of prev.strikes) prevMap.set(s.strike, s);
   340→   340→    }
   341→   341→
   342→   342→    const walls: OIWallsStrike[] = data.strikes.map(s => {
   343→   343→      const p = prevMap.get(s.strike);
   344→   344→      const ceOiChg = p ? s.ceOI - p.ceOI : 0;
   345→   345→      const peOiChg = p ? s.peOI - p.peOI : 0;
   346→   346→      const ceLtpChg = p ? s.ceLTP - p.ceLTP : 0;
   347→   347→      const peLtpChg = p ? s.peLTP - p.peLTP : 0;
   348→   348→
   349→   349→      return {
   350→   350→        strike: s.strike,
   351→   351→        isATM: s.isATM,
   352→   352→        isCeITM: !s.isATM && s.strike < atmStrike,
   353→   353→        isPeITM: !s.isATM && s.strike > atmStrike,
   354→   354→        ceOI: s.ceOI,
   355→   355→        peOI: s.peOI,
   356→   356→        ceLTP: s.ceLTP,
   357→   357→        peLTP: s.peLTP,
   358→   358→        cePct: (s.ceOI / maxOI) * 100,
   359→   359→        pePct: (s.peOI / maxOI) * 100,
   360→   360→        ceOiChange: ceOiChg,
   361→   361→        peOiChange: peOiChg,
   362→   362→        ceColor: getOIColor(ceOiChg, ceLtpChg, false),
   363→   363→        peColor: getOIColor(peOiChg, peLtpChg, true),
   364→   364→      };
   365→   365→    });
   366→   366→
   367→   367→    const totalCEOI = data.strikes.reduce((sum, s) => sum + s.ceOI, 0);
   368→   368→    const totalPEOI = data.strikes.reduce((sum, s) => sum + s.peOI, 0);
   369→   369→    const pcr = totalCEOI > 0 ? totalPEOI / totalCEOI : 0;
   370→   370→    const maxPain = computeMaxPain(data.strikes);
   371→   371→    const maxPainDist = data.spotPrice - maxPain;
   372→   372→
   373→   373→    const m: ComputedMetrics = {
   374→   374→      pcr,
   375→   375→      maxPain,
   376→   376→      maxPainDist,
   377→   377→      totalCEOI,
   378→   378→      totalPEOI,
   379→   379→      totalOI: totalCEOI + totalPEOI,
   380→   380→      ceOIChange: oiChange.ce,
   381→   381→      peOIChange: oiChange.pe,
   382→   382→    };
   383→   383→
   384→   384→    return { walls, metrics: m };
   385→   385→  })();
   386→   386→
   387→   387→  // Append PCR to history when metrics change
   388→   388→  useEffect(() => {
   389→   389→    if (!metrics) return;
   390→   390→    setPcrHistory(prev => {
   391→   391→      const updated = [...prev, metrics.pcr];
   392→   392→      const trimmed = updated.slice(-60);
   393→   393→      savePCRHistory(symbol, trimmed);
   394→   394→      return trimmed;
   395→   395→    });
   396→   396→  }, [isLive ? snapshot?.timestamp : demoData?.timestamp, metrics?.pcr, symbol]);

     1→/**
     2→ * Zerodha Kite API Integration
     3→ *
     4→ * Connects to Kite Connect REST API for real NSE/BSE market data.
     5→ * When KITE_API_KEY + KITE_ACCESS_TOKEN are set in .env, real data flows.
     6→ * Otherwise, falls back to demo data.
     7→ *
     8→ * Setup:
     9→ * 1. Go to https://developers.kite.trade
    10→ * 2. Create an app → get API Key + Secret
    11→ * 3. Login: https://kite.zerodha.com/connect/login?api_key=YOUR_KEY
    12→ * 4. After login, capture request_token from redirect URL
    13→ * 5. Call /api/kite/auth?request_token=xxx to generate access_token
    14→ * 6. Put access_token in .env → KITE_ACCESS_TOKEN=xxx
    15→ * 7. Redeploy
    16→ *
    17→ * Access tokens expire daily at midnight. Re-login each morning.
    18→ */

    31→export function setKiteOverride(apiKey?: string, accessToken?: string): void {

    36→export function isKiteConfigured(): boolean {
    37→  return !!(
    38→    (process.env.KITE_API_KEY && process.env.KITE_ACCESS_TOKEN) ||
    39→    (_overrideApiKey && _overrideAccessToken)
    40→  );

    43→function kiteHeaders() {
    44→  const apiKey = _overrideApiKey || process.env.KITE_API_KEY || '';
    45→  const accessToken = _overrideAccessToken || process.env.KITE_ACCESS_TOKEN || '';
    46→  return {
    47→    'Authorization': `token ${apiKey}:${accessToken}`,
    48→    'X-Kite-Version': '3',
    49→    'Content-Type': 'application/x-www-form-urlencoded',
    50→  };
    51→}

    96→/**
    97→ * Generate access_token from request_token (one-time, after login redirect)
    98→ */
    99→export async function generateSession(requestToken: string): Promise<{
   100→  accessToken: string;
   101→  refreshToken: string;
   102→} | null> {
   103→  const apiKey = process.env.KITE_API_KEY;
   104→  const apiSecret = process.env.KITE_API_SECRET;
   105→  if (!apiKey || !apiSecret) return null;
   106→
   107→  try {
   108→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   109→    const { createHash } = await import('crypto');
   110→    const checksum = createHash('sha256')
   111→      .update(apiKey + requestToken + apiSecret)
   112→      .digest('hex');
   113→
   114→    const res = await fetch(`${KITE_BASE}/session/token`, {
   115→      method: 'POST',
   116→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   117→      body: new URLSearchParams({
   118→        api_key: apiKey,
   119→        request_token: requestToken,
   120→        checksum,
   121→      }).toString(),
   122→    });
   123→
   124→    const data = await res.json();
   125→    if (data.status === 'success' && data.data) {
   126→      return {
   127→        accessToken: data.data.access_token,
   128→        refreshToken: data.data.refresh_token || '',
   129→      };
   130→    }
   131→    console.error('[Kite] Session error:', data.message);
   132→    return null;
   133→  } catch (err) {
   134→    console.error('[Kite] generateSession failed:', err);
   135→    return null;
   136→  }
   137→}

   139→// ─── Instruments ───
   140→
   141→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   142→let instrumentsCache: KiteInstrument[] | null = null;
   143→let instrumentsCacheTime = 0;
   144→
   145→/**
   146→ * Fetch all instruments from Kite (cached for 1 hour)
   147→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   148→ * We filter by exchange after caching
   149→ */
   150→export async function getInstruments(exchange?: string): Promise<KiteInstrument[]> {
   151→  // Return from cache if fresh
   152→  if (instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   153→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   154→  }
   155→
   156→  try {
   157→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   158→    if (!res.ok) {
   159→      const errText = await res.text();
   160→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   161→      return [];
   162→    }
   163→    const text = await res.text();
   164→
   165→    // Kite returns CSV format
   166→    const lines = text.trim().split('\n');
   167→    const instruments: KiteInstrument[] = [];
   168→
   169→    // Kite CSV column order (2025+ format):
   170→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   171→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   172→    for (let i = 1; i < lines.length; i++) { // skip header
   173→      const cols = lines[i].split(',');
   174→      if (cols.length < 12) continue;
   175→      instruments.push({
   176→        instrumentToken: parseInt(cols[0]) || 0,
   177→        exchangeToken: parseInt(cols[1]) || 0,
   178→        tradingSymbol: cols[2],
   179→        name: cols[3] || '',
   180→        exchange: cols[11] || '',
   181→        segment: cols[10] || '',
   182→        instrumentType: cols[9] || '',
   183→        strike: parseFloat(cols[6]) || 0,
   184→        lotSize: parseInt(cols[8]) || 1,
   185→        expiry: cols[5] || '',
   186→      });
   187→    }
   188→
   189→    instrumentsCache = instruments;
   190→    instrumentsCacheTime = Date.now();
   191→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   192→  } catch (err) {
   193→    const errMsg = err instanceof Error ? err.message : String(err);
   194→    console.error('[Kite] instruments fetch failed:', errMsg);
   195→    // Return empty but set error flag for callers to detect
   196→    return [];
   197→  }
   198→}

   200→// ─── Quotes (Real-time) ───
   201→
   202→/**
   203→ * Get real-time quotes for instruments
   204→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   205→ */
   206→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   207→  if (!isKiteConfigured()) return {};
   208→
   209→  try {
   210→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   211→    // Tokens have no encoding issues on any platform.
   212→    // Build a map from token → original instrument key for response mapping.
   213→    const tokenToKey: Record<string, string> = {};
   214→    const tokenList: string[] = [];
   215→
   216→    for (const inst of instruments) {
   217→      // If it's a numeric token, use directly
   218→      if (/^\d+$/.test(inst)) {
   219→        tokenToKey[inst] = inst;
   220→        tokenList.push(inst);
   221→      } else {
   222→        // Look up token from KITE_INDEX_INSTRUMENTS
   223→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   224→          ([, v]) => v.symbol === inst
   225→        );
   226→        if (found) {
   227→          const tok = String(found[1].token);
   228→          tokenToKey[tok] = inst; // map token back to symbol name
   229→          tokenList.push(tok);
   230→        } else {
   231→          // Fallback: use the instrument string as-is with encodeURI
   232→          tokenToKey[inst] = inst;
   233→          tokenList.push(inst);
   234→        }
   235→      }
   236→    }
   237→
   238→    const iList = tokenList.join(',');
   239→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   240→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   241→    const params = tokenList.map(t => `i=${t}`).join('&');
   242→    const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   243→      headers: kiteHeaders(),
   244→    });
   245→
   246→    // Log non-success for debugging
   247→    if (!res.ok) {
   248→      const errText = await res.text();
   249→      console.error(`[Kite] quote API ${res.status}: ${errText}`);
   250→      return { _error: `HTTP ${res.status}: ${errText}` } as any;
   251→    }
   252→
   253→    const data = await res.json();
   254→    if (data.status !== 'success' || !data.data) {
   255→      console.error('[Kite] quote API error:', JSON.stringify(data));
   256→      return { _error: data.message || 'Unknown error' } as any;
   257→    }
   258→
   259→    const quotes: Record<string, KiteQuote> = {};
   260→    for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   261→      // Kite returns the key as the token string (e.g. "256265")
   262→      // Map it back to the original instrument name (e.g. "NSE:NIFTY 50")
   263→      const displayKey = tokenToKey[key] || key;
   264→      quotes[displayKey] = {
   265→        instrumentToken: q.instrument_token || 0,
   266→        lastPrice: q.last_price || 0,
   267→        open: q.ohlc?.open || 0,
   268→        high: q.ohlc?.high || 0,
   269→        low: q.ohlc?.low || 0,
   270→        close: q.ohlc?.close || 0,
   271→        volume: q.volume || 0,
   272→        netChange: q.net_change || 0,
   273→        averagePrice: q.average_price || 0,
   274→        oi: q.oi || 0,
   275→        oiDayHigh: q.oi_day_high || 0,
   276→        oiDayLow: q.oi_day_low || 0,
   277→      };
   278→    }
   279→    return quotes;
   280→  } catch (err) {
   281→    const errMsg = err instanceof Error ? err.message : String(err);
   282→    console.error('[Kite] quote error:', errMsg);
   283→    return { _error: `Exception: ${errMsg}` } as any;
   284→  }
   285→}

   289→/**
   290→ * Get historical candle data for price line chart
   291→ * instrumentToken: numeric token from instruments list
   292→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   293→ */
   294→export async function getCandles(
   295→  instrumentToken: number,
   296→  interval: string = '15minute',
   297→  days: number = 1,
   298→): Promise<KiteHistoricalCandle[]> {
   299→  if (!isKiteConfigured()) return [];
   300→
   301→  // Kite historical API expects IST dates.
   302→  // On Vercel (UTC), we must convert to IST before formatting.
   303→  const IST_OFFSET = 5.5 * 60 * 60 * 1000;
   304→  const toIST = (d: Date) => new Date(d.getTime() + IST_OFFSET);
   305→
   306→  const toDate = toIST(new Date());
   307→  const fromDate = toIST(new Date());
   308→  fromDate.setDate(fromDate.getDate() - days);
   309→
   310→  const fmt = (d: Date) =>
   311→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`;
   312→
   313→  try {
   314→    const fromStr = fmt(fromDate);
   315→    const toStr = fmt(toDate);
   316→    const url = `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${fromStr}&to=${toStr}&continuous=0`;
   317→
   318→    const res = await fetch(url, { headers: kiteHeaders() });
   319→
   320→    const data = await res.json();
   321→    if (data.status !== 'success' || !data.data?.candles) {
   322→      console.error('[Kite] candles: status=%s, from=%s, to=%s', data.status, fromStr, toStr);
   323→      return [];
   324→    }
   325→
   326→    return (data.data.candles as any[][]).map(c => ({
   327→      timestamp: c[0],
   328→      open: c[1],
   329→      high: c[2],
   330→      low: c[3],
   331→      close: c[4],
   332→      volume: c[5],
   333→    }));
   334→  } catch (err) {
   335→    const errMsg = err instanceof Error ? err.message : String(err);
   336→    console.error('[Kite] candles error:', errMsg);
   337→    return [];
   338→  }
   339→}

   341→// ─── Option Chain ───
   342→
   343→/**
   344→ * Get option chain for an index or stock
   345→ * Uses InstrumentSpec for correct exchange, segment
   346→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   347→ * Returns all CE + PE instruments for the current expiry near ATM
   348→ */
   349→export async function getOptionInstruments(
   350→  symbol: string,
   351→  spotPrice: number,
   352→  strikesAroundOverride?: number,
   353→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   354→  const spec = getInstrumentSpec(symbol);
   355→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   356→
   357→  // Fetch instruments from the correct exchange (NSE or BSE)
   358→  const instruments = await getInstruments(spec.exchange);
   359→
   360→  // Filter options using spec
   361→  const indexOptions = instruments.filter(i =>
   362→    i.segment === spec.segment &&
   363→    i.instrumentType === spec.instrumentType &&
   364→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   365→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   366→  );
   367→
   368→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   369→
   370→  // Get unique expiries, sort by nearest
   371→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   372→
   373→  // Use nearest expiry that's not expired
   374→  const today = new Date();
   375→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   376→
   377→  // Filter to nearest expiry
   378→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   379→
   380→  // Get lot size from first option (all same for same underlying)
   381→  const lotSize = expiryOptions[0]?.lotSize || 1;
   382→
   383→  // Derive strike step DYNAMICALLY from actual strike gaps
   384→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   385→  let strikeStep = 50;
   386→  if (strikes.length >= 2) {
   387→    const gaps: Record<number, number> = {};
   388→    for (let i = 1; i < strikes.length; i++) {
   389→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   390→      gaps[gap] = (gaps[gap] || 0) + 1;
   391→    }
   392→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   393→  }
   394→
   395→  // Round ATM strike using the DYNAMIC strike step
   396→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   397→
   398→  // Get strikes around ATM
   399→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   400→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   401→    atmStrike - strikesAround * strikeStep + i * strikeStep
   402→  );
   403→
   404→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   405→
   406→  return {
   407→    instruments: filtered,
   408→    meta: { lotSize, strikeStep },
   409→  };
   410→}

   412→/**
   413→ * Build options flow data from quotes + OI changes
   414→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   415→ */
   416→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   417→  const spec = getInstrumentSpec(symbol);
   418→  if (!spec) return null;
   419→
   420→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   421→  if (optInstruments.length === 0) return null;
   422→
   423→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   424→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   425→
   426→  // Batch fetch quotes
   427→  const quotes = await getQuotes(iKeys);
   428→  if (Object.keys(quotes).length === 0) return null;
   429→
   430→  // Group by strike → build flow data
   431→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   432→
   433→  // Use DYNAMIC strike step and lot size from CSV
   434→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   435→
   436→  for (const inst of optInstruments) {
   437→    // Quotes are keyed by instrument token (string)
   438→    const key = String(inst.instrumentToken);
   439→    const q = quotes[key];
   440→    if (!q) continue;
   441→
   442→    if (!strikes.has(inst.strike)) {
   443→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   444→    }
   445→    const s = strikes.get(inst.strike)!;
   446→
   447→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   448→    const flow = q.volume * meta.lotSize * q.averagePrice;
   449→
   450→    if (inst.tradingSymbol.endsWith('CE')) {
   451→      s.callBuy = flow * 0.6;
   452→      s.callWrite = flow * 0.4;
   453→      s.callOI = q.oi;
   454→      s.callVol = q.volume;
   455→    } else if (inst.tradingSymbol.endsWith('PE')) {
   456→      s.putBuy = flow * 0.6;
   457→      s.putWrite = flow * 0.4;
   458→      s.putOI = q.oi;
   459→      s.putVol = q.volume;
   460→    }
   461→  }
   462→
   463→  // Build result
   464→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   465→    strike,
   466→    isATM: strike === atmStrike,
   467→    ...flow,
   468→    bullishFlow: flow.callBuy + flow.putWrite,
   469→    bearishFlow: flow.putBuy + flow.callWrite,
   470→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   471→  })).sort((a, b) => a.strike - b.strike);
   472→}

   543→// ─── Dynamic Lot Size + Strike Step from Kite CSV ───
   544→// Lot sizes change with NSE revisions (e.g., Nifty went from 50→25→75→65 etc.)
   545→// Strike steps can also change.
   546→// We derive BOTH from Kite's instrument master CSV at runtime.

   548→export interface InstrumentMeta {
   549→  lotSize: number;
   550→  strikeStep: number;
   551→}
   552→
   553→/**
   554→ * Derive lot size and strike step from Kite instruments CSV
   555→ * - lotSize: from the option instrument's lot_size field
   556→ * - strikeStep: derived from the gap between adjacent strikes
   557→ */
   558→export async function getInstrumentMeta(
   559→  symbol: string,
   560→  spotPrice: number,
   561→): Promise<InstrumentMeta> {
   562→  const spec = getInstrumentSpec(symbol);
   563→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   564→
   565→  // Fetch all instruments for this symbol's exchange
   566→  const allInstruments = await getInstruments(spec.exchange);
   567→
   568→  // Filter to this symbol's options for current/near expiry
   569→  const symbolOpts = allInstruments.filter(i =>
   570→    i.segment === spec.segment &&
   571→    i.instrumentType === spec.instrumentType &&
   572→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   573→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   574→  );
   575→
   576→  if (symbolOpts.length === 0) {
   577→    return { lotSize: 1, strikeStep: 50 }; // fallback
   578→  }
   579→
   580→  // Get lot size from the first matching option (all have same lot size for same underlying)
   581→  const lotSize = symbolOpts[0].lotSize || 1;
   582→
   583→  // Derive strike step from unique sorted strikes near ATM
   584→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   585→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   586→
   587→  const expiryOpts = nearestExpiry
   588→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   589→    : symbolOpts;
   590→
   591→  // Get unique strikes, sorted
   592→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   593→
   594→  // Find strikes near ATM and compute step from adjacent gaps
   595→  let strikeStep = 50; // default fallback
   596→  if (strikes.length >= 2) {
   597→    // Use the most common gap between adjacent strikes
   598→    const gaps: Record<number, number> = {};
   599→    for (let i = 1; i < strikes.length; i++) {
   600→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   601→      gaps[gap] = (gaps[gap] || 0) + 1;
   602→    }
   603→    // Most frequent gap = the strike step
   604→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   605→  }
   606→
   607→  return { lotSize, strikeStep };
   608→}

   619→// ─── Black-Scholes Delta Calculator ───
   620→// Approximate delta for options when Kite doesn't provide Greeks.
   621→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   622→
   623→function normCDF(x: number): number {
   624→  // Rational approximation (max error ~7.5e-8)
   625→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   626→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   627→  const sign = x < 0 ? -1 : 1;
   628→  const absX = Math.abs(x);
   629→  const t = 1.0 / (1.0 + p * absX);
   630→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   631→  return 0.5 * (1.0 + sign * y);
   632→}
   633→
   634→/**
   635→ * Calculate Black-Scholes delta for an option
   636→ * @param isCall - true for CE, false for PE
   637→ * @param S - Spot price
   638→ * @param K - Strike price
   639→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   640→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   641→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   642→ */
   643→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   644→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   645→  // If no IV provided, estimate from moneyness
   646→  if (sigma <= 0) {
   647→    const moneyness = (S - K) / S;
   648→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   649→  }
   650→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   651→  const delta = normCDF(d1);
   652→  return isCall ? delta : delta - 1;
   653→}

   677→  peOI: number;

   686→/**
   687→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   688→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   689→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   690→ */
   691→export async function getStrikeFlowSnapshot(
   692→  symbol: string,
   693→  spotPrice: number,
   694→): Promise<StrikeFlowSnapshot | null> {
   695→  const spec = getInstrumentSpec(symbol);
   696→  if (!spec) return null;
   697→
   698→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   699→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   700→  if (optInstruments.length === 0) return null;
   701→
   702→  // Batch fetch quotes using instrument tokens
   703→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   704→  const quotes = await getQuotes(iKeys);
   705→  if ('_error' in quotes) return null;
   706→
   707→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   708→
   709→  // Group by strike
   710→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   711→  for (const inst of optInstruments) {
   712→    if (!strikeMap.has(inst.strike)) {
   713→      strikeMap.set(inst.strike, { ce: null, pe: null });
   714→    }
   715→    const entry = strikeMap.get(inst.strike)!;
   716→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   717→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   718→  }
   719→
   720→  // Calculate time to expiry
   721→  const expiry = optInstruments[0]?.expiry || '';
   722→  const now = new Date();
   723→  const expiryDate = new Date(expiry);
   724→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   725→  const T = daysToExpiry / 365;
   726→
   727→  // Build strike data
   728→  const strikes: StrikeFlowData[] = [];
   729→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   730→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   731→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   732→
   733→    const ceLTP = ceQuote?.lastPrice || 0;
   734→    const peLTP = peQuote?.lastPrice || 0;
   735→    const ceOI = ceQuote?.oi || 0;
   736→    const peOI = peQuote?.oi || 0;
   737→    const ceVol = ceQuote?.volume || 0;
   738→    const peVol = peQuote?.volume || 0;
   739→
   740→    // Compute delta via Black-Scholes
   741→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   742→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   743→
   744→    strikes.push({
   745→      strike,
   746→      isATM: strike === atmStrike,
   747→      ceLTP,
   748→      peLTP,
   749→      ceOI,
   750→      peOI,
   751→      ceVol,
   752→      peVol,
   753→      ceDelta,
   754→      peDelta,
   755→      ceToken: ce?.instrumentToken || 0,
   756→      peToken: pe?.instrumentToken || 0,
   757→    });
   758→  }
   759→
   760→  strikes.sort((a, b) => a.strike - b.strike);
   761→
   762→  return {
   763→    timestamp: new Date().toISOString(),
   764→    symbol,
   765→    spotPrice,
   766→    atmStrike,
   767→    lotSize: meta.lotSize,
   768→    strikeStep: meta.strikeStep,
   769→    expiry,
   770→    strikes,
   771→  };
   772→}

    35→    35→// OI change + Price change → 4 cases
    36→    36→// ═══════════════════════════════════════════
    37→    37→
    38→    38→function computeStrikeFlow(
    39→    39→  prev: StrikeFlowData,
    40→    40→  curr: StrikeFlowData,
    41→    41→  lotSize: number,
    42→    42→): Omit<StrikeFlowCell, 'strike' | 'isATM'> {
    43→    43→  const DIVISOR = 10000000; // Convert to Crores
    44→    44→
    45→    45→  // CE side
    46→    46→  const dceOI = curr.ceOI - prev.ceOI;
    47→    47→  const dceP = curr.ceLTP - prev.ceLTP;
    48→    48→  let ceBuy = 0, ceWrite = 0;
    49→    49→
    50→    50→  if (dceOI > 0 && dceP >= 0) {
    51→    51→    // Case 1: OI up + Price up → CE BUY (writers losing, buyers aggressive)
    52→    52→    ceBuy = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    53→    53→  } else if (dceOI > 0 && dceP < 0) {
    54→    54→    // Case 2: OI up + Price down → CE WRITE (writing at higher premium, price falling)
    55→    55→    ceWrite = (dceOI * curr.ceDelta * lotSize) / DIVISOR;
    56→    56→  } else if (dceOI < 0 && dceP < 0) {
    57→    57→    // Case 3: OI down + Price down → CE WRITE (short covering at lower price = write)
    58→    58→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    59→    59→  } else if (dceOI < 0 && dceP >= 0) {
    60→    60→    // Case 4: OI down + Price up → CE WRITE (long unwinding at higher price = write)
    61→    61→    ceWrite = (Math.abs(dceOI) * curr.ceDelta * lotSize * 0.3) / DIVISOR;
    62→    62→  }
    63→    63→
    64→    64→  // Volume fallback when OI unchanged (from your script v5)
    65→    65→  if (ceBuy === 0 && ceWrite === 0 && curr.ceVol > 0) {
    66→    66→    const vf = 0.4;
    67→    67→    if (dceP >= 0) ceBuy = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    68→    68→    else ceWrite = (curr.ceVol * curr.ceDelta * lotSize * vf) / DIVISOR;
    69→    69→  }
    70→    70→
    71→    71→  // PE side (same 4 cases, mirrored)
    72→    72→  const dpeOI = curr.peOI - prev.peOI;
    73→    73→  const dpeP = curr.peLTP - prev.peLTP;
    74→    74→  let peBuy = 0, peWrite = 0;
    75→    75→
    76→    76→  if (dpeOI > 0 && dpeP >= 0) {
    77→    77→    peBuy = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    78→    78→  } else if (dpeOI > 0 && dpeP < 0) {
    79→    79→    peWrite = (dpeOI * curr.peDelta * lotSize) / DIVISOR;
    80→    80→  } else if (dpeOI < 0 && dpeP < 0) {
    81→    81→    peWrite = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    82→    82→  } else if (dpeOI < 0 && dpeP >= 0) {
    83→    83→    peBuy = (Math.abs(dpeOI) * curr.peDelta * lotSize * 0.3) / DIVISOR;
    84→    84→  }
    85→    85→
    86→    86→  // Volume fallback for PE
    87→    87→  if (peBuy === 0 && peWrite === 0 && curr.peVol > 0) {
    88→    88→    const vf = 0.4;
    89→    89→    if (dpeP >= 0) peBuy = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    90→    90→    else peWrite = (curr.peVol * curr.peDelta * lotSize * vf) / DIVISOR;
    91→    91→  }
    92→    92→
    93→    93→  // Determine display values (show dominant flow per side)
    94→    94→  const ceDisplay = ceBuy >= ceWrite ? ceBuy : ceWrite;
    95→    95→  const peDisplay = peWrite >= peBuy ? peWrite : peBuy;
    96→    96→  const ceType = ceBuy > 0 && ceBuy >= ceWrite ? 'buy' : ceWrite > 0 ? 'write' : 'none';
    97→    97→  const peType = peWrite > 0 && peWrite >= peBuy ? 'write' : peBuy > 0 ? 'buy' : 'none';
    98→    98→
    99→    99→  return { ceBuy, ceWrite, peBuy, peWrite, ceDisplay, peDisplay, ceType, peType };
   100→   100→}

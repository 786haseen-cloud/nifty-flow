     3→ *
     4→ * Connects to Kite Connect REST API for real NSE/BSE market data.

     9→ * 1. Go to https://developers.kite.trade
    10→ * 2. Create an app → get API Key + Secret

    12→ * 4. After login, capture request_token from redirect URL
    13→ * 5. Call /api/kite/auth?request_token=xxx to generate access_token
    14→ * 6. Put access_token in .env → KITE_ACCESS_TOKEN=xxx
    15→ * 7. Redeploy
    16→ *
    17→ * Access tokens expire daily at midnight. Re-login each morning.

   102→/**
   103→ * Generate access_token from request_token (one-time, after login redirect)
   104→ */
   105→export async function generateSession(requestToken: string): Promise<{
   106→  accessToken: string;
   107→  refreshToken: string;
   108→} | null> {
   109→  const apiKey = process.env.KITE_API_KEY;
   110→  const apiSecret = process.env.KITE_API_SECRET;
   111→  if (!apiKey || !apiSecret) return null;
   112→
   113→  try {
   114→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   115→    const { createHash } = await import('crypto');
   116→    const checksum = createHash('sha256')
   117→      .update(apiKey + requestToken + apiSecret)
   118→      .digest('hex');
   119→
   120→    const res = await fetch(`${KITE_BASE}/session/token`, {
   121→      method: 'POST',
   122→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   123→      body: new URLSearchParams({
   124→        api_key: apiKey,
   125→        request_token: requestToken,
   126→        checksum,
   127→      }).toString(),
   128→    });
   129→
   130→    const data = await res.json();
   131→    if (data.status === 'success' && data.data) {
   132→      return {
   133→        accessToken: data.data.access_token,
   134→        refreshToken: data.data.refresh_token || '',
   135→      };
   136→    }
   137→    console.error('[Kite] Session error:', data.message);
   138→    return null;
   139→  } catch (err) {
   140→    console.error('[Kite] generateSession failed:', err);
   141→    return null;
   142→  }
   143→}

   147→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   148→let instrumentsCache: KiteInstrument[] | null = null;
   149→let instrumentsCacheTime = 0;
   150→
   151→/**
   152→ * Invalidate the in-memory instruments cache.
   153→ * Use after credential changes (e.g. user switched from expired → valid token)
   154→ * to force a fresh CSV download on the next getInstruments() call.
   155→ */
   156→export function invalidateInstrumentsCache(): void {
   157→  instrumentsCache = null;
   158→  instrumentsCacheTime = 0;
   159→}

   161→/**
   162→ * Fetch all instruments from Kite (cached for 1 hour)
   163→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   164→ * We filter by exchange after caching
   165→ *
   166→ * @param exchange Optional exchange filter (e.g. 'NFO' returns only NFO F&O instruments)
   167→ * @param forceRefresh Bypass the 1-hour cache and re-download. Useful when creds change.
   168→ */
   169→export async function getInstruments(exchange?: string, forceRefresh?: boolean): Promise<KiteInstrument[]> {
   170→  // Return from cache if fresh (and not forced to refresh)
   171→  if (!forceRefresh && instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   172→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   173→  }
   174→
   175→  try {
   176→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   177→    if (!res.ok) {
   178→      const errText = await res.text();
   179→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   180→      return [];
   181→    }
   182→    const text = await res.text();
   183→
   184→    // Kite returns CSV format
   185→    const lines = text.trim().split('\n');
   186→    const instruments: KiteInstrument[] = [];
   187→
   188→    // Kite CSV column order (2025+ format):
   189→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   190→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   191→    //
   192→    // Kite changed the CSV format in 2025:
   193→    //   OLD: instrument_type was 'OPTIDX'/'OPTSTK'/'FUTIDX'/'FUTSTK'/'EQ'/'INDEX'
   194→    //        segment was 'NFO'/'BFO'/'NSE'/'BSE'
   195→    //   NEW: instrument_type is 'CE'/'PE'/'FUT'/'EQ'
   196→    //        segment is 'NFO-OPT'/'NFO-FUT'/'NFO'/'BFO-OPT'/'BFO-FUT'/'NSE'/'BSE'
   197→    // To avoid breaking the rest of the code (which filters by OPTIDX/OPTSTK/FUTIDX/FUTSTK),
   198→    // we normalize the new format back to the legacy types.
   199→    //
   200→    // Disambiguation: in the new CSV, ALL options (index + stock) are CE/PE — there is no
   201→    // index-vs-stock hint in the row itself. We use the `name` column to decide:
   202→    //   - If the underlying name matches a known F&O index (NIFTY, BANKNIFTY, FINNIFTY,
   203→    //     SENSEX, BANKEX, MIDCPNIFTY, etc.) → OPTIDX / FUTIDX
   204→    //   - Otherwise (RELIANCE, TCS, HDFCBANK, ...) → OPTSTK / FUTSTK
   205→    // The set below is checked at parse time (cheap Set.has lookup).
   206→    const KNOWN_FNO_INDICES = new Set([
   207→      'NIFTY', 'NIFTY 50',
   208→      'BANKNIFTY', 'NIFTY BANK',
   209→      'FINNIFTY', 'NIFTY FIN SERVICE',
   210→      'SENSEX', 'BANKEX',
   211→      'MIDCPNIFTY', 'NIFTY MID SELECT', 'NIFTY PSE',
   212→    ]);
   213→    const normalizeInstrumentType = (rawType: string, segment: string, name: string): string => {
   214→      // Keep legacy values as-is (in case Kite reverts or some entries still use old format)
   215→      if (rawType === 'OPTIDX' || rawType === 'OPTSTK' || rawType === 'FUTIDX' || rawType === 'FUTSTK' ||
   216→          rawType === 'INDEX') {
   217→        return rawType;
   218→      }
   219→      // CRITICAL FIX: In the 2025+ Kite CSV, cash index instruments (NIFTY 50, SENSEX, etc.)
   220→      // have instrument_type='EQ' and segment='INDICES'. We must normalize these to 'INDEX'
   221→      // so that downstream lookups (e.g. highest-bet finding cash instruments) can filter by
   222→      // instrumentType === 'INDEX' instead of the ambiguous 'EQ' (which also matches stocks).
   223→      if (rawType === 'EQ' && segment === 'INDICES') {
   224→        return 'INDEX';
   225→      }
   226→      // Regular EQ stocks stay as 'EQ'
   227→      if (rawType === 'EQ') {
   228→        return rawType;
   229→      }
   230→      const nameUpper = name.toUpperCase();
   231→      const isIndex = KNOWN_FNO_INDICES.has(nameUpper);
   232→      if (rawType === 'CE' || rawType === 'PE') {
   233→        // BFO always has index options only (SENSEX/BANKEX). NFO has both index + stock.
   234→        return isIndex ? 'OPTIDX' : 'OPTSTK';
   235→      }
   236→      if (rawType === 'FUT') {
   237→        return isIndex ? 'FUTIDX' : 'FUTSTK';
   238→      }
   239→      return rawType;
   240→    };
   241→
   242→    for (let i = 1; i < lines.length; i++) { // skip header
   243→      const cols = lines[i].split(',');
   244→      if (cols.length < 12) continue;
   245→      const rawType = cols[9] || '';
   246→      const segment = cols[10] || '';
   247→      // Kite wraps the `name` field in double quotes (e.g. "NIFTY").
   248→      // Strip them so name-based filters (e.g. .includes('NIFTY')) work correctly.
   249→      const rawName = cols[3] || '';
   250→      const name = rawName.startsWith('"') && rawName.endsWith('"')
   251→        ? rawName.slice(1, -1)
   252→        : rawName;
   253→      instruments.push({
   254→        instrumentToken: parseInt(cols[0]) || 0,
   255→        exchangeToken: parseInt(cols[1]) || 0,
   256→        tradingSymbol: cols[2],
   257→        name,
   258→        exchange: cols[11] || '',
   259→        segment,
   260→        instrumentType: normalizeInstrumentType(rawType, segment, name),
   261→        strike: parseFloat(cols[6]) || 0,
   262→        lotSize: parseInt(cols[8]) || 1,
   263→        expiry: cols[5] || '',
   264→      });
   265→    }
   266→
   267→    instrumentsCache = instruments;
   268→    instrumentsCacheTime = Date.now();
   269→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   270→  } catch (err) {
   271→    const errMsg = err instanceof Error ? err.message : String(err);
   272→    console.error('[Kite] instruments fetch failed:', errMsg);
   273→    // Return empty but set error flag for callers to detect
   274→    return [];
   275→  }
   276→}

   280→/**
   281→ * Get real-time quotes for instruments
   282→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   283→ */
   284→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   285→  if (!isKiteConfigured()) return {};
   286→
   287→  try {
   288→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   289→    // Tokens have no encoding issues on any platform.
   290→    // Build a map from token → original instrument key for response mapping.
   291→    const tokenToKey: Record<string, string> = {};
   292→    const tokenList: string[] = [];
   293→
   294→    for (const inst of instruments) {
   295→      // If it's a numeric token, use directly
   296→      if (/^\d+$/.test(inst)) {
   297→        tokenToKey[inst] = inst;
   298→        tokenList.push(inst);
   299→      } else {
   300→        // Look up token from KITE_INDEX_INSTRUMENTS
   301→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   302→          ([, v]) => v.symbol === inst
   303→        );
   304→        if (found) {
   305→          const tok = String(found[1].token);
   306→          tokenToKey[tok] = inst;
   307→          tokenList.push(tok);
   308→        } else {
   309→          // Fallback 1: look up from cached instruments CSV
   310→          const cachedInsts = await getInstruments();
   311→          const csvMatch = cachedInsts.find(
   312→            i => i.tradingSymbol === inst.replace(/^[A-Z]+:/, '') ||
   313→                 `${i.exchange}:${i.tradingSymbol}` === inst
   314→          );
   315→          if (csvMatch) {
   316→            const tok = String(csvMatch.instrumentToken);
   317→            tokenToKey[tok] = inst;
   318→            tokenList.push(tok);
   319→          } else {
   320→            // Fallback 2: use the instrument string as-is
   321→            tokenToKey[inst] = inst;
   322→            tokenList.push(inst);
   323→          }
   324→        }
   325→      }
   326→    }
   327→
   328→    const iList = tokenList.join(',');
   329→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   330→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   331→    // Chunk into batches of 100 to avoid URL length limits and API throttling.
   332→    const CHUNK_SIZE = 100;
   333→    const allQuotes: Record<string, KiteQuote> = {};
   334→
   335→    for (let chunkStart = 0; chunkStart < tokenList.length; chunkStart += CHUNK_SIZE) {
   336→      const chunk = tokenList.slice(chunkStart, chunkStart + CHUNK_SIZE);
   337→      const params = chunk.map(t => `i=${t}`).join('&');
   338→      const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   339→        headers: kiteHeaders(),
   340→      });
   341→
   342→      // Log non-success for debugging
   343→      if (!res.ok) {
   344→        const errText = await res.text();
   345→        console.error(`[Kite] quote API ${res.status}: ${errText}`);
   346→        if (Object.keys(allQuotes).length === 0) {
   347→          return { _error: `HTTP ${res.status}: ${errText}` } as any;
   348→        }
   349→        continue; // partial success — return what we have
   350→      }
   351→
   352→      const data = await res.json();
   353→      if (data.status !== 'success' || !data.data) {
   354→        console.error('[Kite] quote API error:', JSON.stringify(data));
   355→        if (Object.keys(allQuotes).length === 0) {
   356→          return { _error: data.message || 'Unknown error' } as any;
   357→        }
   358→        continue;
   359→      }
   360→
   361→      for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   362→        const displayKey = tokenToKey[key] || key;
   363→        allQuotes[displayKey] = {
   364→          instrumentToken: q.instrument_token || 0,
   365→          lastPrice: q.last_price || 0,
   366→          open: q.ohlc?.open || 0,
   367→          high: q.ohlc?.high || 0,
   368→          low: q.ohlc?.low || 0,
   369→          close: q.ohlc?.close || 0,
   370→          volume: q.volume || 0,
   371→          netChange: q.net_change || 0,
   372→          averagePrice: q.average_price || 0,
   373→          oi: q.oi || 0,
   374→          oiDayHigh: q.oi_day_high || 0,
   375→          oiDayLow: q.oi_day_low || 0,
   376→          dayHigh: q.ohlc?.high || 0,
   377→          dayLow: q.ohlc?.low || 0,
   378→        };
   379→      }
   380→    }
   381→
   382→    return allQuotes;
   383→  } catch (err) {
   384→    const errMsg = err instanceof Error ? err.message : String(err);
   385→    console.error('[Kite] quote error:', errMsg);
   386→    return { _error: `Exception: ${errMsg}` } as any;
   387→  }
   388→}

   392→/**
   393→ * Get historical candle data for price line chart
   394→ * instrumentToken: numeric token from instruments list
   395→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   396→ */
   397→export async function getCandles(
   398→  instrumentToken: number,
   399→  interval: string = '15minute',
   400→  days: number = 1,
   401→): Promise<KiteHistoricalCandle[]> {
   402→  if (!isKiteConfigured()) return [];
   403→
   404→  // Kite historical API expects IST dates. Use centralized IST helpers.
   405→  const toDate = toIST(new Date());
   406→  const fromDate = toIST(new Date());
   407→  fromDate.setDate(fromDate.getDate() - days);
   408→
   409→  try {
   410→    const fromStr = istKiteDateFormat(fromDate);
   411→    const toStr = istKiteDateFormat(toDate);
   412→    // URL-encode the date strings (they contain a space: "2026-08-25 09:15")
   413→    const url = `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${encodeURIComponent(fromStr)}&to=${encodeURIComponent(toStr)}&continuous=0`;
   414→
   415→    const res = await fetch(url, { headers: kiteHeaders() });
   416→
   417→    // Surface HTTP-level errors (403 expired token, 429 rate limit, etc.)
   418→    if (!res.ok) {
   419→      const errText = await res.text();
   420→      console.error(`[Kite] candles HTTP ${res.status} for token ${instrumentToken}: ${errText.substring(0, 300)}`);
   421→      return [];
   422→    }
   423→
   424→    const data = await res.json();
   425→    if (data.status !== 'success' || !data.data?.candles) {
   426→      console.error('[Kite] candles non-success: status=%s, message=%s, from=%s, to=%s',
   427→        data.status, data.message || 'no message', fromStr, toStr);
   428→      return [];
   429→    }
   430→
   431→    return (data.data.candles as any[][]).map(c => ({
   432→      timestamp: c[0],
   433→      open: c[1],
   434→      high: c[2],
   435→      low: c[3],
   436→      close: c[4],
   437→      volume: c[5],
   438→      oi: c[6] || 0,  // c[6] = OI (F&O only), undefined for cash
   439→    }));
   440→  } catch (err) {
   441→    const errMsg = err instanceof Error ? err.message : String(err);
   442→    console.error('[Kite] candles error:', errMsg);
   443→    return [];
   444→  }
   445→}

   447→// ─── Option Chain ───
   448→
   449→/**
   450→ * Get option chain for an index or stock
   451→ * Uses InstrumentSpec for correct exchange, segment
   452→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   453→ * Returns all CE + PE instruments for the current expiry near ATM
   454→ */
   455→export async function getOptionInstruments(
   456→  symbol: string,
   457→  spotPrice: number,
   458→  strikesAroundOverride?: number,
   459→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   460→  const spec = getInstrumentSpec(symbol);
   461→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   462→
   463→  // IMPORTANT: F&O options/futures are NOT on the cash exchange.
   464→  //   - NIFTY index spot is on exchange='NSE', but NIFTY OPTIONS are on exchange='NFO'
   465→  //   - SENSEX spot is on exchange='BSE', but SENSEX OPTIONS are on exchange='BFO'
   466→  //   - Stock (RELIANCE/TCS/...) spot is on exchange='NSE', but their OPTIONS are on exchange='NFO'
   467→  // So we use spec.segment ('NFO' or 'BFO') — which equals the F&O exchange column in
   468→  // Kite's CSV — NOT spec.exchange ('NSE'/'BSE', which is the cash exchange).
   469→  // If you call getInstruments(spec.exchange) you'll only get cash/index instruments,
   470→  // and the option filter below will return ZERO matches.
   471→  const instruments = await getInstruments(spec.segment);
   472→
   473→  // Filter options using spec.
   474→  //
   475→  // Kite changed their CSV format in 2025:
   476→  //   OLD: segment='NFO'/'BFO', instrument_type='OPTIDX'/'OPTSTK'
   477→  //   NEW: segment='NFO-OPT'/'BFO-OPT', instrument_type='CE'/'PE' (we normalize to OPTIDX)
   478→  //
   479→  // So we use a "starts with" match on segment (so 'NFO' matches both 'NFO' and 'NFO-OPT'),
   480→  // and an equality match on instrumentType (we've already normalized new → legacy).
   481→  // Also check searchAliases for symbols whose Kite names don't contain the symbol
   482→  // (e.g. FINNIFTY → Kite uses "NIFTY FIN SERVICE" / "NIFTYFIN" in trading symbols).
   483→  const symUpper = symbol.toUpperCase();
   484→  const aliases = (spec.searchAliases || []).map(a => a.toUpperCase());
   485→  const searchTerms = [symUpper, ...aliases];
   486→
   487→  const indexOptions = instruments.filter(i => {
   488→    if (!(i.segment.startsWith(spec.segment) && i.instrumentType === spec.instrumentType)) {
   489→      return false;
   490→    }
   491→    const nameUp = i.name.toUpperCase();
   492→    const tsUp = i.tradingSymbol.toUpperCase();
   493→    return searchTerms.some(term => nameUp.includes(term) || tsUp.includes(term));
   494→  });
   495→
   496→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   497→
   498→  // Get unique expiries, sort by nearest
   499→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   500→
   501→  // Use nearest expiry that's not expired
   502→  const today = new Date();
   503→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   504→
   505→  // Filter to nearest expiry
   506→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   507→
   508→  // Get lot size from first option (all same for same underlying)
   509→  const lotSize = expiryOptions[0]?.lotSize || 1;
   510→
   511→  // Derive strike step DYNAMICALLY from actual strike gaps
   512→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   513→  let strikeStep = 50;
   514→  if (strikes.length >= 2) {
   515→    const gaps: Record<number, number> = {};
   516→    for (let i = 1; i < strikes.length; i++) {
   517→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   518→      gaps[gap] = (gaps[gap] || 0) + 1;
   519→    }
   520→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   521→  }
   522→
   523→  // Round ATM strike using the DYNAMIC strike step
   524→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   525→
   526→  // Get strikes around ATM
   527→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   528→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   529→    atmStrike - strikesAround * strikeStep + i * strikeStep
   530→  );
   531→
   532→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   533→
   534→  return {
   535→    instruments: filtered,
   536→    meta: { lotSize, strikeStep },
   537→  };
   538→}

   540→/**
   541→ * Build options flow data from quotes + OI changes
   542→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   543→ */
   544→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   545→  const spec = getInstrumentSpec(symbol);
   546→  if (!spec) return null;
   547→
   548→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   549→  if (optInstruments.length === 0) return null;
   550→
   551→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   552→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   553→
   554→  // Batch fetch quotes
   555→  const quotes = await getQuotes(iKeys);
   556→  if (Object.keys(quotes).length === 0) return null;
   557→
   558→  // Group by strike → build flow data
   559→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   560→
   561→  // Use DYNAMIC strike step and lot size from CSV
   562→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   563→
   564→  for (const inst of optInstruments) {
   565→    // Quotes are keyed by instrument token (string)
   566→    const key = String(inst.instrumentToken);
   567→    const q = quotes[key];
   568→    if (!q) continue;
   569→
   570→    if (!strikes.has(inst.strike)) {
   571→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   572→    }
   573→    const s = strikes.get(inst.strike)!;
   574→
   575→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   576→    const flow = q.volume * meta.lotSize * q.averagePrice;
   577→
   578→    if (inst.tradingSymbol.endsWith('CE')) {
   579→      s.callBuy = flow * 0.6;
   580→      s.callWrite = flow * 0.4;
   581→      s.callOI = q.oi;
   582→      s.callVol = q.volume;
   583→    } else if (inst.tradingSymbol.endsWith('PE')) {
   584→      s.putBuy = flow * 0.6;
   585→      s.putWrite = flow * 0.4;
   586→      s.putOI = q.oi;
   587→      s.putVol = q.volume;
   588→    }
   589→  }
   590→
   591→  // Build result
   592→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   593→    strike,
   594→    isATM: strike === atmStrike,
   595→    ...flow,
   596→    bullishFlow: flow.callBuy + flow.putWrite,
   597→    bearishFlow: flow.putBuy + flow.callWrite,
   598→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   599→  })).sort((a, b) => a.strike - b.strike);
   600→}

   693→/**
   694→ * Derive lot size and strike step from Kite instruments CSV
   695→ * - lotSize: from the option instrument's lot_size field
   696→ * - strikeStep: derived from the gap between adjacent strikes
   697→ */
   698→export async function getInstrumentMeta(
   699→  symbol: string,
   700→  spotPrice: number,
   701→): Promise<InstrumentMeta> {
   702→  const spec = getInstrumentSpec(symbol);
   703→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   704→
   705→  // Fetch instruments from the F&O exchange (NFO/BFO), NOT the cash exchange (NSE/BSE).
   706→  // See getOptionInstruments() for the full explanation.
   707→  const allInstruments = await getInstruments(spec.segment);
   708→
   709→  // Filter to this symbol's options for current/near expiry.
   710→  // Use startsWith on segment to match both legacy 'NFO' and new 'NFO-OPT' formats.
   711→  const symbolOpts = allInstruments.filter(i =>
   712→    i.segment.startsWith(spec.segment) &&
   713→    i.instrumentType === spec.instrumentType &&
   714→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   715→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   716→  );
   717→
   718→  if (symbolOpts.length === 0) {
   719→    return { lotSize: 1, strikeStep: 50 }; // fallback
   720→  }
   721→
   722→  // Get lot size from the first matching option (all have same lot size for same underlying)
   723→  const lotSize = symbolOpts[0].lotSize || 1;
   724→
   725→  // Derive strike step from unique sorted strikes near ATM
   726→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   727→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   728→
   729→  const expiryOpts = nearestExpiry
   730→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   731→    : symbolOpts;
   732→
   733→  // Get unique strikes, sorted
   734→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   735→
   736→  // Find strikes near ATM and compute step from adjacent gaps
   737→  let strikeStep = 50; // default fallback
   738→  if (strikes.length >= 2) {
   739→    // Use the most common gap between adjacent strikes
   740→    const gaps: Record<number, number> = {};
   741→    for (let i = 1; i < strikes.length; i++) {
   742→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   743→      gaps[gap] = (gaps[gap] || 0) + 1;
   744→    }
   745→    // Most frequent gap = the strike step
   746→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   747→  }
   748→
   749→  return { lotSize, strikeStep };
   750→}

   761→// ─── Black-Scholes Delta Calculator ───
   762→// Approximate delta for options when Kite doesn't provide Greeks.
   763→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   764→
   765→function normCDF(x: number): number {
   766→  // Rational approximation (max error ~7.5e-8)
   767→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   768→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   769→  const sign = x < 0 ? -1 : 1;
   770→  const absX = Math.abs(x);
   771→  const t = 1.0 / (1.0 + p * absX);
   772→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   773→  return 0.5 * (1.0 + sign * y);
   774→}
   775→
   776→/**
   777→ * Calculate Black-Scholes delta for an option
   778→ * @param isCall - true for CE, false for PE
   779→ * @param S - Spot price
   780→ * @param K - Strike price
   781→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   782→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   783→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   784→ */
   785→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   786→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   787→  // If no IV provided, estimate from moneyness
   788→  if (sigma <= 0) {
   789→    const moneyness = (S - K) / S;
   790→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   791→  }
   792→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   793→  const delta = normCDF(d1);
   794→  return isCall ? delta : delta - 1;
   795→}

   802→export interface StrikeFlowSnapshot {

   815→  isATM: boolean;
   816→  ceLTP: number;
   817→  peLTP: number;
   818→  ceOI: number;
   819→  peOI: number;

   821→  peVol: number;

   828→/**
   829→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   830→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   831→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   832→ */
   833→export async function getStrikeFlowSnapshot(
   834→  symbol: string,
   835→  spotPrice: number,
   836→): Promise<StrikeFlowSnapshot | null> {
   837→  const spec = getInstrumentSpec(symbol);
   838→  if (!spec) return null;
   839→
   840→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   841→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   842→  if (optInstruments.length === 0) return null;
   843→
   844→  // Batch fetch quotes using instrument tokens
   845→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   846→  const quotes = await getQuotes(iKeys);
   847→  if ('_error' in quotes) return null;
   848→
   849→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   850→
   851→  // Group by strike
   852→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   853→  for (const inst of optInstruments) {
   854→    if (!strikeMap.has(inst.strike)) {
   855→      strikeMap.set(inst.strike, { ce: null, pe: null });
   856→    }
   857→    const entry = strikeMap.get(inst.strike)!;
   858→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   859→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   860→  }
   861→
   862→  // Calculate time to expiry
   863→  const expiry = optInstruments[0]?.expiry || '';
   864→  const now = new Date();
   865→  const expiryDate = new Date(expiry);
   866→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   867→  const T = daysToExpiry / 365;
   868→
   869→  // Build strike data
   870→  const strikes: StrikeFlowData[] = [];
   871→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   872→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   873→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   874→
   875→    const ceLTP = ceQuote?.lastPrice || 0;
   876→    const peLTP = peQuote?.lastPrice || 0;
   877→    const ceOI = ceQuote?.oi || 0;
   878→    const peOI = peQuote?.oi || 0;
   879→    const ceVol = ceQuote?.volume || 0;
   880→    const peVol = peQuote?.volume || 0;
   881→
   882→    // Compute delta via Black-Scholes
   883→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   884→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   885→
   886→    strikes.push({
   887→      strike,
   888→      isATM: strike === atmStrike,
   889→      ceLTP,
   890→      peLTP,
   891→      ceOI,
   892→      peOI,
   893→      ceVol,
   894→      peVol,
   895→      ceDelta,
   896→      peDelta,
   897→      ceToken: ce?.instrumentToken || 0,
   898→      peToken: pe?.instrumentToken || 0,
   899→    });
   900→  }
   901→
   902→  strikes.sort((a, b) => a.strike - b.strike);
   903→
   904→  return {
   905→    timestamp: new Date().toISOString(),
   906→    symbol,
   907→    spotPrice,
   908→    atmStrike,
   909→    lotSize: meta.lotSize,
   910→    strikeStep: meta.strikeStep,
   911→    expiry,
   912→    strikes,
   913→  };
   914→}
   915→

    67→function formatLakhs(n: number): string {
    68→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
    69→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
    70→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
    71→  return n.toLocaleString('en-IN');
    72→}

    78→// Compute Max Pain: strike where option buyers lose the most
    79→function computeMaxPain(strikes: StrikeFlowData[]): number {
    80→  if (strikes.length === 0) return 0;
    81→  let minLoss = Infinity;
    82→  let maxPainStrike = strikes[0].strike;
    83→
    84→  for (const k of strikes) {
    85→    let totalLoss = 0;
    86→    for (const s of strikes) {
    87→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
    88→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
    89→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
    90→      // But we only have OI, not premium. Standard Max Pain uses:
    91→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
    92→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
    93→      // Simplified: use intrinsic value approach
    94→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
    95→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
    96→    }
    97→    if (totalLoss < minLoss) {
    98→      minLoss = totalLoss;
    99→      maxPainStrike = k.strike;
   100→    }
   101→  }
   102→  return maxPainStrike;
   103→}

   105→// Generate demo data for 11 strikes
   106→function generateDemoData(symbol: string): StrikeFlowSnapshot {
   107→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   108→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   109→  const atmStrike = Math.round(spot / step) * step;
   110→  const strikes: StrikeFlowData[] = [];
   111→
   112→  for (let i = -5; i <= 5; i++) {
   113→    const strike = atmStrike + i * step;
   114→    const dist = Math.abs(i);
   115→    // Heavier OI near ATM, with some randomness
   116→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   117→    const noise = 0.6 + Math.random() * 0.8;
   118→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   119→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   120→    const moneyness = (spot - strike) / spot;
   121→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   122→    const ceITM = spot > strike;
   123→    const peITM = spot < strike;
   124→
   125→    strikes.push({
   126→      strike,
   127→      isATM: i === 0,
   128→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   129→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   130→      ceOI,
   131→      peOI,
   132→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   133→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   134→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   135→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   136→      ceToken: 100 + i,
   137→      peToken: 200 + i,
   138→    });
   139→  }
   140→
   141→  return {
   142→    timestamp: new Date().toISOString(),
   143→    symbol,
   144→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   145→    atmStrike,
   146→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   147→    strikeStep: step,
   148→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   149→    strikes,
   150→  };
   151→}

   153→// ═══════════════════════════════════════════
   154→// PCR HISTORY (localStorage)
   155→// ═══════════════════════════════════════════
   156→
   157→function getPCRHistoryKey(symbol: string): string {
   158→  const today = new Date().toISOString().split('T')[0];
   159→  return `oi-walls-pcr-history-${today}-${symbol}`;
   160→}
   161→
   162→function loadPCRHistory(symbol: string): number[] {
   163→  if (typeof window === 'undefined') return [];
   164→  try {
   165→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   166→    return raw ? JSON.parse(raw) : [];
   167→  } catch {
   168→    return [];
   169→  }
   170→}
   171→
   172→function savePCRHistory(symbol: string, history: number[]) {
   173→  if (typeof window === 'undefined') return;
   174→  try {
   175→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   176→  } catch {
   177→    // localStorage full or unavailable
   178→  }
   179→}

   185→export default function OIWallsTab() {
   186→  const [symbol, setSymbol] = useState('NIFTY');
   187→  const [snapshot, setSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   188→  const [loading, setLoading] = useState(false);
   189→  const [error, setError] = useState<string | null>(null);
   190→  const [isLive, setIsLive] = useState(false);
   191→  const [lastUpdate, setLastUpdate] = useState<string>('');
   192→  const [pcrHistory, setPcrHistory] = useState<number[]>([]);
   193→  const [demoData, setDemoData] = useState<StrikeFlowSnapshot | null>(null);
   194→
   195→  const prevSnapshotRef = useRef<StrikeFlowSnapshot | null>(null);
   196→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   197→
   198→  // Load PCR history on mount / symbol change
   199→  useEffect(() => {
   200→    setPcrHistory(loadPCRHistory(symbol));
   201→  }, [symbol]);
   202→
   203→  const fetchData = useCallback(async () => {
   204→    setLoading(true);
   205→    setError(null);
   206→    try {
   207→      const res = await fetch(withCreds(`/api/kite/strike-flow?symbol=${symbol}`));
   208→      const data = await res.json();
   209→
   210→      if (data.mode !== 'live') {
   211→        // Generate demo data
   212→        setIsLive(false);
   213→        const demo = generateDemoData(symbol);
   214→        setDemoData(demo);
   215→        setLastUpdate(new Date().toLocaleTimeString('en-IN', {
   216→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   217→        }));
   218→        setLoading(false);
   219→        return;
   220→      }
   221→
   222→      setIsLive(true);
   223→      setDemoData(null);
   224→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   225→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   226→      }));
   227→
   228→      // Compute OI change before updating ref
   229→      const prev = prevSnapshotRef.current;
   230→      if (prev && prev.symbol === data.symbol) {
   231→        // OI change is computed in metrics; we just need to store prev for diff
   232→      }
   233→      prevSnapshotRef.current = data;
   234→      setSnapshot(data);
   235→    } catch (err) {
   236→      setError(err instanceof Error ? err.message : 'Fetch failed');
   237→    } finally {
   238→      setLoading(false);
   239→    }
   240→  }, [symbol]);
   241→
   242→  // Initial fetch + polling
   243→  useEffect(() => {
   244→    fetchData();
   245→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   246→    return () => {
   247→      if (intervalRef.current) clearInterval(intervalRef.current);
   248→    };
   249→  }, [fetchData]);
   250→
   251→  // Compute OI change using ref to prev snapshot
   252→  const oiChange = (() => {
   253→    const prev = prevSnapshotRef.current;
   254→    const curr = snapshot;
   255→    if (!prev || !curr || prev.symbol !== curr.symbol) {
   256→      // Check demo data diff
   257→      if (demoData) return { ce: 0, pe: 0 };
   258→      return { ce: 0, pe: 0 };
   259→    }
   260→    let ceChange = 0, peChange = 0;
   261→    const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   262→    for (const s of curr.strikes) {
   263→      const p = prevMap.get(s.strike);
   264→      if (p) {
   265→        ceChange += s.ceOI - p.ceOI;
   266→        peChange += s.peOI - p.peOI;
   267→      }
   268→    }
   269→    return { ce: ceChange, pe: peChange };
   270→  })();
   271→
   272→  // Use demo or live data
   273→  const data = isLive ? snapshot : demoData;
   274→
   275→  // Compute walls and metrics
   276→  const { walls, metrics } = (() => {
   277→    if (!data || !data.strikes?.length) {
   278→      return { walls: [] as OIWallsStrike[], metrics: null as ComputedMetrics | null };
   279→    }
   280→
   281→    const maxOI = Math.max(
   282→      ...data.strikes.map(s => Math.max(s.ceOI, s.peOI)),
   283→      1
   284→    );
   285→
   286→    const walls: OIWallsStrike[] = data.strikes.map(s => ({
   287→      strike: s.strike,
   288→      isATM: s.isATM,
   289→      ceOI: s.ceOI,
   290→      peOI: s.peOI,
   291→      ceLTP: s.ceLTP,
   292→      peLTP: s.peLTP,
   293→      pePct: (s.peOI / maxOI) * 100,
   294→      cePct: (s.ceOI / maxOI) * 100,
   295→    }));
   296→
   297→    const totalCEOI = data.strikes.reduce((sum, s) => sum + s.ceOI, 0);
   298→    const totalPEOI = data.strikes.reduce((sum, s) => sum + s.peOI, 0);
   299→    const pcr = totalCEOI > 0 ? totalPEOI / totalCEOI : 0;
   300→    const maxPain = computeMaxPain(data.strikes);
   301→    const maxPainDist = data.spotPrice - maxPain;
   302→
   303→    const m: ComputedMetrics = {
   304→      pcr,
   305→      maxPain,
   306→      maxPainDist,
   307→      totalCEOI,
   308→      totalPEOI,
   309→      totalOI: totalCEOI + totalPEOI,
   310→      ceOIChange: oiChange.ce,
   311→      peOIChange: oiChange.pe,
   312→    };
   313→
   314→    return { walls, metrics: m };
   315→  })();
   316→
   317→  // Append PCR to history when metrics change
   318→  useEffect(() => {
   319→    if (!metrics) return;
   320→    setPcrHistory(prev => {
   321→      const updated = [...prev, metrics.pcr];
   322→      const trimmed = updated.slice(-60);
   323→      savePCRHistory(symbol, trimmed);
   324→      return trimmed;
   325→    });
   326→  }, [isLive ? snapshot?.timestamp : demoData?.timestamp, metrics?.pcr, symbol]);
   327→
   328→  // PCR label
   329→  function getPCRLabel(pcr: number): { text: string; color: string } {
   330→    if (pcr >= 1.5) return { text: 'Extreme Bearish', color: 'text-emerald-400' };
   331→    if (pcr >= 1.1) return { text: 'Bearish', color: 'text-emerald-300' };
   332→    if (pcr >= 0.9) return { text: 'Neutral', color: 'text-zinc-400' };
   333→    if (pcr >= 0.6) return { text: 'Bullish', color: 'text-red-400' };
   334→    return { text: 'Extreme Bullish', color: 'text-red-500' };
   335→  }
   336→
   337→  // Spot price position (0-100%) within strike range for the dashed line
   338→  const spotPosition = (() => {
   339→    if (!data || walls.length < 2) return 50;
   340→    const minStrike = walls[0].strike;
   341→    const maxStrike = walls[walls.length - 1].strike;
   342→    const range = maxStrike - minStrike;
   343→    if (range === 0) return 50;
   344→    return Math.max(0, Math.min(100, ((data.spotPrice - minStrike) / range) * 100));
   345→  })();
   346→
   347→  // ═══════════════════════════════════════════
   348→  // PCR SPARKLINE (SVG)
   349→  // ═══════════════════════════════════════════
   350→  function PCRSparkline() {
   351→    if (pcrHistory.length < 2) {
   352→      return (
   353→        <div className="flex items-center justify-center h-10 text-zinc-600 text-[10px]">
   354→          Waiting for data points...
   355→        </div>
   356→      );
   357→    }
   358→
   359→    const w = 280, h = 36;
   360→    const vals = pcrHistory;
   361→    const minV = Math.min(...vals) * 0.95;
   362→    const maxV = Math.max(...vals) * 1.05;
   363→    const rangeV = maxV - minV || 1;
   364→    const step = w / (vals.length - 1);
   365→
   366→    const points = vals.map((v, i) => {
   367→      const x = i * step;
   368→      const y = h - 2 - ((v - minV) / rangeV) * (h - 4);
   369→      return `${x},${y}`;
   370→    }).join(' ');
   371→
   372→    // PCR = 1 reference line
   373→    const pcr1Y = h - 2 - ((1 - minV) / rangeV) * (h - 4);
   374→    const lastVal = vals[vals.length - 1];
   375→    const lastX = (vals.length - 1) * step;
   376→    const lastY = h - 2 - ((lastVal - minV) / rangeV) * (h - 4);
   377→    const lineColor = lastVal >= 1 ? '#34d399' : '#f87171';
   378→
   379→    return (
   380→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   381→        {/* PCR = 1 line */}
   382→        {pcr1Y > 0 && pcr1Y < h && (
   383→          <line x1="0" y1={pcr1Y} x2={w} y2={pcr1Y}
   384→            stroke="#525252" strokeWidth="0.5" strokeDasharray="3,3" />
   385→        )}
   386→        {/* Area fill */}
   387→        <polygon
   388→          points={`0,${h} ${points} ${lastX},${h}`}
   389→          fill={lastVal >= 1 ? 'rgba(52,211,153,0.1)' : 'rgba(248,113,113,0.1)'}
   390→        />
   391→        {/* Line */}
   392→        <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" />
   393→        {/* End dot */}
   394→        <circle cx={lastX} cy={lastY} r="2.5" fill={lineColor} />
   395→      </svg>
   396→    );
   397→  }

   408→          <button
   409→            key={s.symbol}
   410→            onClick={() => {

   412→              setSnapshot(null);

   414→              prevSnapshotRef.current = null;
   415→            }}
   416→            className={
   417→              `text-[10px] px-2 py-1 rounded border transition-all duration-150 cursor-pointer whitespace-nowrap ` +
   418→              (symbol === s.symbol
   419→                ? 'bg-primary/20 border-primary/50 text-primary-foreground font-semibold'

   421→            }
   422→          >
   423→            {s.symbol}
   424→          </button>

   426→      </div>
   427→
   428→      {/* ─── Status Bar ─── */}
   429→      <div className="flex items-center justify-between">
   430→        <div className="flex items-center gap-2">
   431→          {isLive ? (
   432→            <Badge variant="outline" className="gap-1 text-emerald-400 border-emerald-500/30 bg-emerald-500/5">

   449→              Spot: <span className="text-foreground font-semibold">{formatNumber(data.spotPrice)}</span>

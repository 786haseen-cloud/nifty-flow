    14→/**
    15→ * 4-Color OI classification (matches Strike Flow Map convention):
    16→ *
    17→ *  CE  Buy   = ΔOI>0 & ΔLTP≥0  → bright green  #00B050
    18→ *  CE  Write = ΔOI>0 & ΔLTP<0  → light red     #FFCCCC (dark red border)
    19→ *  CE  Close = ΔOI<0           → grey          #6b7280
    20→ *  PE  Write = ΔOI>0 & ΔLTP≥0  → light green   #C6EFCE (dark green border)
    21→ *  PE  Buy   = ΔOI>0 & ΔLTP<0  → bright red    #BD2130
    22→ *  PE  Close = ΔOI<0           → grey          #6b7280
    23→ */
    24→type OIColor = {
    25→  bg: string;        // inline background color
    26→  border?: string;   // inline border color (for Write classes)
    27→  text: string;      // tailwind text class for the OI label
    28→  label?: string;    // short label: "Buy", "Write", "Close"
    29→};
    30→
    31→function getOIColor(oiChange: number, ltpChange: number, isPE: boolean): OIColor {
    32→  // No previous data — neutral
    33→  if (oiChange === 0 && ltpChange === 0) {
    34→    return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    35→  }
    36→  // Unwinding / short covering (ΔOI < 0)
    37→  if (oiChange < 0) {
    38→    return { bg: '#6b7280', text: 'text-zinc-400', label: 'Close' };
    39→  }
    40→  // oiChange >= 0 — new positions opened (or no change)
    41→  if (isPE) {
    42→    // PE: Write = bullish (price up/stable + OI up), Buy = bearish (price down + OI up)
    43→    if (ltpChange >= 0) return { bg: '#C6EFCE', border: '#16a34a', text: 'text-emerald-700', label: 'Write' };
    44→    if (ltpChange < 0)  return { bg: '#BD2130', text: 'text-red-100', label: 'Buy' };
    45→  } else {
    46→    // CE: Buy = bullish (price up/stable + OI up), Write = bearish (price down + OI up)
    47→    if (ltpChange >= 0) return { bg: '#00B050', text: 'text-green-100', label: 'Buy' };
    48→    if (ltpChange < 0)  return { bg: '#FFCCCC', border: '#dc2626', text: 'text-red-700', label: 'Write' };
    49→  }
    50→  // Fallback (shouldn't reach here)
    51→  return { bg: '#3f3f46', text: 'text-zinc-400/80' };
    52→}

   138→function formatLakhs(n: number): string {
   139→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
   140→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
   141→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
   142→  return n.toLocaleString('en-IN');
   143→}

   149→// Compute Max Pain: strike where option buyers lose the most
   150→function computeMaxPain(strikes: StrikeFlowData[]): number {
   151→  if (strikes.length === 0) return 0;
   152→  let minLoss = Infinity;
   153→  let maxPainStrike = strikes[0].strike;
   154→
   155→  for (const k of strikes) {
   156→    let totalLoss = 0;
   157→    for (const s of strikes) {
   158→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
   159→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
   160→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
   161→      // But we only have OI, not premium. Standard Max Pain uses:
   162→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
   163→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
   164→      // Simplified: use intrinsic value approach
   165→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
   166→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
   167→    }
   168→    if (totalLoss < minLoss) {
   169→      minLoss = totalLoss;
   170→      maxPainStrike = k.strike;
   171→    }
   172→  }
   173→  return maxPainStrike;
   174→}

   176→// Generate demo data for 11 strikes
   177→function generateDemoData(symbol: string, prev?: StrikeFlowSnapshot | null): StrikeFlowSnapshot {
   178→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   179→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   180→  const atmStrike = Math.round(spot / step) * step;
   181→
   182→  // If we have a previous snapshot, apply small perturbations to it.
   183→  // This produces realistic per-strike ΔOI and ΔLTP values that
   184→  // feed into the 4-color OI coding (Buy/Write/Close).
   185→  if (prev && prev.strikes.length === 11) {
   186→    const spotJitter = Math.round((Math.random() - 0.5) * step * 0.3);
   187→    const newSpot = prev.spotPrice + spotJitter;
   188→    const newAtm = Math.round(newSpot / step) * step;
   189→
   190→    const strikes: StrikeFlowData[] = prev.strikes.map((s, idx) => {
   191→      // OI change: random walk — some strikes gain, some lose
   192→      const oiDelta = Math.round((Math.random() - 0.45) * s.ceOI * 0.04);
   193→      const peOiDelta = Math.round((Math.random() - 0.45) * s.peOI * 0.04);
   194→
   195→      // LTP change based on spot movement + moneyness
   196→      const ceMoneyness = (newSpot - s.strike) / newSpot;
   197→      const peMoneyness = (s.strike - newSpot) / newSpot;
   198→      const iv = 0.12 + Math.abs(ceMoneyness) * 0.3;
   199→      const ceLTPDelta = spotJitter > 0
   200→        ? iv * Math.abs(spotJitter) * (0.3 + Math.random() * 0.3)
   201→        : -iv * Math.abs(spotJitter) * (0.2 + Math.random() * 0.3);
   202→      const peLTPDelta = -ceLTPDelta * 0.8;
   203→
   204→      const ceITM = newSpot > s.strike;
   205→      const peITM = newSpot < s.strike;
   206→
   207→      return {
   208→        ...s,
   209→        ceLTP: Math.max(0.5, s.ceLTP + ceLTPDelta),
   210→        peLTP: Math.max(0.5, s.peLTP + peLTPDelta),
   211→        ceOI: Math.max(1000, s.ceOI + oiDelta),
   212→        peOI: Math.max(1000, s.peOI + peOiDelta),
   213→        isATM: s.strike === newAtm,
   214→        ceVol: Math.round(Math.abs(oiDelta) * (1 + Math.random())),
   215→        peVol: Math.round(Math.abs(peOiDelta) * (1 + Math.random())),
   216→      };
   217→    });
   218→
   219→    return {
   220→      timestamp: new Date().toISOString(),
   221→      symbol,
   222→      spotPrice: newSpot,
   223→      atmStrike: newAtm,
   224→      lotSize: prev.lotSize,
   225→      strikeStep: step,
   226→      expiry: prev.expiry,
   227→      strikes,
   228→    };
   229→  }
   230→
   231→  // First call (no prev) — generate from scratch with random initial values
   232→  const strikes: StrikeFlowData[] = [];
   233→
   234→  for (let i = -5; i <= 5; i++) {
   235→    const strike = atmStrike + i * step;
   236→    const dist = Math.abs(i);
   237→    // Heavier OI near ATM, with some randomness
   238→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   239→    const noise = 0.6 + Math.random() * 0.8;
   240→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   241→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   242→    const moneyness = (spot - strike) / spot;
   243→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   244→    const ceITM = spot > strike;
   245→    const peITM = spot < strike;
   246→
   247→    strikes.push({
   248→      strike,
   249→      isATM: i === 0,
   250→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   251→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   252→      ceOI,
   253→      peOI,
   254→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   255→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   256→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   257→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   258→      ceToken: 100 + i,
   259→      peToken: 200 + i,
   260→    });
   261→  }
   262→
   263→  return {
   264→    timestamp: new Date().toISOString(),
   265→    symbol,
   266→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   267→    atmStrike,
   268→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   269→    strikeStep: step,
   270→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   271→    strikes,
   272→  };
   273→}

   276→// PCR HISTORY (localStorage)
   277→// ═══════════════════════════════════════════
   278→
   279→function getPCRHistoryKey(symbol: string): string {
   280→  const today = new Date().toISOString().split('T')[0];
   281→  return `oi-walls-pcr-history-${today}-${symbol}`;
   282→}
   283→
   284→function loadPCRHistory(symbol: string): number[] {
   285→  if (typeof window === 'undefined') return [];
   286→  try {
   287→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   288→    return raw ? JSON.parse(raw) : [];
   289→  } catch {
   290→    return [];
   291→  }
   292→}
   293→
   294→function savePCRHistory(symbol: string, history: number[]) {
   295→  if (typeof window === 'undefined') return;
   296→  try {
   297→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   298→  } catch {
   299→    // localStorage full or unavailable
   300→  }
   301→}

   307→export default function OIWallsTab() {
   308→  const [symbol, setSymbol] = useState('NIFTY');
   309→  const [snapshot, setSnapshot] = useState<StrikeFlowSnapshot | null>(null);
   310→  const [loading, setLoading] = useState(false);
   311→  const [error, setError] = useState<string | null>(null);
   312→  const [isLive, setIsLive] = useState(false);
   313→  const [lastUpdate, setLastUpdate] = useState<string>('');
   314→  const [pcrHistory, setPcrHistory] = useState<number[]>([]);
   315→  const [demoData, setDemoData] = useState<StrikeFlowSnapshot | null>(null);
   316→  const [scanData, setScanData] = useState<MaxPainScanItem[]>([]);
   317→  const [scanLoading, setScanLoading] = useState(false);
   318→  const [scanError, setScanError] = useState<string | null>(null);
   319→  const [gravitySignal, setGravitySignal] = useState<GravitySignal | null>(null);
   320→  // Pre-computed per-strike deltas (computed in fetchData, read during render)
   321→  const [strikeDeltas, setStrikeDeltas] = useState<Map<number, { ceOi: number; peOi: number; ceLtp: number; peLtp: number }>>(new Map());
   322→
   323→  const prevSnapshotRef = useRef<StrikeFlowSnapshot | null>(null);
   324→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   325→
   326→  // Load PCR history on mount / symbol change
   327→  useEffect(() => {
   328→    setPcrHistory(loadPCRHistory(symbol));
   329→  }, [symbol]);
   330→
   331→  const fetchData = useCallback(async () => {
   332→    setLoading(true);
   333→    setError(null);
   334→    try {
   335→      const res = await fetch(withCreds(`/api/kite/strike-flow?symbol=${symbol}`));
   336→      const data = await res.json();
   337→
   338→      if (data.mode !== 'live') {
   339→        // Demo mode — store prev snapshot for delta computation,
   340→        // then generate fresh demo data with small perturbations.
   341→        // This enables 4-color OI coding even in demo mode.
   342→        const prev = prevSnapshotRef.current;
   343→        const demo = generateDemoData(symbol, prev);
   344→        // Compute per-strike deltas BEFORE updating the ref
   345→        if (prev && prev.strikes) {
   346→          const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   347→          const d = new Map<number, { ceOi: number; peOi: number; ceLtp: number; peLtp: number }>();
   348→          for (const s of demo.strikes) {
   349→            const p = prevMap.get(s.strike);
   350→            if (p) d.set(s.strike, { ceOi: s.ceOI - p.ceOI, peOi: s.peOI - p.peOI, ceLtp: s.ceLTP - p.ceLTP, peLtp: s.peLTP - p.peLTP });
   351→          }
   352→          setStrikeDeltas(d);
   353→        } else {
   354→          setStrikeDeltas(new Map());
   355→        }
   356→        prevSnapshotRef.current = demo;
   357→        setIsLive(false);
   358→        setDemoData(demo);
   359→        setLastUpdate(new Date().toLocaleTimeString('en-IN', {
   360→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   361→        }));
   362→        setLoading(false);
   363→        return;
   364→      }
   365→
   366→      setIsLive(true);
   367→      setDemoData(null);
   368→      setLastUpdate(new Date(data.timestamp).toLocaleTimeString('en-IN', {
   369→        hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
   370→      }));
   371→
   372→      // Compute per-strike deltas BEFORE updating the ref
   373→      const prev = prevSnapshotRef.current;
   374→      if (prev && prev.symbol === data.symbol && prev.strikes) {
   375→        const prevMap = new Map(prev.strikes.map(s => [s.strike, s]));
   376→        const d = new Map<number, { ceOi: number; peOi: number; ceLtp: number; peLtp: number }>();
   377→        for (const s of data.strikes) {
   378→          const p = prevMap.get(s.strike);
   379→          if (p) d.set(s.strike, { ceOi: s.ceOI - p.ceOI, peOi: s.peOI - p.peOI, ceLtp: s.ceLTP - p.ceLTP, peLtp: s.peLTP - p.peLTP });
   380→        }
   381→        setStrikeDeltas(d);
   382→      } else {
   383→        setStrikeDeltas(new Map());
   384→      }
   385→      prevSnapshotRef.current = data;
   386→      setSnapshot(data);
   387→    } catch (err) {
   388→      setError(err instanceof Error ? err.message : 'Fetch failed');
   389→    } finally {
   390→      setLoading(false);
   391→    }
   392→  }, [symbol]);
   393→
   394→  // Initial fetch + polling for per-symbol OI walls
   395→  useEffect(() => {
   396→    fetchData();
   397→    intervalRef.current = setInterval(fetchData, REFRESH_INTERVAL);
   398→    return () => {
   399→      if (intervalRef.current) clearInterval(intervalRef.current);
   400→    };
   401→  }, [fetchData]);
   402→
   403→  // Fetch max pain scan for all symbols (longer interval — 120s)
   404→  const scanIntervalRef = useRef<NodeJS.Timeout | null>(null);
   405→  const fetchScan = useCallback(async () => {
   406→    setScanLoading(true);
   407→    setScanError(null);
   408→    try {
   409→      const res = await fetch(withCreds('/api/kite/max-pain-scan'));
   410→      const json = await res.json();
   411→
   412→      if (json.mode === 'live' && json.symbols?.length > 0) {
   413→        const items: MaxPainScanItem[] = json.symbols;
   414→        setScanData(items);
   415→
   416→        // Compute gravity signal — works BOTH directions
   417→        const indices = items.filter(s => s.type === 'index');
   418→        const stocks = items.filter(s => s.type === 'stock');
   419→        const indicesAbove = indices.filter(s => s.dist > 0).length;
   420→        const indicesBelow = indices.filter(s => s.dist < 0).length;
   421→        const stocksAbove = stocks.filter(s => s.dist > 0).length;
   422→        const stocksBelow = stocks.filter(s => s.dist < 0).length;
   423→        const avgIdxDist = indices.length > 0 ? indices.reduce((s, i) => s + i.distPct, 0) / indices.length : 0;
   424→        const avgStkDist = stocks.length > 0 ? stocks.reduce((s, i) => s + i.distPct, 0) / stocks.length : 0;
   425→
   426→        // Determine direction: which side has more index alignment?
   427→        let direction: GravitySignal['direction'] = 'mixed';
   428→        if (indicesAbove > indicesBelow) direction = 'down';   // spot above MP → gravity pulls DOWN
   429→        else if (indicesBelow > indicesAbove) direction = 'up'; // spot below MP → gravity pulls UP
   430→
   431→        // Determine aligned count (whichever direction is dominant)
   432→        const alignedIndices = Math.max(indicesAbove, indicesBelow);
   433→        const alignedStocks = direction === 'down' ? stocksAbove : stocksBelow;
   434→        const stockAvgConfirms = direction === 'down' ? avgStkDist > 0.1 : avgStkDist < -0.1;
   435→
   436→        let strength: GravitySignal['strength'] = 'divergent';
   437→        if (alignedIndices === 4 && stockAvgConfirms) {
   438→          strength = 'strong';
   439→        } else if (alignedIndices >= 3 && stockAvgConfirms) {
   440→          strength = 'moderate';
   441→        } else if (alignedIndices >= 2 && direction !== 'mixed') {
   442→          strength = 'weak';
   443→        }
   444→
   445→        setGravitySignal({
   446→          indicesAbove,
   447→          indicesBelow,
   448→          indicesTotal: indices.length,
   449→          stocksAbove,
   450→          stocksBelow,
   451→          stocksTotal: stocks.length,
   452→          avgIndexDistPct: avgIdxDist,
   453→          avgStockDistPct: avgStkDist,
   454→          direction,
   455→          strength,
   456→        });
   457→      } else if (json.mode === 'demo') {
   458→        setScanData([]);
   459→        setGravitySignal(null);
   460→      } else if (json.error) {
   461→        setScanError(json.error);
   462→      }
   463→    } catch (err) {
   464→      setScanError(err instanceof Error ? err.message : 'Scan failed');
   465→    } finally {
   466→      setScanLoading(false);
   467→    }
   468→  }, []);
   469→
   470→  useEffect(() => {
   471→    fetchScan();
   472→    scanIntervalRef.current = setInterval(fetchScan, 120000); // every 2 min
   473→    return () => {
   474→      if (scanIntervalRef.current) clearInterval(scanIntervalRef.current);
   475→    };
   476→  }, [fetchScan]);
   477→
   478→  // Compute OI change using ref to prev snapshot
   479→  const oiChange = (() => {

   522→        peOiChg = d.peOi;
   523→        ceLtpChg = d.ceLtp;

   527→        // Infer activity from OI concentration patterns.

   529→        const avgPeOI = data.strikes.reduce((sum, x) => sum + x.peOI, 0) / data.strikes.length;
   530→        const ceRatio = s.ceOI / avgCeOI;

   539→        peLtpChg = s.peLTP * 0.002 * (Math.random() - 0.7);

   544→        isATM: s.isATM,

   603→  function PCRSparkline() {

   615→    const maxV = Math.max(...vals) * 1.05;
   616→    const rangeV = maxV - minV || 1;
   617→    const step = w / (vals.length - 1);
   618→
   619→    const points = vals.map((v, i) => {
   620→      const x = i * step;
   621→      const y = h - 2 - ((v - minV) / rangeV) * (h - 4);
   622→      return `${x},${y}`;
   623→    }).join(' ');
   624→
   625→    // PCR = 1 reference line
   626→    const pcr1Y = h - 2 - ((1 - minV) / rangeV) * (h - 4);
   627→    const lastVal = vals[vals.length - 1];
   628→    const lastX = (vals.length - 1) * step;
   629→    const lastY = h - 2 - ((lastVal - minV) / rangeV) * (h - 4);
   630→    const lineColor = lastVal >= 1 ? '#34d399' : '#f87171';
   631→
   632→    return (
   633→      <svg width="100%" height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   634→        {/* PCR = 1 line */}
   635→        {pcr1Y > 0 && pcr1Y < h && (
   636→          <line x1="0" y1={pcr1Y} x2={w} y2={pcr1Y}
   637→            stroke="#525252" strokeWidth="0.5" strokeDasharray="3,3" />
   638→        )}
   639→        {/* Area fill */}
   640→        <polygon
   641→          points={`0,${h} ${points} ${lastX},${h}`}
   642→          fill={lastVal >= 1 ? 'rgba(52,211,153,0.1)' : 'rgba(248,113,113,0.1)'}
   643→        />
   644→        {/* Line */}
   645→        <polyline points={points} fill="none" stroke={lineColor} strokeWidth="1.5" />
   646→        {/* End dot */}
   647→        <circle cx={lastX} cy={lastY} r="2.5" fill={lineColor} />
   648→      </svg>

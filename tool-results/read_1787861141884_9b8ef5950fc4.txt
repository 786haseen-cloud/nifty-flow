    67→    67→function formatLakhs(n: number): string {
    68→    68→  if (n >= 10000000) return (n / 10000000).toFixed(2) + ' Cr';
    69→    69→  if (n >= 100000) return (n / 100000).toFixed(2) + ' L';
    70→    70→  if (n >= 1000) return (n / 1000).toFixed(1) + 'K';
    71→    71→  return n.toLocaleString('en-IN');
    72→    72→}

    78→    78→// Compute Max Pain: strike where option buyers lose the most
    79→    79→function computeMaxPain(strikes: StrikeFlowData[]): number {
    80→    80→  if (strikes.length === 0) return 0;
    81→    81→  let minLoss = Infinity;
    82→    82→  let maxPainStrike = strikes[0].strike;
    83→    83→
    84→    84→  for (const k of strikes) {
    85→    85→    let totalLoss = 0;
    86→    86→    for (const s of strikes) {
    87→    87→      // CE buyer loses if spot expires at K below the CE strike: CE worthless
    88→    88→      // Actually: CE buyer loss = max(0, K_expiry - S_strike) * CE_OI... no.
    89→    89→      // At expiry price K, CE with strike S > K is worthless → buyer loses premium paid.
    90→    90→      // But we only have OI, not premium. Standard Max Pain uses:
    91→    91→      // CE buyer loss at expiry K = Σ max(0, strike_i - K) * CE_OI_i
    92→    92→      // (CE is ITM and worth (strike - K), but buyer already paid more than intrinsic...)
    93→    93→      // Simplified: use intrinsic value approach
    94→    94→      totalLoss += Math.max(0, s.strike - k.strike) * s.ceOI;
    95→    95→      totalLoss += Math.max(0, k.strike - s.strike) * s.peOI;
    96→    96→    }
    97→    97→    if (totalLoss < minLoss) {
    98→    98→      minLoss = totalLoss;
    99→    99→      maxPainStrike = k.strike;
   100→   100→    }
   101→   101→  }
   102→   102→  return maxPainStrike;
   103→   103→}

   105→   105→// Generate demo data for 11 strikes
   106→   106→function generateDemoData(symbol: string): StrikeFlowSnapshot {
   107→   107→  const spot = DEMO_BASE_PRICES[symbol] || 24350;
   108→   108→  const step = DEMO_STRIKE_STEPS[symbol] || 50;
   109→   109→  const atmStrike = Math.round(spot / step) * step;
   110→   110→  const strikes: StrikeFlowData[] = [];
   111→   111→
   112→   112→  for (let i = -5; i <= 5; i++) {
   113→   113→    const strike = atmStrike + i * step;
   114→   114→    const dist = Math.abs(i);
   115→   115→    // Heavier OI near ATM, with some randomness
   116→   116→    const baseFactor = Math.max(0.15, 1 - dist * 0.15);
   117→   117→    const noise = 0.6 + Math.random() * 0.8;
   118→   118→    const ceOI = Math.round(800000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   119→   119→    const peOI = Math.round(700000 * baseFactor * noise * (0.8 + Math.random() * 0.4));
   120→   120→    const moneyness = (spot - strike) / spot;
   121→   121→    const iv = 0.12 + Math.abs(moneyness) * 0.3;
   122→   122→    const ceITM = spot > strike;
   123→   123→    const peITM = spot < strike;
   124→   124→
   125→   125→    strikes.push({
   126→   126→      strike,
   127→   127→      isATM: i === 0,
   128→   128→      ceLTP: ceITM ? (spot - strike) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   129→   129→      peLTP: peITM ? (strike - spot) + iv * spot * 0.1 : iv * spot * 0.05 * (1 - dist * 0.1),
   130→   130→      ceOI,
   131→   131→      peOI,
   132→   132→      ceVol: Math.round(ceOI * (0.02 + Math.random() * 0.08)),
   133→   133→      peVol: Math.round(peOI * (0.02 + Math.random() * 0.08)),
   134→   134→      ceDelta: 0.5 + (i === 0 ? 0 : (spot > strike ? 0.15 : -0.15)),
   135→   135→      peDelta: 0.5 + (i === 0 ? 0 : (spot < strike ? 0.15 : -0.15)),
   136→   136→      ceToken: 100 + i,
   137→   137→      peToken: 200 + i,
   138→   138→    });
   139→   139→  }
   140→   140→
   141→   141→  return {
   142→   142→    timestamp: new Date().toISOString(),
   143→   143→    symbol,
   144→   144→    spotPrice: spot + Math.round((Math.random() - 0.5) * step),
   145→   145→    atmStrike,
   146→   146→    lotSize: symbol === 'BANKNIFTY' ? 30 : symbol === 'NIFTY' ? 75 : 1,
   147→   147→    strikeStep: step,
   148→   148→    expiry: new Date(Date.now() + 3 * 86400000).toISOString().split('T')[0],
   149→   149→    strikes,
   150→   150→  };
   151→   151→}

   157→   157→function getPCRHistoryKey(symbol: string): string {
   158→   158→  const today = new Date().toISOString().split('T')[0];
   159→   159→  return `oi-walls-pcr-history-${today}-${symbol}`;

   161→   161→
   162→   162→function loadPCRHistory(symbol: string): number[] {
   163→   163→  if (typeof window === 'undefined') return [];
   164→   164→  try {
   165→   165→    const raw = localStorage.getItem(getPCRHistoryKey(symbol));
   166→   166→    return raw ? JSON.parse(raw) : [];
   167→   167→  } catch {
   168→   168→    return [];
   169→   169→  }
   170→   170→}
   171→   171→
   172→   172→function savePCRHistory(symbol: string, history: number[]) {
   173→   173→  if (typeof window === 'undefined') return;
   174→   174→  try {
   175→   175→    localStorage.setItem(getPCRHistoryKey(symbol), JSON.stringify(history.slice(-60)));
   176→   176→  } catch {
   177→   177→    // localStorage full or unavailable
   178→   178→  }
   179→   179→}

   384→   384→            stroke="#525252" strokeWidth="0.5" strokeDasharray="3,3" />
   385→   385→        )}
   386→   386→        {/* Area fill */}

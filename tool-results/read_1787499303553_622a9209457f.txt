    94→    94→/**
    95→    95→ * Generate access_token from request_token (one-time, after login redirect)
    96→    96→ */
    97→    97→export async function generateSession(requestToken: string): Promise<{
    98→    98→  accessToken: string;
    99→    99→  refreshToken: string;
   100→   100→} | null> {
   101→   101→  const apiKey = process.env.KITE_API_KEY;
   102→   102→  const apiSecret = process.env.KITE_API_SECRET;
   103→   103→  if (!apiKey || !apiSecret) return null;
   104→   104→
   105→   105→  try {
   106→   106→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   107→   107→    const { createHash } = await import('crypto');
   108→   108→    const checksum = createHash('sha256')
   109→   109→      .update(apiKey + requestToken + apiSecret)
   110→   110→      .digest('hex');
   111→   111→
   112→   112→    const res = await fetch(`${KITE_BASE}/session/token`, {
   113→   113→      method: 'POST',
   114→   114→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   115→   115→      body: new URLSearchParams({
   116→   116→        api_key: apiKey,
   117→   117→        request_token: requestToken,
   118→   118→        checksum,
   119→   119→      }).toString(),
   120→   120→    });
   121→   121→
   122→   122→    const data = await res.json();
   123→   123→    if (data.status === 'success' && data.data) {
   124→   124→      return {
   125→   125→        accessToken: data.data.access_token,
   126→   126→        refreshToken: data.data.refresh_token || '',
   127→   127→      };
   128→   128→    }
   129→   129→    console.error('[Kite] Session error:', data.message);
   130→   130→    return null;
   131→   131→  } catch (err) {
   132→   132→    console.error('[Kite] generateSession failed:', err);
   133→   133→    return null;
   134→   134→  }
   135→   135→}

   137→   137→// ─── Instruments ───
   138→   138→
   139→   139→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   140→   140→let instrumentsCache: KiteInstrument[] | null = null;
   141→   141→let instrumentsCacheTime = 0;
   142→   142→
   143→   143→/**
   144→   144→ * Fetch all instruments from Kite (cached for 1 hour)
   145→   145→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   146→   146→ * We filter by exchange after caching
   147→   147→ */
   148→   148→export async function getInstruments(exchange?: string): Promise<KiteInstrument[]> {
   149→   149→  // Return from cache if fresh
   150→   150→  if (instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   151→   151→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   152→   152→  }
   153→   153→
   154→   154→  try {
   155→   155→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   156→   156→    if (!res.ok) {
   157→   157→      const errText = await res.text();
   158→   158→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   159→   159→      return [];
   160→   160→    }
   161→   161→    const text = await res.text();
   162→   162→
   163→   163→    // Kite returns CSV format
   164→   164→    const lines = text.trim().split('\n');
   165→   165→    const instruments: KiteInstrument[] = [];
   166→   166→
   167→   167→    // Kite CSV column order (2025+ format):
   168→   168→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   169→   169→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   170→   170→    for (let i = 1; i < lines.length; i++) { // skip header
   171→   171→      const cols = lines[i].split(',');
   172→   172→      if (cols.length < 12) continue;
   173→   173→      instruments.push({
   174→   174→        instrumentToken: parseInt(cols[0]) || 0,
   175→   175→        exchangeToken: parseInt(cols[1]) || 0,
   176→   176→        tradingSymbol: cols[2],
   177→   177→        name: cols[3] || '',
   178→   178→        exchange: cols[11] || '',
   179→   179→        segment: cols[10] || '',
   180→   180→        instrumentType: cols[9] || '',
   181→   181→        strike: parseFloat(cols[6]) || 0,
   182→   182→        lotSize: parseInt(cols[8]) || 1,
   183→   183→        expiry: cols[5] || '',
   184→   184→      });
   185→   185→    }
   186→   186→
   187→   187→    instrumentsCache = instruments;
   188→   188→    instrumentsCacheTime = Date.now();
   189→   189→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   190→   190→  } catch (err) {
   191→   191→    const errMsg = err instanceof Error ? err.message : String(err);
   192→   192→    console.error('[Kite] instruments fetch failed:', errMsg);
   193→   193→    // Return empty but set error flag for callers to detect
   194→   194→    return [];
   195→   195→  }
   196→   196→}

   198→   198→// ─── Quotes (Real-time) ───
   199→   199→
   200→   200→/**
   201→   201→ * Get real-time quotes for instruments
   202→   202→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   203→   203→ */
   204→   204→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   205→   205→  if (!isKiteConfigured()) return {};
   206→   206→
   207→   207→  try {
   208→   208→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   209→   209→    // Tokens have no encoding issues on any platform.
   210→   210→    // Build a map from token → original instrument key for response mapping.
   211→   211→    const tokenToKey: Record<string, string> = {};
   212→   212→    const tokenList: string[] = [];
   213→   213→
   214→   214→    for (const inst of instruments) {
   215→   215→      // If it's a numeric token, use directly
   216→   216→      if (/^\d+$/.test(inst)) {
   217→   217→        tokenToKey[inst] = inst;
   218→   218→        tokenList.push(inst);
   219→   219→      } else {
   220→   220→        // Look up token from KITE_INDEX_INSTRUMENTS
   221→   221→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   222→   222→          ([, v]) => v.symbol === inst
   223→   223→        );
   224→   224→        if (found) {
   225→   225→          const tok = String(found[1].token);
   226→   226→          tokenToKey[tok] = inst; // map token back to symbol name
   227→   227→          tokenList.push(tok);
   228→   228→        } else {
   229→   229→          // Fallback: use the instrument string as-is with encodeURI
   230→   230→          tokenToKey[inst] = inst;
   231→   231→          tokenList.push(inst);
   232→   232→        }
   233→   233→      }
   234→   234→    }
   235→   235→
   236→   236→    const iList = tokenList.join(',');
   237→   237→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   238→   238→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   239→   239→    const params = tokenList.map(t => `i=${t}`).join('&');
   240→   240→    const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   241→   241→      headers: kiteHeaders(),
   242→   242→    });
   243→   243→
   244→   244→    // Log non-success for debugging
   245→   245→    if (!res.ok) {
   246→   246→      const errText = await res.text();
   247→   247→      console.error(`[Kite] quote API ${res.status}: ${errText}`);
   248→   248→      return { _error: `HTTP ${res.status}: ${errText}` } as any;
   249→   249→    }
   250→   250→
   251→   251→    const data = await res.json();
   252→   252→    if (data.status !== 'success' || !data.data) {
   253→   253→      console.error('[Kite] quote API error:', JSON.stringify(data));
   254→   254→      return { _error: data.message || 'Unknown error' } as any;
   255→   255→    }
   256→   256→
   257→   257→    const quotes: Record<string, KiteQuote> = {};
   258→   258→    for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   259→   259→      // Kite returns the key as the token string (e.g. "256265")
   260→   260→      // Map it back to the original instrument name (e.g. "NSE:NIFTY 50")
   261→   261→      const displayKey = tokenToKey[key] || key;
   262→   262→      quotes[displayKey] = {
   263→   263→        instrumentToken: q.instrument_token || 0,
   264→   264→        lastPrice: q.last_price || 0,
   265→   265→        open: q.ohlc?.open || 0,
   266→   266→        high: q.ohlc?.high || 0,
   267→   267→        low: q.ohlc?.low || 0,
   268→   268→        close: q.ohlc?.close || 0,
   269→   269→        volume: q.volume || 0,
   270→   270→        netChange: q.net_change || 0,
   271→   271→        averagePrice: q.average_price || 0,
   272→   272→        oi: q.oi || 0,
   273→   273→        oiDayHigh: q.oi_day_high || 0,
   274→   274→        oiDayLow: q.oi_day_low || 0,
   275→   275→      };
   276→   276→    }
   277→   277→    return quotes;
   278→   278→  } catch (err) {
   279→   279→    const errMsg = err instanceof Error ? err.message : String(err);
   280→   280→    console.error('[Kite] quote error:', errMsg);
   281→   281→    return { _error: `Exception: ${errMsg}` } as any;
   282→   282→  }
   283→   283→}

   327→   327→  }

   330→   330→// ─── Option Chain ───
   331→   331→
   332→   332→/**
   333→   333→ * Get option chain for an index or stock
   334→   334→ * Uses InstrumentSpec for correct exchange, segment
   335→   335→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   336→   336→ * Returns all CE + PE instruments for the current expiry near ATM
   337→   337→ */
   338→   338→export async function getOptionInstruments(
   339→   339→  symbol: string,
   340→   340→  spotPrice: number,
   341→   341→  strikesAroundOverride?: number,
   342→   342→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   343→   343→  const spec = getInstrumentSpec(symbol);
   344→   344→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   345→   345→
   346→   346→  // Fetch instruments from the correct exchange (NSE or BSE)
   347→   347→  const instruments = await getInstruments(spec.exchange);
   348→   348→
   349→   349→  // Filter options using spec
   350→   350→  const indexOptions = instruments.filter(i =>
   351→   351→    i.segment === spec.segment &&
   352→   352→    i.instrumentType === spec.instrumentType &&
   353→   353→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   354→   354→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   355→   355→  );
   356→   356→
   357→   357→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   358→   358→
   359→   359→  // Get unique expiries, sort by nearest
   360→   360→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   361→   361→
   362→   362→  // Use nearest expiry that's not expired
   363→   363→  const today = new Date();
   364→   364→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   365→   365→
   366→   366→  // Filter to nearest expiry
   367→   367→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   368→   368→
   369→   369→  // Get lot size from first option (all same for same underlying)
   370→   370→  const lotSize = expiryOptions[0]?.lotSize || 1;
   371→   371→
   372→   372→  // Derive strike step DYNAMICALLY from actual strike gaps
   373→   373→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   374→   374→  let strikeStep = 50;
   375→   375→  if (strikes.length >= 2) {
   376→   376→    const gaps: Record<number, number> = {};
   377→   377→    for (let i = 1; i < strikes.length; i++) {
   378→   378→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   379→   379→      gaps[gap] = (gaps[gap] || 0) + 1;
   380→   380→    }
   381→   381→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   382→   382→  }
   383→   383→
   384→   384→  // Round ATM strike using the DYNAMIC strike step
   385→   385→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   386→   386→
   387→   387→  // Get strikes around ATM
   388→   388→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   389→   389→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   390→   390→    atmStrike - strikesAround * strikeStep + i * strikeStep
   391→   391→  );
   392→   392→
   393→   393→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   394→   394→
   395→   395→  return {
   396→   396→    instruments: filtered,
   397→   397→    meta: { lotSize, strikeStep },
   398→   398→  };
   399→   399→}
   400→   400→
   401→   401→/**
   402→   402→ * Build options flow data from quotes + OI changes
   403→   403→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   404→   404→ */
   405→   405→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   406→   406→  const spec = getInstrumentSpec(symbol);
   407→   407→  if (!spec) return null;
   408→   408→
   409→   409→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   410→   410→  if (optInstruments.length === 0) return null;
   411→   411→
   412→   412→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   413→   413→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   414→   414→
   415→   415→  // Batch fetch quotes
   416→   416→  const quotes = await getQuotes(iKeys);
   417→   417→  if (Object.keys(quotes).length === 0) return null;
   418→   418→
   419→   419→  // Group by strike → build flow data
   420→   420→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   421→   421→
   422→   422→  // Use DYNAMIC strike step and lot size from CSV
   423→   423→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   424→   424→
   425→   425→  for (const inst of optInstruments) {
   426→   426→    // Quotes are keyed by instrument token (string)
   427→   427→    const key = String(inst.instrumentToken);
   428→   428→    const q = quotes[key];
   429→   429→    if (!q) continue;
   430→   430→
   431→   431→    if (!strikes.has(inst.strike)) {
   432→   432→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   433→   433→    }
   434→   434→    const s = strikes.get(inst.strike)!;
   435→   435→
   436→   436→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   437→   437→    const flow = q.volume * meta.lotSize * q.averagePrice;
   438→   438→
   439→   439→    if (inst.tradingSymbol.endsWith('CE')) {
   440→   440→      s.callBuy = flow * 0.6;
   441→   441→      s.callWrite = flow * 0.4;
   442→   442→      s.callOI = q.oi;
   443→   443→      s.callVol = q.volume;
   444→   444→    } else if (inst.tradingSymbol.endsWith('PE')) {
   445→   445→      s.putBuy = flow * 0.6;
   446→   446→      s.putWrite = flow * 0.4;
   447→   447→      s.putOI = q.oi;
   448→   448→      s.putVol = q.volume;
   449→   449→    }
   450→   450→  }
   451→   451→
   452→   452→  // Build result
   453→   453→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   454→   454→    strike,
   455→   455→    isATM: strike === atmStrike,
   456→   456→    ...flow,
   457→   457→    bullishFlow: flow.callBuy + flow.putWrite,
   458→   458→    bearishFlow: flow.putBuy + flow.callWrite,
   459→   459→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   460→   460→  })).sort((a, b) => a.strike - b.strike);

   542→   542→/**
   543→   543→ * Derive lot size and strike step from Kite instruments CSV
   544→   544→ * - lotSize: from the option instrument's lot_size field
   545→   545→ * - strikeStep: derived from the gap between adjacent strikes
   546→   546→ */
   547→   547→export async function getInstrumentMeta(
   548→   548→  symbol: string,
   549→   549→  spotPrice: number,
   550→   550→): Promise<InstrumentMeta> {
   551→   551→  const spec = getInstrumentSpec(symbol);
   552→   552→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   553→   553→
   554→   554→  // Fetch all instruments for this symbol's exchange
   555→   555→  const allInstruments = await getInstruments(spec.exchange);
   556→   556→
   557→   557→  // Filter to this symbol's options for current/near expiry
   558→   558→  const symbolOpts = allInstruments.filter(i =>
   559→   559→    i.segment === spec.segment &&
   560→   560→    i.instrumentType === spec.instrumentType &&
   561→   561→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   562→   562→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   563→   563→  );
   564→   564→
   565→   565→  if (symbolOpts.length === 0) {
   566→   566→    return { lotSize: 1, strikeStep: 50 }; // fallback
   567→   567→  }
   568→   568→
   569→   569→  // Get lot size from the first matching option (all have same lot size for same underlying)
   570→   570→  const lotSize = symbolOpts[0].lotSize || 1;
   571→   571→
   572→   572→  // Derive strike step from unique sorted strikes near ATM
   573→   573→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   574→   574→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   575→   575→
   576→   576→  const expiryOpts = nearestExpiry
   577→   577→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   578→   578→    : symbolOpts;
   579→   579→
   580→   580→  // Get unique strikes, sorted
   581→   581→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   582→   582→
   583→   583→  // Find strikes near ATM and compute step from adjacent gaps
   584→   584→  let strikeStep = 50; // default fallback
   585→   585→  if (strikes.length >= 2) {
   586→   586→    // Use the most common gap between adjacent strikes
   587→   587→    const gaps: Record<number, number> = {};
   588→   588→    for (let i = 1; i < strikes.length; i++) {
   589→   589→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   590→   590→      gaps[gap] = (gaps[gap] || 0) + 1;
   591→   591→    }
   592→   592→    // Most frequent gap = the strike step
   593→   593→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   594→   594→  }
   595→   595→
   596→   596→  return { lotSize, strikeStep };
   597→   597→}

   608→   608→// ─── Black-Scholes Delta Calculator ───
   609→   609→// Approximate delta for options when Kite doesn't provide Greeks.
   610→   610→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   611→   611→
   612→   612→function normCDF(x: number): number {
   613→   613→  // Rational approximation (max error ~7.5e-8)
   614→   614→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   615→   615→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   616→   616→  const sign = x < 0 ? -1 : 1;
   617→   617→  const absX = Math.abs(x);
   618→   618→  const t = 1.0 / (1.0 + p * absX);
   619→   619→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   620→   620→  return 0.5 * (1.0 + sign * y);
   621→   621→}
   622→   622→
   623→   623→/**
   624→   624→ * Calculate Black-Scholes delta for an option
   625→   625→ * @param isCall - true for CE, false for PE
   626→   626→ * @param S - Spot price
   627→   627→ * @param K - Strike price
   628→   628→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   629→   629→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   630→   630→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   631→   631→ */
   632→   632→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   633→   633→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   634→   634→  // If no IV provided, estimate from moneyness
   635→   635→  if (sigma <= 0) {
   636→   636→    const moneyness = (S - K) / S;
   637→   637→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   638→   638→  }
   639→   639→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   640→   640→  const delta = normCDF(d1);
   641→   641→  return isCall ? delta : delta - 1;
   642→   642→}

   644→   644→// ─── Strike Flow Snapshot ───
   645→   645→// Returns raw per-strike data for the frontend to compute 4-color flow.

   649→   649→export interface StrikeFlowSnapshot {

   654→   654→  lotSize: number;

   661→   661→  strike: number;
   662→   662→  isATM: boolean;
   663→   663→  ceLTP: number;
   664→   664→  peLTP: number;
   665→   665→  ceOI: number;
   666→   666→  peOI: number;
   667→   667→  ceVol: number;
   668→   668→  peVol: number;
   669→   669→  ceDelta: number;

   675→   675→/**
   676→   676→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   677→   677→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   678→   678→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   679→   679→ */
   680→   680→export async function getStrikeFlowSnapshot(
   681→   681→  symbol: string,
   682→   682→  spotPrice: number,
   683→   683→): Promise<StrikeFlowSnapshot | null> {
   684→   684→  const spec = getInstrumentSpec(symbol);
   685→   685→  if (!spec) return null;
   686→   686→
   687→   687→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   688→   688→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   689→   689→  if (optInstruments.length === 0) return null;
   690→   690→
   691→   691→  // Batch fetch quotes using instrument tokens
   692→   692→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   693→   693→  const quotes = await getQuotes(iKeys);
   694→   694→  if ('_error' in quotes) return null;
   695→   695→
   696→   696→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   697→   697→
   698→   698→  // Group by strike
   699→   699→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   700→   700→  for (const inst of optInstruments) {
   701→   701→    if (!strikeMap.has(inst.strike)) {
   702→   702→      strikeMap.set(inst.strike, { ce: null, pe: null });
   703→   703→    }
   704→   704→    const entry = strikeMap.get(inst.strike)!;
   705→   705→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   706→   706→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   707→   707→  }
   708→   708→
   709→   709→  // Calculate time to expiry
   710→   710→  const expiry = optInstruments[0]?.expiry || '';
   711→   711→  const now = new Date();
   712→   712→  const expiryDate = new Date(expiry);
   713→   713→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   714→   714→  const T = daysToExpiry / 365;
   715→   715→
   716→   716→  // Build strike data
   717→   717→  const strikes: StrikeFlowData[] = [];
   718→   718→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   719→   719→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   720→   720→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   721→   721→
   722→   722→    const ceLTP = ceQuote?.lastPrice || 0;
   723→   723→    const peLTP = peQuote?.lastPrice || 0;
   724→   724→    const ceOI = ceQuote?.oi || 0;
   725→   725→    const peOI = peQuote?.oi || 0;
   726→   726→    const ceVol = ceQuote?.volume || 0;
   727→   727→    const peVol = peQuote?.volume || 0;
   728→   728→
   729→   729→    // Compute delta via Black-Scholes
   730→   730→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   731→   731→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   732→   732→
   733→   733→    strikes.push({
   734→   734→      strike,
   735→   735→      isATM: strike === atmStrike,
   736→   736→      ceLTP,
   737→   737→      peLTP,
   738→   738→      ceOI,
   739→   739→      peOI,
   740→   740→      ceVol,
   741→   741→      peVol,
   742→   742→      ceDelta,
   743→   743→      peDelta,
   744→   744→      ceToken: ce?.instrumentToken || 0,
   745→   745→      peToken: pe?.instrumentToken || 0,
   746→   746→    });
   747→   747→  }
   748→   748→
   749→   749→  strikes.sort((a, b) => a.strike - b.strike);
   750→   750→
   751→   751→  return {
   752→   752→    timestamp: new Date().toISOString(),
   753→   753→    symbol,
   754→   754→    spotPrice,
   755→   755→    atmStrike,
   756→   756→    lotSize: meta.lotSize,
   757→   757→    strikeStep: meta.strikeStep,
   758→   758→    expiry,
   759→   759→    strikes,
   760→   760→  };
   761→   761→}

     1→/**
     2→ * Zerodha Kite API Integration
     3→ *
     4→ * Connects to Kite Connect REST API for real NSE/BSE market data.
     5→ * When KITE_API_KEY + KITE_ACCESS_TOKEN are set in .env, real data flows.
     6→ * Otherwise, falls back to demo data.
     7→ *
     8→ * Setup:
     9→ * 1. Go to https://developers.kite.trade
    10→ * 2. Create an app → get API Key + Secret
    11→ * 3. Login: https://kite.zerodha.com/connect/login?api_key=YOUR_KEY
    12→ * 4. After login, capture request_token from redirect URL
    13→ * 5. Call /api/kite/auth?request_token=xxx to generate access_token
    14→ * 6. Put access_token in .env → KITE_ACCESS_TOKEN=xxx
    15→ * 7. Redeploy
    16→ *
    17→ * Access tokens expire daily at midnight. Re-login each morning.
    18→ */

    47→    'Authorization': `token ${apiKey}:${accessToken}`,
    48→    'X-Kite-Version': '3',
    49→    'Content-Type': 'application/x-www-form-urlencoded',

   100→/**
   101→ * Generate access_token from request_token (one-time, after login redirect)
   102→ */
   103→export async function generateSession(requestToken: string): Promise<{
   104→  accessToken: string;
   105→  refreshToken: string;
   106→} | null> {
   107→  const apiKey = process.env.KITE_API_KEY;
   108→  const apiSecret = process.env.KITE_API_SECRET;
   109→  if (!apiKey || !apiSecret) return null;
   110→
   111→  try {
   112→    // Kite uses checksum = sha256(api_key + request_token + api_secret)
   113→    const { createHash } = await import('crypto');
   114→    const checksum = createHash('sha256')
   115→      .update(apiKey + requestToken + apiSecret)
   116→      .digest('hex');
   117→
   118→    const res = await fetch(`${KITE_BASE}/session/token`, {
   119→      method: 'POST',
   120→      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   121→      body: new URLSearchParams({
   122→        api_key: apiKey,
   123→        request_token: requestToken,
   124→        checksum,
   125→      }).toString(),
   126→    });
   127→
   128→    const data = await res.json();
   129→    if (data.status === 'success' && data.data) {
   130→      return {
   131→        accessToken: data.data.access_token,
   132→        refreshToken: data.data.refresh_token || '',
   133→      };
   134→    }
   135→    console.error('[Kite] Session error:', data.message);
   136→    return null;
   137→  } catch (err) {
   138→    console.error('[Kite] generateSession failed:', err);
   139→    return null;
   140→  }
   141→}

   145→// Cache all instruments from ALL exchanges (Kite returns one CSV with NSE+BSE+NFO+BFO+CDS+MCX)
   146→let instrumentsCache: KiteInstrument[] | null = null;
   147→let instrumentsCacheTime = 0;
   148→
   149→/**
   150→ * Invalidate the in-memory instruments cache.
   151→ * Use after credential changes (e.g. user switched from expired → valid token)
   152→ * to force a fresh CSV download on the next getInstruments() call.
   153→ */
   154→export function invalidateInstrumentsCache(): void {
   155→  instrumentsCache = null;
   156→  instrumentsCacheTime = 0;
   157→}
   158→
   159→/**
   160→ * Fetch all instruments from Kite (cached for 1 hour)
   161→ * Kite returns a single CSV with all exchanges: NSE, BSE, NFO, BFO, CDS, MCX
   162→ * We filter by exchange after caching
   163→ *
   164→ * @param exchange Optional exchange filter (e.g. 'NFO' returns only NFO F&O instruments)
   165→ * @param forceRefresh Bypass the 1-hour cache and re-download. Useful when creds change.
   166→ */
   167→export async function getInstruments(exchange?: string, forceRefresh?: boolean): Promise<KiteInstrument[]> {
   168→  // Return from cache if fresh (and not forced to refresh)
   169→  if (!forceRefresh && instrumentsCache && Date.now() - instrumentsCacheTime < 3600000) {
   170→    return exchange ? instrumentsCache.filter(i => i.exchange === exchange) : instrumentsCache;
   171→  }
   172→
   173→  try {
   174→    const res = await fetch(`${KITE_BASE}/instruments`, { headers: kiteHeaders() });
   175→    if (!res.ok) {
   176→      const errText = await res.text();
   177→      console.error(`[Kite] instruments API ${res.status}: ${errText}`);
   178→      return [];
   179→    }
   180→    const text = await res.text();
   181→
   182→    // Kite returns CSV format
   183→    const lines = text.trim().split('\n');
   184→    const instruments: KiteInstrument[] = [];
   185→
   186→    // Kite CSV column order (2025+ format):
   187→    // 0:instrument_token 1:exchange_token 2:tradingsymbol 3:name 4:last_price
   188→    // 5:expiry 6:strike 7:tick_size 8:lot_size 9:instrument_type 10:segment 11:exchange
   189→    //
   190→    // Kite changed the CSV format in 2025:
   191→    //   OLD: instrument_type was 'OPTIDX'/'OPTSTK'/'FUTIDX'/'FUTSTK'/'EQ'/'INDEX'
   192→    //        segment was 'NFO'/'BFO'/'NSE'/'BSE'
   193→    //   NEW: instrument_type is 'CE'/'PE'/'FUT'/'EQ'
   194→    //        segment is 'NFO-OPT'/'NFO-FUT'/'NFO'/'BFO-OPT'/'BFO-FUT'/'NSE'/'BSE'
   195→    // To avoid breaking the rest of the code (which filters by OPTIDX/OPTSTK/FUTIDX/FUTSTK),
   196→    // we normalize the new format back to the legacy types.
   197→    //
   198→    // Disambiguation: in the new CSV, ALL options (index + stock) are CE/PE — there is no
   199→    // index-vs-stock hint in the row itself. We use the `name` column to decide:
   200→    //   - If the underlying name matches a known F&O index (NIFTY, BANKNIFTY, FINNIFTY,
   201→    //     SENSEX, BANKEX, MIDCPNIFTY, etc.) → OPTIDX / FUTIDX
   202→    //   - Otherwise (RELIANCE, TCS, HDFCBANK, ...) → OPTSTK / FUTSTK
   203→    // The set below is checked at parse time (cheap Set.has lookup).
   204→    const KNOWN_FNO_INDICES = new Set([
   205→      'NIFTY', 'BANKNIFTY', 'FINNIFTY', 'SENSEX', 'BANKEX',
   206→      'MIDCPNIFTY', 'NIFTY MID SELECT', 'NIFTY PSE',
   207→    ]);
   208→    const normalizeInstrumentType = (rawType: string, segment: string, name: string): string => {
   209→      // Keep legacy values as-is (in case Kite reverts or some entries still use old format)
   210→      if (rawType === 'OPTIDX' || rawType === 'OPTSTK' || rawType === 'FUTIDX' || rawType === 'FUTSTK' ||
   211→          rawType === 'INDEX') {
   212→        return rawType;
   213→      }
   214→      // CRITICAL FIX: In the 2025+ Kite CSV, cash index instruments (NIFTY 50, SENSEX, etc.)
   215→      // have instrument_type='EQ' and segment='INDICES'. We must normalize these to 'INDEX'
   216→      // so that downstream lookups (e.g. highest-bet finding cash instruments) can filter by
   217→      // instrumentType === 'INDEX' instead of the ambiguous 'EQ' (which also matches stocks).
   218→      if (rawType === 'EQ' && segment === 'INDICES') {
   219→        return 'INDEX';
   220→      }
   221→      // Regular EQ stocks stay as 'EQ'
   222→      if (rawType === 'EQ') {
   223→        return rawType;
   224→      }
   225→      const nameUpper = name.toUpperCase();
   226→      const isIndex = KNOWN_FNO_INDICES.has(nameUpper);
   227→      if (rawType === 'CE' || rawType === 'PE') {
   228→        // BFO always has index options only (SENSEX/BANKEX). NFO has both index + stock.
   229→        return isIndex ? 'OPTIDX' : 'OPTSTK';
   230→      }
   231→      if (rawType === 'FUT') {
   232→        return isIndex ? 'FUTIDX' : 'FUTSTK';
   233→      }
   234→      return rawType;
   235→    };
   236→
   237→    for (let i = 1; i < lines.length; i++) { // skip header
   238→      const cols = lines[i].split(',');
   239→      if (cols.length < 12) continue;
   240→      const rawType = cols[9] || '';
   241→      const segment = cols[10] || '';
   242→      // Kite wraps the `name` field in double quotes (e.g. "NIFTY").
   243→      // Strip them so name-based filters (e.g. .includes('NIFTY')) work correctly.
   244→      const rawName = cols[3] || '';
   245→      const name = rawName.startsWith('"') && rawName.endsWith('"')
   246→        ? rawName.slice(1, -1)
   247→        : rawName;
   248→      instruments.push({
   249→        instrumentToken: parseInt(cols[0]) || 0,
   250→        exchangeToken: parseInt(cols[1]) || 0,
   251→        tradingSymbol: cols[2],
   252→        name,
   253→        exchange: cols[11] || '',
   254→        segment,
   255→        instrumentType: normalizeInstrumentType(rawType, segment, name),
   256→        strike: parseFloat(cols[6]) || 0,
   257→        lotSize: parseInt(cols[8]) || 1,
   258→        expiry: cols[5] || '',
   259→      });
   260→    }
   261→
   262→    instrumentsCache = instruments;
   263→    instrumentsCacheTime = Date.now();
   264→    return exchange ? instruments.filter(i => i.exchange === exchange) : instruments;
   265→  } catch (err) {
   266→    const errMsg = err instanceof Error ? err.message : String(err);
   267→    console.error('[Kite] instruments fetch failed:', errMsg);
   268→    // Return empty but set error flag for callers to detect
   269→    return [];
   270→  }
   271→}

   275→/**
   276→ * Get real-time quotes for instruments
   277→ * iTokens: comma-separated instrument tokens or "NSE:NIFTY 50" format
   278→ */
   279→export async function getQuotes(instruments: string[]): Promise<Record<string, KiteQuote>> {
   280→  if (!isKiteConfigured()) return {};
   281→
   282→  try {
   283→    // Strategy: Use instrument tokens (numeric) instead of trading symbols.
   284→    // Tokens have no encoding issues on any platform.
   285→    // Build a map from token → original instrument key for response mapping.
   286→    const tokenToKey: Record<string, string> = {};
   287→    const tokenList: string[] = [];
   288→
   289→    for (const inst of instruments) {
   290→      // If it's a numeric token, use directly
   291→      if (/^\d+$/.test(inst)) {
   292→        tokenToKey[inst] = inst;
   293→        tokenList.push(inst);
   294→      } else {
   295→        // Look up token from KITE_INDEX_INSTRUMENTS
   296→        const found = Object.entries(KITE_INDEX_INSTRUMENTS).find(
   297→          ([, v]) => v.symbol === inst
   298→        );
   299→        if (found) {
   300→          const tok = String(found[1].token);
   301→          tokenToKey[tok] = inst; // map token back to symbol name
   302→          tokenList.push(tok);
   303→        } else {
   304→          // Fallback: use the instrument string as-is with encodeURI
   305→          tokenToKey[inst] = inst;
   306→          tokenList.push(inst);
   307→        }
   308→      }
   309→    }
   310→
   311→    const iList = tokenList.join(',');
   312→    // Kite requires SEPARATE 'i' query params for multiple instruments.
   313→    // ?i=256265&i=260105 works, but ?i=256265,260105 returns empty data.
   314→    const params = tokenList.map(t => `i=${t}`).join('&');
   315→    const res = await fetch(`${KITE_BASE}/quote?${params}`, {
   316→      headers: kiteHeaders(),
   317→    });
   318→
   319→    // Log non-success for debugging
   320→    if (!res.ok) {
   321→      const errText = await res.text();
   322→      console.error(`[Kite] quote API ${res.status}: ${errText}`);
   323→      return { _error: `HTTP ${res.status}: ${errText}` } as any;
   324→    }
   325→
   326→    const data = await res.json();
   327→    if (data.status !== 'success' || !data.data) {
   328→      console.error('[Kite] quote API error:', JSON.stringify(data));
   329→      return { _error: data.message || 'Unknown error' } as any;
   330→    }
   331→
   332→    const quotes: Record<string, KiteQuote> = {};
   333→    for (const [key, q] of Object.entries(data.data as Record<string, any>)) {
   334→      // Kite returns the key as the token string (e.g. "256265")
   335→      // Map it back to the original instrument name (e.g. "NSE:NIFTY 50")
   336→      const displayKey = tokenToKey[key] || key;
   337→      quotes[displayKey] = {
   338→        instrumentToken: q.instrument_token || 0,
   339→        lastPrice: q.last_price || 0,
   340→        open: q.ohlc?.open || 0,
   341→        high: q.ohlc?.high || 0,
   342→        low: q.ohlc?.low || 0,
   343→        close: q.ohlc?.close || 0,
   344→        volume: q.volume || 0,
   345→        netChange: q.net_change || 0,
   346→        averagePrice: q.average_price || 0,
   347→        oi: q.oi || 0,
   348→        oiDayHigh: q.oi_day_high || 0,
   349→        oiDayLow: q.oi_day_low || 0,
   350→        dayHigh: q.ohlc?.high || 0,
   351→        dayLow: q.ohlc?.low || 0,
   352→      };
   353→    }
   354→    return quotes;
   355→  } catch (err) {
   356→    const errMsg = err instanceof Error ? err.message : String(err);
   357→    console.error('[Kite] quote error:', errMsg);
   358→    return { _error: `Exception: ${errMsg}` } as any;
   359→  }
   360→}

   364→/**
   365→ * Get historical candle data for price line chart
   366→ * instrumentToken: numeric token from instruments list
   367→ * interval: "minute", "3minute", "5minute", "15minute", "30minute", "hour", "day"
   368→ */
   369→export async function getCandles(
   370→  instrumentToken: number,
   371→  interval: string = '15minute',
   372→  days: number = 1,
   373→): Promise<KiteHistoricalCandle[]> {
   374→  if (!isKiteConfigured()) return [];
   375→
   376→  // Kite historical API expects IST dates.
   377→  // On Vercel (UTC), we must convert to IST before formatting.
   378→  const IST_OFFSET = 5.5 * 60 * 60 * 1000;
   379→  const toIST = (d: Date) => new Date(d.getTime() + IST_OFFSET);
   380→
   381→  const toDate = toIST(new Date());
   382→  const fromDate = toIST(new Date());
   383→  fromDate.setDate(fromDate.getDate() - days);
   384→
   385→  // Kite historical API requires format "YYYY-MM-DD HH:MM:SS" (WITH seconds).
   386→  // Sending "YYYY-MM-DD HH:MM" (no seconds) returns HTTP 400 "invalid from date".
   387→  const fmt = (d: Date) =>
   388→    `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}:00`;
   389→
   390→  try {
   391→    const fromStr = fmt(fromDate);
   392→    const toStr = fmt(toDate);
   393→    // URL-encode the date strings (they contain a space: "2026-08-25 09:15")
   394→    const url = `${KITE_BASE}/instruments/historical/${instrumentToken}/${interval}?from=${encodeURIComponent(fromStr)}&to=${encodeURIComponent(toStr)}&continuous=0`;
   395→
   396→    const res = await fetch(url, { headers: kiteHeaders() });
   397→
   398→    // Surface HTTP-level errors (403 expired token, 429 rate limit, etc.)
   399→    if (!res.ok) {
   400→      const errText = await res.text();
   401→      console.error(`[Kite] candles HTTP ${res.status} for token ${instrumentToken}: ${errText.substring(0, 300)}`);
   402→      return [];
   403→    }
   404→
   405→    const data = await res.json();
   406→    if (data.status !== 'success' || !data.data?.candles) {
   407→      console.error('[Kite] candles non-success: status=%s, message=%s, from=%s, to=%s',
   408→        data.status, data.message || 'no message', fromStr, toStr);
   409→      return [];
   410→    }
   411→
   412→    return (data.data.candles as any[][]).map(c => ({
   413→      timestamp: c[0],
   414→      open: c[1],
   415→      high: c[2],
   416→      low: c[3],
   417→      close: c[4],
   418→      volume: c[5],
   419→      oi: c[6] || 0,  // c[6] = OI (F&O only), undefined for cash
   420→    }));
   421→  } catch (err) {
   422→    const errMsg = err instanceof Error ? err.message : String(err);
   423→    console.error('[Kite] candles error:', errMsg);
   424→    return [];
   425→  }
   426→}

   428→// ─── Option Chain ───

   430→/**
   431→ * Get option chain for an index or stock
   432→ * Uses InstrumentSpec for correct exchange, segment
   433→ * Fetches lot size + strike step DYNAMICALLY from Kite CSV
   434→ * Returns all CE + PE instruments for the current expiry near ATM
   435→ */
   436→export async function getOptionInstruments(
   437→  symbol: string,
   438→  spotPrice: number,
   439→  strikesAroundOverride?: number,
   440→): Promise<{ instruments: KiteInstrument[]; meta: InstrumentMeta }> {
   441→  const spec = getInstrumentSpec(symbol);
   442→  if (!spec) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   443→
   444→  // IMPORTANT: F&O options/futures are NOT on the cash exchange.
   445→  //   - NIFTY index spot is on exchange='NSE', but NIFTY OPTIONS are on exchange='NFO'
   446→  //   - SENSEX spot is on exchange='BSE', but SENSEX OPTIONS are on exchange='BFO'
   447→  //   - Stock (RELIANCE/TCS/...) spot is on exchange='NSE', but their OPTIONS are on exchange='NFO'
   448→  // So we use spec.segment ('NFO' or 'BFO') — which equals the F&O exchange column in
   449→  // Kite's CSV — NOT spec.exchange ('NSE'/'BSE', which is the cash exchange).
   450→  // If you call getInstruments(spec.exchange) you'll only get cash/index instruments,
   451→  // and the option filter below will return ZERO matches.
   452→  const instruments = await getInstruments(spec.segment);
   453→
   454→  // Filter options using spec.
   455→  //
   456→  // Kite changed their CSV format in 2025:
   457→  //   OLD: segment='NFO'/'BFO', instrument_type='OPTIDX'/'OPTSTK'
   458→  //   NEW: segment='NFO-OPT'/'BFO-OPT', instrument_type='CE'/'PE' (we normalize to OPTIDX)
   459→  //
   460→  // So we use a "starts with" match on segment (so 'NFO' matches both 'NFO' and 'NFO-OPT'),
   461→  // and an equality match on instrumentType (we've already normalized new → legacy).
   462→  const symUpper = symbol.toUpperCase();
   463→  const indexOptions = instruments.filter(i =>
   464→    i.segment.startsWith(spec.segment) &&
   465→    i.instrumentType === spec.instrumentType &&
   466→    (i.name.toUpperCase().includes(symUpper) ||
   467→     i.tradingSymbol.toUpperCase().includes(symUpper))
   468→  );
   469→
   470→  if (indexOptions.length === 0) return { instruments: [], meta: { lotSize: 1, strikeStep: 50 } };
   471→
   472→  // Get unique expiries, sort by nearest
   473→  const expiries = [...new Set(indexOptions.map(i => i.expiry))].sort();
   474→
   475→  // Use nearest expiry that's not expired
   476→  const today = new Date();
   477→  const nearestExpiry = expiries.find(e => new Date(e) >= new Date(today.toDateString())) || expiries[0];
   478→
   479→  // Filter to nearest expiry
   480→  const expiryOptions = indexOptions.filter(i => i.expiry === nearestExpiry);
   481→
   482→  // Get lot size from first option (all same for same underlying)
   483→  const lotSize = expiryOptions[0]?.lotSize || 1;
   484→
   485→  // Derive strike step DYNAMICALLY from actual strike gaps
   486→  const strikes = [...new Set(expiryOptions.map(i => i.strike))].sort((a, b) => a - b);
   487→  let strikeStep = 50;
   488→  if (strikes.length >= 2) {
   489→    const gaps: Record<number, number> = {};
   490→    for (let i = 1; i < strikes.length; i++) {
   491→      const gap = Math.round((strikes[i] - strikes[i - 1]) * 100) / 100; // handle 2.5 etc
   492→      gaps[gap] = (gaps[gap] || 0) + 1;
   493→    }
   494→    strikeStep = parseFloat(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   495→  }
   496→
   497→  // Round ATM strike using the DYNAMIC strike step
   498→  const atmStrike = Math.round(spotPrice / strikeStep) * strikeStep;
   499→
   500→  // Get strikes around ATM
   501→  const strikesAround = strikesAroundOverride ?? spec.strikesAround;
   502→  const strikeList = Array.from({ length: strikesAround * 2 + 1 }, (_, i) =>
   503→    atmStrike - strikesAround * strikeStep + i * strikeStep
   504→  );
   505→
   506→  const filtered = expiryOptions.filter(i => strikeList.includes(i.strike));
   507→
   508→  return {
   509→    instruments: filtered,
   510→    meta: { lotSize, strikeStep },
   511→  };
   512→}
   513→
   514→/**
   515→ * Build options flow data from quotes + OI changes
   516→ * Lot size and strike step come DYNAMICALLY from Kite's CSV
   517→ */
   518→export async function getOptionsFlow(symbol: string, spotPrice: number) {
   519→  const spec = getInstrumentSpec(symbol);
   520→  if (!spec) return null;
   521→
   522→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice);
   523→  if (optInstruments.length === 0) return null;
   524→
   525→  // Build instrument keys for batch quote — use instrument TOKENS (numeric, no encoding issues)
   526→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   527→
   528→  // Batch fetch quotes
   529→  const quotes = await getQuotes(iKeys);
   530→  if (Object.keys(quotes).length === 0) return null;
   531→
   532→  // Group by strike → build flow data
   533→  const strikes = new Map<number, { callBuy: number; putBuy: number; callWrite: number; putWrite: number; callOI: number; putOI: number; callVol: number; putVol: number; }>();
   534→
   535→  // Use DYNAMIC strike step and lot size from CSV
   536→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   537→
   538→  for (const inst of optInstruments) {
   539→    // Quotes are keyed by instrument token (string)
   540→    const key = String(inst.instrumentToken);
   541→    const q = quotes[key];
   542→    if (!q) continue;
   543→
   544→    if (!strikes.has(inst.strike)) {
   545→      strikes.set(inst.strike, { callBuy: 0, putBuy: 0, callWrite: 0, putWrite: 0, callOI: 0, putOI: 0, callVol: 0, putVol: 0 });
   546→    }
   547→    const s = strikes.get(inst.strike)!;
   548→
   549→    // Flow = volume × lotSize (from CSV) × avgPrice (real money flow)
   550→    const flow = q.volume * meta.lotSize * q.averagePrice;
   551→
   552→    if (inst.tradingSymbol.endsWith('CE')) {
   553→      s.callBuy = flow * 0.6;
   554→      s.callWrite = flow * 0.4;
   555→      s.callOI = q.oi;
   556→      s.callVol = q.volume;
   557→    } else if (inst.tradingSymbol.endsWith('PE')) {
   558→      s.putBuy = flow * 0.6;
   559→      s.putWrite = flow * 0.4;
   560→      s.putOI = q.oi;
   561→      s.putVol = q.volume;
   562→    }
   563→  }
   564→
   565→  // Build result
   566→  return Array.from(strikes.entries()).map(([strike, flow]) => ({
   567→    strike,
   568→    isATM: strike === atmStrike,
   569→    ...flow,
   570→    bullishFlow: flow.callBuy + flow.putWrite,
   571→    bearishFlow: flow.putBuy + flow.callWrite,
   572→    netFlow: (flow.callBuy + flow.putWrite) - (flow.putBuy + flow.callWrite),
   573→  })).sort((a, b) => a.strike - b.strike);
   574→}

   655→/**
   656→ * Derive lot size and strike step from Kite instruments CSV
   657→ * - lotSize: from the option instrument's lot_size field
   658→ * - strikeStep: derived from the gap between adjacent strikes
   659→ */
   660→export async function getInstrumentMeta(
   661→  symbol: string,
   662→  spotPrice: number,
   663→): Promise<InstrumentMeta> {
   664→  const spec = getInstrumentSpec(symbol);
   665→  if (!spec) return { lotSize: 1, strikeStep: 50 }; // fallback
   666→
   667→  // Fetch instruments from the F&O exchange (NFO/BFO), NOT the cash exchange (NSE/BSE).
   668→  // See getOptionInstruments() for the full explanation.
   669→  const allInstruments = await getInstruments(spec.segment);
   670→
   671→  // Filter to this symbol's options for current/near expiry.
   672→  // Use startsWith on segment to match both legacy 'NFO' and new 'NFO-OPT' formats.
   673→  const symbolOpts = allInstruments.filter(i =>
   674→    i.segment.startsWith(spec.segment) &&
   675→    i.instrumentType === spec.instrumentType &&
   676→    (i.name.toUpperCase().includes(symbol.toUpperCase()) ||
   677→     i.tradingSymbol.toUpperCase().includes(symbol.toUpperCase()))
   678→  );
   679→
   680→  if (symbolOpts.length === 0) {
   681→    return { lotSize: 1, strikeStep: 50 }; // fallback
   682→  }
   683→
   684→  // Get lot size from the first matching option (all have same lot size for same underlying)
   685→  const lotSize = symbolOpts[0].lotSize || 1;
   686→
   687→  // Derive strike step from unique sorted strikes near ATM
   688→  const nearestExpiry = [...new Set(symbolOpts.map(i => i.expiry))].sort()
   689→    .find(e => new Date(e) >= new Date(new Date().toDateString()));
   690→
   691→  const expiryOpts = nearestExpiry
   692→    ? symbolOpts.filter(i => i.expiry === nearestExpiry)
   693→    : symbolOpts;
   694→
   695→  // Get unique strikes, sorted
   696→  const strikes = [...new Set(expiryOpts.map(i => i.strike))].sort((a, b) => a - b);
   697→
   698→  // Find strikes near ATM and compute step from adjacent gaps
   699→  let strikeStep = 50; // default fallback
   700→  if (strikes.length >= 2) {
   701→    // Use the most common gap between adjacent strikes
   702→    const gaps: Record<number, number> = {};
   703→    for (let i = 1; i < strikes.length; i++) {
   704→      const gap = Math.round(strikes[i] - strikes[i - 1]);
   705→      gaps[gap] = (gaps[gap] || 0) + 1;
   706→    }
   707→    // Most frequent gap = the strike step
   708→    strikeStep = parseInt(Object.entries(gaps).sort((a, b) => b[1] - a[1])[0][0]) || 50;
   709→  }
   710→
   711→  return { lotSize, strikeStep };
   712→}

   723→// ─── Black-Scholes Delta Calculator ───
   724→// Approximate delta for options when Kite doesn't provide Greeks.
   725→// Uses the standard normal CDF approximation (Abramowitz & Stegun).
   726→
   727→function normCDF(x: number): number {
   728→  // Rational approximation (max error ~7.5e-8)
   729→  const a1 = 0.254829592, a2 = -0.284496736, a3 = 1.421413741;
   730→  const a4 = -1.453152027, a5 = 1.061405429, p = 0.3275911;
   731→  const sign = x < 0 ? -1 : 1;
   732→  const absX = Math.abs(x);
   733→  const t = 1.0 / (1.0 + p * absX);
   734→  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX / 2);
   735→  return 0.5 * (1.0 + sign * y);
   736→}
   737→
   738→/**
   739→ * Calculate Black-Scholes delta for an option
   740→ * @param isCall - true for CE, false for PE
   741→ * @param S - Spot price
   742→ * @param K - Strike price
   743→ * @param T - Time to expiry in years (e.g., 3/365 for 3 days)
   744→ * @param r - Risk-free rate (default 0.065 = 6.5%)
   745→ * @param sigma - Implied volatility (if 0, estimate from moneyness)
   746→ */
   747→function bsDelta(isCall: boolean, S: number, K: number, T: number, r: number = 0.065, sigma: number = 0): number {
   748→  if (T <= 0 || S <= 0 || K <= 0) return isCall ? 0.5 : -0.5;
   749→  // If no IV provided, estimate from moneyness
   750→  if (sigma <= 0) {
   751→    const moneyness = (S - K) / S;
   752→    sigma = 0.12 + Math.abs(moneyness) * 0.3; // rough estimate: 12-42%
   753→  }
   754→  const d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
   755→  const delta = normCDF(d1);
   756→  return isCall ? delta : delta - 1;
   757→}

   781→  peOI: number;

   790→/**
   791→ * Get raw strike flow snapshot — per-strike OI, LTP, volume, delta.
   792→ * Frontend diffs consecutive snapshots to compute 4-color flow.
   793→ * Fetches 11 strikes around ATM (±5 × strikeStep).
   794→ */
   795→export async function getStrikeFlowSnapshot(
   796→  symbol: string,
   797→  spotPrice: number,
   798→): Promise<StrikeFlowSnapshot | null> {
   799→  const spec = getInstrumentSpec(symbol);
   800→  if (!spec) return null;
   801→
   802→  // Get instruments for nearest expiry, 5 strikes each side = 11 total
   803→  const { instruments: optInstruments, meta } = await getOptionInstruments(symbol, spotPrice, 5);
   804→  if (optInstruments.length === 0) return null;
   805→
   806→  // Batch fetch quotes using instrument tokens
   807→  const iKeys = optInstruments.map(i => String(i.instrumentToken));
   808→  const quotes = await getQuotes(iKeys);
   809→  if ('_error' in quotes) return null;
   810→
   811→  const atmStrike = Math.round(spotPrice / meta.strikeStep) * meta.strikeStep;
   812→
   813→  // Group by strike
   814→  const strikeMap = new Map<number, { ce: KiteInstrument | null; pe: KiteInstrument | null }>();
   815→  for (const inst of optInstruments) {
   816→    if (!strikeMap.has(inst.strike)) {
   817→      strikeMap.set(inst.strike, { ce: null, pe: null });
   818→    }
   819→    const entry = strikeMap.get(inst.strike)!;
   820→    if (inst.tradingSymbol.endsWith('CE')) entry.ce = inst;
   821→    else if (inst.tradingSymbol.endsWith('PE')) entry.pe = inst;
   822→  }
   823→
   824→  // Calculate time to expiry
   825→  const expiry = optInstruments[0]?.expiry || '';
   826→  const now = new Date();
   827→  const expiryDate = new Date(expiry);
   828→  const daysToExpiry = Math.max(0.5, (expiryDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   829→  const T = daysToExpiry / 365;
   830→
   831→  // Build strike data
   832→  const strikes: StrikeFlowData[] = [];
   833→  for (const [strike, { ce, pe }] of strikeMap.entries()) {
   834→    const ceQuote = ce ? quotes[String(ce.instrumentToken)] : null;
   835→    const peQuote = pe ? quotes[String(pe.instrumentToken)] : null;
   836→
   837→    const ceLTP = ceQuote?.lastPrice || 0;
   838→    const peLTP = peQuote?.lastPrice || 0;
   839→    const ceOI = ceQuote?.oi || 0;
   840→    const peOI = peQuote?.oi || 0;
   841→    const ceVol = ceQuote?.volume || 0;
   842→    const peVol = peQuote?.volume || 0;
   843→
   844→    // Compute delta via Black-Scholes
   845→    const ceDelta = Math.abs(bsDelta(true, spotPrice, strike, T));
   846→    const peDelta = Math.abs(bsDelta(false, spotPrice, strike, T));
   847→
   848→    strikes.push({
   849→      strike,
   850→      isATM: strike === atmStrike,
   851→      ceLTP,
   852→      peLTP,
   853→      ceOI,
   854→      peOI,
   855→      ceVol,
   856→      peVol,
   857→      ceDelta,
   858→      peDelta,
   859→      ceToken: ce?.instrumentToken || 0,
   860→      peToken: pe?.instrumentToken || 0,
   861→    });
   862→  }
   863→
   864→  strikes.sort((a, b) => a.strike - b.strike);
   865→
   866→  return {
   867→    timestamp: new Date().toISOString(),
   868→    symbol,
   869→    spotPrice,
   870→    atmStrike,
   871→    lotSize: meta.lotSize,
   872→    strikeStep: meta.strikeStep,
   873→    expiry,
   874→    strikes,
   875→  };
   876→}

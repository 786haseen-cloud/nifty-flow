//@version=6

// --- 1. Configuration: Top 12 Nifty Stocks with Actual Market Cap Weighting ---
// Weighting method selection
useMarketCapWeight = input.bool(true, "Use Market Cap Weighting (Recommended)")
useEqualWeight = input.bool(false, "Use Equal Weight (Alternative)")
manualMode = input.bool(false, "Manual Weight Mode (Override All)")

// Actual market cap weights for top 12 Nifty stocks (based on free-float market cap as of 2025)
manualWeight1 = input.float(13.2, "HDFCBANK Weight %", minval=0, maxval=30, step=0.1)
manualWeight2 = input.float(9.0, "ICICIBANK Weight %", minval=0, maxval=30, step=0.1)
manualWeight3 = input.float(8.7, "RELIANCE Weight %", minval=0, maxval=30, step=0.1)
manualWeight4 = input.float(5.0, "INFY Weight %", minval=0, maxval=30, step=0.1)
manualWeight5 = input.float(4.6, "BHARTIARTL Weight %", minval=0, maxval=30, step=0.1)
manualWeight6 = input.float(3.6, "ITC Weight %", minval=0, maxval=30, step=0.1)
manualWeight7 = input.float(3.5, "LT Weight %", minval=0, maxval=30, step=0.1)
manualWeight8 = input.float(3.2, "TCS Weight %", minval=0, maxval=30, step=0.1)
manualWeight9 = input.float(3.1, "AXISBANK Weight %", minval=0, maxval=30, step=0.1)
manualWeight10 = input.float(3.0, "KOTAKBANK Weight %", minval=0, maxval=30, step=0.1)
manualWeight11 = input.float(2.8, "HINDUNILVR Weight %", minval=0, maxval=30, step=0.1)
manualWeight12 = input.float(2.5, "SUNPHARMA Weight %", minval=0, maxval=30, step=0.1)

// --- 2. Helper function to get weight based on symbol ---
getWeight(symbol) =>
    weight = 0.0
    if useMarketCapWeight
        if symbol == "NSE:HDFCBANK"
            weight := 13.2
        else if symbol == "NSE:ICICIBANK"
            weight := 9.0
        else if symbol == "NSE:RELIANCE"
            weight := 8.7
        else if symbol == "NSE:INFY"
            weight := 5.0
        else if symbol == "NSE:BHARTIARTL"
            weight := 4.6
        else if symbol == "NSE:ITC"
            weight := 3.6
        else if symbol == "NSE:LT"
            weight := 3.5
        else if symbol == "NSE:TCS"
            weight := 3.2
        else if symbol == "NSE:AXISBANK"
            weight := 3.1
        else if symbol == "NSE:KOTAKBANK"
            weight := 3.0
        else if symbol == "NSE:HINDUNILVR"
            weight := 2.8
        else if symbol == "NSE:SUNPHARMA"
            weight := 2.5
        else
            weight := 0.0
    else if useEqualWeight
        weight := 8.3333
    else
        weight := 0.0
    weight

// --- 3. Declare the main variable ---
var float niftyCashFlowTrend = 0.0

// --- 4. Get weighted cash flow for each stock ---
// HDFCBANK
close1 = request.security("NSE:HDFCBANK", timeframe.period, close)
open1 = request.security("NSE:HDFCBANK", timeframe.period, open)
volume1 = request.security("NSE:HDFCBANK", timeframe.period, volume)
cf1 = (close1 - open1) * volume1
weight1 = manualMode ? manualWeight1 : getWeight("NSE:HDFCBANK")
weightedCF1 = cf1 * weight1

// ICICIBANK
close2 = request.security("NSE:ICICIBANK", timeframe.period, close)
open2 = request.security("NSE:ICICIBANK", timeframe.period, open)
volume2 = request.security("NSE:ICICIBANK", timeframe.period, volume)
cf2 = (close2 - open2) * volume2
weight2 = manualMode ? manualWeight2 : getWeight("NSE:ICICIBANK")
weightedCF2 = cf2 * weight2

// RELIANCE

open3 = request.security("NSE:RELIANCE", timeframe.period, open)
volume3 = request.security("NSE:RELIANCE", timeframe.period, volume)
cf3 = (close3 - open3) * volume3
weight3 = manualMode ? manualWeight3 : getWeight("NSE:RELIANCE")
weightedCF3 = cf3 * weight3

// INFY

volume4 = request.security("NSE:INFY", timeframe.period, volume)

weightedCF4 = cf4 * weight4

// BHARTIARTL

open5 = request.security("NSE:BHARTIARTL", timeframe.period, open)
volume5 = request.security("NSE:BHARTIARTL", timeframe.period, volume)
cf5 = (close5 - open5) * volume5

// AXISBANK

cf9 = (close9 - open9) * volume9
weight9 = manualMode ? manualWeight9 : getWeight("NSE:AXISBANK")

// KOTAKBANK
close10 = request.security("NSE:KOTAKBANK", timeframe.period, close)

// SUNPHARMA

// --- 5. Sum all weighted cash flows ---
totalWeight = 100.0
totalWeightedCF = weightedCF1 + weightedCF2 + weightedCF3 + weightedCF4 + weightedCF5 + 
                  weightedCF6 + weightedCF7 + weightedCF8 + weightedCF9 + weightedCF10 +
                  weightedCF11 + weightedCF12

// --- 6. Calculate final trend line ---
niftyCashFlowTrend := totalWeightedCF / totalWeight

// --- 7. Smoothing and Bands ---
smoothLength = input.int(14, "Smoothing Period")
cfSmooth = ta.sma(niftyCashFlowTrend, smoothLength)
cfVolatility = ta.stdev(niftyCashFlowTrend, smoothLength)

upperBand = cfSmooth + (cfVolatility * 0.5)
lowerBand = cfSmooth - (cfVolatility * 0.5)

// --- 8. Trading Signals ---
isStrongInflow = niftyCashFlowTrend > upperBand and niftyCashFlowTrend > niftyCashFlowTrend[1]
isStrongOutflow = niftyCashFlowTrend < lowerBand and niftyCashFlowTrend < niftyCashFlowTrend[1]

// --- 9. Plotting ---
plot(niftyCashFlowTrend, "Combined Cash Flow", color.new(#2962FF, 0), 2)

p1 = plot(upperBand, "Inflow Zone", color.new(#00C853, 70))
p2 = plot(lowerBand, "Outflow Zone", color.new(#FF1744, 70))
fill(p1, p2, color.new(#2962FF, 95))

hline(0, "Zero Line", color.new(#888888, 50))

plotshape(isStrongInflow, "INFLOW", shape.triangleup, location.belowbar, 
         color.new(#00C853, 0), size=size.small, text="IN")

plotshape(isStrongOutflow, "OUTFLOW", shape.triangledown, location.abovebar, 
         color.new(#FF1744, 0), size=size.small, text="OUT")

bgcolor(isStrongInflow ? color.new(#00C853, 90) : 
        isStrongOutflow ? color.new(#FF1744, 90) : na)

    weightText = manualMode ? "Manual" : (useMarketCapWeight ? "Market Cap" : "Equal (8.33%)")

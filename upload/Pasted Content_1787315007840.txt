  function calculateRealIV(marketPrice, strike, T, r, optionPrice, type) {
    if (optionPrice <= 0 || marketPrice <= 0 || T <= 0.001) return 0.18;
    
    var sigma = 0.20;
    var tolerance = 0.0001;
    var maxIterations = 30;
    
    for (var iter = 0; iter < maxIterations; iter++) {
      var d1 = (Math.log(marketPrice / strike) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
      var d2 = d1 - sigma * Math.sqrt(T);
      
      var priceDiff;
      if (type === "CE") {
        priceDiff = marketPrice * normCDF(d1) - strike * Math.exp(-r * T) * normCDF(d2) - optionPrice;
      } else {
        priceDiff = strike * Math.exp(-r * T) * normCDF(-d2) - marketPrice * normCDF(-d1) - optionPrice;
      }
      
      if (Math.abs(priceDiff) < tolerance) break;
      
      var vega = marketPrice * Math.sqrt(T) * normPDF(d1);
      if (vega < 0.0001) vega = 0.0001;
      
      var newSigma = sigma - priceDiff / vega;
      if (newSigma <= 0.05) newSigma = 0.05;
      if (newSigma > 1.5) newSigma = 1.5;
      
      sigma = newSigma;
    }
    
    return sigma;
  }

  function calculateCompleteGreeks(S, K, T, r, sigma, type) {
    if (T <= 0.001) T = 0.001;
    if (sigma <= 0.01) sigma = 0.15;
    
    var d1 = (Math.log(S / K) + (r + sigma * sigma / 2) * T) / (sigma * Math.sqrt(T));
    var d2 = d1 - sigma * Math.sqrt(T);
    
    var delta, gamma, theta, vega, rho;
    var pdfD1 = normPDF(d1);
    
    if (type === "CE") {
      delta = normCDF(d1);
      theta = (-S * pdfD1 * sigma / (2 * Math.sqrt(T))) - (r * K * Math.exp(-r * T) * normCDF(d2));
      rho = K * T * Math.exp(-r * T) * normCDF(d2);
    } else {
      delta = normCDF(d1) - 1;
      theta = (-S * pdfD1 * sigma / (2 * Math.sqrt(T))) + (r * K * Math.exp(-r * T) * normCDF(-d2));
      rho = -K * T * Math.exp(-r * T) * normCDF(-d2);
    }
    
    gamma = pdfD1 / (S * sigma * Math.sqrt(T));
    vega = S * pdfD1 * Math.sqrt(T);
    
    var tradingDays = 252;
    theta = theta / tradingDays;
    vega = vega / 100;
    
    return {
      delta: delta,
      gamma: gamma,
      theta: theta,
      vega: vega,
      rho: rho,
      iv: sigma
    };
  }

function getNiftyExpiries() {
  var ist = new Date(new Date().toLocaleString("en-US", { timeZone: "Asia/Kolkata" }));

  var day     = ist.getDay();
  var hours   = ist.getHours();
  var minutes = ist.getMinutes();

  var expired = (day === 2 && (hours > 15 || (hours === 15 && minutes >= 30)));
  var daysUntilTuesday = (2 - day + 7) % 7;

  if (day === 2 && !expired) {
    daysUntilTuesday = 0;
  } else if (day === 2 && expired) {
    daysUntilTuesday = 7;
  }

  var currentExpDate = new Date(ist);
  currentExpDate.setDate(ist.getDate() + daysUntilTuesday);

  var nextWeeklyDate = new Date(currentExpDate);
  nextWeeklyDate.setDate(currentExpDate.getDate() + 7);

  var lastTuesdayOfMonth = new Date(currentExpDate.getFullYear(), currentExpDate.getMonth() + 1, 0);
  while (lastTuesdayOfMonth.getDay() !== 2) {
    lastTuesdayOfMonth.setDate(lastTuesdayOfMonth.getDate() - 1);
  }

  function resetTime(date) {
    return new Date(date.getFullYear(), date.getMonth(), date.getDate());
  }

  var monthNames = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN",
                    "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"];

  function formatWeekly(date) {
    var yy = String(date.getFullYear()).slice(-2);
    var m = String(date.getMonth() + 1);
    var dd = String(date.getDate()).padStart(2, '0');
    return yy + m + dd;
  }

  function formatMonthly(date) {
    var yy = String(date.getFullYear()).slice(-2); 
    var month = monthNames[date.getMonth()];       
    return yy + month; 
  }

  var currentExpiryCode = formatWeekly(currentExpDate);
  var nextExpiryCode;
  var isNextMonthly = false;
  
  var nextWeeklyDateOnly = resetTime(nextWeeklyDate);
  var lastTuesdayOfMonthOnly = resetTime(lastTuesdayOfMonth);
  
  if (nextWeeklyDateOnly.getTime() === lastTuesdayOfMonthOnly.getTime()) {
    nextExpiryCode = formatMonthly(nextWeeklyDate); 
    isNextMonthly = true;
    Logger.log("Next expiry is MONTHLY: " + nextExpiryCode);
  } else {
    nextExpiryCode = formatWeekly(nextWeeklyDate);
    Logger.log("Next expiry is WEEKLY: " + nextExpiryCode);
  }

  Logger.log("Current expiry: " + currentExpiryCode);
  Logger.log("Next expiry: " + nextExpiryCode);

  return {
    current: currentExpiryCode,
    next: nextExpiryCode,
    currentDate: currentExpDate,
    nextDate: nextWeeklyDate,
    isNextMonthly: isNextMonthly
  };
}

function normCDF(x) {
  return (1.0 + erf(x / Math.sqrt(2.0))) / 2.0;
}

function erf(x) {
  var sign = (x >= 0) ? 1 : -1;
  x = Math.abs(x);

  var a1 = 0.254829592;
  var a2 = -0.284496736;
  var a3 = 1.421413741;
  var a4 = -1.453152027;
  var a5 = 1.061405429;
  var p  = 0.3275911;

  var t = 1.0 / (1.0 + p * x);
  var y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);

  return sign * y;
}

function normPDF(x) {
  return Math.exp(-x * x / 2) / Math.sqrt(2 * Math.PI);
}

                Utilities.formatDate(rawCellVal, tz, "HH:mm:ss") : rawCellVal.toString().trim();

    // -------------------------------------------------------------------------

        } else if (Number(totalBearPremium) > Number(totalBullPremium)) {
          netExternalBear = (Number(totalBearPremium) - Number(totalBullPremium)).toFixed(2);

        var emaPutWriteVol  = (Number(cleanPutWriteVol) * k) + (prevPutWriteVolEMA * (1 - k));
        var emaCashFlow     = (Number(cashCr || 0) * k) + (prevCashFlowEMA * (1 - k));

        // =========================================================================

        var intradayLow = liveSpot;

            intradayLow = Math.min.apply(null, prices);

          minuteSignal = "BULLISH"; 
        }
        // 🟥 Condition C: Standard Continuous Bearish Trend
        else if (emaCallWriteVol > emaPutWriteVol && meltRatio > 1.12 && emaCashFlow < 0.5) {
          minuteSignal = "BEARISH";
        }

        else if (liveSpot >= (intradayHigh * 0.9995) && callWriteExpanding > (emaPutWriteVol * 0.10) && cashFlowImproving < -0.3) {
          minuteSignal = "BEARISH"; 
        }

        else if (volDifferencePct < 0.12 && ceMelt > 15 && peMelt > 15) {

        if (mainSignalUpper.indexOf("BEARISH") > -1) combinationScore -= 1.5;

        if (minuteSignal === "BULLISH") combinationScore += 1.5;
        if (minuteSignal === "BEARISH") combinationScore -= 1.5;
        if (minuteSignal === "TRAP")    combinationScore = 0; // Trap forces a flat center line

        // Final score mapping text tag for easy scanning alongside numbers
        var displayArrow = "➡️ SIDEWAYS / TRAP";
        if (combinationScore >= 2)       displayArrow = "▲▲ SUPER BULLISH (" + combinationScore + ")";
        else if (combinationScore > 0)  displayArrow = "▲ MILD BULLISH (" + combinationScore + ")";
        else if (combinationScore <= -2) displayArrow = "▼▼ SUPER BEARISH (" + combinationScore + ")";
        else if (combinationScore < 0)  displayArrow = "▼ MILD BEARISH (" + combinationScore + ")";

        // =========================================================================
        // ARRAY COMPILATION & DISPATCH PIPELINE
        // =========================================================================
        var dynamicsRow = [
          currentMinute,                                     // A. TIME
          niftyPrice,                                        // B. NIFTY_PRICE
          percentChangeStr,                                  // C. NIFTY_%_CHANGE
          score + "/5",                                      // D. ALGO_SCORE
          signal,                                            // E. ALGO_SIGNAL
          cashCr,                                            // F. CASH_FLOW_CR
          
          // --- CURRENT EXPIRY ---
          cBuyCr,                                            // G. CURR_CALL_BUY
          cWriteCr,                                          // H. CURR_CALL_WRITE
          pBuyCr,                                            // I. CURR_PUT_BUY
          pWriteCr,                                          // J. CURR_PUT_WRITE
          (numFutCurr >= 0 ? futCurrCr : "0.00"),            // K. FUT_CURR_IN
          (numFutCurr < 0 ? Math.abs(numFutCurr).toFixed(2) : "0.00"), // L. FUT_CURR_OUT

          nBuyCr,                                            // M. NEXT_CALL_BUY
          nWriteCr,                                          // N. NEXT_CALL_WRITE
          npBuyCr,                                           // O. NEXT_PUT_BUY
          npWriteCr,                                         // P. NEXT_PUT_WRITE
          (numFutNext >= 0 ? futNextCr : "0.00"),            // Q. FUT_NEXT_IN

          netExternalBear,                                   // T. TOTAL_BEAR_PREMIUM

                  Utilities.formatDate(rawTimeVal, tz, "HH:mm:ss") : rawTimeVal.toString().trim();

              }

        dynamicsSheet.getRange(targetDynamicsRow, 1, 1, dynamicsRow.length).setValues([dynamicsRow]);

      }
    } catch (externalErr) {

    }

  // This closes the outer pipeline wrapper safely
  } catch (err) {

function runAdvancedInstitutionalIndicators(sheet, rowNum, niftyPrice, volumeCr, cashCr, futCurrCr, futNextCr) {
  try {
    if (rowNum < 6) return; 
    
    // ---------------------------------------------------------------------
    // MODULE 1: THE MULTI-EXPIRY OI MULTIPLIER (EXPIRY ROLL VALUE)
    // ---------------------------------------------------------------------
    var cBuy   = Number(sheet.getRange(rowNum, 9).getValue())  || 0;
    var cWrite = Number(sheet.getRange(rowNum, 10).getValue()) || 0;
    var pBuy   = Number(sheet.getRange(rowNum, 11).getValue()) || 0;
    var pWrite = Number(sheet.getRange(rowNum, 12).getValue()) || 0;
    
    var nBuy   = Number(sheet.getRange(rowNum, 13).getValue()) || 0;
    var nWrite = Number(sheet.getRange(rowNum, 14).getValue()) || 0;
    var npBuy  = Number(sheet.getRange(rowNum, 15).getValue()) || 0;
    var npWrite = Number(sheet.getRange(rowNum, 16).getValue()) || 0;

    var curOptNet = (cBuy + pWrite) - (cWrite + pBuy);
    var nxtOptNet = (nBuy + npWrite) - (nWrite + npBuy);
    
    var expiryRollStatus = "⚡ CALM";
    if (Math.abs(nxtOptNet) > (Math.abs(curOptNet) * 1.5) && Math.abs(nxtOptNet) > 0) {
      expiryRollStatus = "🌀 INSTITUTIONAL ROLL OVER";
    } else if (Math.abs(nxtOptNet) > Math.abs(curOptNet) && Math.abs(nxtOptNet) > 0) {
      expiryRollStatus = "🔄 ACCUMULATING FAR MONTH";
    }

    // ---------------------------------------------------------------------
    // MODULE 2: ORDER BOOK FLOW TOXICITY (VPIN LITE)
    // ---------------------------------------------------------------------
    var toxicityStatus = "⚪ RETAIL CHURN";
    var priceMoveSum = 0;
    
    if (rowNum >= 9) { 
      var historicalPrices = sheet.getRange(rowNum - 3, 2, 4, 1).getValues(); 
      for (var i = 1; i < historicalPrices.length; i++) {
        priceMoveSum += (historicalPrices[i][0] - historicalPrices[i-1][0]);
      }
      
      var netFlows = Number(cashCr) + Number(futCurrCr) + Number(futNextCr);
      
      if (priceMoveSum > 0 && netFlows > 0) {
        toxicityStatus = "🟢 INST. ACCUMULATION";
      } else if (priceMoveSum < 0 && netFlows < 0) {
        toxicityStatus = "🔴 INST. DISTRIBUTION";
      }
    }

    // ---------------------------------------------------------------------
    // MODULE 3: OPTIONS PREMIUM DIVERGENCE INDEX (PDI)
    // ---------------------------------------------------------------------
    var pdiStatus = "✅ ALIGNED";
    if (rowNum >= 15) { 
      var priceWindow = sheet.getRange(rowNum - 9, 2, 10, 1).getValues().map(function(r){return r[0];});
      var callBuyWindow = sheet.getRange(rowNum - 9, 9, 10, 1).getValues().map(function(r){return r[0];});
      
      var maxPriceIndex = priceWindow.indexOf(Math.max.apply(null, priceWindow));
      var maxCallIndex  = callBuyWindow.indexOf(Math.max.apply(null, callBuyWindow));
      
      if (maxPriceIndex === 9 && maxCallIndex !== 9) {
        pdiStatus = "⚠️ BEARISH DIVERGENCE";
      } else if (priceWindow[9] < priceWindow[0] && callBuyWindow[9] > callBuyWindow[0]) {
        pdiStatus = "🚀 BULLISH DIVERGENCE";
      }
    }

    // ---------------------------------------------------------------------
    // MODULE 4: HIGH-CONVICTION CONFLUENCE SIGNAL ENGINE
    // ---------------------------------------------------------------------
    var finalTradeSignal = "🟡 NEUTRAL / WAIT";
    var scoreNum = 2; 

    if (rowNum >= 6) {
      var coreScoreText = sheet.getRange(rowNum, 21).getValue();
      if (coreScoreText !== null && coreScoreText !== undefined) {
        if (typeof coreScoreText === 'string' && coreScoreText.indexOf("/") > -1) {
          scoreNum = parseInt(coreScoreText.split("/")[0]);
        } else {
          scoreNum = parseInt(coreScoreText); 
        }
      }
    }

    if (scoreNum >= 4 && toxicityStatus.indexOf("ACCUMULATION") > -1) {
      if (pdiStatus.indexOf("BULLISH") > -1 || expiryRollStatus !== "⚡ CALM") {
        finalTradeSignal = "🚀 HIGH-CONVICTION BUY";
      } else {
        finalTradeSignal = "🟢 SQUEEZE LONG";
      }
    }
    else if (scoreNum <= 1 && toxicityStatus.indexOf("DISTRIBUTION") > -1) {
      if (pdiStatus.indexOf("BEARISH") > -1 || expiryRollStatus !== "⚡ CALM") {
        finalTradeSignal = "⚠️ HIGH-CONVICTION SELL";
      } else {
        finalTradeSignal = "🔴 SQUEEZE SHORT";
      }
    }
    else if (pdiStatus.indexOf("BULLISH") > -1 && priceMoveSum < 0) {
      finalTradeSignal = "🔄 SCALP CALL BUY (REVERSAL)";
    } 
    else if (pdiStatus.indexOf("BEARISH") > -1 && priceMoveSum > 0) {
      finalTradeSignal = "🔄 SCALP PUT BUY (REVERSAL)";
    }

    // ---------------------------------------------------------------------
    // WRITE OUTPUT MATRIX TO DASHBOARD (COLUMNS W, X, Y, Z)
    // ---------------------------------------------------------------------
    var outputRange = sheet.getRange(rowNum, 23, 1, 4); 
    outputRange.setValues([[expiryRollStatus, toxicityStatus, pdiStatus, finalTradeSignal]]);
    outputRange.setHorizontalAlignment("center").setFontWeight("bold");

    // ---------------------------------------------------------------------
    // FIXED: TARGET EXACT SAME ROW (ROWNUM) FOR METRICS (COLUMNS AA, AB, AC)
    // ---------------------------------------------------------------------
    // Uses rowNum to ensure it prints directly beside your trade signals from row 6 onwards
    var meltOutputRange = sheet.getRange(rowNum, 27, 1, 3); // AA (27), AB (28), AC (29)
    meltOutputRange.setValues([[
      typeof totalCeExtrinsicMelt !== 'undefined' ? Number(totalCeExtrinsicMelt) : 0, 
      typeof totalPeExtrinsicMelt !== 'undefined' ? Number(totalPeExtrinsicMelt) : 0, 
      typeof finalMeltRatio !== 'undefined' ? Number(finalMeltRatio) : 1
    ]]);
    meltOutputRange.setHorizontalAlignment("center").setFontWeight("bold").setFontColor("#2f5597");
    
    // ---------------------------------------------------------------------
    // VISUAL COLOR-CODING LOGIC (COLUMNS W, X, Y, Z)
    // ---------------------------------------------------------------------
    sheet.getRange(rowNum, 23).setBackground(expiryRollStatus.indexOf("⚡") > -1 ? "#ffffff" : "#d9e1f2").setFontColor("#1f4e78");
    
    var toxCell = sheet.getRange(rowNum, 24);
    if (toxicityStatus.indexOf("ACCUMULATION") > -1) toxCell.setBackground("#c6efce").setFontColor("#006100");
    else if (toxicityStatus.indexOf("DISTRIBUTION") > -1) toxCell.setBackground("#ffc7ce").setFontColor("#9c0006");
    else toxCell.setBackground("#ffffff").setFontColor("#7f7f7f");

    var pdiCell = sheet.getRange(rowNum, 25);
    if (pdiStatus.indexOf("BEARISH") > -1) pdiCell.setBackground("#ffeb9c").setFontColor("#9c6500");
    else if (pdiStatus.indexOf("BULLISH") > -1) pdiCell.setBackground("#b4c6e7").setFontColor("#1f4e78");
    else pdiCell.setBackground("#ffffff").setFontColor("#595959");

    var sigCell = sheet.getRange(rowNum, 26);
    if (finalTradeSignal.indexOf("HIGH-CONVICTION BUY") > -1) {
      sigCell.setBackground("#00b050").setFontColor("#ffffff"); 
    } else if (finalTradeSignal.indexOf("HIGH-CONVICTION SELL") > -1) {
      sigCell.setBackground("#c00000").setFontColor("#ffffff"); 
    } else if (finalTradeSignal.indexOf("SQUEEZE LONG") > -1) {
      sigCell.setBackground("#e2f0d9").setFontColor("#385723"); 
    } else if (finalTradeSignal.indexOf("SQUEEZE SHORT") > -1) {
      sigCell.setBackground("#fce4d6").setFontColor("#c65911"); 
    } else if (finalTradeSignal.indexOf("SCALP") > -1) {
      sigCell.setBackground("#fff2cc").setFontColor("#7f6000"); 
    } else {
      sigCell.setBackground("#f2f2f2").setFontColor("#a6a6a6"); 
    }

  } catch(err) {
    Logger.log("Advanced indicators engine exception: " + err.message);
  }
}

    var callWriteVol = Number(rowData[24]) || 0;     // Col Y
    var putWriteVol = Number(rowData[25]) || 0;      // Col Z
    
    // --- THRESHOLDS FOR BIG PLAYERS ---
    var minPremiumCr = 3.0;       
    var minWriteVol = 50000;      
    
    var signal = "WAIT / RETAIL NOISE";
    var reason = "No institutional alignment";

    // LOGIC 1: BULLISH BIG PLAYER BET (BUY CALL)

    var isCashBullish = cashFlow > 0;
    var isFutBullish = futCurrIn > futCurrOut;

    var heavyCallBuy = (currCallBuy > minPremiumCr);

        reason = "Price down, but Bull Premium > Bear Premium. Institutions buying the dip.";
      } else {
        signal = "📉 BUY PUT (BIG PLAYER BEARISH)";
        reason = "Cash + Fut Outflow + Heavy Call Writing/Put Buying. Smart Money Short.";
      }
    }
    
    // Write Signals to Columns 30 (AD) and 31 (AE)
    sheet.getRange(lastRow, 30).setValue(signal);
    sheet.getRange(lastRow, 31).setValue(reason);
    
    // Color Code the Signal Cell
    var signalCell = sheet.getRange(lastRow, 30);
    if (signal.indexOf("BUY CALL") > -1) {
      signalCell.setBackground("#008000").setFontColor("white").setFontWeight("bold");
    } else if (signal.indexOf("BUY PUT") > -1) {
      signalCell.setBackground("#FF0000").setFontColor("white").setFontWeight("bold");
    } else if (signal.indexOf("TRAP") > -1) {
      signalCell.setBackground("#FFA500").setFontColor("black").setFontWeight("bold");
    } else {
      signalCell.setBackground("#808080").setFontColor("white");
    }
    
  } catch (e) {
    Logger.log("Big Player Detection Error: " + e.toString());
    runStrikeHeatMap();
  }
}

var EXTERNAL_WB_ID = "1YC-seFY6uQRGOQEielLnoFyEYuFtjOrnQb7JlnIi_W4";

var START_ROW = 5;

var STRIKE_STEP = 50;

var HEAT_THRESHOLD_MULT = 1.2;  // × standard deviation
var MIN_HEAT_THRESHOLD = 2;     // minimum ₹2

// =========================================================================
// HELPER: Normalize any timestamp value (Date object or string) to "HH:mm:ss"
// =========================================================================
function _hmExtractTimeOnly(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, "Asia/Kolkata", "HH:mm:ss");
  }
  var s = String(val).trim();
  var match = s.match(/(\d{2}:\d{2}:\d{2})/);
  return match ? match[1] : s;
}

// =========================================================================
// HELPER: Normalize timestamp to "dd/MM/yyyy HH:mm:ss" for consistent comparison
// =========================================================================
function _hmNormalizeTime(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, "Asia/Kolkata", "dd/MM/yyyy HH:mm:ss");
  }
  return String(val).trim();
}

function runStrikeHeatMap() {

  Logger.log("=== STRIKE HEAT MAP v2.1 STARTED ===");

  // =========================
  // 1. GET SOURCE RAW_OPTIONS SHEET
  // =========================
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var rawSheet = ss.getSheetByName("RAW_OPTIONS");
  if (!rawSheet) {
    Logger.log("ERROR: RAW_OPTIONS sheet not found");
    return;
  }

  var lastRow = rawSheet.getLastRow();
  if (lastRow < START_ROW + STRIKE_COUNT) {
    Logger.log("Not enough data yet (need 2+ snapshots). Rows: " + lastRow);
    return;
  }

  // =========================
  // 2. READ THE LAST ~50 ROWS (covers 2+ full snapshots of 22 rows each)
  // =========================
  var scanRows = Math.min(50, lastRow - START_ROW + 1);
  var scanStart = lastRow - scanRows + 1;
  var rawData = rawSheet.getRange(scanStart, 1, scanRows, 7).getValues();
  // Cols: 0=Time, 1=Expiry, 2=Strike, 3=CE LTP, 4=PE LTP, 5=CE OI, 6=PE OI

  // =========================
  // 3. FIND UNIQUE TIMESTAMPS (using normalized strings for Date object safety)
  // =========================
  var timestamps = [];
  for (var i = 0; i < rawData.length; i++) {
    var tNorm = _hmNormalizeTime(rawData[i][0]);
    if (tNorm === "") continue;
    if (timestamps.length === 0 || timestamps[timestamps.length - 1] !== tNorm) {
      timestamps.push(tNorm);
    }
  }

  if (timestamps.length < 2) {
    Logger.log("Only 1 snapshot available. Need at least 2. Waiting...");
    return;
  }

  var prevTimestamp = timestamps[timestamps.length - 2];
  var currTimestamp = timestamps[timestamps.length - 1];
  var currTimeOnly = _hmExtractTimeOnly(rawData[rawData.length - 1][0]);
  Logger.log("Previous: " + prevTimestamp + " → Current: " + currTimestamp);

  // =========================
  // 4. EXTRACT CURRENT WEEK EXPIRY (FIXED v2.1)
  // =========================
  // BUG FIX: Don't use "first non-empty expiry in scan window" because the
  // scan window may start mid-snapshot at a next-week row (26714).
  // Instead, find the FIRST expiry within the CURRENT timestamp's block.
  // Within each timestamp, current week rows (26707) ALWAYS come before
  // next week rows (26714), so the first expiry seen = current week.
  var currentWeekExpiry = "";
  for (var i = 0; i < rawData.length; i++) {
    var tNorm = _hmNormalizeTime(rawData[i][0]);
    if (tNorm === currTimestamp) {
      var exp = String(rawData[i][1]).trim();
      if (exp !== "" && currentWeekExpiry === "") {
        currentWeekExpiry = exp;
        break; // First expiry in current timestamp = current week
      }
    }
  }

  if (currentWeekExpiry === "") {
    Logger.log("ERROR: Could not detect current week expiry");
    return;
  }
  Logger.log("Current Week Expiry: " + currentWeekExpiry);

  // =========================
  // 5. PARSE PREVIOUS AND CURRENT SNAPSHOT DATA
  // =========================
  var prevData = {};
  var currData = {};

  for (var i = 0; i < rawData.length; i++) {
    var tNorm = _hmNormalizeTime(rawData[i][0]);
    var exp = String(rawData[i][1]).trim();
    var strike = Number(rawData[i][2]);
    var ceLTP = Number(rawData[i][3]) || 0;
    var peLTP = Number(rawData[i][4]) || 0;
    var ceOI = Number(rawData[i][5]) || 0;
    var peOI = Number(rawData[i][6]) || 0;

    // Only current week expiry
    if (exp !== currentWeekExpiry) continue;
    if (strike <= 0) continue;

    var entry = { ceLTP: ceLTP, peLTP: peLTP, ceOI: ceOI, peOI: peOI };

    if (tNorm === prevTimestamp) {
      prevData[strike] = entry;
    } else if (tNorm === currTimestamp) {
      currData[strike] = entry;
    }
  }

  // Verify we have all 11 strikes in both snapshots
  var strikes = Object.keys(currData).map(Number).sort(function(a, b) { return a - b; });

  if (strikes.length < STRIKE_COUNT) {
    Logger.log("Only " + strikes.length + " strikes in current snapshot for expiry " +
      currentWeekExpiry + ". Need " + STRIKE_COUNT);
    return;
  }

  var prevStrikes = Object.keys(prevData).map(Number).sort(function(a, b) { return a - b; });
  if (prevStrikes.length < STRIKE_COUNT) {
    Logger.log("Previous snapshot only has " + prevStrikes.length + " strikes for expiry " +
      currentWeekExpiry + ". Need " + STRIKE_COUNT);
    return;
  }

  Logger.log("Strikes found: " + strikes.join(", "));

  // =========================
  // 6. CALCULATE PREMIUM CHANGES & OI CHANGES
  // =========================
  var cePremiumChange = {};
  var pePremiumChange = {};
  var ceOIChange = {};
  var peOIChange = {};

  for (var s = 0; s < strikes.length; s++) {
    var stk = strikes[s];
    if (!prevData[stk]) continue;

    cePremiumChange[stk] = currData[stk].ceLTP - prevData[stk].ceLTP;
    pePremiumChange[stk] = currData[stk].peLTP - prevData[stk].peLTP;
    ceOIChange[stk] = currData[stk].ceOI - prevData[stk].ceOI;
    peOIChange[stk] = currData[stk].peOI - prevData[stk].peOI;
  }

  // =========================
  // 7. CALCULATE HEAT SCORES
  // =========================
  var ceHeatScore = {};
  var peHeatScore = {};

  for (var s = 0; s < strikes.length; s++) {
    var stk = strikes[s];
    if (cePremiumChange[stk] === undefined) continue;

    var ceNeighborSum = 0;
    var peNeighborSum = 0;
    var neighborCount = 0;

    for (var n = -NEIGHBOR_RANGE; n <= NEIGHBOR_RANGE; n++) {
      if (n === 0) continue;
      var neighborStrike = stk + (n * STRIKE_STEP);
      if (cePremiumChange[neighborStrike] !== undefined) {
        ceNeighborSum += cePremiumChange[neighborStrike];
        peNeighborSum += pePremiumChange[neighborStrike];
        neighborCount++;
      }
    }

    if (neighborCount > 0) {
      ceHeatScore[stk] = cePremiumChange[stk] - (ceNeighborSum / neighborCount);
      peHeatScore[stk] = pePremiumChange[stk] - (peNeighborSum / neighborCount);
    }
  }

  // =========================
  // 8. DYNAMIC THRESHOLD
  // =========================
  var allCeHeats = [];
  var allPeHeats = [];
  for (var stk in ceHeatScore) {
    allCeHeats.push(ceHeatScore[stk]);
    allPeHeats.push(peHeatScore[stk]);
  }

  function calcStdDev(arr) {
    if (arr.length < 2) return 0;
    var sum = 0;
    for (var i = 0; i < arr.length; i++) sum += arr[i];
    var mean = sum / arr.length;
    var sqDiffSum = 0;
    for (var i = 0; i < arr.length; i++) sqDiffSum += (arr[i] - mean) * (arr[i] - mean);
    return Math.sqrt(sqDiffSum / arr.length);
  }

  var ceStdDev = calcStdDev(allCeHeats);
  var peStdDev = calcStdDev(allPeHeats);

  var ceThreshold = Math.max(MIN_HEAT_THRESHOLD, HEAT_THRESHOLD_MULT * ceStdDev);
  var peThreshold = Math.max(MIN_HEAT_THRESHOLD, HEAT_THRESHOLD_MULT * peStdDev);

  Logger.log("CE Threshold: " + ceThreshold.toFixed(2) + " (stdDev=" + ceStdDev.toFixed(2) + ")");
  Logger.log("PE Threshold: " + peThreshold.toFixed(2) + " (stdDev=" + peStdDev.toFixed(2) + ")");

  // =========================
  // 9. DETERMINE SIGNALS
  // =========================
  function getSignal(heatScore, oiChange, optionType) {
    var oiRising = oiChange > 0;
    var oiFalling = oiChange < 0;

    if (optionType === "CE") {
      if (heatScore < -ceThreshold && oiRising) return "🔴 BIG SHORT (Call Writers)";
      if (heatScore > ceThreshold && oiRising) return "🟢 BIG LONG (Call Buyers)";
      if (heatScore < -ceThreshold && oiFalling) return "🟡 SHORT COVERING (Calls)";
      if (heatScore > ceThreshold && oiFalling) return "🟡 LONG UNWINDING (Calls)";
      return "⚪ Normal";
    } else {
      if (heatScore < -peThreshold && oiRising) return "🟢 BIG SHORT (Put Writers)";
      if (heatScore > peThreshold && oiRising) return "🔴 BIG LONG (Put Buyers)";
      if (heatScore < -peThreshold && oiFalling) return "🟡 SHORT COVERING (Puts)";
      if (heatScore > peThreshold && oiFalling) return "🟡 LONG UNWINDING (Puts)";
      return "⚪ Normal";
    }
  }

  // =========================
  // 10. WRITE TO EXTERNAL WORKBOOK
  // =========================
  try {
    var extWb = SpreadsheetApp.openById(EXTERNAL_WB_ID);
  } catch (e) {
    Logger.log("ERROR: Cannot open external workbook: " + e.toString());
    return;
  }

  var hmSheet = extWb.getSheetByName(HEATMAP_SHEET_NAME);

  // Create sheet + headers if it doesn't exist
  if (!hmSheet) {
    hmSheet = extWb.insertSheet(HEATMAP_SHEET_NAME);
    hmSheet.getRange(1, 1, 1, 18).setValues([[
      "STRIKE HEAT MAP — NIFTY50 CURRENT WEEK EXPIRY",
      "", "", "", "", "", "", "", "", "", "", "", "", "", "", "", "", ""
    ]]);
    hmSheet.getRange(1, 1, 1, 18).merge().setFontWeight("bold").setFontSize(12)
      .setBackground("#1a1a2e").setFontColor("#e94560");

    hmSheet.getRange(2, 1, 1, 18).setValues([[
      "Time", "NIFTY Spot", "Expiry",
      "CE ₹ Change", "CE Prev LTP", "CE LTP", "CE ΔOI", "CE OI", "CE Heat Score", "CE Signal",
      "Strike",
      "PE Signal", "PE Heat Score", "PE OI", "PE ΔOI", "PE LTP", "PE Prev LTP", "PE ₹ Change"
    ]]);
    hmSheet.getRange(2, 1, 1, 18).setFontWeight("bold").setFontSize(9)
      .setBackground("#16213e").setFontColor("#eee");

    hmSheet.getRange(3, 1, 1, 18).setValues([[
      "Thresholds →", "", "",
      "CE Thr: (auto)", "", "", "", "", "(1.2×σ=...)", "Signal Logic:",
      "Heat = Premium Change vs Neighbor Avg",
      "Signal Logic:", "(1.2×σ=...)", "", "", "", "", "PE Thr: (auto)"
    ]]);
    hmSheet.getRange(3, 1, 1, 18).setFontWeight("bold").setFontSize(8)
      .setBackground("#0f3460").setFontColor("#a8d8ea");

    hmSheet.getRange(4, 1, 1, 18).setValues([["","","","","","","","","","","","","","","","","","",""]]);
  }

  // Get NIFTY spot price from DYNAMICS tab
  var niftySpot = 0;
  try {
    var dynSheet = extWb.getSheetByName("NIFTY50 DYNAMICS");
    if (dynSheet) {
      var dynLastRow = dynSheet.getLastRow();
      if (dynLastRow >= START_ROW) {
        niftySpot = Number(dynSheet.getRange(dynLastRow, 2).getValue()) || 0;
      }
    }
  } catch (e) {
    niftySpot = strikes[Math.floor(strikes.length / 2)];
  }

  if (niftySpot === 0) {
    niftySpot = strikes[Math.floor(strikes.length / 2)];
  }

  // =========================
  // 11. SAME-MINUTE OVERWRITE (FIXED v2.1)
  // =========================
  // BUG FIX: Old code used String(Date).split(" ")[1] which gave "Jul" instead
  // of time. Now uses _hmExtractTimeOnly() for proper "HH:mm:ss" comparison.
  // Also FIX: Overwrite starts at block START (lastRow - 10), not at lastRow,
  // to prevent duplicate rows stacking up.
  var hmLastRow = hmSheet.getLastRow();
  var targetRow = (hmLastRow < START_ROW - 1) ? START_ROW : hmLastRow + 1;

  if (hmLastRow >= START_ROW) {
    var lastHmTimeRaw = hmSheet.getRange(hmLastRow, 1).getValue();
    var lastHmTimeOnly = _hmExtractTimeOnly(lastHmTimeRaw);

    if (lastHmTimeOnly === currTimeOnly) {
      // Same minute — overwrite the entire previous block
      // Previous block of 11 rows ends at hmLastRow, so starts at hmLastRow - 10
      targetRow = Math.max(START_ROW, hmLastRow - STRIKE_COUNT + 1);
      Logger.log("Same-minute overwrite: rows " + targetRow + " to " +
        (targetRow + STRIKE_COUNT - 1));
    }
  }

  // =========================
  // 12. BUILD OUTPUT ROWS
  // =========================
  var outputRows = [];
  for (var s = 0; s < strikes.length; s++) {
    var stk = strikes[s];
    var ceLTP = currData[stk] ? currData[stk].ceLTP : 0;
    var peLTP = currData[stk] ? currData[stk].peLTP : 0;
    var ceOI = currData[stk] ? currData[stk].ceOI : 0;
    var peOI = currData[stk] ? currData[stk].peOI : 0;
    var prevCeLTP = prevData[stk] ? prevData[stk].ceLTP : 0;
    var prevPeLTP = prevData[stk] ? prevData[stk].peLTP : 0;

    var ceChange = cePremiumChange[stk] !== undefined ? cePremiumChange[stk] : 0;
    var peChange = pePremiumChange[stk] !== undefined ? pePremiumChange[stk] : 0;
    var ceHeat = ceHeatScore[stk] !== undefined ? ceHeatScore[stk] : 0;
    var peHeat = peHeatScore[stk] !== undefined ? peHeatScore[stk] : 0;
    var ceDO = ceOIChange[stk] !== undefined ? ceOIChange[stk] : 0;
    var peDO = peOIChange[stk] !== undefined ? peOIChange[stk] : 0;

    var ceSignal = getSignal(ceHeat, ceDO, "CE");
    var peSignal = getSignal(peHeat, peDO, "PE");

    outputRows.push([
      currTimeOnly,                          // Col A: Time
      s === 0 ? niftySpot : "",              // Col B: Spot (first row only)
      s === 0 ? currentWeekExpiry : "",      // Col C: Expiry (first row only)
      parseFloat(ceChange.toFixed(2)),       // Col D: CE ₹ Change
      prevCeLTP,                             // Col E: CE Prev LTP
      ceLTP,                                 // Col F: CE LTP
      ceDO,                                  // Col G: CE ΔOI
      ceOI,                                  // Col H: CE OI
      parseFloat(ceHeat.toFixed(2)),         // Col I: CE Heat Score
      ceSignal,                              // Col J: CE Signal
      stk,                                   // Col K: Strike (CENTER)
      peSignal,                              // Col L: PE Signal
      parseFloat(peHeat.toFixed(2)),         // Col M: PE Heat Score
      peOI,                                  // Col N: PE OI
      peDO,                                  // Col O: PE ΔOI
      peLTP,                                 // Col P: PE LTP
      prevPeLTP,                             // Col Q: PE Prev LTP
      parseFloat(peChange.toFixed(2))        // Col R: PE ₹ Change
    ]);
  }

  // Write batch
  var writeStartRow = targetRow;
  hmSheet.getRange(writeStartRow, 1, outputRows.length, 18).setValues(outputRows);

  // =========================
  // 13. COLOR CODE THE SIGNALS
  // =========================
  var redBg = "#ffcccc";
  var greenBg = "#ccffcc";
  var yellowBg = "#ffffcc";
  var greyBg = "#f0f0f0";

  for (var s = 0; s < outputRows.length; s++) {
    var row = writeStartRow + s;
    var ceSig = outputRows[s][9];     // CE Signal at index 9 (col J)
    var peSig = outputRows[s][11];    // PE Signal at index 11 (col L)

    // CE Signal coloring (column J = 10)
    if (ceSig.indexOf("BIG SHORT") !== -1) {
      hmSheet.getRange(row, 10).setBackground(redBg).setFontWeight("bold");
      hmSheet.getRange(row, 9).setBackground("#ff6666");    // CE Heat Score col I
    } else if (ceSig.indexOf("BIG LONG") !== -1) {
      hmSheet.getRange(row, 10).setBackground(greenBg).setFontWeight("bold");
      hmSheet.getRange(row, 9).setBackground("#66ff66");
    } else if (ceSig.indexOf("COVERING") !== -1 || ceSig.indexOf("UNWINDING") !== -1) {
      hmSheet.getRange(row, 10).setBackground(yellowBg);
    } else {
      hmSheet.getRange(row, 10).setBackground(greyBg);
    }

    // PE Signal coloring (column L = 12)
    if (peSig.indexOf("BIG SHORT") !== -1) {
      hmSheet.getRange(row, 12).setBackground(greenBg).setFontWeight("bold");
      hmSheet.getRange(row, 13).setBackground("#66ff66");   // PE Heat Score col M
    } else if (peSig.indexOf("BIG LONG") !== -1) {
      hmSheet.getRange(row, 12).setBackground(redBg).setFontWeight("bold");
      hmSheet.getRange(row, 13).setBackground("#ff6666");
    } else if (peSig.indexOf("COVERING") !== -1 || peSig.indexOf("UNWINDING") !== -1) {
      hmSheet.getRange(row, 12).setBackground(yellowBg);
    } else {
      hmSheet.getRange(row, 12).setBackground(greyBg);
    }

    if (ceSig.indexOf("BIG") !== -1 || peSig.indexOf("BIG") !== -1) {
      hmSheet.getRange(row, 1, 1, 18).setFontWeight("bold");
    }
  }

  // =========================
  // 14. UPDATE HEADER WITH LIVE VALUES
  // =========================
  hmSheet.getRange(3, 4).setValue("CE Thr: ₹" + ceThreshold.toFixed(2));
  hmSheet.getRange(3, 9).setValue("(1.2\u00D7\u03C3=" + ceStdDev.toFixed(2) + ")");
  hmSheet.getRange(3, 18).setValue("PE Thr: ₹" + peThreshold.toFixed(2));
  hmSheet.getRange(3, 13).setValue("(1.2\u00D7\u03C3=" + peStdDev.toFixed(2) + ")");

  // Format expiry for readable display (26707 → 2026-07-07)
  var expiryReadable = currentWeekExpiry;
  if (currentWeekExpiry.length === 5 && isNaN(currentWeekExpiry) === false) {
    var yy = "20" + currentWeekExpiry.substring(0, 2);
    var mm = currentWeekExpiry.substring(2, 3);
    var dd = currentWeekExpiry.substring(3, 5);
    expiryReadable = yy + "-" + mm + "-" + dd;
  }
  hmSheet.getRange(1, 1).setValue(
    "STRIKE HEAT MAP \u2014 NIFTY50 | Expiry: " + expiryReadable +
    " | " + currTimeOnly + " | Spot: " + niftySpot
  );

  Logger.log("=== STRIKE HEAT MAP v2.1 COMPLETE ===");
  Logger.log("Expiry: " + currentWeekExpiry + " | Strikes: " + strikes.length +
    " | Rows " + writeStartRow + "-" + (writeStartRow + outputRows.length - 1));

  for (var s = 0; s < outputRows.length; s++) {
    var stk = outputRows[s][10];    // Strike at index 10 (col K)
    var ceSig = outputRows[s][9];    // CE Signal at index 9 (col J)
    var peSig = outputRows[s][11];   // PE Signal at index 11 (col L)
    if (ceSig.indexOf("BIG") !== -1) Logger.log("\u26A1 " + stk + " CE: " + ceSig);
    if (peSig.indexOf("BIG") !== -1) Logger.log("\u26A1 " + stk + " PE: " + peSig);
  }
}
